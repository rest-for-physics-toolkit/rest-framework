113bool TRestDetectorSignalToHitsProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
114 const auto* signalEvent = dynamic_cast<const TRestDetectorSignalEvent*>(&input);
115 auto* hitsEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
116 if (!signalEvent || !hitsEvent) {
117 throw std::runtime_error(
118 "TRestDetectorSignalToHitsProcess: input/output is not a TRestDetectorSignalEvent/"
119 "TRestDetectorHitsEvent");
120 }
121
122 hitsEvent->Initialize();
123 hitsEvent->SetEventInfo(&input);
124
125 for (int s = 0; s < signalEvent->GetNumberOfSignals(); s++) {
126 const auto& signal = signalEvent->GetSignal(s);
127 if (signal.GetNPoints() == 0) continue;
128 const int daqID = signal.GetSignalID();
129
130 auto channelPos = fReadout->GetPositionFromChannel(daqID);
131 if (std::isnan(channelPos.X()) && std::isnan(channelPos.Y()) && std::isnan(channelPos.Z())) {
132 RESTDebug << "TRestDetectorSignalToHitsProcess: channel " << daqID << " not found in readout"
133 << RESTendl;
134 continue;
135 }
136
137 // (time, energy) pairs of the hits created by this signal
138 std::vector<std::pair<double, double>> hits;
139 auto graph = signal.GetGraph();
140 if (fMethod == "intWindow") {
141 for (const auto& hit : GetIntWindow(graph, fIntWindow)) {
142 if (hit.second > fThreshold) hits.push_back(hit);
143 }
144 } else if (fMethod == "integral") {
145 const bool wholeSignal = fIntegralRange.first >= fIntegralRange.second;
146 // Negative times are valid (e.g. a trigger delay): a flag marks the maximum found
147 double charge = 0, maxCharge = 0, maxTime = 0;
148 bool found = false;
149 for (int j = 0; j < signal.GetNPoints(); j++) {
150 const auto [time, value] = signal.GetPoint(j);
151 if (!wholeSignal && (time < fIntegralRange.first || time > fIntegralRange.second)) continue;
152 charge += value;
153 if (!found || value > maxCharge) {
154 maxCharge = value;
155 maxTime = time;
156 found = true;
157 }
158 }
159 if (found && charge > 0) hits.emplace_back(maxTime, charge);
160 } else if (fMethod == "qCenter") {
161 double charge = 0, weightedTime = 0;
162 for (int j = 0; j < signal.GetNPoints(); j++) {
163 const auto [time, value] = signal.GetPoint(j);
164 charge += value;
165 weightedTime += time * value;
166 }
167 if (charge > 0) hits.emplace_back(weightedTime / charge, charge / signal.GetNPoints());
168 } else if (fMethod == "all") {
169 for (int j = 0; j < signal.GetNPoints(); j++) hits.push_back(signal.GetPoint(j));
170 } else {
171 const bool smooth = fSmoothingPoints > 1 && (fMethod == "onlyMax" || fMethod == "tripleMax" ||
172 fMethod == "tripleMaxAverage");
173 if (smooth) {
174 const auto smoothed = GetSignalSmoothed(signal.GetCharge(), fSmoothingPoints);
175 for (int j = 0; j < graph.GetN(); j++) graph.SetPointY(j, smoothed[j]);
176 }
177 const auto hit = GetMax(graph, fMethod);
178 // Failures (empty signal, failed agetFit) return (-1, -1), rejected by the amplitude: negative
179 // times are valid. A failed gaussFit or landauFit gives the maximum point
180 if (std::isfinite(hit.first) && hit.second > 0) hits.push_back(hit);
181 }
182
183 const auto type = TRestHitsData::GetHitType(channelPos.X(), channelPos.Y(), channelPos.Z());
184 for (const auto& [time, energy] : hits) hitsEvent->AddHit(channelPos, energy, time, type);
185 }
186
187 return hitsEvent->GetNumberOfHits() > 0;
188}
