44void PeriodicPrint(SimulationManager* simulationManager) {
45 const auto restG4Metadata = simulationManager->GetRestMetadata();
46
47 while (!simulationManager->GetPeriodicPrintThreadEndFlag()) {
48 std::this_thread::sleep_for(std::chrono::seconds(2));
49
50 if (!restG4Metadata->PrintProgress() &&
51 restG4Metadata->GetVerboseLevel() < TRestLogManager::REST_Verbose_Level::REST_Silent) {
52 continue;
53 }
54
55 for (auto& outputManager : simulationManager->GetOutputManagerContainer()) {
56 simulationManager->SyncStatsFromChild(outputManager);
57 }
58
59 string outputPercentageType = "events";
60 double completionPercentage = double(simulationManager->GetNumberOfProcessedEvents()) /
61 double(simulationManager->GetNumberOfEventsInRun()) * 100;
62 if (restG4Metadata->GetNumberOfRequestedEntries() > 0) {
63 auto completionPercentageEntries = double(simulationManager->GetNumberOfStoredEvents()) /
64 double(restG4Metadata->GetNumberOfRequestedEntries()) * 100;
65 if (completionPercentageEntries > completionPercentage) {
66 completionPercentage = completionPercentageEntries;
67 outputPercentageType = "entries";
68 }
69 }
70 if (restG4Metadata->GetSimulationMaxTimeSeconds() > 0) {
71 auto completionPercentageTime = double(simulationManager->GetElapsedTime()) /
72 double(restG4Metadata->GetSimulationMaxTimeSeconds()) * 100;
73 if (completionPercentageTime > completionPercentage) {
74 completionPercentage = completionPercentageTime;
75 outputPercentageType = "time";
76 }
77 }
78
79 string progress;
80 if (outputPercentageType == "events") {
81 progress = " | " + to_string(simulationManager->GetNumberOfProcessedEvents()) +
82 " events processed / " + to_string(simulationManager->GetNumberOfEventsInRun()) +
83 " requested (" +
84 TRestTools::Format("%.2e", simulationManager->GetNumberOfProcessedEvents() /
85 simulationManager->GetElapsedTime()) +
86 "/s)";
87 } else if (outputPercentageType == "entries") {
88 progress = " | " + to_string(simulationManager->GetNumberOfStoredEvents()) + " entries / " +
89 to_string(restG4Metadata->GetNumberOfRequestedEntries()) + " requested";
90 } else if (outputPercentageType == "time") {
91 progress = " | " + TRestTools::ToTimeStringLong(simulationManager->GetElapsedTime()) +
92 " elapsed / " +
93 TRestTools::ToTimeStringLong(
94 simulationManager->GetRestMetadata()->GetSimulationMaxTimeSeconds());
95 }
96
97 string timeInfo;
98 if (outputPercentageType != "time") {
99 timeInfo = " | " + TRestTools::ToTimeStringLong(simulationManager->GetElapsedTime()) + " elapsed";
100 }
101
102 G4cout << TRestTools::Format("%5.2f", completionPercentage) << "% | "
103 << simulationManager->GetNumberOfStoredEvents() << " events stored / "
104 << simulationManager->GetNumberOfProcessedEvents() << " processed ("
105 << TRestTools::Format("%.2e", simulationManager->GetNumberOfStoredEvents() /
106 simulationManager->GetElapsedTime())
107 << "/s)" << progress << timeInfo << G4endl;
108 }

121void SimulationManager::BeginOfRunAction() {
122 if (G4Threading::IsMultithreadedApplication() && G4Threading::G4GetThreadId() != -1) {
123 return;
124 }
125
126 gCurrentSimulationManager = this;
127
128 std::signal(SIGINT, interruptSignalHandler);
129
130 // The start of the simulation: the biasing stages are several runs, the time limit is for all of them
131 if (fTimeStartUnix == 0) fTimeStartUnix = chrono::steady_clock::now().time_since_epoch().count();
132 // The counters of the threads were taken at the end of the previous run
133 fNumberOfProcessedEvents = 0;

135 fPeriodicPrintThreadEndFlag = false;

138 if (GetRestMetadata()->PrintProgress() ||
139 GetRestMetadata()->GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Silent) {

145void SimulationManager::EndOfRunAction() {
146 if (G4Threading::IsMultithreadedApplication() && G4Threading::G4GetThreadId() != -1) {
147 return; // Only call this once from the main thread
148 }
149
150 fPeriodicPrintThreadEndFlag = true;
151
152 WriteEvents();
153
154 if (fPeriodicPrintThread != nullptr) {
155 if (fPeriodicPrintThread->joinable()) {
156 fPeriodicPrintThread->join(); // need to join thread, it may block for up to 1 thread period
157 }
158 }
159
160 // The output managers (one per worker thread, built once by ActionInitialization::Build) are kept for
161 // the next run (/run/beamOn again), deleted by the destructor
162 for (auto& outputManager : fOutputManagerContainer) {
163 fNumberOfProcessedEvents += outputManager->TakeEventCounter();
164 }
165 // The primaries of the generator: the first biasing stage (the only one without biasing)
166 const int stage = fBiasingStage;
167 if (stage == 0) GetRestMetadata()->SetNumberOfEvents(fNumberOfProcessedEvents);
168 if (IsBiasingActive()) {
169 GetRestMetadata()->SetBiasingStageEvents(stage, fNumberOfProcessedEvents);
170 lock_guard<mutex> guard(fBiasingMutex);
171 GetRestMetadata()->SetBiasingStageParticles(stage, static_cast<int64_t>(fBiasingParticles.size()));
172 }
173
174 // cry generator: the showers simulated and their exposure, to normalize to a rate
175 auto& generatorInfo = GetRestMetadata()->GetGeant4PrimaryGeneratorInfo();
176 if (generatorInfo.IsCRY() && stage == 0) {
177 generatorInfo.UpdateCRYExposure();
178 G4cout << "CRY showers simulated: " << generatorInfo.GetCRYShowersUsed() << ", exposure "
179 << generatorInfo.GetCRYExposureTime() / 1e6 << " s" << G4endl;
180 }
181}

198void SimulationManager::WriteEvents(bool blocking) {
199 unique_lock<mutex> writeLock(fWriteMutex, defer_lock);
200 if (blocking) {
201 writeLock.lock();
202 } else if (!writeLock.try_lock()) {
203 return;
204 }
205
206 bool wrote = false;
207 while (true) {
208 unique_ptr<TRestGeant4Event> event;
209 {
210 lock_guard<mutex> guard(fSimulationManagerMutex);
211 if (fEventContainer.empty()) {
212 break;
213 }
214 event = std::move(fEventContainer.front());
215 fEventContainer.pop();
216 }
217
218 fEvent.MoveFrom(std::move(*event));
219 event.reset();
220
221 if (!fRestGeant4Metadata->GetStoreTracks()) {
222 fEvent.ClearTracks();
223 }
224
225 fRestRun->Fill();
226 wrote = true;
227 }
228
229 if (!wrote) {
230 return;
231 }
232
233 fNumberOfStoredEvents = fRestRun->GetSavedEntries();
234
235 const auto nRequestedEntries = GetRestMetadata()->GetNumberOfRequestedEntries();
236 if (nRequestedEntries > 0 && !fAbortFlag && fNumberOfStoredEvents >= nRequestedEntries) {
237 G4cout << "Stopping Run! We have reached the number of requested entries (" << nRequestedEntries
238 << ")" << endl;
239 StopSimulation();
240 }

302int64_t SimulationManager::StartNextBiasingStage() {
303 const auto& volumes = fRestGeant4Metadata->GetBiasingVolumes();
304 const int stage = fBiasingStage;
305 lock_guard<mutex> guard(fBiasingMutex);
306 // The order of the threads is not reproducible, the event IDs (and the order inside each event) are
307 std::sort(fBiasingParticles.begin(), fBiasingParticles.end(),
308 [](const BiasingParticle& a, const BiasingParticle& b) {
309 return a.eventID != b.eventID ? a.eventID < b.eventID : a.sequence < b.sequence;
310 });
311 fBiasingReplayParticles = std::move(fBiasingParticles);
312 fBiasingParticles.clear();
313 fBiasingReplicas = static_cast<int>(volumes.at(stage).GetBiasingFactor());
314 fBiasingStage = stage + 1;
315 return static_cast<int64_t>(fBiasingReplayParticles.size()) * fBiasingReplicas;
316}

340void OutputManager::BeginOfEventAction() {
341 // This should only be executed once at BeginOfEventAction
342 UpdateEvent();
343 fProcessedEventsCounter++;
344 fBiasingKilledSecondaries.clear();
345 fBiasingSequence = 0;
346
347 if (fSimulationManager->GetAbortFlag()) {
348 G4RunManager::GetRunManager()->AbortRun(true);
349 }
350
351 // activeVolumes chance: the hits of each active volume are stored in this event with its probability
352 // (legacy semantics; the Geant4 random engine of the thread, reproducible with the seed)
353 fVolumesNotStored.clear();
354 const auto* metadata = fSimulationManager->GetRestMetadata();
355 for (unsigned int i = 0; i < metadata->GetNumberOfActiveVolumes(); i++) {
356 const auto& volume = metadata->GetActiveVolumeName(i);
357 const double chance = metadata->GetStorageChance(volume);
358 if (chance < 1 && G4UniformRand() >= chance) fVolumesNotStored.insert(volume);
359 }
360
361 if (fSimulationManager->GetRestMetadata()->GetSimulationMaxTimeSeconds() != 0 &&
362 !fSimulationManager->GetAbortFlag() &&
363 fSimulationManager->GetElapsedTime() >
364 fSimulationManager->GetRestMetadata()->GetSimulationMaxTimeSeconds()) {
365 G4cout << "Stopping Run! We have reached the time limit of "
366 << TRestTools::ToTimeStringLong(
367 fSimulationManager->GetRestMetadata()->GetSimulationMaxTimeSeconds())
368 << endl;
369 fSimulationManager->StopSimulation();
370 }

373void OutputManager::UpdateEvent() {
374 auto event = G4EventManager::GetEventManager()->GetConstCurrentEvent();
375
376 if (!fEvent) {
377 fEvent = make_unique<TRestGeant4Event>();
378 } else {
379 // The event not submitted (not valid) is reused: everything is reset (tracks, hits, energies,
380 // volumes, sub-event data and sub-event ID, the caller sets it for the next sub-event)
381 fEvent->Initialize();
382 }
383 fEvent->SetID(event->GetEventID());
384 fEvent->SetOK(true);
385 // Unix time with nanoseconds: a rate in bins shorter than a second
386 const std::chrono::duration<double> now = std::chrono::system_clock::now().time_since_epoch();
387 fEvent->SetTime(now.count());
388
389 if (fSimulationManager->GetRestMetadata() != nullptr) {
390 fEvent->SetGeant4Metadata(fSimulationManager->GetRestMetadata());
391 }
392
393 if (fSimulationManager->GetRestRun() != nullptr) {
394 fEvent->InitializeReferences(fSimulationManager->GetRestRun());
395 }
396
397 // The primaries of the current Geant4 event (also set by the primary generator, before)
398 fEvent->UpdatePrimaryData(event);
399
400 const auto metadata = fSimulationManager->GetRestMetadata();
401 if (metadata == nullptr) {
402 std::ostringstream errMsg;
403 errMsg << "\n---------------------------------------------------------\n"
404 << "TRestGeant4Event: Missing Geant4 metadata for event ID: " << fEvent->GetID();
405 G4Exception("OutputManager::UpdateEvent", "REST_G4_MISSING_METADATA", FatalException,
406 errMsg.str().c_str());
407 }
408
409 // The primary generator gives a vertex per particle: the particles of all the vertices
410 int primaryParticles = 0;
411 for (int v = 0; v < event->GetNumberOfPrimaryVertex(); ++v) {
412 const auto* primaryVertex = event->GetPrimaryVertex(v);
413 if (primaryVertex != nullptr) primaryParticles += primaryVertex->GetNumberOfParticle();
414 }
415 const int numberOfSources = metadata->GetNumberOfSources();
416
417 // A Decay0 source and the cry generator give the particles of their event; a biasing stage after the
418 // first one replays a stored particle
419 const auto& generatorInfo = metadata->GetGeant4PrimaryGeneratorInfo();
420 const bool variableParticles = generatorInfo.HasDecay0Sources() || generatorInfo.IsCRY() ||
421 fSimulationManager->GetBiasingStage() > 0;
422 if (!variableParticles && primaryParticles != numberOfSources &&
423 int(fEvent->GetNumberOfPrimaries()) != numberOfSources) {
424 std::ostringstream errMsg;
425 errMsg << "\n---------------------------------------------------------\n"
426 << "TRestGeant4Event: Number of particles on primary vertex does not match number of sources "
427 "for event ID: "
428 << fEvent->GetID();
429 G4Exception("OutputManager::UpdateEvent", "REST_G4_SOURCE_MISTMATCH", FatalException,
430 errMsg.str().c_str());
431 }
432}

455 // Only the events of the last biasing stage are written

503void OutputManager::ApplyBiasingVolume(const G4Step* step, const TRestGeant4BiasingVolume& volume) {
504 const auto* pre = step->GetPreStepPoint();
505 const auto* post = step->GetPostStepPoint();
506 const G4ThreeVector& from = pre->GetPosition();
507 const G4ThreeVector& to = post->GetPosition();
508 const double t = volume.GetEntryFraction({from.x() / mm, from.y() / mm, from.z() / mm},
509 {to.x() / mm, to.y() / mm, to.z() / mm});
510 if (t < 0) return;
511
512 // A kill volume (TRestGeant4Event::InsertStep) has already stopped the track: nothing to bias
513 const auto* metadata = fSimulationManager->GetRestMetadata();
514 const auto* preVolume = pre->GetPhysicalVolume();
515 if (preVolume != nullptr && metadata->IsKillVolume(preVolume->GetName())) return;
516
517 auto* track = const_cast<G4Track*>(step->GetTrack());
518 track->SetTrackStatus(fStopAndKill);
519
520 // The secondaries created after the crossing point (the ones of the interaction at the post-step point)
521 // are part of the history replayed by the next stage
522 const G4ThreeVector segment = to - from;
523 const double length2 = segment.mag2();
524 if (const auto* secondaries = step->GetSecondaryInCurrentStep()) {
525 for (const auto* secondary : *secondaries) {
526 const double u = length2 > 0 ? (secondary->GetPosition() - from).dot(segment) / length2 : 1.0;
527 if (u >= t) fBiasingKilledSecondaries.insert(secondary);
528 }
529 }
530
531 // The state before the crossing point: exact for neutral particles, for charged ones the energy loss
532 // and the deflection in the part of the step outside the volume are neglected
533 const double energy = pre->GetKineticEnergy();
534 if (energy / keV < volume.GetMinEnergy() || energy / keV > volume.GetMaxEnergy()) return;
535
536 BiasingParticle particle;
537 particle.definition = track->GetParticleDefinition();
538 particle.charge = track->GetDynamicParticle()->GetCharge();
539 particle.energy = energy;
540 particle.time = pre->GetGlobalTime() + t * (post->GetGlobalTime() - pre->GetGlobalTime());
541 particle.position = from + t * segment;
542 particle.direction = pre->GetMomentumDirection();
543 particle.polarization = pre->GetPolarization();
544 particle.eventID = G4EventManager::GetEventManager()->GetConstCurrentEvent()->GetEventID();
545 particle.sequence = fBiasingSequence++;
546 fSimulationManager->StoreBiasingParticle(particle);
547}
