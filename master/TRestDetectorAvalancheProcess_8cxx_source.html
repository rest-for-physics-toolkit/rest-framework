76double Gamma(TRandom& random, double shape, double scale) {
77 if (!(shape > 0) || !(scale > 0)) return 0;
78 if (shape < 1) {
79 // Gamma(shape + 1) U^(1 / shape)
80 const double gamma = Gamma(random, shape + 1, scale);
81 return gamma * std::pow(random.Rndm(), 1 / shape);
82 }
83 const double d = shape - 1. / 3;
84 const double c = 1 / std::sqrt(9 * d);
85 while (true) {
86 const double x = random.Gaus(0, 1);
87 double v = 1 + c * x;
88 if (v <= 0) continue;
89 v = v * v * v;
90 const double u = random.Rndm();
91 if (u < 1 - 0.0331 * x * x * x * x || std::log(u) < 0.5 * x * x + d * (1 - v + std::log(v))) {
92 return d * v * scale;
93 }
94 }
95}

151void TRestDetectorAvalancheProcess::InitProcess() {
152 const std::string user = "TRestDetectorAvalancheProcess";
153 if (!fResponseFile.empty() && (fAmplificationVoltage != 0 || fGap != 0)) {
154 throw std::runtime_error(user + ": responseFile or amplificationVoltage and gap, not both");
155 }
156 if (fResponseFile.empty() && (fAmplificationVoltage <= 0 || fGap <= 0)) {
157 throw std::runtime_error(user + ": amplificationVoltage and gap must be > 0 (or responseFile)");
158 }
159 if (!fManager) throw std::runtime_error(user + ": no TRestManager");
160 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "charge", user);
161 if (!medium) throw std::runtime_error(user + ": needs a TRestDetectorMedium with charge");
162 fCharge = &medium->GetChargeResponse(user);
163 if (fCharge->GetWValue() <= 0) {
164 throw std::runtime_error(user + ": the charge of the medium '" + medium->GetName() +
165 "' needs wValue > 0");
166 }
167
168 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
169 fRandom.SetSeed(fSeed);
170 fMediumName = medium->GetName();
171
172 if (!fResponseFile.empty()) {
173 // The amplification in a field map: the response instead of the gain table
174 std::unique_ptr<TFile> file(TFile::Open(fResponseFile.c_str(), "READ"));
175 if (!file || file->IsZombie()) throw std::runtime_error(user + ": cannot open " + fResponseFile);
176 fResponse = std::make_shared<const TRestDetectorAmplificationResponse>(
177 TRestDetectorAmplificationResponse::Read(file.get()));
178 if (fResponse->fCollected <= 0) {
179 throw std::runtime_error(user + ": the amplification response of " + fResponseFile +
180 " has no avalanche at the anode");
181 }
182 if (!fResponse->fMedium.empty() && fResponse->fMedium != fMediumName) {
183 RESTWarning << user << ": the amplification response of " << fResponseFile
184 << " is for the medium '" << fResponse->fMedium << "', not '" << fMediumName << "'"
185 << RESTendl;
186 }
187 fGain = fResponse->fMeanSize;
188 fTheta = fResponse->GetTheta();
189 fTransparency = fResponse->GetTransparency();
190 fSharingSigma = fResponse->GetSharingSigma();
191 fReadout = nullptr;
192 fCellChecked = fNormalWarned = false;
193 if (fResponse->HasCellMap() || fSharingSigma > 0) {
194 // The hits located in the pixels of the readout, displaced in its plane
195 auto* readouts = dynamic_cast<TRestDetectorReadoutManager*>(
196 fManager->GetMetadataClass("TRestDetectorReadoutManager"));
197 if (!readouts) {
198 throw std::runtime_error(user + ": the amplification response of " + fResponseFile +
199 " has the map of the cell or charge sharing: a "
200 "TRestDetectorReadoutManager is needed");
201 }
202 fReadout = readouts->SelectReadout(fReadoutName, user);
203 }
204 UpdateYAMLFromParams<TRestDetectorAvalancheProcess>(fNode);
205 return;
206 }
207 fResponse.reset();
208 fReadout = nullptr;
209 fSharingSigma = 0;
210 fTransparency = 1;
211
212 const auto& table = fCharge->GetGainTable(fGap, fTable);
213 const double field = fAmplificationVoltage / fGap; // V/mm
214 fGain = table.GetGain(field);
215 fTheta = table.GetTheta(field);
216 if (fGain <= 0 || fTheta <= -1) {
217 throw std::runtime_error(user + ": table " + table.GetKey() + " gives gain " + std::to_string(fGain) +
218 " and theta " + std::to_string(fTheta) +
219 " (gain > 0 and theta > -1 needed)");
220 }
221 fTable = table.GetKey();
222 // The stored metadata keeps the medium, the table and the gain used
223 UpdateYAMLFromParams<TRestDetectorAvalancheProcess>(fNode);
224}

226void TRestDetectorAvalancheProcess::CheckCell(const TRestDetectorReadout::PixelPosition& pixel) {
227 if (fCellChecked) return;
228 fCellChecked = true;
229 const double cellX = fResponse->fCellHigh[0] - fResponse->fCellLow[0];
230 const double cellY = fResponse->fCellHigh[1] - fResponse->fCellLow[1];
231 const double pixelX = 2 * pixel.halfSize.X(), pixelY = 2 * pixel.halfSize.Y();
232 if (std::abs(cellX / pixelX - 1) > 0.05 || std::abs(cellY / pixelY - 1) > 0.05) {
233 RESTWarning << "TRestDetectorAvalancheProcess: the cell of the amplification response of "
234 << fResponseFile << " (" << cellX << " x " << cellY
235 << " mm) is not the size of the pixel "
236 << "of the readout (" << pixelX << " x " << pixelY << " mm): the map of the cell is "
237 << "stretched to the pixel" << RESTendl;
238 }

241ROOT::Math::XYZVector TRestDetectorAvalancheProcess::Share(const ROOT::Math::XYZVector& position) {
242 using V = ROOT::Math::XYZVector;
243 const V normal = fReadout->GetReadoutNormal(position);
244 if (!(normal.R() > 0)) {
245 if (!fNormalWarned) {
246 fNormalWarned = true;
247 RESTWarning << "TRestDetectorAvalancheProcess: the readout has no normal at (" << position.X()
248 << ", " << position.Y() << ", " << position.Z()
249 << ") mm: no charge sharing for the hits there" << RESTendl;
250 }
251 return position;
252 }
253 // Two orthogonal axes of the plane (the Gaussian is the same along any axis of it)
254 const V axis = std::abs(normal.X()) < 0.9 ? V(1, 0, 0) : V(0, 1, 0);
255 const V first = axis.Cross(normal).Unit();
256 const V second = normal.Cross(first);
257 const double along = fRandom.Gaus(0, fSharingSigma);
258 const double across = fRandom.Gaus(0, fSharingSigma);
259 return position + along * first + across * second;
260}

262bool TRestDetectorAvalancheProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
263 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
264 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
265 if (!inputEvent || !outputEvent) {
266 throw std::runtime_error(
267 "TRestDetectorAvalancheProcess: input/output is not a TRestDetectorHitsEvent");
268 }
269
270 outputEvent->Initialize();
271 outputEvent->SetEventInfo(inputEvent);
272 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
273
274 const double wValue = fCharge->GetWValue();
275 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
276 const double energy = inputEvent->GetEnergy(n);
277 if (energy <= 0) continue;
278 const double electrons = energy / wValue;
279 if (fResponse) {
280 // Each electron amplified with the transparency, its gain and the time of the hit from the
281 // response; with the map, those of the bin of the hit in the cell, its pixel
282 double transparency = fTransparency, factor = 1;
283 TRestDetectorReadout::PixelPosition pixel;
284 if (fReadout && fReadout->GetPixelPosition(inputEvent->GetPosition(n), pixel) &&
285 pixel.halfSize.X() > 0 && pixel.halfSize.Y() > 0) {
286 CheckCell(pixel);
287 const int bin = fResponse->GetCellBin(0.5 + pixel.local.X() / (2 * pixel.halfSize.X()),
288 0.5 + pixel.local.Y() / (2 * pixel.halfSize.Y()));
289 transparency = fResponse->GetCellTransparency(bin);
290 factor = fResponse->GetCellGainFactor(bin);
291 }
292 const long count = std::max(1L, std::lround(electrons));
293 double gain = 0;
294 for (long e = 0; e < count; e++) {
295 if (fRandom.Uniform() < transparency) gain += factor * fResponse->SampleSize(fRandom);
296 }
297 if (!(gain > 0)) continue;
298 const double time = inputEvent->GetTime(n) + fResponse->SampleTime(fRandom);
299 // Charge sharing: the diffusion below the start plane, after the bin of the hit in the cell
300 const auto position =
301 fSharingSigma > 0 ? Share(inputEvent->GetPosition(n)) : inputEvent->GetPosition(n);
302 outputEvent->AddHit(position, energy * gain / count, time, inputEvent->GetType(n),
303 inputEvent->GetParticleID(n));
304 continue;
305 }
306 // Sum of the Polya gains of the electrons of the hit
307 const double gain = Gamma(fRandom, electrons * (fTheta + 1), fGain / (fTheta + 1));
308 outputEvent->AddHit(inputEvent->GetPosition(n), energy * gain / electrons, inputEvent->GetTime(n),
309 inputEvent->GetType(n), inputEvent->GetParticleID(n));
310 }
311
312 return outputEvent->GetNumberOfHits() > 0;
313}
