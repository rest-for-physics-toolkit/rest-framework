18// FFTW only guarantees thread safety when executing a plan, not when creating
19// or destroying it. Plan handling is serialized; execution runs unlocked.
20// Option K: each plan owns its transform (without it TVirtualFFT::FFT returns and
21// reinitializes the shared global fgFFT, used by every thread).
22std::mutex gPlanMutex;
23
24class FFTPlan {
25 public:
26 FFTPlan(int n, const char* option) {
27 std::lock_guard<std::mutex> lock(gPlanMutex);
28 fFFT = TVirtualFFT::FFT(1, &n, option);
29 }
30 ~FFTPlan() {
31 std::lock_guard<std::mutex> lock(gPlanMutex);
32 delete fFFT;
33 }
34 FFTPlan(const FFTPlan&) = delete;
35 FFTPlan& operator=(const FFTPlan&) = delete;
36
37 TVirtualFFT* Get() const { return fFFT; }
38
39 private:
40 TVirtualFFT* fFFT = nullptr;
41};

44Spectrum RunC2C(const std::vector<double>& re, const std::vector<double>& im, bool forward) {
45 const int n = (int)re.size();
46 if (n <= 0) return {};
47
48 FFTPlan plan(n, forward ? "C2CFORWARD ES K" : "C2CBACKWARD ES K");
49 if (!plan.Get()) {
50 RESTError << "TRestFFTAnalysis: could not create the FFT plan (is FFTW available in this ROOT?)"
51 << RESTendl;
52 return {};
53 }
54
55 plan.Get()->SetPointsComplex(re.data(), im.data());
56 plan.Get()->Transform();
57
58 std::vector<double> outRe(n), outIm(n);
59 plan.Get()->GetPointsComplex(outRe.data(), outIm.data());
60
61 Spectrum out(n);
62 for (int i = 0; i < n; i++) out[i] = {outRe[i], outIm[i]};
63 return out;
64}

67void SetHermitian(Spectrum& s, int k, std::complex<double> value) {
68 const int n = (int)s.size();
69 if (k == 0 || 2 * k == n) {
70 s[k] = value.real();
71 return;
72 }
73 s[k] = value;
74 s[n - k] = std::conj(value);
75}

77template <typename Op>
78bool CombineSpectra(Spectrum& s, const Spectrum& other, int from, int to, const char* caller, Op op) {
79 if (s.size() != other.size()) {
80 RESTError << "TRestFFTAnalysis::" << caller << ": spectra have different sizes (" << s.size()
81 << " vs " << other.size() << ")" << RESTendl;
82 return false;
83 }
84 const int n = (int)s.size();
85 // Up to the Nyquist bin (n / 2) included
86 if (to <= 0 || to > n / 2 + 1) to = n / 2 + 1;
87 for (int i = std::max(0, from); i < to; i++) SetHermitian(s, i, op(s[i], other[i]));
88 return true;
89}

100Spectrum ForwardFFT(const std::vector<T>& signal, int nStart, int nEnd) {
101 const int n = (int)signal.size() - nStart - nEnd;
102 if (nStart < 0 || nEnd < 0 || n <= 0) {
103 RESTError << "TRestFFTAnalysis::ForwardFFT: invalid range (size " << signal.size() << ", nStart "
104 << nStart << ", nEnd " << nEnd << ")" << RESTendl;
105 return {};
106 }
107
108 std::vector<double> re(signal.begin() + nStart, signal.begin() + nStart + n);
109 std::vector<double> im(n, 0.0);
110 return RunC2C(re, im, true);
111}

117std::vector<double> BackwardFFT(const Spectrum& spectrum) {
118 const int n = (int)spectrum.size();
119 std::vector<double> re(n), im(n);
120 for (int i = 0; i < n; i++) {
121 re[i] = spectrum[i].real();
122 im[i] = spectrum[i].imag();
123 }
124
125 const Spectrum out = RunC2C(re, im, false);
126
127 std::vector<double> signal(out.size());
128 for (size_t i = 0; i < out.size(); i++) signal[i] = out[i].real() / n;
129 return signal;
130}

142Spectrum GaussianSecondOrderResponse(int nfft, double f1, double f2, double Ao, double sigma) {
143 Spectrum s(std::max(nfft, 0));
144 const double a = std::sqrt(Ao);
145 for (int i = 0; i <= nfft / 2; i++) {
146 const double w = 2. * i / 3; // scale factor of the legacy code
147 std::complex<double> value =
148 std::complex<double>(f1, 0) / std::complex<double>(f1 * f2 - w * w, f1 * w);
149 value *= a * std::exp(-sigma * w * w);
150 SetHermitian(s, i, value);
151 }
152 return s;
153}

155Spectrum SecondOrderAnalyticalResponse(int nfft, double f1, double f2, double to) {
156 Spectrum s(std::max(nfft, 0));
157 for (int i = 0; i <= nfft / 2; i++) {
158 const double w = 2.59 * i; // scale factor of the legacy code
159 std::complex<double> value =
160 std::complex<double>(f1, 0) / std::complex<double>(f1 * w, f1 * f2 - w * w);
161 value *= std::exp(std::complex<double>(0, -w * to));
162 SetHermitian(s, i, value);
163 }
164 return s;
165}

171void ButterWorthFilter(Spectrum& s, int cutOff, int order) {
172 if (cutOff == 0) {
173 throw std::invalid_argument(
174 "TRestFFTAnalysis::ButterWorthFilter: cutOff must be nonzero (the gain divides by it)");
175 }
176 const int n = (int)s.size();
177 for (int i = std::max(0, cutOff + 1); i <= n / 2; i++) {
178 const double gain = 1.0 / std::sqrt(1.0 + std::pow((double)i / cutOff, 2.0 * order));
179 SetHermitian(s, i, s[i] * gain);
180 }
181}

218bool ApplyResponse(Spectrum& s, const Spectrum& response, int cutOff) {
219 const int n = (int)s.size();
220 if (cutOff <= 0 || cutOff >= n / 2) {
221 RESTError << "TRestFFTAnalysis::ApplyResponse: cutOff must be in (0, " << n / 2 << "), got " << cutOff
222 << RESTendl;
223 return false;
224 }
225
226 if (!DivideBy(s, response, 0, cutOff)) return false;
227
228 // Above the cutOff the spectrum is not deconvolved: it is rescaled so its
229 // amplitude is continuous with the last deconvolved bin.
230 const double denom = std::norm(s[cutOff]);
231 if (denom <= 0) return true; // nothing above cutOff to rescale
232
233 const double scale = std::sqrt(std::norm(s[cutOff - 1]) / denom);
234 for (int i = cutOff; i <= n / 2; i++) SetHermitian(s, i, s[i] * scale);
235 return true;
236}
