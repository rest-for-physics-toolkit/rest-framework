100bool TRestDetectorHitsTransformationProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
101 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
102 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
103 if (!inputEvent || !outputEvent) {
104 throw std::runtime_error(
105 "TRestDetectorHitsTransformationProcess: input/output is not a TRestDetectorHitsEvent");
106 }
107
108 outputEvent->Initialize();
109 outputEvent->SetEventInfo(inputEvent);
110
111 const XYZVector mirrorPosition = ToVector(fMirrorPosition);
112 XYZVector mirrorNormal = ToVector(fMirrorNormal);
113 const bool mirror = mirrorNormal.R() > 0;
114 if (mirror) mirrorNormal = mirrorNormal.Unit();
115
116 const bool rotate = fRotationAngle != 0;
117 const XYZVector rotationCenter = ToVector(fRotationCenter);
118 const ROOT::Math::AxisAngle rotation(ToVector(fRotationAxis), rotate ? (double)fRotationAngle : 0.);
119
120 const XYZVector translation = ToVector(fTranslation);
121
122 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
123 XYZVector position = inputEvent->GetPosition(n);
124 const auto type = inputEvent->GetType(n);
125
126 if (type == TRestHitsData::REST_HitType::XYZ) {
127 if (mirror) {
128 const XYZVector relative = position - mirrorPosition;
129 position = mirrorPosition + relative - 2 * relative.Dot(mirrorNormal) * mirrorNormal;
130 }
131 if (rotate) position = rotationCenter + rotation * (position - rotationCenter);
132 position += translation;
133 }
134
135 outputEvent->AddHit(position, inputEvent->GetEnergy(n) * fEnergyFactor, inputEvent->GetTime(n), type,
136 inputEvent->GetParticleID(n));
137 }
138
139 return true;
140}
