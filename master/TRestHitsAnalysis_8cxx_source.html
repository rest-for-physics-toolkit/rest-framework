67// Held-Karp: shortest path visiting the subset `mask` and ending at `j`, built subset by subset. Open
68// paths start anywhere, cyclic ones at point 0 (and come back to it)
69std::vector<size_t> GetShortestPathExact(const std::vector<std::vector<double>>& distance, bool cyclic) {
70 const size_t n = distance.size();
71 const size_t full = (size_t)1 << n;
72 const double inf = std::numeric_limits<double>::infinity();
73 std::vector<double> cost(full * n, inf);
74 std::vector<int> previous(full * n, -1);
75 if (cyclic) {
76 cost[1 * n + 0] = 0;
77 } else {
78 for (size_t j = 0; j < n; j++) cost[((size_t)1 << j) * n + j] = 0;
79 }
80
81 for (size_t mask = 1; mask < full; mask++) {
82 for (size_t j = 0; j < n; j++) {
83 const double c = cost[mask * n + j];
84 if (c == inf) continue;
85 for (size_t k = 0; k < n; k++) {
86 if (mask & ((size_t)1 << k)) continue;
87 const size_t next = (mask | ((size_t)1 << k)) * n + k;
88 if (c + distance[j][k] < cost[next]) {
89 cost[next] = c + distance[j][k];
90 previous[next] = (int)j;
91 }
92 }
93 }
94 }
95
96 size_t last = 0;
97 double best = inf;
98 for (size_t j = 0; j < n; j++) {
99 const double c = cost[(full - 1) * n + j] + (cyclic ? distance[j][0] : 0);
100 if (c < best) {
101 best = c;
102 last = j;
103 }
104 }
105
106 std::vector<size_t> path;
107 size_t mask = full - 1;
108 int j = (int)last;
109 while (j >= 0) {
110 path.push_back(j);
111 const int p = previous[mask * n + j];
112 mask &= ~((size_t)1 << j);
113 j = p;
114 }
115 std::reverse(path.begin(), path.end());
116 return path;
117}

119// Nearest neighbour path from `start`
120std::vector<size_t> GetNearestNeighbourPath(const std::vector<std::vector<double>>& distance, size_t start) {
121 const size_t n = distance.size();
122 std::vector<bool> visited(n, false);
123 std::vector<size_t> path = {start};
124 visited[start] = true;
125 while (path.size() < n) {
126 const size_t current = path.back();
127 size_t closest = n;
128 for (size_t k = 0; k < n; k++) {
129 if (!visited[k] && (closest == n || distance[current][k] < distance[current][closest]))
130 closest = k;
131 }
132 visited[closest] = true;
133 path.push_back(closest);
134 }
135 return path;

138// 2-opt: reverses the segments [i, j] of the path while that makes it shorter. A cyclic path keeps its
139// first point
140void ImprovePath2Opt(std::vector<size_t>& path, const std::vector<std::vector<double>>& distance,
141 bool cyclic) {
142 const size_t n = path.size();
143 bool improved = true;
144 while (improved) {
145 improved = false;
146 for (size_t i = cyclic ? 1 : 0; i + 1 < n; i++) {
147 for (size_t j = i + 1; j < n; j++) {
148 // Edges before i and after j, if they exist (the cyclic path closes at the first point)
149 const bool hasBefore = i > 0;
150 const bool hasAfter = j + 1 < n || cyclic;
151 const size_t before = hasBefore ? path[i - 1] : 0;
152 const size_t after = j + 1 < n ? path[j + 1] : path[0];
153 double delta = 0;
154 if (hasBefore) delta += distance[before][path[j]] - distance[before][path[i]];
155 if (hasAfter) delta += distance[path[i]][after] - distance[path[j]][after];
156 if (delta < -1e-12) {
157 std::reverse(path.begin() + i, path.begin() + j + 1);
158 improved = true;
159 }
160 }
161 }
162 }

165// Line that best fits the points: energy weighted centroid and principal axis of the energy weighted
166// covariance (unit vector, null for a single point). False if there are no points or no energy
167bool FitLine(const std::vector<XYZVector>& points, const std::vector<double>& energies, XYZVector& centroid,
168 XYZVector& axis) {
169 if (points.empty() || points.size() != energies.size()) return false;
170
171 double totalEnergy = 0;
172 centroid = XYZVector();
173 for (size_t i = 0; i < points.size(); i++) {
174 const double w = std::max(energies[i], 0.);
175 totalEnergy += w;
176 centroid += w * points[i];
177 }
178 if (totalEnergy <= 0) return false;
179 centroid /= totalEnergy;
180
181 double covariance[3][3] = {};
182 for (size_t i = 0; i < points.size(); i++) {
183 const double w = std::max(energies[i], 0.) / totalEnergy;
184 const XYZVector d = points[i] - centroid;
185 const double c[3] = {d.X(), d.Y(), d.Z()};
186 for (int a = 0; a < 3; a++) {
187 for (int b = 0; b < 3; b++) covariance[a][b] += w * c[a] * c[b];
188 }
189 }
190 // Direction of largest spread: eigenvector of the largest eigenvalue (TMatrixDSymEigen sorts them in
191 // descending order), with the sign of the column of the covariance with the largest norm. A single
192 // point (no spread) gives no direction
193 TMatrixDSym matrix(3);
194 for (int a = 0; a < 3; a++) {
195 for (int b = 0; b < 3; b++) matrix(a, b) = covariance[a][b];
196 }
197 const TMatrixDSymEigen eigen(matrix);
198 const TVectorD& eigenValues = eigen.GetEigenValues();
199 const TMatrixD& eigenVectors = eigen.GetEigenVectors();
200 axis = XYZVector();
201 if (eigenValues[0] > 0) {
202 axis = XYZVector(eigenVectors(0, 0), eigenVectors(1, 0), eigenVectors(2, 0)).Unit();
203 XYZVector largestColumn;
204 for (int a = 0; a < 3; a++) {
205 const XYZVector column(covariance[0][a], covariance[1][a], covariance[2][a]);
206 if (column.R() > largestColumn.R()) largestColumn = column;
207 }
208 if (axis.Dot(largestColumn) < 0) axis = -axis;
209 }
210 return true;
211}

214std::vector<std::vector<size_t>> TRestHitsAnalysis::FindClusters(const std::vector<XYZVector>& points,
215 double distance,
216 const std::array<bool, 3>& coordinates) {
217 // distance <= 0 (or NaN): no points are joined, every point is a cluster
218 if (!(distance > 0)) {
219 std::vector<std::vector<size_t>> clusters(points.size());
220 for (size_t n = 0; n < points.size(); n++) clusters[n] = {n};
221 return clusters;
222 }
223 const double maxDistance2 = distance * distance;
224 auto distance2 = [&](size_t a, size_t b) {
225 double d2 = 0;
226 for (int axis = 0; axis < 3; axis++) {
227 if (!coordinates[axis]) continue;
228 const double d = GetCoordinate(points[a], axis) - GetCoordinate(points[b], axis);
229 d2 += d * d;
230 }
231 return d2;
232 };
233
234 // Cells of side distance along the enabled coordinates (0 along the others): the points closer than
235 // distance to a point are in its cell or in the adjacent ones
236 const double cellSize = distance;
237 auto getCell = [&](size_t n) { return GetCell(points[n], cellSize, coordinates); };
238
239 std::unordered_map<Cell, std::vector<size_t>, CellHash> cells;
240 for (size_t n = 0; n < points.size(); n++) cells[getCell(n)].push_back(n);
241
242 const std::vector<Cell> offsets = GetNeighbourOffsets(coordinates);
243
244 std::vector<std::vector<size_t>> clusters;
245 std::vector<bool> assigned(points.size(), false);
246 for (size_t seed = 0; seed < points.size(); seed++) {
247 if (assigned[seed]) continue;
248
249 // Breadth-first search of the points closer than distance to any point already in the cluster
250 std::vector<size_t> cluster = {seed};
251 assigned[seed] = true;
252 for (size_t q = 0; q < cluster.size(); q++) {
253 const size_t point = cluster[q];
254 const Cell cell = getCell(point);
255 for (const auto& offset : offsets) {
256 auto it = cells.find({cell[0] + offset[0], cell[1] + offset[1], cell[2] + offset[2]});
257 if (it == cells.end()) continue;
258
259 // Assigned points leave the cell, so they are not compared again
260 auto& members = it->second;
261 for (size_t m = 0; m < members.size();) {
262 const size_t j = members[m];
263 if (!assigned[j]) {
264 if (distance2(point, j) >= maxDistance2) {
265 m++;
266 continue;
267 }
268 assigned[j] = true;
269 cluster.push_back(j);
270 }
271 members[m] = members.back();
272 members.pop_back();
273 }
274 }
275 }
276 std::sort(cluster.begin(), cluster.end());
277 clusters.push_back(std::move(cluster));
278 }
279 return clusters;
280}

282std::vector<size_t> TRestHitsAnalysis::GetShortestPath(const std::vector<XYZVector>& points, bool cyclic,
283 size_t maxExactPoints) {
284 const size_t n = points.size();
285 std::vector<size_t> order(n);
286 std::iota(order.begin(), order.end(), 0);
287 if (n < 3) return order;
288
289 std::vector<std::vector<double>> distance(n, std::vector<double>(n, 0));
290 for (size_t i = 0; i < n; i++) {
291 for (size_t j = i + 1; j < n; j++) distance[i][j] = distance[j][i] = (points[i] - points[j]).R();
292 }
293
294 if (n <= std::min<size_t>(maxExactPoints, 16)) return GetShortestPathExact(distance, cyclic);
295
296 // Nearest neighbour from every point, or from the farthest point from the centroid
297 std::vector<size_t> starts;
298 if (n <= 200) {
299 starts = order;
300 } else {
301 XYZVector centroid;
302 for (const auto& p : points) centroid += p;
303 centroid /= (double)n;
304 size_t farthest = 0;
305 for (size_t i = 1; i < n; i++) {
306 if ((points[i] - centroid).Mag2() > (points[farthest] - centroid).Mag2()) farthest = i;
307 }
308 starts = {farthest};
309 }
310
311 std::vector<size_t> best;
312 double bestLength = std::numeric_limits<double>::infinity();
313 for (size_t start : starts) {
314 std::vector<size_t> path = GetNearestNeighbourPath(distance, start);
315 const double length = GetPathLength(path, distance, cyclic);
316 if (length < bestLength) {
317 bestLength = length;
318 best = std::move(path);
319 }
320 }
321 ImprovePath2Opt(best, distance, cyclic);
322 return best;
323}

325std::vector<TRestHitsAnalysis::Node> TRestHitsAnalysis::KMeansClustering(
326 const std::vector<XYZVector>& points, const std::vector<double>& energies,
327 const std::vector<XYZVector>& centroids, int maxIt, bool fixBoundaries) {
328 const int nPoints = (int)points.size();
329 const int nodes = (int)centroids.size();
330 if (nodes < 2 || nPoints == 0) return {};
331
332 auto isFixed = [&](int n) { return fixBoundaries && (n == 0 || n == nodes - 1); };
333
334 std::vector<XYZVector> centroid = centroids, centroidOld(nodes);
335
336 std::vector<std::vector<int>> clusterPoints(nodes);
337 for (int it = 0; it < maxIt; it++) {
338 for (auto& cluster : clusterPoints) cluster.clear();
339
340 for (int i = 0; i < nPoints; i++) {
341 double minDistance = std::numeric_limits<double>::max();
342 int closest = -1;
343 for (int n = 0; n < nodes; n++) {
344 if (isFixed(n)) continue;
345 const double d = (centroid[n] - points[i]).R();
346 if (d < minDistance) {
347 minDistance = d;
348 closest = n;
349 }
350 }
351 if (closest >= 0) clusterPoints[closest].push_back(i);
352 }
353
354 bool converge = true;
355 for (int n = 0; n < nodes; n++) {
356 if (isFixed(n)) continue;
357 XYZVector mean;
358 double totalEnergy = 0;
359 for (int i : clusterPoints[n]) {
360 mean += energies[i] * points[i];
361 totalEnergy += energies[i];
362 }
363 if (totalEnergy > 0) centroid[n] = mean / totalEnergy;
364 converge &= (centroid[n] - centroidOld[n]).R() < 1e-9;
365 centroidOld[n] = centroid[n];
366 }
367 if (converge) break;
368 }
369
370 std::vector<Node> result;
371 for (int n = 0; n < nodes; n++) {
372 if (isFixed(n)) {
373 result.push_back({centroid[n], 0, XYZVector()});
374 } else if (!clusterPoints[n].empty()) {
375 double energy = 0;
376 double variance[3] = {};
377 for (int i : clusterPoints[n]) {
378 energy += energies[i];
379 const XYZVector d = points[i] - centroid[n];
380 variance[0] += energies[i] * d.X() * d.X();
381 variance[1] += energies[i] * d.Y() * d.Y();
382 variance[2] += energies[i] * d.Z() * d.Z();
383 }
384 auto sigma = [&](int a) {
385 return energy > 0 ? std::sqrt(std::max(variance[a] / energy, 0.)) : 0;
386 };
387 result.push_back({centroid[n], energy, XYZVector(sigma(0), sigma(1), sigma(2))});
388 }
389 }
390 return result;
391}

393std::vector<TRestHitsAnalysis::Node> TRestHitsAnalysis::MergePoints(
394 const std::vector<XYZVector>& points, const std::vector<double>& energies, double startingDistance,
395 double minimumDistance, double distanceStepFactor, size_t maxNodes,
396 const std::vector<XYZVector>& sigmas) {
397 if (startingDistance <= 0 || distanceStepFactor <= 1) {
398 throw std::invalid_argument(
399 "TRestHitsAnalysis::MergePoints: startingDistance must be > 0 and distanceStepFactor > 1");
400 }
401 if (points.empty() || points.size() != energies.size()) return {};
402 maxNodes = std::max<size_t>(maxNodes, 1);
403
404 struct Group {
405 XYZVector position;
406 double energy;
407 std::vector<size_t> members;
408 };
409 std::vector<Group> groups;
410 groups.reserve(points.size());
411 for (size_t i = 0; i < points.size(); i++) groups.push_back({points[i], energies[i], {i}});
412
413 // As legacy TRestHits::MergeHits: energy weighted position, the first one kept without energy
414 auto merge = [](Group& a, Group& b) {
415 const double totalEnergy = a.energy + b.energy;
416 if (totalEnergy > 0) a.position = (a.energy * a.position + b.energy * b.position) / totalEnergy;
417 a.energy = totalEnergy;
418 a.members.insert(a.members.end(), b.members.begin(), b.members.end());
419 };
420
421 double distance = startingDistance;
422 while (distance < minimumDistance || groups.size() > maxNodes) {
423 const double distance2 = distance * distance;
424 bool merged = true;
425 while (merged) {
426 merged = false;
427 for (size_t i = 0; i < groups.size(); i++) {
428 for (size_t j = i + 1; j < groups.size();) {
429 if ((groups[i].position - groups[j].position).Mag2() < distance2) {
430 merge(groups[i], groups[j]);
431 groups.erase(groups.begin() + j);
432 merged = true;
433 } else {
434 j++;
435 }
436 }
437 }
438 }
439 distance *= distanceStepFactor;
440 }
441
442 std::vector<Node> nodes;
443 nodes.reserve(groups.size());
444 for (const auto& group : groups) {
445 double variance[3] = {};
446 for (size_t i : group.members) {
447 const XYZVector d = points[i] - group.position;
448 const XYZVector s = i < sigmas.size() ? sigmas[i] : XYZVector();
449 variance[0] += energies[i] * (d.X() * d.X() + s.X() * s.X());
450 variance[1] += energies[i] * (d.Y() * d.Y() + s.Y() * s.Y());
451 variance[2] += energies[i] * (d.Z() * d.Z() + s.Z() * s.Z());
452 }
453 auto sigma = [&](int a) {
454 return group.energy > 0 ? std::sqrt(std::max(variance[a] / group.energy, 0.)) : 0;
455 };
456 nodes.push_back({group.position, group.energy, XYZVector(sigma(0), sigma(1), sigma(2))});
457 }
458 return nodes;
459}

461bool TRestHitsAnalysis::GetLine(const std::vector<XYZVector>& points, const std::vector<double>& energies,
462 XYZVector& first, XYZVector& last) {
463 XYZVector centroid, axis;
464 if (!FitLine(points, energies, centroid, axis)) return false;
465
466 double tMin = 0, tMax = 0;
467 for (size_t i = 0; i < points.size(); i++) {
468 const double t = (points[i] - centroid).Dot(axis);
469 if (i == 0 || t < tMin) tMin = t;
470 if (i == 0 || t > tMax) tMax = t;
471 }
472 first = centroid + tMin * axis;
473 last = centroid + tMax * axis;
474 return true;
475}

487double TRestHitsAnalysis::GetLineResidual(const std::vector<XYZVector>& points,
488 const std::vector<double>& energies) {
489 XYZVector centroid, axis;
490 if (!FitLine(points, energies, centroid, axis)) return -1;
491
492 double sum = 0, totalEnergy = 0;
493 for (size_t i = 0; i < points.size(); i++) {
494 const double w = std::max(energies[i], 0.);
495 const XYZVector d = points[i] - centroid;
496 const XYZVector perpendicular = d - d.Dot(axis) * axis;
497 sum += w * perpendicular.Mag2();
498 totalEnergy += w;
499 }
500 return std::sqrt(sum / totalEnergy);
501}

503double TRestHitsAnalysis::GetTrackEnds(const std::vector<XYZVector>& points,
504 const std::vector<double>& energies, XYZVector& origin,
505 XYZVector& end) {
506 if (points.empty()) return 0;
507 const XYZVector& first = points.front();
508 const XYZVector& last = points.back();
509 const XYZVector segment = last - first;
510 const double length2 = segment.Mag2();
511
512 // Energy of the halves closer to the first and to the last point
513 double firstEnergy = 0, lastEnergy = 0;
514 for (size_t i = 0; i < points.size(); i++) {
515 const double fraction = length2 > 0 ? (points[i] - first).Dot(segment) / length2
516 : (points.size() > 1 ? (double)i / (points.size() - 1) : 0);
517 if (fraction < 0.5) {
518 firstEnergy += energies[i];
519 } else if (fraction > 0.5) {
520 lastEnergy += energies[i];
521 } else {
522 firstEnergy += energies[i] / 2.;
523 lastEnergy += energies[i] / 2.;
524 }
525 }
526
527 if (firstEnergy > lastEnergy) {
528 end = first;
529 origin = last;
530 } else {
531 origin = first;
532 end = last;
533 }
534 const double total = firstEnergy + lastEnergy;
535 return total > 0 ? std::abs(lastEnergy - firstEnergy) / total : 0;
536}

541std::pair<double, double> MinimizeOnInterval(const std::function<double(double)>& f, double low, double high,
542 int nScan, int nGolden, int& best) {
543 const double step = (high - low) / nScan;
544 best = 0;
545 double bestValue = std::numeric_limits<double>::max();
546 for (int i = 0; i <= nScan; i++) {
547 const double value = f(low + i * step);
548 if (value < bestValue) {
549 bestValue = value;
550 best = i;
551 }
552 }
553 const double ratio = (std::sqrt(5.0) - 1) / 2;
554 double a = low + std::max(best - 1, 0) * step;
555 double b = low + std::min(best + 1, nScan) * step;
556 double c = b - ratio * (b - a);
557 double d = a + ratio * (b - a);
558 double fC = f(c);
559 double fD = f(d);
560 for (int i = 0; i < nGolden; i++) {
561 if (fC < fD) {
562 b = d;
563 d = c;
564 fD = fC;
565 c = b - ratio * (b - a);
566 fC = f(c);
567 } else {
568 a = c;
569 c = d;
570 fC = fD;
571 d = a + ratio * (b - a);
572 fD = f(d);
573 }
574 }
575 const double x = (a + b) / 2;
576 return {x, f(x)};
577}

580double TRestHitsAnalysis::GetGaussSigma(const std::vector<double>& values,
581 const std::vector<double>& energies, double resolution) {
582 if (values.empty()) return 0;
583
584 // Energy of each position (several hits in a strip are added); positions closer than a millionth
585 // of their size are the same one
586 std::map<double, double> byPosition;
587 for (size_t n = 0; n < values.size() && n < energies.size(); n++) byPosition[values[n]] += energies[n];
588 std::vector<std::pair<double, double>> positions;
589 for (const auto& [x, e] : byPosition) {
590 if (!positions.empty() && x - positions.back().first <= 1e-6 * std::max(1.0, std::abs(x))) {
591 positions.back().second += e;
592 } else {
593 positions.emplace_back(x, e);
594 }
595 }
596
597 double total = 0;
598 double largest = 0;
599 double pitch = std::numeric_limits<double>::max();
600 for (size_t n = 0; n < positions.size(); n++) {
601 total += positions[n].second;
602 largest = std::max(largest, positions[n].second);
603 if (n > 0) pitch = std::min(pitch, positions[n].first - positions[n - 1].first);
604 }
605 if (!(total > 0)) return -1;
606 // With two positions any sigma below the pitch fits (the centre moving with it): three with more than
607 // 1% of the energy of the largest one are needed
608 const auto significant = std::count_if(positions.begin(), positions.end(),
609 [largest](const auto& p) { return p.second > 0.01 * largest; });
610 if (significant < 3) return -1;
611 const double xMin = positions.front().first;
612 const double xMax = positions.back().first;
613
614 // Least squares of the energies against the gaussian of integral total, as a function of its centre
615 // and sigma. The width of a position: the pitch (strips) or the resolution (continuous)
616 const bool continuous = resolution > 0;
617 const double width = continuous ? resolution : pitch;
618 std::function<double(double, double)> chi2;
619 std::vector<double> stripEnergy;
620 std::vector<double> cumulativeE2;
621 if (continuous) {
622 // A continuous coordinate (e.g. the drift distance, from the time of each pulse) has no strips:
623 // with strips of width -> 0 a position with a third of the energy is fitted by sigma -> 0, so the
624 // positions and the gaussian are both smoothed with a gaussian of width resolution (r). The least
625 // squares of the smoothed densities, up to a constant, with s1 = sqrt(sigma^2 + r^2) and
626 // s2 = sqrt(sigma^2 + 2 r^2):
627 // total^2 / (2 sqrt(pi) s1) - 2 total sum_i e_i exp(-(x_i - mean)^2 / (2 s2^2)) / (sqrt(2 pi) s2)
628 // sigma is the width of the distribution (without the resolution); the time of each evaluation
629 // depends on the number of positions only
630 chi2 = [&positions, total, resolution](double mean, double sigma) {
631 const double s1 = std::sqrt(sigma * sigma + resolution * resolution);
632 const double s2 = std::sqrt(sigma * sigma + 2 * resolution * resolution);
633 double sum = 0;
634 for (const auto& [x, e] : positions) {
635 const double d = (x - mean) / s2;
636 sum += e * std::exp(-0.5 * d * d);
637 }
638 constexpr double kPi = 3.14159265358979323846;
639 return total * total / (2 * std::sqrt(kPi) * s1) - 2 * total * sum / (std::sqrt(2 * kPi) * s2);
640 };
641 } else {
642 // Strips of width pitch from one before the first to one after the last (0 where there is no hit).
643 // Strip k covers [xMin + (k - 1.5) pitch, xMin + (k - 0.5) pitch]; the energy expected in it is the
644 // gaussian integrated over it. The strips farther than kTail sigmas from the centre expect no energy
645 // (less than 1e-18 of it): they add their energy squared, from cumulative sums, and only the strips
646 // near the centre are evaluated, each edge once (the upper edge of a strip is the lower edge of the
647 // next), in a time that depends on sigma / pitch, not on the extent
648 const long nStrips = std::lround((xMax - xMin) / pitch) + 3;
649 if (nStrips > 100000) return -1;
650 stripEnergy.assign(nStrips, 0);
651 for (const auto& [x, e] : positions) stripEnergy[std::lround((x - xMin) / pitch) + 1] += e;
652 cumulativeE2.assign(nStrips + 1, 0);
653 for (long k = 0; k < nStrips; k++) {
654 cumulativeE2[k + 1] = cumulativeE2[k] + stripEnergy[k] * stripEnergy[k];
655 }
656 chi2 = [&stripEnergy, &cumulativeE2, nStrips, xMin, pitch, total](double mean, double sigma) {
657 constexpr double kTail = 9;
658 const double scale = 1 / (sigma * std::sqrt(2.0));
659 const double reach = kTail * sigma;
660 const double low = std::floor((mean - reach - xMin) / pitch + 0.5);
661 const double high = std::ceil((mean + reach - xMin) / pitch + 1.5);
662 const long first = std::clamp(static_cast<long>(low), 0L, nStrips - 1);
663 const long last = std::clamp(static_cast<long>(high), first, nStrips - 1);
664 double sum = cumulativeE2[first] + (cumulativeE2[nStrips] - cumulativeE2[last + 1]);
665 double lowerErf = std::erf((xMin + (first - 1.5) * pitch - mean) * scale);
666 for (long k = first; k <= last; k++) {
667 const double upperErf = std::erf((xMin + (k - 0.5) * pitch - mean) * scale);
668 const double expected = total * 0.5 * (upperErf - lowerErf);
669 sum += (stripEnergy[k] - expected) * (stripEnergy[k] - expected);
670 lowerErf = upperErf;
671 }
672 return sum;
673 };
674 }
675
676 // For each sigma (in log, from width / 100 to the extent) the best centre, within the positions and
677 // half a strip (or a resolution) beyond them. The golden section search after each scan (two scan
678 // steps, divided by 0.618 at each step) ends within 1e-6 of the interval scanned
679 constexpr int nScan = 40;
680 constexpr int nGolden = 25;
681 const double margin = continuous ? resolution : pitch / 2;
682 const double extent = xMax - xMin + 2 * width;
683 const auto bestForSigma = [&](double logSigma) {
684 const double sigma = std::exp(logSigma);
685 int best = 0;
686 return MinimizeOnInterval([&](double mean) { return chi2(mean, sigma); }, xMin - margin,
687 xMax + margin, nScan, nGolden, best)
688 .second;
689 };
690 const double logMin = std::log(width / 100);
691 const double logMax = std::log(extent);
692 int best = 0;
693 const double logSigma = MinimizeOnInterval(bestForSigma, logMin, logMax, nScan, nGolden, best).first;
694 // At the extent: the energy does not decrease away from the centre (no gaussian)
695 if (best == nScan) return -1;
696 return std::exp(logSigma);
697}

699double TRestHitsAnalysis::GetTwist(const std::vector<XYZVector>& points, size_t n, size_t m) {
700 if (m == 0) m = points.size();
701 if (m < n + 2) return 0;
702
703 double sum = 0;
704 int count = 0;
705 for (size_t N = n + 1; N + 1 < m; N++) {
706 // Unit: null vector if the points coincide (as TVector3::Unit in legacy)
707 const XYZVector a = (points[N - 1] - points[N + 1]).Unit();
708 const XYZVector b = (points[N] - points[N + 1]).Unit();
709 sum += (1 - a.Dot(b)) / 2.;
710 count++;
711 }
712 return count == 0 ? -1 : sum / count;
