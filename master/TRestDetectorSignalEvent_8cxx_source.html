55void TRestDetectorSignal::Sort() {
56 int start = fData->offsets[fSignalIdx];
57 int nPoints = GetNPoints();
58 if (nPoints <= 1) return;
59
60 std::vector<int> indices(nPoints);
61 std::iota(indices.begin(), indices.end(), 0);
62
63 std::stable_sort(indices.begin(), indices.end(), [this, start](int a, int b) {
64 return fData->allSignalsTime[start + a] < fData->allSignalsTime[start + b];
65 });
66
67 bool alreadySorted = true;
68 for (int i = 0; i < nPoints; ++i) {
69 if (indices[i] != i) {
70 alreadySorted = false;
71 break;
72 }
73 }
74 if (alreadySorted) return;
75
76 std::vector<double> sortedTime(nPoints);
77 std::vector<double> sortedCharge(nPoints);
78
79 for (int i = 0; i < nPoints; ++i) {
80 sortedTime[i] = fData->allSignalsTime[start + indices[i]];
81 sortedCharge[i] = fData->allSignalsCharge[start + indices[i]];
82 }
83
84 std::copy(sortedTime.begin(), sortedTime.end(), fData->allSignalsTime.begin() + start);
85 std::copy(sortedCharge.begin(), sortedCharge.end(), fData->allSignalsCharge.begin() + start);
86}
