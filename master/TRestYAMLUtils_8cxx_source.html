23YAML::Node ResolveEnvVars(const YAML::Node& node) {
24 if (node.IsScalar()) {
25 std::string str = node.as<std::string>();
26 std::regex re(R"(\$\{ENV:([^}]+)\})");
27 std::smatch match;
28 std::string::const_iterator searchStart(str.cbegin());
29
30 while (std::regex_search(searchStart, str.cend(), match, re)) {
31 std::string varName = match[1].str();
32 const char* envValue = std::getenv(varName.c_str());
33 std::string replacement = envValue ? envValue : "";
34 // match.position is relative to searchStart; the value is not searched again
35 const size_t position = (searchStart - str.cbegin()) + match.position(0);
36 str.replace(position, match.length(0), replacement);
37 searchStart = str.cbegin() + position + replacement.size();
38 }
39 return YAML::Node(str);
40 } else if (node.IsSequence()) {
41 YAML::Node out(YAML::NodeType::Sequence);
42 for (auto n : node) {
43 out.push_back(ResolveEnvVars(n));
44 }
45 return out;
46 } else if (node.IsMap()) {
47 YAML::Node out(YAML::NodeType::Map);
48 for (auto it : node) {
49 out[it.first.Scalar()] = ResolveEnvVars(it.second);
50 }
51 return out;
52 }
53 return node;
54}

76YAML::Node ResolveParameters(const YAML::Node& node) {
77 if (node.IsScalar()) {
78 std::string str = node.as<std::string>();
79 const std::string original = str;
80 const std::regex re(R"(\$\{PARAM:([^}]+)\})");
81 std::smatch match;
82 bool replaced = false;
83 int substitutions = 0;
84 while (std::regex_search(str, match, re)) {
85 CheckSubstitutions(substitutions, original);
86 const std::string name = match[1].str();
87 const auto parameter = gParameters.find(name);
88 if (parameter == gParameters.end()) {
89 throw std::runtime_error("${PARAM:" + name + "}: parameter '" + name +
90 "' not defined in the section `parameters`");
91 }
92 gUsedParameters.insert(name);
93 str.replace(match.position(0), match.length(0), parameter->second);
94 replaced = true;
95 }
96 return replaced ? YAML::Node(str) : node;
97 } else if (node.IsSequence()) {
98 YAML::Node out(YAML::NodeType::Sequence);
99 for (auto n : node) out.push_back(ResolveParameters(n));
100 return out;
101 } else if (node.IsMap()) {
102 YAML::Node out(YAML::NodeType::Map);
103 for (auto it : node) out[it.first.Scalar()] = ResolveParameters(it.second);
104 return out;
105 }
106 return node;
107}

111YAML::Node ResolveYamlRefs(const YAML::Node& root, const YAML::Node& node) {
112 if (node.IsScalar()) {
113 std::string str = node.as<std::string>();
114 const std::string original = str;
115 std::smatch match;
116 int substitutions = 0;
117
118 std::regex yamlRe(R"(\$\{([^}]+)\})");
119 while (std::regex_search(str, match, yamlRe)) {
120 CheckSubstitutions(substitutions, original);
121 std::string keyPath = match[1].str();
122
123 YAML::Node refNode = YAML::Clone(root);
124 for (auto key : TRestTools::Split(keyPath, '.')) {
125 if (!refNode[key]) throw std::runtime_error("key not found: " + keyPath);
126 refNode = refNode[key];
127 }
128
129 if (!refNode.IsScalar())
130 throw std::runtime_error("Cannot substitute non-scalar node: " + keyPath);
131
132 std::string replacement = refNode.as<std::string>();
133 str.replace(match.position(0), match.length(0), replacement);
134 }
135 return YAML::Node(str);
136 } else if (node.IsSequence()) {
137 YAML::Node out(YAML::NodeType::Sequence);
138 for (auto n : node) out.push_back(ResolveYamlRefs(root, n));
139 return out;
140 } else if (node.IsMap()) {
141 YAML::Node out(YAML::NodeType::Map);
142 for (auto it : node) out[it.first.Scalar()] = ResolveYamlRefs(root, it.second);
143 return out;
144 }
145
146 return node;
147}

151YAML::Node MergeNodes(const YAML::Node& base, const YAML::Node& local) {
152 if (!base.IsMap() || !local.IsMap()) return YAML::Clone(local);
153 YAML::Node out = YAML::Clone(base);
154 for (const auto& element : local) {
155 const std::string key = element.first.Scalar();
156 const YAML::Node& constOut = out; // Const access: a missing key is not added
157 const YAML::Node baseValue = constOut[key];
158 if (baseValue && baseValue.IsMap() && element.second.IsMap())
159 out[key] = MergeNodes(baseValue, element.second);
160 else
161 out[key] = YAML::Clone(element.second);
162 }
163 return out;
164}

171YAML::Node ExpandSelects(const YAML::Node& node) {
172 if (node.IsSequence()) {
173 YAML::Node out(YAML::NodeType::Sequence);
174 for (const auto& element : node) out.push_back(ExpandSelects(element));
175 return out;
176 }
177 if (!node.IsMap()) return node;
178
179 const YAML::Node select = node["select"];
180 if (!select) {
181 YAML::Node out(YAML::NodeType::Map);
182 for (const auto& element : node) out[element.first.Scalar()] = ExpandSelects(element.second);
183 return out;
184 }
185
186 const YAML::Node cases = node["cases"];
187 if (!select.IsScalar()) throw std::runtime_error("select: a value expected (e.g. \"${PARAM:name}\")");
188 if (!cases || !cases.IsMap()) throw std::runtime_error("select: a map `cases` (value: content) expected");
189 const std::string value = select.Scalar();
190 // Const access (a missing key is not added) and reset (assigning a node would write into the case); a
191 // default YAML::Node is a defined (null) node: whether a case is found is kept apart
192 const YAML::Node& constCases = cases;
193 YAML::Node chosen;
194 bool found = false;
195 for (const std::string& name : {value, std::string("default")}) {
196 if (!found && constCases[name]) {
197 chosen.reset(constCases[name]);
198 found = true;
199 }
200 }
201 if (!found) {
202 std::string names;
203 for (const auto& element : cases) names += (names.empty() ? "" : ", ") + element.first.Scalar();
204 throw std::runtime_error("select: no case '" + value + "' nor `default` in the cases (" + names +
205 ")");
206 }
207
208 YAML::Node common(YAML::NodeType::Map);
209 for (const auto& element : node) {
210 const std::string key = element.first.Scalar();
211 if (key != "select" && key != "cases") common[key] = element.second;
212 }
213 if (!chosen.IsMap()) {
214 if (common.size() > 0) {
215 throw std::runtime_error("select: the case '" + value +
216 "' is not a map, the block cannot have other keys");
217 }
218 return ExpandSelects(chosen);
219 }
220 return ExpandSelects(MergeNodes(common, chosen));
221}

254YAML::Node LoadYAMLFile(const std::string& fileName) {
255 std::ifstream stream(TRestTools::GetLocalFile(fileName));
256 if (!stream) throw std::runtime_error("Cannot open the YAML file " + fileName);
257 std::vector<std::string> lines;
258 std::string text;
259 for (std::string line; std::getline(stream, line);) {
260 lines.push_back(line);
261 text += line + "\n";
262 }
263
264 for (size_t i = 0; i < lines.size(); i++) {
265 const auto start = lines[i].find_first_not_of(" \t");
266 if (start == std::string::npos || lines[i][start] == '#') continue;
267 if (lines[i].substr(0, start).find('\t') != std::string::npos) {
268 throw std::runtime_error(fileName + ", line " + std::to_string(i + 1) +
269 ": indented with a tab, YAML is indented with spaces\n " + lines[i]);
270 }
271 }
272
273 YAML::Node node;
274 try {
275 node = YAML::Load(text);
276 } catch (const YAML::ParserException& error) {
277 const int line = error.mark.line;
278 std::string message = fileName + ", line " + std::to_string(line + 1) + ", column " +
279 std::to_string(error.mark.column + 1) + ": " + error.msg;
280 if (line >= 0 && line < static_cast<int>(lines.size())) message += "\n " + lines[line];
281 message +=
282 "\n(check the indentation of this line and of the previous ones: the keys of a section are "
283 "aligned, deeper than the key of the section)";
284 throw std::runtime_error(message);
285 }
286 CheckRepeatedKeys(node, fileName);
287 return node;
288}

293YAML::Node ExpandIncludes(const YAML::Node& node, const std::string& directory,
294 std::vector<std::string>& stack) {
295 if (node.IsSequence()) {
296 YAML::Node out(YAML::NodeType::Sequence);
297 for (const auto& element : node) out.push_back(ExpandIncludes(element, directory, stack));
298 return out;
299 }
300 if (!node.IsMap()) return YAML::Clone(node);
301
302 YAML::Node local(YAML::NodeType::Map);
303 // A default YAML::Node is a defined (null) node: whether there is an include is kept apart
304 bool hasInclude = false;
305 YAML::Node include;
306 for (const auto& element : node) {
307 const std::string key = element.first.Scalar();
308 if (key == "include") {
309 hasInclude = true;
310 include.reset(element.second);
311 } else {
312 local[key] = ExpandIncludes(element.second, directory, stack);
313 }
314 }
315 if (!hasInclude) return local;
316
317 if (!include.IsScalar() || include.Scalar().empty())
318 throw std::runtime_error("include: a file name expected (\"file.yaml\" or \"file.yaml:key\")");
319 // "file.yaml:key": the section `key` of the file
320 std::string fileName = ResolveEnvVars(include).Scalar();
321 std::string sectionKey;
322 const size_t colon = fileName.rfind(':');
323 if (colon != std::string::npos) {
324 const std::string prefix = fileName.substr(0, colon);
325 auto endsWith = [&prefix](const std::string& end) {
326 return prefix.size() > end.size() &&
327 prefix.compare(prefix.size() - end.size(), end.size(), end) == 0;
328 };
329 if (endsWith(".yaml") || endsWith(".yml")) {
330 sectionKey = fileName.substr(colon + 1);
331 fileName = prefix;
332 }
333 }
334
335 // A URL (e.g. a versioned file of rest-common) is downloaded; relative to a URL, a URL
336 std::string pathString;
337 if (TRestTools::IsURL(fileName)) {
338 pathString = fileName;
339 } else if (TRestTools::IsURL(directory) && std::filesystem::path(fileName).is_relative()) {
340 pathString = directory + fileName;
341 } else {
342 std::filesystem::path path(fileName);
343 if (path.is_relative()) path = std::filesystem::path(directory) / path;
344 pathString = std::filesystem::weakly_canonical(path).string();
345 if (!std::filesystem::exists(path))
346 throw std::runtime_error("include: file " + pathString +
347 " not found (relative paths are relative to " +
348 "the file that includes it)");
349 }
350 if (std::find(stack.begin(), stack.end(), pathString) != stack.end())
351 throw std::runtime_error("include: " + pathString + " includes itself (directly or not)");
352
353 stack.push_back(pathString);
354 YAML::Node included = ExpandIncludes(LoadYAMLFile(pathString), IncludeBase(pathString), stack);
355 stack.pop_back();
356
357 if (!sectionKey.empty()) {
358 const YAML::Node& constIncluded = included;
359 if (!constIncluded.IsMap() || !constIncluded[sectionKey])
360 throw std::runtime_error("include: no key '" + sectionKey + "' in " + pathString);
361 included.reset(YAML::Clone(constIncluded[sectionKey]));
362 }
363 if (!included.IsMap())
364 throw std::runtime_error("include: the content of " + include.Scalar() +
365 " must be a map (the keys of the section)");
366 return MergeNodes(included, local);
367}

383YAML::Node TRestYAMLUtils::ResolveRefs(const YAML::Node& root,
384 const std::map<std::string, std::string>& overrides,
385 bool setCurrentParameters) {
386 YAML::Node cfg = YAML::Clone(root);
387 std::lock_guard<std::mutex> lock(gParametersMutex);
388 // Without setCurrentParameters the current ones are restored when leaving (also on errors)
389 struct Restore {
390 bool active;
391 std::map<std::string, std::string> parameters;
392 std::set<std::string> used;
393 ~Restore() {
394 if (!active) return;
395 gParameters = std::move(parameters);
396 gUsedParameters = std::move(used);
397 }
398 } restore{!setCurrentParameters, setCurrentParameters ? decltype(gParameters){} : gParameters,
399 setCurrentParameters ? decltype(gUsedParameters){} : gUsedParameters};
400 gParameters.clear();
401 gUsedParameters.clear();
402 YAML::Node parameters;
403 if (cfg.IsMap() && cfg["parameters"]) {
404 parameters = cfg["parameters"];
405 cfg.remove("parameters");
406 }
407 if (parameters && !parameters.IsNull() && !parameters.IsMap()) {
408 throw std::runtime_error("`parameters` must be a map name: value");
409 }
410 for (const auto& [name, value] : overrides) {
411 if (!parameters || !parameters.IsMap() || !parameters[name]) {
412 throw std::runtime_error("--set " + name + ": no parameter '" + name +
413 "' in the section `parameters` of the configuration");
414 }
415 parameters[name] = value;
416 }
417 if (parameters && parameters.IsMap()) {
418 for (const auto& parameter : ResolveEnvVars(parameters)) {
419 const std::string name = parameter.first.as<std::string>();
420 if (!parameter.second.IsScalar()) {
421 throw std::runtime_error("parameter '" + name + "': the value must be a number or a text");
422 }
423 gParameters[name] = parameter.second.Scalar();
424 }
425 }
426 // The parameters are resolved in all the cases of the select blocks (those of the cases not chosen are
427 // also defined and used), then the blocks are replaced by their case
428 cfg = ExpandSelects(ResolveParameters(ResolveEnvVars(cfg)));
429 return ResolveYamlRefs(cfg, cfg);
430}

572std::string ScalarText(const std::string& value) {
573 try {
574 const YAML::Node node = YAML::Load("key: " + value);
575 if (!value.empty() && node.IsMap() && node.size() == 1 && node["key"].IsScalar() &&
576 node["key"].Tag() == "?" && node["key"].Scalar() == value)
577 return value;
578 } catch (const YAML::Exception&) {
579 }
580 std::string quoted = "\"";
581 for (const char c : value) {
582 if (c == '"' || c == '\\') quoted += '\\';
583 quoted += c;
584 }
585 return quoted + "\"";
586}

590size_t ScalarEnd(const std::string& line, size_t start) {
591 if (start < line.size() && (line[start] == '"' || line[start] == '\'')) {
592 const char quote = line[start];
593 for (size_t i = start + 1; i < line.size(); i++) {
594 if (quote == '"' && line[i] == '\\') {
595 i++; // Escaped character
596 } else if (line[i] == quote) {
597 if (quote == '\'' && i + 1 < line.size() && line[i + 1] == '\'') {
598 i++; // '' inside a single quoted scalar
599 } else {
600 return i + 1;
601 }
602 }
603 }
604 return std::string::npos;
605 }
606 size_t end = line.find(" #", start);
607 if (end == std::string::npos) end = line.size();
608 while (end > start && std::isspace(static_cast<unsigned char>(line[end - 1]))) end--;
609 return end;
610}

613void TRestYAMLUtils::SetValuesInFile(const std::string& fileName,
614 const std::vector<std::pair<std::string, std::string>>& values) {
615 if (values.empty()) return;
616
617 std::vector<std::string> lines;
618 {
619 std::ifstream in(fileName);
620 if (!in.is_open()) throw std::runtime_error("SetValuesInFile: cannot read " + fileName);
621 std::string line;
622 while (std::getline(in, line)) lines.push_back(line);
623 }
624
625 for (const auto& [path, value] : values) {
626 auto error = [&, &path = path](const std::string& reason) {
627 return std::runtime_error("SetValuesInFile: cannot set '" + path + "' in " + fileName + ": " +
628 reason);
629 };
630
631 const auto keys = TRestTools::Split(path, '.');
632 if (keys.empty()) throw error("empty path");
633 const std::string& key = keys.back();
634 const std::string scalar = ScalarText(value);
635
636 // A missing parent key is added (e.g. runInfo of a section whose runInfo is in an included file),
637 // then the text is parsed again: at most one more attempt per key
638 bool done = false;
639 for (size_t attempt = 0; !done; attempt++) {
640 if (attempt > keys.size()) throw error("its parent keys cannot be added");
641
642 // Parsed again after each change: the positions of the nodes in the text (lines and columns
643 // from 0)
644 std::string text;
645 for (const auto& line : lines) text += line + "\n";
646 const YAML::Node root = YAML::Load(text);
647
648 YAML::Node parent;
649 parent.reset(root);
650 YAML::Mark parentKeyMark; // Key of `parent` in its map (not for the root)
651 bool parentIsRoot = true;
652 bool added = false;
653 for (size_t k = 0; k + 1 < keys.size(); k++) {
654 const YAML::Node current = parent; // Const access: a missing key is not added
655 if (current.IsNull() && !parentIsRoot) {
656 // A key without value: its first key added below it
657 lines.insert(lines.begin() + parentKeyMark.line + 1,
658 std::string(parentKeyMark.column + 2, ' ') + ScalarText(keys[k]) + ":");
659 added = true;
660 break;
661 }
662 if (!current.IsMap() || current.size() == 0 || current.Style() == YAML::EmitterStyle::Flow)
663 throw error("the parent of " + keys[k] + " is not a block map");
664 if (!current[keys[k]]) {
665 // Added before the first key of the map, with its indentation
666 const YAML::Mark first = current.begin()->first.Mark();
667 lines.insert(lines.begin() + first.line,
668 std::string(first.column, ' ') + ScalarText(keys[k]) + ":");
669 added = true;
670 break;
671 }
672 for (const auto& element : current) {
673 if (element.first.Scalar() != keys[k]) continue;
674 parentKeyMark = element.first.Mark();
675 break;
676 }
677 parent.reset(current[keys[k]]);
678 parentIsRoot = false;
679 }
680 if (added) continue;
681 done = true;
682
683 if (parent.IsNull() && !parentIsRoot) {
684 lines.insert(lines.begin() + parentKeyMark.line + 1,
685 std::string(parentKeyMark.column + 2, ' ') + ScalarText(key) + ": " + scalar);
686 break;
687 }
688 if (!parent.IsMap() || parent.size() == 0 || parent.Style() == YAML::EmitterStyle::Flow)
689 throw error("its parent is not a block map");
690
691 bool found = false;
692 for (const auto& element : parent) {
693 if (element.first.Scalar() != key) continue;
694 found = true;
695 const YAML::Node valueNode = element.second;
696 if (!valueNode.IsScalar() && !valueNode.IsNull()) throw error("not a scalar");
697
698 const YAML::Mark keyMark = element.first.Mark();
699 std::string& line = lines.at(keyMark.line);
700 const bool quotedKey = line[keyMark.column] == '"' || line[keyMark.column] == '\'';
701 const size_t keyEnd =
702 quotedKey ? ScalarEnd(line, keyMark.column) : keyMark.column + key.size();
703 const size_t colon = keyEnd == std::string::npos ? std::string::npos : line.find(':', keyEnd);
704 if (colon == std::string::npos) throw error("key not found in its line");
705
706 size_t start = line.find_first_not_of(" \t", colon + 1);
707 if (start == std::string::npos || line[start] == '#') {
708 // No value (null): written after the colon, before the comment
709 const std::string comment = start == std::string::npos ? "" : " " + line.substr(start);
710 line = line.substr(0, colon + 1) + " " + scalar + comment;
711 break;
712 }
713 if (valueNode.Mark().line != keyMark.line)
714 throw error("the value is not in the line of its key");
715 start = valueNode.Mark().column;
716 if (line[start] == '|' || line[start] == '>') throw error("block scalar");
717 const size_t end = ScalarEnd(line, start);
718 if (end == std::string::npos) throw error("value of several lines");
719 line = line.substr(0, start) + scalar + line.substr(end);
720 break;
721 }
722
723 if (!found) {
724 // Added before the first key of the map, with its indentation
725 const YAML::Mark first = parent.begin()->first.Mark();
726 lines.insert(lines.begin() + first.line,
727 std::string(first.column, ' ') + ScalarText(key) + ": " + scalar);
728 }
729 }
730 }
731
732 std::ofstream out(fileName, std::ios::trunc);
733 if (!out.is_open()) throw std::runtime_error("SetValuesInFile: cannot write " + fileName);
734 for (const auto& line : lines) out << line << "\n";
