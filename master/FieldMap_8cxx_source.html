73 CSTRegion RegionNear(double x, double y, double z, unsigned int reach) const {
74 const double point[3] = {x, y, z};
75 unsigned int first[3], last[3];
76 for (int a = 0; a < 3; a++) {
77 const double step = fNodes[a] > 1 ? (fHigh[a] - fLow[a]) / (fNodes[a] - 1) : 1;
78 const double f = std::clamp((point[a] - fLow[a]) / step, 0.0, static_cast<double>(fNodes[a] - 1));
79 const auto cell = static_cast<unsigned int>(std::floor(f));
80 first[a] = cell > reach ? cell - reach : 0;
81 last[a] = std::min(cell + 1 + reach, fNodes[a] - 1);
82 }
83 unsigned char region = static_cast<unsigned char>(CSTRegion::kGas);
84 for (unsigned int k = first[2]; k <= last[2]; k++) {
85 for (unsigned int j = first[1]; j <= last[1]; j++) {
86 for (unsigned int i = first[0]; i <= last[0]; i++) {
87 const unsigned char node =
88 fRegions[(static_cast<size_t>(k) * fNodes[1] + j) * fNodes[0] + i];
89 if (node == static_cast<unsigned char>(CSTRegion::kDielectric))
90 return CSTRegion::kDielectric;
91 region = std::max(region, node);
92 }
93 }
94 }
95 return static_cast<CSTRegion>(region);

156void ReadCSTExport(const std::string& fileName, const std::string& error, Callback&& onNode) {
157 std::ifstream input(fileName);
158 if (!input) throw std::runtime_error(error + "cannot be read");
159 CSTUnits units;
160 std::string line;
161 size_t lineNumber = 0;
162 while (std::getline(input, line)) {
163 lineNumber++;
164 const auto first = line.find_first_not_of(" \t\r");
165 if (first == std::string::npos) continue;
166 line = line.substr(first);
167 if (!IsNode(line)) {
168 // The header: the units of the columns between brackets, x y z and the 3 components
169 std::vector<std::string> columns;
170 const std::regex brackets(R"(\[([^\]]*)\])");
171 for (auto it = std::sregex_iterator(line.begin(), line.end(), brackets);
172 it != std::sregex_iterator(); ++it) {
173 columns.push_back((*it)[1]);
174 }
175 if (columns.empty()) continue;
176 if (columns.size() < 6 || columns[0] != columns[1] || columns[0] != columns[2] ||
177 columns[3] != columns[4] || columns[3] != columns[5]) {
178 throw std::runtime_error(error + "line " + std::to_string(lineNumber) +
179 ": the header must give the units of x, y, z and of the 3 field "
180 "components");
181 }
182 units = {columns[0], columns[3]};
183 continue;
184 }
185 if (units.fLength.empty()) {
186 throw std::runtime_error(error + "no header with the units before the nodes");
187 }
188 std::istringstream values(line);
189 double x[3], value[3];
190 values >> x[0] >> x[1] >> x[2] >> value[0] >> value[1] >> value[2];
191 if (values.fail()) {
192 throw std::runtime_error(error + "line " + std::to_string(lineNumber) +
193 ": x y z and 3 field components expected");
194 }
195 onNode(units, x, value);
196 }
197}

209double Reduce(double x, double low, double high, bool periodic, bool mirror, bool& mirrored) {
210 const double length = high - low;
211 if (!(length > 0) || (!periodic && !mirror)) return x;
212 double reduced = low + std::fmod(x - low, length);
213 if (reduced < low) reduced += length;
214 if (mirror) {
215 const long cells = std::lround(std::floor((x - low) / length));
216 if (cells % 2 != 0) {
217 reduced = low + high - reduced;
218 mirrored = true;
219 }
220 }
221 return reduced;
222}

225std::unique_ptr<Garfield::ComponentGrid> LoadCSTFieldMap(const std::string& fileName,
226 const std::array<double, 3>& offset,
227 const std::string& dielectricFile) {
228 const std::string error = "restGarfield: field map " + fileName + ": ";
229
230 // The nodes in Garfield units, moved to the detector frame, written to a file for ComponentGrid (it only
231 // reads files)
232 TemporaryFile grid;
233 std::ofstream output(grid.Path());
234 output.precision(12);
235 double lengthFactor = 0, fieldFactor = 0; // to cm and V/cm
236 // The nodes (cm), whether they are in the gas and the magnitude of their field (V/cm), for the mesh, the
237 // inactive nodes and the dielectrics
238 std::vector<std::array<double, 3>> positions;
239 std::vector<bool> inGas;
240 std::vector<float> magnitudes;
241 ReadCSTExport(fileName, error, [&](const CSTUnits& units, const double* x, const double* e) {
242 if (lengthFactor == 0) {
243 lengthFactor = ParseUnit(units.fLength, error) * kCmPerMm;
244 fieldFactor = ParseUnit(units.fField, error) * kVcmPerVmm;
245 }
246 std::array<double, 3> position;
247 for (int i = 0; i < 3; i++) position[i] = x[i] * lengthFactor + offset[i] * kCmPerMm;
248 for (int i = 0; i < 3; i++) output << " " << position[i];
249 for (int i = 0; i < 3; i++) output << " " << e[i] * fieldFactor;
250 output << "\n";
251 positions.push_back(position);
252 // In the gas unless the field is exactly 0: CST writes 0 out of the solved volume (the corners of
253 // the box around a cylinder) and inside the conductors
254 inGas.push_back(e[0] != 0 || e[1] != 0 || e[2] != 0);
255 magnitudes.push_back(std::sqrt(e[0] * e[0] + e[1] * e[1] + e[2] * e[2]) * fieldFactor);
256 });
257 output.close();
258 const size_t nodes = positions.size();
259 if (nodes == 0) throw std::runtime_error(error + "no nodes");
260
261 // The mesh: the coordinates of the nodes along each axis, equally spaced, every node given once
262 const std::string regular = "a regular grid is needed (every node once, equal steps)";
263 std::array<unsigned int, 3> n;
264 std::array<double, 3> low, high, step;
265 for (int a = 0; a < 3; a++) {
266 std::set<double> values;
267 for (const auto& position : positions) values.insert(position[a]);
268 n[a] = static_cast<unsigned int>(values.size());
269 low[a] = *values.begin();
270 high[a] = *values.rbegin();
271 step[a] = n[a] > 1 ? (high[a] - low[a]) / (n[a] - 1) : 0;
272 size_t index = 0;
273 for (const double value : values) {
274 if (std::abs(value - (low[a] + index * step[a])) > 1e-6 * std::max(1., std::abs(step[a]))) {
275 throw std::runtime_error(error + regular);
276 }
277 index++;
278 }
279 if (step[a] == 0) step[a] = 1;
280 }
281 if (static_cast<size_t>(n[0]) * n[1] * n[2] != nodes) throw std::runtime_error(error + regular);
282 std::vector<unsigned char> regions(nodes, static_cast<unsigned char>(CSTRegion::kConductor));
283 std::vector<bool> given(nodes, false);
284 std::vector<float> magnitude(nodes, 0);
285 for (size_t node = 0; node < nodes; node++) {
286 size_t index = 0;
287 for (int a = 2; a >= 0; a--) {
288 index = index * n[a] + static_cast<size_t>(std::lround((positions[node][a] - low[a]) / step[a]));
289 }
290 if (given[index]) throw std::runtime_error(error + regular);
291 given[index] = true;
292 if (inGas[node]) regions[index] = static_cast<unsigned char>(CSTRegion::kGas);
293 magnitude[index] = magnitudes[node];
294 }
295 positions = {};
296 inGas = {};
297 magnitudes = {};
298
299 // The dielectrics: nodes where |D| > 1.5 eps0 |E|
300 if (!dielectricFile.empty()) {
301 const std::string dielectricError =
302 "restGarfield: dielectric map " + dielectricFile + " of " + fileName + ": ";
303 double dielectricLength = 0; // to cm
304 ReadCSTExport(
305 dielectricFile, dielectricError, [&](const CSTUnits& units, const double* x, const double* d) {
306 if (dielectricLength == 0) {
307 dielectricLength = ParseUnit(units.fLength, dielectricError) * kCmPerMm;
308 if (units.fField != "C/m^2" && units.fField != "As/m^2") {
309 throw std::runtime_error(dielectricError +
310 "the displacement field D in C/m^2 is needed");
311 }
312 }
313 size_t index = 0;
314 for (int a = 2; a >= 0; a--) {
315 const double position = x[a] * dielectricLength + offset[a] * kCmPerMm;
316 const double f = (position - low[a]) / step[a];
317 const long i = std::lround(f);
318 if (i < 0 || i >= static_cast<long>(n[a]) || std::abs(f - i) > 1e-3) {
319 throw std::runtime_error(dielectricError + "a node is not a node of the field map");
320 }
321 index = index * n[a] + static_cast<size_t>(i);
322 }
323 // |D| / (eps0 |E|): D in C/m^2, E in V/cm (100 V/m)
324 const double displacement = std::sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
325 if (displacement > kDielectricPermittivity * kEpsilon0 * magnitude[index] * 100) {
326 regions[index] = static_cast<unsigned char>(CSTRegion::kDielectric);
327 }
328 });
329 }
330
331 // The mesh set before loading: ComponentGrid sizes its own flags with it (deduced from the file, they
332 // were sized 1 x 1 x 1 in Garfield++ 4.0)
333 auto component = std::make_unique<CSTFieldGrid>(n, low, high, std::move(regions));
334 if (!component->SetMesh(n[0], n[1], n[2], low[0], high[0], low[1], high[1], low[2], high[2]) ||
335 !component->LoadElectricField(grid.Path(), "XYZ", false, false)) {
336 throw std::runtime_error(error + "rejected by Garfield++ ComponentGrid (" + regular + ")");
337 }
338 return component;
339}

365int CSTFieldMap::Locate(double x, double y, double z, std::array<double, 3>& cell,
366 std::array<bool, 3>& mirrored) const {
367 const double point[3] = {x, y, z};
368 for (int a = 0; a < 3; a++) {
369 mirrored[a] = false;
370 cell[a] = Reduce(point[a], fLow[a], fHigh[a], m_periodic[a], m_mirrorPeriodic[a], mirrored[a]);
371 }
372 // The first map containing the point; else, within the rounding of the point (e.g. 0.1 mm in cm is above
373 // a face at 0.01 cm), the first map with the point moved onto its face (ComponentGrid checks its limits
374 // exactly: a point just above them has no medium)
375 constexpr double tolerance = 1e-9; // cm
376 for (const double margin : {0.0, tolerance}) {
377 for (size_t map = 0; map < fMaps.size(); map++) {
378 const auto& limits = fLimits[map];
379 bool inside = true;
380 for (int a = 0; a < 3 && inside; a++) {
381 inside = cell[a] >= limits[2 * a] - margin && cell[a] <= limits[2 * a + 1] + margin;
382 }
383 if (!inside) continue;
384 for (int a = 0; a < 3; a++) cell[a] = std::clamp(cell[a], limits[2 * a], limits[2 * a + 1]);
385 return static_cast<int>(map);
386 }
387 }
388 return -1;
389}
