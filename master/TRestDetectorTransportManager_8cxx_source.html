124bool ReadGraph(TDirectory* directory, const std::string& name, std::vector<double>& fields,
125 std::vector<double>& values, const std::string& error) {
126 auto* graph = dynamic_cast<TGraph*>(directory->Get(name.c_str()));
127 if (!graph) return false;
128 std::vector<double> graphFields(graph->GetX(), graph->GetX() + graph->GetN());
129 for (auto& field : graphFields) field /= kVcmPerVmm;
130 if (fields.empty()) {
131 fields = graphFields;
132 } else if (graphFields != fields) {
133 throw std::runtime_error(error + "the graphs have different fields");
134 }
135 values = std::vector<double>(graph->GetY(), graph->GetY() + graph->GetN());
136 delete graph;
137 return true;

163 std::vector<std::unique_ptr<T>> tables;
164 if (!node || node.IsNull()) return tables;

172 if (!configurations[i].IsMap()) configurations.clear();

177 }

179 const std::string name = configurations.size() == 1 ? key : key + "_" + std::to_string(i);
180 tables.push_back(std::make_unique<T>(name, configurations[i]));
181 tables.back()->CheckConfig();

275double TRestDetectorTransportTable::Interpolate(const std::vector<double>& values, double field,
276 double pressure, bool logValues) const {
277 const size_t n = std::min(values.size(), fFields.size());
278 if (n == 0) return 0;
279 // Field of the table with the same E/p
280 const double x = fPressureScaling ? field * fPressure / pressure : field;
281 if (n == 1 || x <= fFields.front()) return values.front();
282 if (x >= fFields[n - 1]) return values[n - 1];
283 const size_t i = std::upper_bound(fFields.begin(), fFields.begin() + n, x) - fFields.begin() - 1;
284 const double t = std::log(x / fFields[i]) / std::log(fFields[i + 1] / fFields[i]);
285 if (logValues && values[i] > 0 && values[i + 1] > 0) {
286 return values[i] * std::pow(values[i + 1] / values[i], t);
287 }
288 return values[i] + t * (values[i + 1] - values[i]);
289}

367void TRestDetectorTransportTable::Write(TFile* file, const YAML::Node& medium,
368 const YAML::Node& generator) const {
369 const std::string key = GetKey();
370 YAML::Node entry;
371 entry["key"] = key;
372 entry["pressure"] = fPressure;
373 entry["temperature"] = fTemperature;
374 entry["fieldRange"] = std::vector<double>{fFields.front(), fFields.back()};
375 entry["pressureScaling"] = fPressureScaling;
376 entry["penningRate"] = fPenning;
377 std::vector<std::string> carriers;
378 for (const auto& [carrier, values] : fCarriers) carriers.push_back(carrier);
379 entry["carriers"] = carriers;
380 entry["generator"] = YAML::Clone(generator);
381 AddEntry(file, fMedium, medium, kTransportEntries, entry, kTablesDir, "TRestDetectorTransportTable");
382
383 std::string composition;
384 if (medium["components"]) {
385 for (const auto& [component, fraction] : medium["components"].as<std::map<std::string, double>>()) {
386 composition += (composition.empty() ? " (" : ", ") + component + " " + Format(fraction) + "%";
387 }
388 if (!composition.empty()) composition += ")";
389 }
390 const std::string condition = fMedium + composition + ", " + Format(fPressure) + " bar, " +
391 Format(fTemperature) + " K" +
392 (fPenning > 0 ? ", Penning " + Format(fPenning) : "");
393 for (const auto& [carrier, carrierValues] : fCarriers) {
394 for (const auto& quantity : kQuantities) {
395 const auto& values = carrierValues.*quantity.values;
396 if (values.size() != fFields.size()) continue;
397 WriteGraph(fMedium + "_" + key + "_" + carrier + "_" + quantity.name,
398 condition + ";E (V/cm);" + carrier + " " + quantity.title, fFields, values);
399 }
400 }

480double InterpolateLinear(const std::vector<double>& fields, const std::vector<double>& values, double x) {
481 if (fields.size() == 1 || x <= fields.front()) return values.front();
482 if (x >= fields.back()) return values.back();
483 const size_t i = std::upper_bound(fields.begin(), fields.end(), x) - fields.begin() - 1;
484 const double t = (x - fields[i]) / (fields[i + 1] - fields[i]);
485 return values[i] + t * (values[i + 1] - values[i]);
486}

489double TRestDetectorGainTable::GetGain(double field) const {
490 if (!InRange(field)) {
491 throw std::runtime_error("TRestDetectorGainTable: amplification field " + Format(field * kVcmPerVmm) +
492 " V/cm out of the range of table " + GetKey());

494 std::vector<double> logGain;
495 for (const double gain : fGain) logGain.push_back(std::log(std::max(gain, 1e-300)));
496 return std::exp(InterpolateLinear(fFields, logGain, field));

551 entry["temperature"] = fTemperature;
552 entry["gap"] = fTableGap;
553 entry["fieldRange"] = std::vector<double>{fFields.front(), fFields.back()};
554 entry["penningRate"] = fPenning;
555 entry["avalanches"] = sizes.empty() ? 0 : static_cast<int>(sizes.front().size());
556 entry["generator"] = YAML::Clone(generator);
557 AddEntry(file, fMedium, medium, kGainEntries, entry, kTablesDir, "TRestDetectorGainTable");
558

560 " K, gap " + Format(fTableGap * 1000) + " #mum" +
561 (fPenning > 0 ? ", Penning " + Format(fPenning) : "");
562 const std::tuple<const char*, const char*, const std::vector<double>*> quantities[] = {
563 {"gain", "mean gain", &fGain}, {"theta", "Polya #theta", &fTheta}};
564 for (const auto& [quantity, title, values] : quantities) {
565 WriteGraph(fMedium + "_" + key + "_" + quantity, condition + ";E (V/cm);" + title, fFields, *values);
566 }

568 if (sizes[i].empty()) continue;
569 const double maxSize = *std::max_element(sizes[i].begin(), sizes[i].end());

571 TH1D histogram(
572 name.c_str(),
573 (condition + ", E " + Format(fFields[i] * kVcmPerVmm) + " V/cm;avalanche size;avalanches")
574 .c_str(),
575 100, 0, 1.05 * maxSize + 1);
576 histogram.SetDirectory(nullptr);
577 for (const double size : sizes[i]) histogram.Fill(size);
578 histogram.Write(name.c_str(), TObject::kOverwrite);
579 }
