27bool CaseInsensitiveCompare(const std::string& str1, const std::string& str2) {
28 if (str1.size() != str2.size()) return false;
29 return std::equal(str1.begin(), str1.end(), str2.begin(), [](unsigned char c1, unsigned char c2) {
30 return std::tolower(c1) == std::tolower(c2);
31 });
32}

75TF1 EnergyDistributionFormulasToRootFormula(EnergyDistributionFormulas formula) {
76 switch (formula) {
77 case EnergyDistributionFormulas::COSMIC_NEUTRONS: {
78 const char* title = "Cosmic Neutrons at Sea Level";
79 auto cosmicNeutrons = [](double* xs, double*) {
80 const double logE = std::log(xs[0] * 1E-3);
81 return 1.006E-6 * std::exp(-0.3500 * logE * logE + 2.1451 * logE) +
82 1.011E-3 * std::exp(-0.4106 * logE * logE - 0.6670 * logE);
83 };
84 auto distribution = TF1(title, cosmicNeutrons, 1.0E2, 1.0E7, 0);
85 distribution.SetNormalized(true);
86 distribution.SetTitle(title);
87 distribution.GetXaxis()->SetTitle("Energy (keV)");
88 return distribution;
89 }

91 throw invalid_argument(

96 const char* title = "Watt formula: Neutrons from U238 fission";

224TF2 EnergyAndAngularDistributionFormulasToRootFormula(EnergyAndAngularDistributionFormulas type) {
225 if (type == EnergyAndAngularDistributionFormulas::COSMIC_MUONS) {
226 // Guan modified Gaisser parametrization formula for sea-level cosmic-ray muon flux
227 // Source document: arXiv:1509.06176 (Chinese Physics C Vol. 39, No. 6)
228 // Input parameter x: muon kinetic energy (processed inside simulation in units of keV)
229 // Input parameter y: zenith angle theta (in radians)
230 // The formula is differential in solid angle: the last factor integrates over the azimuth (dOmega =
231 // 2 pi sin(theta) dtheta), without another 2 pi (legacy geant4lib#151): counts / s / cm2 / keV / rad
232 const char* title = "Cosmic Muons Energy and Angular";
233
234 // Compiled (not a formula string): no interpreter needed, e.g. in the generator of each thread
235 auto guan = [](double* xs, double*) {
236 const double energy = xs[0] * 1.0E-6; // GeV
237 const double c = std::cos(xs[1]);
238 if (c < 0) return 0.0;
239 const double cosStar = std::sqrt(
240 (c * c + 0.0105212 - 0.068287 * std::pow(c, 0.958633) + 0.0407253 * std::pow(c, 0.817285)) /
241 0.982960);
242 const double spectrum =
243 0.14 * std::pow(energy * (1.0 + 3.64 / (energy * std::pow(cosStar, 1.29))), -2.7);
244 const double pions = 1.0 / (1.0 + 1.1 * energy * cosStar / 115.0);
245 const double kaons = 0.054 / (1.0 + 1.1 * energy * cosStar / 850.0);
246 return 1.0E-6 * spectrum * (pions + kaons) * 2.0 * TMath::Pi() * std::sin(xs[1]);
247 };
248 auto f = TF2(title, guan, 2.0E5, 5.0E9, 0.0, TMath::Pi() / 2.0, 0);
249
250 f.SetTitle(title);
251 return f;
252 }
253 throw invalid_argument(
254 "EnergyAndAngularDistributionFormulasToRootFormula: "
255 "Unknown energy and angular distribution formula");
256}

260HistogramContentUnits ParseHistogramContentUnits(const string& title) {
261 HistogramContentUnits units;
262 bool perSecond = false, perArea = false, perAngle = false;
263 // The words of the title: letters, digits, ^, braces (cm^{2}) and the sign of an exponent (cm^-2); the
264 // base of a word is before ^ or {
265 string word;
266 auto parse = [&]() {
267 if (word.empty()) return;
268 const string base = word.substr(0, word.find_first_of("^{"));
269 const bool squared = word.find('2') != string::npos;
270 static const map<string, double> energies = {
271 {"eV", 1e-3}, {"keV", 1}, {"MeV", 1e3}, {"GeV", 1e6}, {"TeV", 1e9}};
272 if (energies.count(base)) {
273 units.perEnergy = energies.at(base);
274 units.known = true;
275 } else if (base == "s" || base == "sec") {
276 perSecond = units.known = true;
277 } else if (base == "cm2" || (base == "cm" && squared)) {
278 perArea = units.known = true;
279 } else if (base == "sr" || base == "deg" || base == "rad") {
280 perAngle = units.known = true;
281 }
282 word.clear();
283 };
284 for (const char c : title) {
285 const bool exponentSign = c == '-' && !word.empty() && (word.back() == '^' || word.back() == '{');
286 if (std::isalnum(static_cast<unsigned char>(c)) || c == '^' || c == '{' || c == '}' || exponentSign) {
287 word += c;
288 } else {
289 parse();
290 }
291 }
292 parse();
293 units.flux = perSecond && perArea && !perAngle;
294 return units;
295}

297std::unique_ptr<TH1> ReadHistogram(const string& fileName, const string& name) {
298 // The current directory is restored at the end (closing the file makes it gROOT): restG4 writes the
299 // geometry in the current directory, its output file
300 TDirectory::TContext context;
301 unique_ptr<TFile> file(TFile::Open(TRestTools::GetFullPath(fileName).c_str(), "READ"));
302 if (!file || file->IsZombie()) {
303 throw invalid_argument("ReadHistogram: cannot open the file '" + fileName + "'");
304 }
305 const auto* histogram = file->Get<TH1>(name.c_str());
306 if (!histogram) {
307 throw invalid_argument("ReadHistogram: histogram '" + name + "' not found in '" + fileName + "'");
308 }
309 unique_ptr<TH1> copy(static_cast<TH1*>(histogram->Clone()));
310 copy->SetDirectory(nullptr);
311 return copy;

323std::unique_ptr<TH1D> ReadHistogram1D(const string& fileName, const string& name) {
324 auto histogram = ReadHistogram(fileName, name);
325 if (histogram->GetDimension() != 1) {
326 throw invalid_argument("ReadHistogram1D: '" + name + "' of '" + fileName + "' is not a 1D histogram");
327 }
328 if (auto* h = dynamic_cast<TH1D*>(histogram.get())) {
329 histogram.release();
330 return std::unique_ptr<TH1D>(h);
331 }
332 const auto edges = BinEdges(*histogram->GetXaxis());
333 auto out = std::make_unique<TH1D>(histogram->GetName(), histogram->GetTitle(), histogram->GetNbinsX(),
334 edges.data());
335 out->SetDirectory(nullptr);
336 // The axis titles give the units (e.g. "Energy (GeV)")
337 out->GetXaxis()->SetTitle(histogram->GetXaxis()->GetTitle());
338 out->GetYaxis()->SetTitle(histogram->GetYaxis()->GetTitle());
339 for (int b = 0; b <= histogram->GetNbinsX() + 1; b++) {
340 out->SetBinContent(b, histogram->GetBinContent(b));
341 out->SetBinError(b, histogram->GetBinError(b));
342 }
343 out->SetEntries(histogram->GetEntries());
344 return out;
345}

374ROOT::Math::XYZVector IsotropicDirection(TRandom& random) {
375 const double cosTheta = random.Uniform(-1, 1);
376 const double sinTheta = std::sqrt(1 - cosTheta * cosTheta);
377 const double phi = random.Uniform(0, 2 * M_PI);
378 return {sinTheta * std::cos(phi), sinTheta * std::sin(phi), cosTheta};
379}

381ROOT::Math::XYZVector DirectionFromAxis(TRandom& random, const ROOT::Math::XYZVector& axis, double angle) {
382 // Two unit vectors normal to the axis and to each other
383 const ROOT::Math::XYZVector reference =
384 std::abs(axis.Z()) < 0.9 ? ROOT::Math::XYZVector(0, 0, 1) : ROOT::Math::XYZVector(1, 0, 0);
385 const auto v = axis.Cross(reference).Unit();
386 const auto w = axis.Cross(v);
387 const double phi = random.Uniform(0, 2 * M_PI);
388 return std::cos(angle) * axis + std::sin(angle) * (std::cos(phi) * v + std::sin(phi) * w);
389}
