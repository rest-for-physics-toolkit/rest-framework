122template <typename T>
123inline std::string FormatAs(T value, Physical_Unit quantity) {
124 double val = static_cast<double>(value);
125 if (val == 0.0) return "0";
126
127 std::string baseUnitSymbol;
128 double factorToBase = 1.0;
129
130 if (quantity == Energy) {
131 baseUnitSymbol = "eV";
132 factorToBase = 1.0 / REST_Units_Map["eV"].second; // keV -> eV
133 } else if (quantity == Time) {
134 baseUnitSymbol = "s";
135 factorToBase = 1.0 / REST_Units_Map["s"].second; // us -> s
136 } else if (quantity == Length) {
137 baseUnitSymbol = "m";
138 factorToBase = 1.0 / REST_Units_Map["m"].second; // mm -> m
139 } else if (quantity == Voltage) {
140 baseUnitSymbol = "V";
141 factorToBase = 1.0 / REST_Units_Map["V"].second; // V -> V (1.0)
142 } else if (quantity == Mass) {
143 baseUnitSymbol = "g";
144 factorToBase = 1.0 / REST_Units_Map["g"].second; // kg -> g
145 } else if (quantity == MagneticField) {
146 baseUnitSymbol = "T";
147 factorToBase = 1.0 / REST_Units_Map["T"].second; // T -> T (1.0)
148 } else if (quantity == Pressure) {
149 baseUnitSymbol = "bar";
150 factorToBase = 1.0 / REST_Units_Map["bar"].second; // bar -> bar (1.0)
151 } else if (quantity == Angle) {
152 baseUnitSymbol = "rad";
153 factorToBase = 1.0 / REST_Units_Map["rad"].second; // rad -> rad (1.0)
154 } else if (quantity == Frequency) {
155 baseUnitSymbol = "Hz";
156 factorToBase = 1.0 / REST_Units_Map["Hz"].second; // MHz -> Hz
157 } else {
158 return std::to_string(val);
159 }
160
161 double valueInSI = val * factorToBase;
162 const double absValue = std::abs(valueInSI);
163
164 std::string prefix = "";
165 double scaleFactor = 1.0;
166
167 if (absValue < 1e-6) {
168 prefix = "n";
169 scaleFactor = 1e9;
170 } else if (absValue < 1e-3) {
171 prefix = "u";
172 scaleFactor = 1e6;
173 } else if (absValue < 1.0) {
174 prefix = "m";
175 scaleFactor = 1e3;
176 } else if (absValue < 1e3) {
177 prefix = "";
178 scaleFactor = 1.0;
179 } else if (absValue < 1e6) {
180 prefix = "k";
181 scaleFactor = 1e-3;
182 } else if (absValue < 1e9) {
183 prefix = "M";
184 scaleFactor = 1e-6;
185 } else if (absValue < 1e12) {
186 prefix = "G";
187 scaleFactor = 1e-9;
188 } else {
189 prefix = "T";
190 scaleFactor = 1e-12;
191 }
192
193 std::ostringstream ss;
194 ss << std::fixed << std::setprecision(2) << (valueInSI * scaleFactor) << " " << prefix << baseUnitSymbol;
195 return ss.str();
196}
