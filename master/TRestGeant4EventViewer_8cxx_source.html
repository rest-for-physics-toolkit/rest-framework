53TrackVisualConfiguration GetTrackVisualConfiguration(const TRestGeant4Track& track) {
54 auto config = TrackVisualConfiguration();
55
56 const auto particleName = track.GetParticleName();
57
58 if (particleName == "geantino") {
59 config.fColor = kRed;
60 config.fLineStyle = 9;
61 config.fLineWidth = 2;
62 return config;
63 }
64
65 if (particleName.find('-') != std::string::npos) {
66 config.fColor = kMagenta;
67 } else if (particleName.find('+') != std::string::npos) {
68 config.fColor = kAzure;
69 }
70
71 if (particleName == "neutron") {
72 config.fColor = kOrange;
73 } else if (particleName == "gamma") {
74 config.fColor = kGreen;
75 } else if (particleName == "e-") {
76 config.fColor = kRed;
77 } else if (particleName == "mu-") {
78 config.fColor = kGray;
79 } else if (particleName == "alpha") {
80 config.fColor = kYellow;
81 }
82
83 Width_t width = TMath::Log10(track.GetInitialEnergy() / 10);
84 width = (width > 10 ? 10 : width);
85 width = (width < 1 ? 1 : width);
86
87 // At least 1, a track below 10 keV had a width <= 0 and was not drawn
88 config.fLineWidth = width;
89
90 if (track.GetCreatorProcess() == "nCapture") {
91 config.fLineStyle = 2;
92 }
93
94 return config;
95}

97TEveStraightLineSet* GetTrackEveDrawable(const TRestGeant4Track& track, double geomScale) {
98 auto lineSet = new TEveStraightLineSet(
99 TRestTools::Format("ID %d | %s | Created by %s | KE: %s", track.GetTrackID(),
100 track.GetParticleName().c_str(), track.GetCreatorProcess().c_str(),
101 REST_Units::FormatAs(track.GetInitialEnergy(), REST_Units::Energy).c_str())
102 .c_str());
103
104 const size_t nHits = track.GetNumberOfHits();
105
106 if (nHits > 1) {
107 for (size_t i = 0; i < nHits - 1; ++i) {
108 const auto posCurrent = track.GetHitPosition(i);
109 const auto posNext = track.GetHitPosition(i + 1);
110
111 lineSet->AddLine(
112 static_cast<float>(geomScale * posCurrent.X()),
113 static_cast<float>(geomScale * posCurrent.Y()),
114 static_cast<float>(geomScale * posCurrent.Z()), static_cast<float>(geomScale * posNext.X()),
115 static_cast<float>(geomScale * posNext.Y()), static_cast<float>(geomScale * posNext.Z()));
116 }
117 }
118
119 const auto config = GetTrackVisualConfiguration(track);
120
121 lineSet->SetMainColor(config.fColor);
122 lineSet->SetLineColor(config.fColor);
123 lineSet->SetLineStyle(config.fLineStyle);
124 lineSet->SetLineWidth(config.fLineWidth);
125
126 return lineSet;
127}

162void TRestGeant4EventViewer::AddEvent(TRestEvent* event) {
163 DeleteCurrentEvent();
164
165 if (!event) {
166 RESTError << "TRestGeant4EventViewer::AddEvent, received event is NULL!!!" << RESTendl;
167 return;
168 }
169
170 fG4Event = static_cast<TRestGeant4Event*>(event);
171 fEvent = fG4Event;
172 fG4Metadata = fG4Event->GetGeant4Metadata();
173
174 if (fG4Metadata == nullptr) {
175 RESTError << "TRestGeant4EventViewer::AddEvent. No TRestGeant4Metadata found." << RESTendl;
176 return;
177 }
178
179 // The geometry of the simulation (gGeoManager, given by the browser, may be another one, e.g. a readout)
180 if (auto* geometry = fG4Metadata->GetGeant4GeometryInfo().GetGeometry()) SetGeometry(geometry);
181
182 const int nTracks = fG4Event->GetNumberOfTracks();
183
184 map<int, TEveStraightLineSet*> linesSet;
185 map<std::string, TEvePointSet*> hitsPoints;
186 map<std::string, TEveElementList*> hitsType;
187
188 auto trackList = new TEveElementList("Tracks");
189 gEve->AddElement(trackList);
190
191 auto hitsList = new TEveElementList("Hits");
192 gEve->AddElement(hitsList);
193
194 const auto& physicsInfo = fG4Metadata->GetGeant4PhysicsInfo();
195
196 for (int i = 0; i < nTracks; ++i) {
197 const auto track = fG4Event->GetTrack(i);
198
199 if (track.GetInitialEnergy() < 1.0 || track.GetLength() < 0.1) {
200 continue;
201 }
202
203 auto line = GetTrackEveDrawable(track, fGeomScale);
204 linesSet[track.GetTrackID()] = line;
205
206 TEveElement* parentLine = trackList;
207
208 if (linesSet.count(track.GetParentID())) {
209 parentLine = linesSet.at(track.GetParentID());
210 }
211
212 gEve->AddElement(line, parentLine);
213
214 const size_t nTrackHits = track.GetNumberOfHits();
215
216 for (size_t j = 0; j < nTrackHits; ++j) {
217 const auto processName = physicsInfo.GetProcessName(track.GetHitProcess(j));
218
219 const auto processType = physicsInfo.GetProcessType(processName);
220
221 const XYZVector position = track.GetHitPosition(j);
222
223 if (hitsType.count(processType) == 0) {
224 hitsType[processType] = new TEveElementList(processType.c_str());
225
226 gEve->AddElement(hitsType[processType], hitsList);
227 }
228
229 if (hitsPoints.count(processName) == 0) {
230 hitsPoints[processName] = new TEvePointSet(processName.c_str());
231
232 auto hitPoints = hitsPoints.at(processName);
233
234 auto hitsVisualConfig = GetHitsVisualConfiguration(processName, processType);
235
236 hitPoints->SetMarkerColor(hitsVisualConfig.fColor);
237 hitPoints->SetMarkerStyle(hitsVisualConfig.fMarkerStyle);
238 hitPoints->SetMarkerSize(hitsVisualConfig.fMarkerSize);
239
240 gEve->AddElement(hitPoints, hitsType[processType]);
241 }
242
243 hitsPoints.at(processName)
244 ->SetNextPoint(fGeomScale * position.X(), fGeomScale * position.Y(),
245 fGeomScale * position.Z());
246 }
247 }
248
249 if (nTracks > 0) {
250 const auto firstTrack = fG4Event->GetTrack(0);
251
252 const XYZVector position = firstTrack.GetInitialPosition();
253
254 const std::string subID =
255 fG4Event->GetSubID() > 0 ? TRestTools::Format(" (SubID: %d)", fG4Event->GetSubID()) : "";
256 AddText(
257 TRestTools::Format("Event ID: %d%s | Primary origin: (%4.2lf, %4.2lf, %4.2lf) mm",
258 fG4Event->GetID(), subID.c_str(), position.X(), position.Y(), position.Z()),
259 position);
260 } else {
261 AddText(TRestTools::Format("Event ID: %d | No tracks found", fG4Event->GetID()), XYZVector(0, 0, 0));
262 }
263
264 Update();
265}

302void TRestGeant4EventViewer::AddTrack(int trkID, int parentID, XYZVector from, const std::string& name) {
303 TEveLine* evLine = new TEveLine();
304
305 evLine->SetName(name.c_str());
306 if (fHitConnectors.count(trkID) > 0) {
307 RESTWarning << "TRestGeant4EventViewer::AddTrack: track ID " << trkID << " added twice" << RESTendl;
308 }
309 fHitConnectors[trkID] = evLine;
310
311 evLine->SetMainColor(kWhite);
312 evLine->SetLineWidth(4);
313
314 if (name.find("gamma") != std::string::npos) evLine->SetMainColor(kGreen);
315
316 if (name.find("e-") != std::string::npos) evLine->SetMainColor(kRed);
317
318 if (name.find("mu-") != std::string::npos) evLine->SetMainColor(kGray);
319
320 if (name.find("alpha") != std::string::npos) evLine->SetMainColor(kYellow);
321
322 if (name.find("neutron") != std::string::npos) evLine->SetMainColor(kBlue);
323
324 evLine->SetNextPoint(from.X() * fGeomScale, from.Y() * fGeomScale, from.Z() * fGeomScale);
325
326 // A primary (parent ID 0) or a track whose parent has no line is drawn at the top level
327 const auto parent = fHitConnectors.find(parentID);
328 if (parentID != 0 && parent != fHitConnectors.end() && parent->second != nullptr) {
329 parent->second->AddElement(evLine);
330 return;
331 }
332 if (parentID != 0) {
333 RESTWarning << "Parent ID: " << parentID << " of track " << trkID << " was not found!" << RESTendl;
334
335 RESTWarning << "This might be solved by enabling registerEmptyTracks in the TRestGeant4Metadata"
336 << RESTendl;
337 }
338 gEve->AddElement(evLine);
339}
