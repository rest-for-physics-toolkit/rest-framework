73std::string TRestRunInfo::ResolveFilePattern(const std::string& pattern) const {
74 const YAML::Node parameters = GetParameters();
75 return ReplaceDelimited(pattern, '[', ']', [&](const std::string& name) -> std::optional<std::string> {
76 char number[32];
77 if (name == "runNumber") {
78 snprintf(number, sizeof(number), "%05d", fRunNumber);
79 return number;
80 }
81 if (name == "subRunNumber") {
82 snprintf(number, sizeof(number), "%03d", fSubRunNumber);
83 return number;
84 }
85 if (name == "version") return REST_VERSION;
86 const YAML::Node node = TRestYAMLUtils::GetNodeAtPath(parameters, name);
87 if (!node || !node.IsScalar()) {
88 throw std::runtime_error("TRestRunInfo: [" + name + "] in \"" + pattern +
89 "\" is not a parameter of the run (its keys, e.g. [runNumber] or "
90 "[runTag], or [runInfo.key])");
91 }
92 return CleanString(node.Scalar());
93 });
