119void TRestRawCommonNoiseReductionProcess::ProcessBlock(TRestRawSignalEvent& signalEvent,
120 const TRestCommonNoiseBlock& block) {
121 std::vector<int> members;
122 int nBins = -1;
123 for (int s = 0; s < signalEvent.GetNumberOfSignals(); s++) {
124 const auto& sig = signalEvent.GetSignal(s);
125 if (!block.ContainsID(sig.GetSignalID())) continue;
126 const int n = sig.GetNPoints();
127 if (nBins < 0) nBins = n;
128 if (n != nBins) continue;
129 members.push_back(s);
130 }
131 if (nBins <= 0 || (int)members.size() < block.fMinSignalsRequired) return;
132
133 std::vector<int> active;
134 std::vector<double> activeBaseline;
135 std::vector<std::vector<short>> activeData;
136 std::vector<size_t> cleanIdx;
137
138 for (int pos : members) {
139 std::vector<short> data = signalEvent.GetSignal(pos).GetData();
140
141 double baseline = 0, sigma = 0;
142 GetBaselineSigma(data, block.fBaselineRange.first, block.fBaselineRange.second, baseline, sigma);
143
144 if (sigma < block.fBaselineSigmaThreshold) continue;
145
146 bool hasSignal = false;
147 if (sigma > 0) {
148 const double limit = block.fSignalSigmaThreshold * sigma;
149 const double hi = baseline + limit, lo = baseline - limit;
150 for (short v : data) {
151 if (v > hi || v < lo) {
152 hasSignal = true;
153 break;
154 }
155 }
156 }
157
158 if (!hasSignal) cleanIdx.push_back(active.size());
159 active.push_back(pos);
160 activeBaseline.push_back(baseline);
161 activeData.push_back(std::move(data));
162 }
163
164 const int nClean = (int)cleanIdx.size();
165 if (nClean < block.fMinSignalsRequired) return;
166
167 double baselineMean = 0;
168 for (double b : activeBaseline) baselineMean += b;
169 baselineMean /= (double)active.size();
170
171 std::vector<short> matrix((size_t)nBins * nClean);
172 for (int k = 0; k < nClean; k++) {
173 const auto& d = activeData[cleanIdx[k]];
174 for (int b = 0; b < nBins; b++) matrix[(size_t)b * nClean + k] = d[b];
175 }
176
177 int begin = nClean / 2, end = nClean / 2;
178 if (block.fMode == 1) {
179 const int middle = nClean / 2;
180 const int half = (int)(nClean * block.fCenterWidth * 0.01);
181 begin = std::max(0, middle - half);
182 end = std::min(nClean - 1, middle + half);
183 }
184 const double norm = (double)(end - begin + 1);
185
186 std::vector<int> deltas(nBins);
187 for (int b = 0; b < nBins; b++) {
188 short* row = matrix.data() + (size_t)b * nClean;
189
190 std::nth_element(row, row + begin, row + nClean);
191 double binCorrection;
192 if (end == begin) {
193 binCorrection = row[begin];
194 } else {
195 std::nth_element(row + begin + 1, row + end, row + nClean);
196 double sum = 0;
197 for (int i = begin; i <= end; i++) sum += row[i];
198 binCorrection = sum / norm;
199 }
200
201 deltas[b] = (int)std::lround(baselineMean - binCorrection);
202 }
203
204 for (int pos : active) {
205 auto& sig = signalEvent.GetSignal(pos);
206 for (int b = 0; b < nBins; b++) sig.IncreaseBinByClamped(b, deltas[b]);
207 }
208}
