86void TRestGeant4ToDetectorHitsProcess::InitProcess() {
87 fVolumeIDs.clear();
88
89 const auto* metadata =
90 fRunInfo ? dynamic_cast<TRestGeant4Metadata*>(fRunInfo->GetMetadataClass("TRestGeant4Metadata"))
91 : nullptr;
92 if (!metadata) {
93 throw std::runtime_error(
94 "TRestGeant4ToDetectorHitsProcess: no TRestGeant4Metadata found, needed to select volumes");
95 }
96 const auto& geometryInfo = metadata->GetGeant4GeometryInfo();
97
98 if (fVolumes.empty()) {
99 const auto& sensitive = metadata->GetSensitiveVolumes();
100 fVolumeIDs = geometryInfo.GetVolumeIDsFromNames({sensitive.begin(), sensitive.end()});
101 if (fVolumeIDs.empty()) {
102 throw std::runtime_error(
103 "TRestGeant4ToDetectorHitsProcess: no 'volumes' and no sensitive volume found");
104 }
105 }
106
107 // Volume selection as TRestGeant4QuenchingProcess: the names matching each expression, compared
108 // with the physical and filtered names of the hit volumes (resolved here once per volume ID)
109 for (const auto& expression : fVolumes) {
110 const auto ids = geometryInfo.GetVolumeIDsMatchingExpression(expression);
111 if (ids.empty()) {
112 RESTWarning << "TRestGeant4ToDetectorHitsProcess: no volume matches '" << expression << "'"
113 << RESTendl;
114 }
115 fVolumeIDs.insert(ids.begin(), ids.end());
116 }
117 for (const auto id : fVolumeIDs) {
118 RESTDebug << "TRestGeant4ToDetectorHitsProcess: volume " << geometryInfo.GetVolumeFromID(id)
119 << " (id " << id << ") selected" << RESTendl;
120 }
121
122 if (fVolumeIDs.empty()) {
123 throw std::runtime_error("TRestGeant4ToDetectorHitsProcess: no volume matches 'volumes'");
124 }
125
126 RegisterObservable("TimeOffset", fTimeOffset);
127}

129bool TRestGeant4ToDetectorHitsProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
130 const auto* g4Event = dynamic_cast<const TRestGeant4Event*>(&input);
131 auto* hitsEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
132 if (!g4Event || !hitsEvent) {
133 throw std::runtime_error(
134 "TRestGeant4ToDetectorHitsProcess: input/output is not a TRestGeant4Event/"
135 "TRestDetectorHitsEvent");
136 }
137
138 hitsEvent->Initialize();
139 hitsEvent->SetEventInfo(&input);
140
141 const auto& data = g4Event->fEventData;
142 const auto& hits = data.hitsStorage;
143 // Times relative to the first deposit of the event
144 fTimeOffset = g4Event->GetFirstDepositTime();
145 // The hits of the track t are [trackStartIndices[t], trackStartIndices[t] + trackNHits[t])
146 for (size_t t = 0; t < data.trackParticleIDs.size(); t++) {
147 const size_t start = data.trackStartIndices.at(t);
148 const size_t end = start + data.trackNHits.at(t);
149 for (size_t i = start; i < end; i++) {
150 if (hits.energy[i] <= 0) continue;
151 if (fVolumeIDs.count(data.hitVolumeID.at(i)) == 0) continue;
152 hitsEvent->AddHit(hits.x[i], hits.y[i], hits.z[i], hits.energy[i], hits.time[i] - fTimeOffset,
153 TRestDetectorHitsEvent::REST_HitType::XYZ, data.trackParticleIDs[t]);
154 }
155 }
156
157 RESTDebug << "TRestGeant4ToDetectorHitsProcess: " << hitsEvent->GetNumberOfHits() << " hits added"
158 << RESTendl;
159
160 return hitsEvent->GetNumberOfHits() > 0;
161}
