47bool InitializeRestRuntimeFromRestPath() {
48 const char* restPathEnv = gSystem->Getenv("REST_PATH");
49 if (!restPathEnv || restPathEnv[0] == '\0') {
50 std::cerr << "restRoot: REST_PATH is not set.\n";
51 return false;
52 }
53
54 const std::filesystem::path restPath(restPathEnv);
55 const std::filesystem::path libDir = restPath / "lib";
56 const std::filesystem::path includeDir = restPath / "include";
57
58 if (!std::filesystem::exists(libDir) || !std::filesystem::is_directory(libDir)) {
59 std::cerr << "restRoot: library directory not found: " << libDir << "\n";
60 return false;
61 }
62
63 std::vector<std::filesystem::path> libraries;
64 for (const auto& entry : std::filesystem::directory_iterator(libDir)) {
65 if (!entry.is_regular_file()) continue;
66 const auto fileName = entry.path().filename().string();
67 if (fileName.rfind("libRest", 0) == 0 && entry.path().extension() == REST_SHARED_LIBRARY_SUFFIX) {
68 if (entry.path().stem() == "libRestG4") continue;
69 libraries.push_back(entry.path());
70 }
71 }
72
73 std::sort(libraries.begin(), libraries.end());
74
75 bool loadedAny = false;
76 for (const auto& lib : libraries) {
77 std::cout << "Loading " << lib << std::endl;
78 if (gSystem->Load(lib.string().c_str()) >= 0) {
79 loadedAny = true;
80 }
81 }
82
83 if (!loadedAny) {
84 RESTError << "restRoot: no REST shared libraries could be loaded from " << libDir << RESTendl;
85 return false;
86 }
87
88 if (std::filesystem::exists(includeDir) && std::filesystem::is_directory(includeDir)) {
89 // ACLiC and the interpreter
90 gSystem->AddIncludePath((" -I" + includeDir.string()).c_str());
91 gInterpreter->AddIncludePath(includeDir.string().c_str());
92 }
93
94 // The macros are compiled in libraries (lib<macro>_C.so, without dictionary) and REST_Macros.h,
95 // generated by CMake, declares their functions and global pointers: loading them is immediate
96 // (interpreting all the .C at startup took several seconds). The .C are installed as reference,
97 // out of the macro path: .x/.L of them would redefine the declared functions
98 const std::filesystem::path macrosDir = restPath / "macros";
99 if (std::filesystem::exists(macrosDir) && std::filesystem::is_directory(macrosDir)) {
100 std::vector<std::filesystem::path> macroLibraries;
101 for (const auto& entry : std::filesystem::directory_iterator(macrosDir)) {
102 if (!entry.is_regular_file()) continue;
103 const auto fileName = entry.path().filename().string();
104 const std::string suffix = std::string("_C") + REST_SHARED_LIBRARY_SUFFIX;
105 if (fileName.rfind("lib", 0) == 0 && fileName.size() > suffix.size() &&
106 fileName.compare(fileName.size() - suffix.size(), suffix.size(), suffix) == 0) {
107 macroLibraries.push_back(entry.path());
108 }
109 }
110 std::sort(macroLibraries.begin(), macroLibraries.end());
111 for (const auto& lib : macroLibraries) {
112 if (gSystem->Load(lib.string().c_str()) < 0) {
113 RESTError << "restRoot: the macro library " << lib << " could not be loaded" << RESTendl;
114 }
115 }
116
117 const std::filesystem::path declarations = macrosDir / "REST_Macros.h";
118 if (!std::filesystem::exists(declarations) ||
119 !gInterpreter->Declare(("#include \"" + declarations.string() + "\"").c_str())) {
120 RESTError << "restRoot: the macro declarations " << declarations << " could not be loaded"
121 << RESTendl;
122 }
123 }
124
125 return true;
126}

133int main(int argc, char* argv[]) {
134 std::cout << "---------------------Welcome to RESTRoot--------------------\n";
135 std::cout << REST_Version::GetSummary() << "\n";
136
137 REST_Units::SetGeoManagerDefaultUnits();
138
139 if (!InitializeRestRuntimeFromRestPath()) {

141 }
142
143 // argv[0] (the program) is passed to TRint as it is; the REST files are looked for from argv[1]
144 std::vector<std::string> restFiles;
145 std::vector<char*> cleanArgv;
146 if (argc > 0) cleanArgv.push_back(argv[0]);
147
148 for (int i = 1; i < argc; ++i) {
149 std::string arg = argv[i];
150 std::string fullPath = TRestTools::GetFullPath(arg);
151
152 if (TRestTools::isDataSet(fullPath) || TRestTools::isValidTRestRun(fullPath)) {
153 restFiles.push_back(fullPath);
154 continue;
155 }
156
157 cleanArgv.push_back(argv[i]);
158 }
159
160 // A single REST file is opened (the globals _run, _ev and _dataSet hold one file): the last one
161 const std::string fileToOpen = restFiles.empty() ? "" : restFiles.back();
162 if (restFiles.size() > 1) {
163 RESTWarning << "restRoot: " << restFiles.size() << " REST files given, only the last one is opened ("
164 << fileToOpen << ")" << RESTendl;
165 }
166
167 cleanArgv.push_back(nullptr);
168
169 int cleanArgc = static_cast<int>(cleanArgv.size()) - 1;
170 char** cleanArgvPtr = cleanArgv.data();
171
172 gStyle->SetPalette(1);
173 gStyle->SetTimeOffset(0);
174
175 TRint theApp("App", &cleanArgc, cleanArgvPtr);
176
177 if (!fileToOpen.empty()) {
178 std::string command = "REST_OpenInputFile(\"" + fileToOpen + "\");";
179 gROOT->ProcessLine(command.c_str());
180 }
181
182 theApp.Run();
183 return 0;
184}
