145void TRestDetectorPhotosensorProcess::LoadConfig() {
146 TRestEventProcess::LoadConfig();
147 if (!fNode || fNode.IsNull()) {
148 RESTError << "TRestDetectorPhotosensorProcess::LoadConfig - YAML node is missing" << RESTendl;
149 return;
150 }
151 UpdateParamsFromYAML<TRestDetectorPhotosensorProcess>(fNode);
152 const std::string prefix = "TRestDetectorPhotosensorProcess: ";
153 if (fTimeJitter < 0 || fGainSpread < 0 || fAfterpulseSpread < 0 || fCellRecoveryTime < 0) {
154 throw std::runtime_error(
155 prefix + "timeJitter, gainSpread, afterpulseSpread and cellRecoveryTime must be >= 0");
156 }
157 if (fDarkCountRate < 0) throw std::runtime_error(prefix + "darkCountRate must be >= 0");
158 if (fDarkCountRate > 0 && !(fDarkCountWindow.second > fDarkCountWindow.first)) {
159 throw std::runtime_error(prefix +
160 "darkCountRate needs a darkCountWindow [start, end] with end > start");
161 }
162 for (double probability : {fCrosstalkProbability, fAfterpulseProbability}) {
163 if (probability < 0 || probability >= 1) {
164 throw std::runtime_error(prefix +
165 "crosstalkProbability and afterpulseProbability must be in [0, 1)");
166 }
167 }
168 // Mean number of avalanches triggered by one: below 1, or the chain does not end
169 if (fCrosstalkProbability + fAfterpulseProbability >= 1) {
170 throw std::runtime_error(prefix + "crosstalkProbability + afterpulseProbability must be below 1");
171 }
172 if (fAfterpulseProbability > 0 && fAfterpulseDelay <= 0) {
173 throw std::runtime_error(prefix + "afterpulseProbability needs afterpulseDelay > 0");
174 }
175 if (fCells < 0) throw std::runtime_error(prefix + "cells must be >= 0");
176 UpdateYAMLFromParams<TRestDetectorPhotosensorProcess>(fNode);
177}

179void TRestDetectorPhotosensorProcess::InitProcess() {
180 const std::string user = "TRestDetectorPhotosensorProcess";
181 if (!fManager) throw std::runtime_error(user + ": no TRestManager");
182 auto* readouts =
183 dynamic_cast<TRestDetectorReadoutManager*>(fManager->GetMetadataClass("TRestDetectorReadoutManager"));
184 if (!readouts) throw std::runtime_error(user + ": no TRestDetectorReadoutManager found");
185 fReadout = readouts->SelectReadout(fReadoutName, user);
186 // The stored metadata keeps the readout used
187 UpdateYAMLFromParams<TRestDetectorPhotosensorProcess>(fNode);
188
189 // A position of each channel for its dark counts: one that the readout gives back as the channel (as
190 // TRestDetectorHitsToSignalProcess finds it), the read end or else the centre
191 fChannelPositions.clear();
192 if (fDarkCountRate > 0) {
193 const auto inChannel = [this](int daqID, const XYZVector& p) {
194 return std::isfinite(p.X()) && std::isfinite(p.Y()) && std::isfinite(p.Z()) &&
195 fReadout->GetChannelFromPosition(p.X(), p.Y(), p.Z()) == daqID;
196 };
197 for (int daqID : fReadout->GetDAQChannels()) {
198 const XYZVector end = fReadout->GetReadoutPosition(daqID);
199 const XYZVector centre = fReadout->GetPositionFromChannel(daqID);
200 if (inChannel(daqID, end)) {
201 fChannelPositions[daqID] = end;
202 } else if (inChannel(daqID, centre)) {
203 fChannelPositions[daqID] = centre;
204 } else {
205 throw std::runtime_error(user + ": no position of the channel " + std::to_string(daqID) +
206 " of readout " + fReadout->GetName() + " for its dark counts");
207 }
208 }
209 if (fChannelPositions.empty()) {
210 throw std::runtime_error(user + ": the readout " + fReadout->GetName() +
211 " has no channels for the dark counts");
212 }
213 }
214
215 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
216 fRandom.SetSeed(fSeed);
217
218 RegisterObservable("NAvalanches", fNAvalanches);
219 RegisterObservable("NDarkCounts", fNDarkCounts);
220 RegisterObservable("NCrosstalk", fNCrosstalk);
221 RegisterObservable("NAfterpulses", fNAfterpulses);
222 RegisterObservable("NSaturationLosses", fNSaturationLosses);
223}

225double TRestDetectorPhotosensorProcess::AfterpulseDelay() {
226 if (fAfterpulseSpread > 0) return std::max(0.0, fRandom.Gaus(fAfterpulseDelay, fAfterpulseSpread));
227 return fRandom.Exp(fAfterpulseDelay);
228}

230bool TRestDetectorPhotosensorProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
231 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
232 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
233 if (!inputEvent || !outputEvent) {
234 throw std::runtime_error(
235 "TRestDetectorPhotosensorProcess: input/output is not a TRestDetectorHitsEvent");
236 }
237
238 outputEvent->Initialize();
239 outputEvent->SetEventInfo(inputEvent);
240 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
241 fNAvalanches = 0;
242 fNDarkCounts = 0;
243 fNCrosstalk = 0;
244 fNAfterpulses = 0;
245 fNSaturationLosses = 0;
246
247 // The avalanches of each channel, earliest first: the photoelectrons and the dark counts
248 std::map<int, std::priority_queue<Avalanche>> channels;
249 int lastChannel = -1;
250 XYZVector lastPosition;
251 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
252 const XYZVector position = inputEvent->GetPosition(n);
253 // The photoelectrons of a channel share its position (TRestDetectorLightCollectionProcess)
254 if (n == 0 || position != lastPosition) {
255 lastChannel = fReadout->GetChannelFromPosition(position.X(), position.Y(), position.Z());
256 lastPosition = position;
257 }
258 if (lastChannel < 0) continue;
259 const long photoelectrons = std::lround(inputEvent->GetEnergy(n));
260 auto& queue = channels[lastChannel];
261 for (long p = 0; p < photoelectrons; p++) {
262 queue.push({inputEvent->GetTime(n), position, inputEvent->GetParticleID(n), Kind::Photoelectron});
263 }
264 }
265 if (fDarkCountRate > 0) {
266 const double start = fDarkCountWindow.first, length = fDarkCountWindow.second - start;
267 for (const auto& [daqID, position] : fChannelPositions) {
268 const int darkCounts = fRandom.Poisson(fDarkCountRate * length);
269 if (darkCounts == 0) continue;
270 auto& queue = channels[daqID];
271 for (int d = 0; d < darkCounts; d++) {
272 queue.push({start + length * fRandom.Rndm(), position, 0, Kind::DarkCount});
273 }
274 }
275 }
276
277 // Each channel in time order: saturation, then the crosstalk and afterpulses of each avalanche
278 std::unordered_map<int, double> firedCells; // cell: time of its last avalanche
279 for (auto& [daqID, queue] : channels) {
280 firedCells.clear();
281 while (!queue.empty()) {
282 const Avalanche avalanche = queue.top();
283 queue.pop();
284 if (fCells > 0) {
285 const int cell = static_cast<int>(fRandom.Integer(fCells));
286 const auto fired = firedCells.find(cell);
287 if (fired != firedCells.end() &&
288 (fCellRecoveryTime <= 0 || avalanche.time - fired->second < fCellRecoveryTime)) {
289 fNSaturationLosses++;
290 continue;
291 }
292 firedCells[cell] = avalanche.time;
293 }
294
295 double time = avalanche.time;
296 if (fTimeJitter > 0) time += fRandom.Gaus(0, fTimeJitter);
297 const double charge = fGainSpread > 0 ? std::max(0.0, fRandom.Gaus(1, fGainSpread)) : 1.0;
298 outputEvent->AddHit(avalanche.position, charge, time, TRestDetectorHitsEvent::REST_HitType::XYZ,
299 avalanche.particleID);
300 fNAvalanches++;
301 if (avalanche.kind == Kind::DarkCount) fNDarkCounts++;
302 if (avalanche.kind == Kind::Crosstalk) fNCrosstalk++;
303 if (avalanche.kind == Kind::Afterpulse) fNAfterpulses++;
304
305 if (fCrosstalkProbability > 0 && fRandom.Rndm() < fCrosstalkProbability) {
306 queue.push({avalanche.time, avalanche.position, avalanche.particleID, Kind::Crosstalk});
307 }
308 if (fAfterpulseProbability > 0 && fRandom.Rndm() < fAfterpulseProbability) {
309 queue.push({avalanche.time + AfterpulseDelay(), avalanche.position, avalanche.particleID,
310 Kind::Afterpulse});
311 }
312 }
313 }
314
315 RESTDebug << "TRestDetectorPhotosensorProcess: event " << inputEvent->GetID() << ", " << fNAvalanches
316 << " avalanches (" << fNDarkCounts << " dark counts, " << fNCrosstalk << " crosstalk, "
317 << fNAfterpulses << " afterpulses), " << fNSaturationLosses << " lost by saturation"
318 << RESTendl;
319 return outputEvent->GetNumberOfHits() > 0;
320}
