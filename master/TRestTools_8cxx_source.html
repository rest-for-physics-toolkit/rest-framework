45std::vector<std::string> TRestTools::GetFilesMatchingPattern(const std::string& filePattern) {
46 std::vector<std::string> outputFileNames;
47
48 if (filePattern.empty()) {
49 RESTWarning << "Warning pattern cannot be empty" << RESTendl;
50 return outputFileNames;

52
53 fs::path path(filePattern);
54
55 // Separate path and file pattern
56 fs::path directory = path.parent_path();

62
63 std::regex regex_pattern(TRestTools::PatternToRegex(filenamePattern), std::regex::icase);
64
65 for (const auto& entry : fs::directory_iterator(directory)) {

67 std::string filename = entry.path().filename().string();

69 outputFileNames.push_back(entry.path().string());

107double TRestTools::StringToTimeStamp(const std::string& date) {
108 std::tm time{};
109 const char separator = date.find('/') != std::string::npos ? '/' : '-';
110 const std::string format = std::string("%d") + separator + "%d" + separator + "%d %d:%d:%d";
111 const int fields = std::sscanf(date.c_str(), format.c_str(), &time.tm_year, &time.tm_mon, &time.tm_mday,
112 &time.tm_hour, &time.tm_min, &time.tm_sec);
113 if (fields != 3 && fields != 5 && fields != 6) {
114 throw std::runtime_error("TRestTools::StringToTimeStamp: not a valid date \"" + date +
115 "\" (YYYY/MM/DD or YYYY-MM-DD, optionally HH:MM or HH:MM:SS)");
116 }
117 time.tm_year -= 1900;
118 time.tm_mon--;
119 time.tm_isdst = -1;
120 return static_cast<double>(std::mktime(&time));
121}

171 RESTError << "isValidTRestRun: " << filename << " is zombie or invalid" << RESTendl;
172 return false;

174
175 TObject* analysisTree = file->Get("AnalysisTree");
176 if (!analysisTree) {
177 RESTWarning << "isValidTRestRun: " << filename << ": AnalysisTree not found" << RESTendl;
178 file->Close();

189 metadataDir->cd();
190
191 TList* keysInDir = metadataDir->GetListOfKeys();
192 bool hasRestRunMetadata = false;
193
194 if (keysInDir) {

196 TKey* key = dynamic_cast<TKey*>(keysInDir->At(i));
197 if (!key) continue;
198
199 std::string instanceName = key->GetName();

204 if (!yamlObj) continue;

209 testConfig["class"].as<std::string>() == "TRestRun") {

214 // Not valid YAML metadata: ignored
215 }
216 }
217 }

272void TRestTools::CopyDirectory(TDirectory* source, TDirectory* target) {
273 if (!source || !target) return;
274 for (auto* object : *source->GetListOfKeys()) {
275 auto* key = static_cast<TKey*>(object);
276 const TClass* type = TClass::GetClass(key->GetClassName());
277 if (!type) continue;
278 if (type->InheritsFrom(TDirectory::Class())) {
279 TDirectory* folder = target->GetDirectory(key->GetName());
280 if (!folder) folder = target->mkdir(key->GetName());
281 CopyDirectory(source->GetDirectory(key->GetName()), folder);
282 continue;
283 }
284 if (type->InheritsFrom(TGeoManager::Class())) {
285 // Reading a geometry deletes gGeoManager (TGeoManager::Init) and makes it the current one: read
286 // and deleted with the geometry lock, gGeoManager and gGeoIdentity kept
287 std::lock_guard<std::mutex> lock(gGeometryMutex);
288 TGeoManager* currentGeometry = gGeoManager;
289 TGeoIdentity* currentIdentity = gGeoIdentity;
290 gGeoManager = nullptr;
291 auto* geometry = dynamic_cast<TGeoManager*>(key->ReadObj());
292 gGeoManager = currentGeometry;
293 if (geometry) {
294 target->cd();
295 geometry->Write(key->GetName(), TObject::kOverwrite);
296 }
297 DeleteGeometryLocked(geometry);
298 gGeoIdentity = currentIdentity;
299 continue;
300 }
301 std::unique_ptr<TObject> copy(key->ReadObj());
302 if (copy) {
303 target->cd();
304 copy->Write(key->GetName(), TObject::kOverwrite);
305 }
306 }

378int TRestTools::GetRunNumberAuto() {
379 // Same directory as the history, settings and downloads: $REST_HOME/.rest, or $HOME/.rest
380 std::filesystem::path runFilePath;
381 try {
382 runFilePath = std::filesystem::path(GetRESTUserDirectory()) / "runNumber";
383 } catch (const std::exception& e) {
384 throw std::runtime_error(std::string("TRestTools::GetRunNumberAuto: could not access the REST user "
385 "directory (") +
386 e.what() + ")");
387 }
388
389 int fd = ::open(runFilePath.c_str(), O_RDWR | O_CREAT, 0666);
390 if (fd == -1) {
391 throw std::runtime_error("TRestTools::GetRunNumberAuto: Failed to open or create runNumber file: " +
392 runFilePath.string() + " (" + std::strerror(errno) + ")");
393 }
394
395 // Lock the file exclusively to handle multi-process concurrency safely
396 ::flock(fd, LOCK_EX);
397
398 int currentRunNr = 0;
399 if (std::filesystem::exists(runFilePath) && std::filesystem::file_size(runFilePath) > 0) {
400 std::ifstream in(runFilePath);
401 if (in.is_open()) in >> currentRunNr;
402 }
403
404 // Increment the run number for the next execution
405 int nextRunNr = currentRunNr + 1;
406
407 // Overwrite the file with the new incremented value
408 std::ofstream out(runFilePath, std::ios::trunc);
409 if (out.is_open()) {
410 out << nextRunNr << "\n";
411 out.flush();
412 } else {
413 RESTError << "Failed to update runNumber file: " << runFilePath << RESTendl;
414 ::flock(fd, LOCK_UN);
415 ::close(fd);
416 throw std::runtime_error("TRestTools::GetRunNumberAuto: Failed to update runNumber file: " +
417 runFilePath.string());
418 }
419
420 ::flock(fd, LOCK_UN);
421 ::close(fd);
422
423 // Return the current allocated run number to the process
424 return currentRunNr;
425}

427std::string TRestTools::SearchFileInPath(const std::vector<std::string>& paths, const std::string& filename) {
428 if (fs::exists(filename)) return filename;
429
430 for (const auto& p : paths) {
431 fs::path basePath(p);
432 if (fs::exists(basePath / filename)) return (basePath / filename).string();
433
434 if (fs::is_directory(basePath)) {
435 // Use explicit iterator to access depth() control
436 for (fs::recursive_directory_iterator it(basePath, fs::directory_options::skip_permission_denied),
437 end;
438 it != end; ++it) {
439 // Keeps the 5-level recursion limit using the iterator depth
440 if (it.depth() >= 5) {
441 it.disable_recursion_pending(); // Optimization: stop descending deeper into this branch
442 continue;
443 }
444
445 const auto& entry = *it;
446 if (entry.is_directory() && fs::exists(entry.path() / filename)) {
447 return (entry.path() / filename).string();
448 }
449 }
450 }
451 }
452 return "";
453}

514std::string TRestTools::PatternToRegex(const std::string& pattern) {
515 std::string regex_pattern;
516 for (char c : pattern) {
517 if (c == '*') {
518 regex_pattern += ".*"; // '*' -> Wildcard
519 } else if (std::string_view(".^$|()[]{}+?\\").find(c) != std::string_view::npos) {
520 // Special characters of a regex only: escaping another one (e.g. \_) is an error in libc++
521 regex_pattern += "\\";
522 regex_pattern += c;
523 } else {
524 regex_pattern += c;
525 }
526 }
527 return regex_pattern;

530std::string TRestTools::ReplaceDelimited(const std::string& text, char open, char close,
531 const TextReplacement& replacement) {
532 std::string result;
533 size_t position = 0;
534 while (true) {
535 const size_t first = text.find(open, position);
536 const size_t last = first == std::string::npos ? first : text.find(close, first + 1);
537 if (last == std::string::npos) {
538 result += text.substr(position);
539 break;
540 }
541 result += text.substr(position, first - position);
542 const auto value = replacement(text.substr(first + 1, last - first - 1));
543 result += value ? *value : text.substr(first, last - first + 1);
544 position = last + 1;
545 }
546 return result;
547}

549std::string TRestTools::ReplaceIdentifiers(const std::string& expression, const std::string& identifier,
550 const TextReplacement& replacement) {
551 const std::regex regex(identifier);
552 std::string result;
553 size_t last = 0;
554 for (auto it = std::sregex_iterator(expression.begin(), expression.end(), regex);
555 it != std::sregex_iterator(); ++it) {
556 const size_t position = it->position();
557 const char previous = position > 0 ? expression[position - 1] : ' ';
558 if (std::isdigit(static_cast<unsigned char>(previous)) || previous == '.') continue;
559 const auto value = replacement(it->str());
560 if (!value) continue;
561 result += expression.substr(last, position - last) + *value;
562 last = position + it->length();
563 }
564 return result + expression.substr(last);
565}

625std::string TRestTools::ToTimeStringLong(double seconds) {
626 const auto absSeconds = std::abs(seconds);
627
628 char buffer[32];
629 double valueToFormat = seconds;
630 std::string_view unit;
631
632 if (absSeconds < 60) {
633 unit = " seconds";
634 } else if (absSeconds < 3600) {
635 valueToFormat = seconds / 60.0;
636 unit = " minutes";
637 } else if (absSeconds < 86400) {
638 valueToFormat = seconds / 3600.0;
639 unit = " hours";
640 } else if (absSeconds < 604800) {
641 valueToFormat = seconds / 86400.0;
642 unit = " days";
643 } else {
644 valueToFormat = seconds / 604800.0;
645 unit = " weeks";
646 }
647
648 auto [ptr, ec] =
649 std::to_chars(buffer, buffer + sizeof(buffer), valueToFormat, std::chars_format::fixed, 2);
650
651 if (ec == std::errc{}) {
652 return std::string(buffer, ptr - buffer) + std::string(unit);
653 }
654
655 return "0.00 seconds";
656}

685std::string TRestTools::DownloadFile(const std::string& url) {
686 // The URL goes into a shell command (between single quotes): only the URL characters (RFC 3986) without
687 // ' (nor $, `, \, spaces or control characters) are accepted, with a scheme (http://...) first (not an
688 // option of curl)
689 const size_t scheme = url.find("://");
690 if (scheme == std::string::npos || scheme == 0 || scheme + 3 >= url.size() ||
691 !std::isalpha(static_cast<unsigned char>(url[0])) ||
692 !std::all_of(url.begin(), url.begin() + scheme,
693 [](unsigned char c) { return std::isalnum(c) || c == '+' || c == '-' || c == '.'; })) {
694 throw std::runtime_error("DownloadFile: invalid URL " + url);
695 }
696 static const std::string allowed = "-._~:/?#[]@!&()*+,;=%";
697 for (const char character : url) {
698 if (!std::isalnum(static_cast<unsigned char>(character)) &&
699 allowed.find(character) == std::string::npos) {
700 throw std::runtime_error("DownloadFile: URL with an unsafe character ('" +
701 std::string(1, character) + "'): " + url);
702 }
703 }
704 const std::filesystem::path directory = std::filesystem::path(GetRESTUserDirectory()) / "download";
705 std::string name = url.substr(scheme + 3);
706 for (char& character : name) {
707 if (character == '/' || character == ':' || character == '?' || character == '&') character = '_';
708 }
709 const std::filesystem::path fileName = directory / name;
710 if (std::filesystem::exists(fileName)) return fileName.string();
711
712 std::filesystem::create_directories(directory);
713 // Unique per process and call (threads): pid, thread and counter
714 static std::atomic<unsigned long> downloads{0};
715 const std::filesystem::path partial =
716 fileName.string() + ".part" + std::to_string(getpid()) + "_" +
717 std::to_string(std::hash<std::thread::id>{}(std::this_thread::get_id())) + "_" +
718 std::to_string(downloads++);
719 if (partial.string().find('\'') != std::string::npos) {
720 throw std::runtime_error("DownloadFile: download path with a quote: " + partial.string());
721 }
722 const std::string command = "curl -sSfL -o '" + partial.string() + "' '" + url + "'";
723 std::cout << "Downloading " << url << " to " << fileName.string() << std::endl;
724 if (std::system(command.c_str()) != 0 || !std::filesystem::exists(partial)) {
725 std::filesystem::remove(partial);
726 throw std::runtime_error("DownloadFile: cannot download " + url + " (curl)");
727 }
728 std::filesystem::rename(partial, fileName);
729 return fileName.string();
730}
