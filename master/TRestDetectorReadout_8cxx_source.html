103static bool ParseDecodingString(const std::string& text, std::map<int, int>& decodingMap) {
104 std::stringstream stream(text);
105 std::string line;
106 int linesParsed = 0;
107
108 while (std::getline(stream, line)) {
109 if (line.empty()) continue;
110
111 size_t firstChar = line.find_first_not_of(" ");
112 if (firstChar == std::string::npos || line[firstChar] == '#') continue;
113
114 // As legacy decoding files: first column DAQ channel, second readout (physical) channel. A
115 // negative physical ID marks an unconnected DAQ channel and is skipped
116 std::stringstream ss(line);
117 int daqID = -1;
118 int physicalID = -1;
119 if (ss >> daqID >> physicalID) {
120 if (physicalID >= 0) {
121 const auto [it, inserted] = decodingMap.emplace(physicalID, daqID);
122 if (!inserted) {
123 RESTWarning << "Decoding: physical ID " << physicalID << " repeated (DAQ channels "
124 << it->second << " and " << daqID << "), DAQ channel " << daqID << " kept"
125 << RESTendl;
126 it->second = daqID;
127 }
128 }
129 linesParsed++;
130 } else {
131 RESTWarning
132 << "Decoding: malformed line skipped (two integers, daqChannel physicalID, expected): '"
133 << line << "'" << RESTendl;
134 }
135 }
136
137 return (linesParsed > 0);
138}

210void TRestDetectorReadout::CollectViewItems(const std::set<int>& activeIDs,
211 std::vector<ViewItem>& items) const {
212 const bool usePathNavigation = !fPathToPhysicalIDMap.empty();
213
214 auto add = [&](TGeoNode* n, const TGeoMatrix* m, int physID) {
215 if (!n || !m || !n->GetVolume() || !n->GetVolume()->GetShape()) return;
216 if (n->GetVolume()->IsAssembly()) return;
217
218 TGeoVolume* vol = n->GetVolume();
219 const bool active = activeIDs.count(physID) > 0;
220
221 ViewItem item;
222 item.shape = vol->GetShape();
223 item.medium = vol->GetMedium();
224 item.matrix = std::make_unique<TGeoHMatrix>(*m);
225 item.color = active ? static_cast<Color_t>(kRed) : vol->GetLineColor();
226 item.transparency = active ? 0 : vol->GetTransparency();
227 item.name = n->GetName();
228 item.physicalID = physID;
229 items.push_back(std::move(item));
230 };
231
232 if (usePathNavigation) {
233 for (const auto& [path, physID] : fPathToPhysicalIDMap) {
234 if (!fGeoManager->cd(path.c_str())) continue;
235 add(fGeoManager->GetCurrentNode(), fGeoManager->GetCurrentMatrix(), physID);
236 }
237 } else {
238 TGeoIterator it(fGeoManager->GetTopVolume());
239 while (TGeoNode* node = it()) {
240 add(node, it.GetCurrentMatrix(), static_cast<int>(node->GetUniqueID()));
241 }
242 }

245void TRestDetectorReadout::DrawViewItems(std::vector<ViewItem>& items, TGeoManager*& viewGeo,
246 const std::string& option) {
247 if (items.empty()) return;
248
249 TGeoMedium* topMedium = nullptr;
250 for (const auto& item : items)
251 if (item.medium) {
252 topMedium = item.medium;
253 break;
254 }
255 if (!topMedium) return;
256
257 double half = 0.0;
258 for (const auto& item : items) {
259 auto* bbox = dynamic_cast<TGeoBBox*>(item.shape);
260 if (!bbox) continue;
261 const double* org = bbox->GetOrigin();
262 for (int sx = -1; sx <= 1; sx += 2)
263 for (int sy = -1; sy <= 1; sy += 2)
264 for (int sz = -1; sz <= 1; sz += 2) {
265 double local[3] = {org[0] + sx * bbox->GetDX(), org[1] + sy * bbox->GetDY(),
266 org[2] + sz * bbox->GetDZ()};
267 double global[3];
268 item.matrix->LocalToMaster(local, global);
269 half = std::max<double>(
270 {half, std::abs(global[0]), std::abs(global[1]), std::abs(global[2])});
271 }
272 }
273 half = (half > 0.0) ? 1.1 * half : 1.e5;
274
275 TGeoManager* prevGeo = gGeoManager;
276 if (viewGeo) {
277 if (prevGeo == viewGeo) prevGeo = nullptr;
278 TRestTools::DeleteGeometry(viewGeo);
279 viewGeo = nullptr;
280 }
281
282 gGeoManager = nullptr; // the constructor would delete the existing global geometry
283 viewGeo = new TGeoManager("activeEventView", "Active event view");
284
285 TGeoVolume* top = viewGeo->MakeBox("TOP", topMedium, half, half, half);
286 viewGeo->SetTopVolume(top);
287
288 // Items are grouped in assemblies: a single volume with too many daughters (~1e5) overflows the
289 // int buffer size computed in TGeoVoxelFinder::SortAll, corrupting the heap at CloseGeometry
290 const int kItemsPerGroup = 1000;
291 TGeoVolumeAssembly* group = nullptr;
292 int i = 0;
293 for (auto& item : items) {
294 if (i % kItemsPerGroup == 0) {
295 group = new TGeoVolumeAssembly(Form("VIEW_GROUP_%d", i / kItemsPerGroup));
296 top->AddNode(group, i / kItemsPerGroup);
297 }
298 TGeoMedium* med = item.medium ? item.medium : topMedium;
299 auto* copy = new TGeoVolume(Form("%s_v%d", item.name.c_str(), i), item.shape, med);
300 copy->SetLineColor(item.color);
301 copy->SetVisibility(kTRUE);
302 group->AddNode(copy, i++, item.matrix.release());
303 }
304
305 viewGeo->CloseGeometry();
306 viewGeo->SetVisLevel(2);
307 viewGeo->SetVisOption(0);
308 top->SetVisibility(kFALSE);
309 top->VisibleDaughters(kTRUE);
310 top->Draw(option.c_str());
311
312 gGeoManager = prevGeo;
313}

315ROOT::Math::XYZVector TRestDetectorReadout::GetPositionFromChannel(int daqID) const {
316 if (!fGeoManager) return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
317
318 const int targetPhysicalID = GetPhysicalID(daqID);
319 if (targetPhysicalID < 0) return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
320
321 bool usePathNavigation = !fPathToPhysicalIDMap.empty();
322
323 if (usePathNavigation) {
324 std::string targetPath = "";
325 for (const auto& [path, id] : fPathToPhysicalIDMap) {
326 if (id == targetPhysicalID) {
327 targetPath = path;
328 break;
329 }
330 }
331 if (targetPath.empty()) return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
332
333 if (!fGeoManager->cd(targetPath.c_str())) {
334 return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
335 }
336
337 const TGeoMatrix* currentGlobalMatrix = fGeoManager->GetCurrentMatrix();
338 if (!currentGlobalMatrix) return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
339
340 double localOrigin[] = {0.0, 0.0, 0.0};
341 double globalMasterOrigin[] = {0.0, 0.0, 0.0};
342 currentGlobalMatrix->LocalToMaster(localOrigin, globalMasterOrigin);
343
344 return ROOT::Math::XYZVector(globalMasterOrigin[0], globalMasterOrigin[1], globalMasterOrigin[2]);
345 }
346
347 TGeoIterator it(fGeoManager->GetTopVolume());
348 TGeoNode* node = nullptr;
349 while ((node = it())) {
350 if (static_cast<int>(node->GetUniqueID()) == targetPhysicalID) {
351 const TGeoMatrix* currentGlobalMatrix = it.GetCurrentMatrix();
352 if (!currentGlobalMatrix) continue;
353
354 double localOrigin[] = {0.0, 0.0, 0.0};
355 double globalMasterOrigin[] = {0.0, 0.0, 0.0};
356 currentGlobalMatrix->LocalToMaster(localOrigin, globalMasterOrigin);
357
358 return ROOT::Math::XYZVector(globalMasterOrigin[0], globalMasterOrigin[1], globalMasterOrigin[2]);
359 }
360 }
361 return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
362}

364std::pair<ROOT::Math::XYZVector, ROOT::Math::XYZVector> TRestDetectorReadout::GetChannelsBoundingBox() const {
365 const double nan = std::numeric_limits<double>::quiet_NaN();
366 const double inf = std::numeric_limits<double>::infinity();
367 double minimum[3] = {inf, inf, inf}, maximum[3] = {-inf, -inf, -inf};
368 bool found = false;
369 if (fGeoManager) {
370 // The nodes of the channels: by path (fPathToPhysicalIDMap) or by unique ID, as
371 // GetPositionFromChannel
372 TGeoIterator it(fGeoManager->GetTopVolume());
373 TGeoNode* node = nullptr;
374 while ((node = it())) {
375 int physicalID = -1;
376 if (!fPathToPhysicalIDMap.empty()) {
377 TString path; // TGeoIterator::GetPath needs a TString
378 it.GetPath(path);
379 const auto pathIt = fPathToPhysicalIDMap.find(path.Data());
380 if (pathIt != fPathToPhysicalIDMap.end()) physicalID = pathIt->second;
381 } else {
382 physicalID = static_cast<int>(node->GetUniqueID());
383 }
384 if (physicalID < 0 || fPhysicalToDAQMap.count(physicalID) == 0) continue;
385 const auto* box = dynamic_cast<const TGeoBBox*>(node->GetVolume()->GetShape());
386 const TGeoMatrix* matrix = it.GetCurrentMatrix();
387 if (!box || !matrix) continue;
388 // The corners of its box in the global frame
389 const double* origin = box->GetOrigin();
390 const double half[3] = {box->GetDX(), box->GetDY(), box->GetDZ()};
391 for (int corner = 0; corner < 8; corner++) {
392 double local[3], global[3];
393 for (int axis = 0; axis < 3; axis++)
394 local[axis] = origin[axis] + ((corner >> axis) & 1 ? half[axis] : -half[axis]);
395 matrix->LocalToMaster(local, global);
396 for (int axis = 0; axis < 3; axis++) {
397 minimum[axis] = std::min(minimum[axis], global[axis]);
398 maximum[axis] = std::max(maximum[axis], global[axis]);
399 }
400 }
401 found = true;
402 }
403 }
404 if (!found) return {ROOT::Math::XYZVector(nan, nan, nan), ROOT::Math::XYZVector(nan, nan, nan)};
405 return {ROOT::Math::XYZVector(minimum[0], minimum[1], minimum[2]),
406 ROOT::Math::XYZVector(maximum[0], maximum[1], maximum[2])};
407}

433bool TRestDetectorReadout::GetPixelPosition(const ROOT::Math::XYZVector& position,
434 PixelPosition& pixel) const {
435 pixel = PixelPosition();
436 if (!fGeoManager) return false;
437 TGeoNode* node = fGeoManager->FindNode(position.X(), position.Y(), position.Z());
438 if (!node || node == fGeoManager->GetTopNode()) return false;
439 const auto* box = dynamic_cast<const TGeoBBox*>(node->GetVolume()->GetShape());
440 const TGeoHMatrix* matrix = fGeoManager->GetCurrentMatrix();
441 if (!box || !matrix) return false;
442 const double master[3] = {position.X(), position.Y(), position.Z()};
443 double local[3];
444 matrix->MasterToLocal(master, local);
445 // Every shape is a TGeoBBox: its bounding box, centred at its origin
446 const double* origin = box->GetOrigin();
447 pixel.daqID = CurrentNodeChannel(node);
448 pixel.local = ROOT::Math::XYZVector(local[0] - origin[0], local[1] - origin[1], local[2] - origin[2]);
449 pixel.halfSize = ROOT::Math::XYZVector(box->GetDX(), box->GetDY(), box->GetDZ());
450 return true;
451}

569
570 const std::string resolvedGeometryName = geometryName.empty() ? GetName() : geometryName;

574 if (!readoutDir) return false;

576
577 // With its hash: the datasets compare the readouts of their runs without reading the geometry
578 if (GetGeoManager()) TRestTools::WriteGeometry(readoutDir, GetGeoManager(), kGeometryKey);
579
580 // Export the decoding payload (original logic preserved).

587 rootDecString.Write(decodingName.c_str(), TObject::kOverwrite);
588
589 fOut->cd();
590
591 // The additional decodings (`decodings`)
592 for (const auto& [name, fileName] : fDecodings) {
593 std::map<int, int> decodingMap;
594 if (name == decodingName) {
595 RESTError << "TRestDetectorReadout: the decoding '" << name
596 << "' of `decodings` has the name of the one of decodingFile" << RESTendl;
597 return false;
598 }

601 return false;

608 RESTError << "TRestDetectorReadout: error exporting the metadata: " << e.what() << RESTendl;
