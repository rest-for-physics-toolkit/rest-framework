322 template <typename Class, typename T, std::enable_if_t<std::is_base_of_v<TRestMetadata, T>, int> = 0>
323 void RegisterField(const std::string& yamlKey, std::vector<T> Class::* memberPtr) {
324 auto readFunc = [this, yamlKey, memberPtr](TRestMetadata* base, const YAML::Node& node) {
325 auto* object = dynamic_cast<Class*>(base);
326 if (!object || !node || !node[yamlKey]) return;
327 const YAML::Node source = node[yamlKey];
328 auto& values = object->*memberPtr;
329 values.clear();
330 auto add = [&](const YAML::Node& child) {
331 if (!child.IsMap()) throw std::runtime_error("each element must be a map");
332 values.emplace_back(yamlKey, YAML::Node(child));
333 CheckFields(&values.back(), child);
334 };
335 if (source.IsSequence()) {
336 for (const auto& child : source) add(child);
337 } else {
338 add(source);
339 }
340 };
341
342 auto writeFunc = [yamlKey, memberPtr](TRestMetadata* base, YAML::Node& n) {
343 auto* object = dynamic_cast<Class*>(base);
344 if (!object) return;
345 // Each element is written in its own node (the one it keeps): assigning n[yamlKey] redirects
346 // the node already there, so a single map element is not replaced by a sequence
347 auto& values = object->*memberPtr;
348 YAML::Node existing = n[yamlKey];
349 YAML::Node sequence(YAML::NodeType::Sequence);
350 for (auto& value : values) {
351 YAML::Node child = value.GetYAMLNode();
352 if (!child) child = YAML::Node(YAML::NodeType::Map);
353 value.template UpdateYAMLFromParams<T>(child);
354 if (values.size() == 1 && child.is(existing)) return;
355 sequence.push_back(child);
356 }
357 n[yamlKey] = sequence;
358 };
359
360 Add<Class>(yamlKey, readFunc, writeFunc);
361 }

365 template <typename Class, typename MemberClass, typename T>
366 void RegisterField(const std::string& yamlKey, T MemberClass::* memberPtr) {
367 auto readFunc = [yamlKey, memberPtr](TRestMetadata* base, const YAML::Node& n) {
368 auto* object = dynamic_cast<MemberClass*>(base);
369 if (object && n && n[yamlKey]) object->*memberPtr = n[yamlKey].as<T>();
370 };
371
372 auto writeFunc = [yamlKey, memberPtr](TRestMetadata* base, YAML::Node& n) {
373 auto* object = dynamic_cast<MemberClass*>(base);
374 if (!object) return;
375 if constexpr (std::is_same_v<T, YAML::Node>) {
376 if (!(object->*memberPtr) || (object->*memberPtr).IsNull()) return; // not given
377 }
378 n[yamlKey] = object->*memberPtr;
379 };
