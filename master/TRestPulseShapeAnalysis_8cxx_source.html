28std::pair<double, double> FitAroundMaximum(TGraph& signal, const char* label, double amplitudeScale,
29 const TF1Maker& make) {
30 const int n = signal.GetN();
31 if (n == 0) return {-1, -1};
32
33 const int maxBin = TMath::LocMax(n, signal.GetY());
34 const double maxTime = signal.GetPointX(maxBin);
35 const double maxValue = signal.GetPointY(maxBin);
36
37 int left = maxBin, right = maxBin;
38 while (left > 0 && signal.GetPointY(left - 1) >= maxValue / 2) left--;
39 while (right < n - 1 && signal.GetPointY(right + 1) >= maxValue / 2) right++;
40 left = std::max(0, left - 1);
41 right = std::min(n - 1, right + 1);
42 const double lower = signal.GetPointX(left), upper = signal.GetPointX(right);
43 if (right - left + 1 < 4 || !(upper > lower)) return {maxTime, maxValue};
44
45 const double width = (upper - lower) / 2.355; // FWHM (plus a point on each side) to sigma
46 std::unique_ptr<TF1> f = make(lower, upper);
47 f->SetParameters(maxValue * amplitudeScale, maxTime, width);
48 f->SetParLimits(1, lower, upper);
49 f->SetParLimits(2, width / 20, width * 20);
50 // B: the initial values and limits are used also with the predefined functions (gaus, landau)
51 TFitResultPtr fitResult = signal.Fit(f.get(), "QNRSB");
52
53 if (!fitResult.Get() || !fitResult->IsValid()) {
54 std::cout << "WARNING: bad " << label << " fit with maximum at time = " << maxTime
55 << ". Failed parameters:";
56 for (int i = 0; i < f->GetNpar(); i++) std::cout << " " << f->GetParameter(i);
57 std::cout << ". The maximum point is used" << std::endl;
58 return {maxTime, maxValue};
59 }
60 return {f->GetParameter(1), f->GetParameter(0)};
61}

76template <typename T>
77bool BaselineSigmaSD(const std::vector<T>& signal, int startBin, int endBin, double& baseLine,
78 double& baseLineSigma) {
79 baseLine = 0;
80 baseLineSigma = 0;
81
82 double sum = 0, sum2 = 0;
83 int nPoints = 0;
84 for (int i = startBin; i < endBin; i++) {
85 if (i < 0 || i >= (int)signal.size()) continue;
86 const double v = signal[i];
87 sum += v;
88 sum2 += v * v;
89 nPoints++;
90 }
91
92 if (nPoints > 0) {
93 baseLine = sum / nPoints;
94 baseLineSigma = std::sqrt(std::max(0.0, sum2 / nPoints - baseLine * baseLine));
95 }
96 return true;
97}

106template <typename T>
107bool BaselineSigmaIQR(const std::vector<T>& signal, int startBin, int endBin, double& baseLine,
108 double& baseLineSigma) {
109 baseLine = 0;
110 baseLineSigma = 0;
111
112 // [startBin, endBin) clamped to the signal, as the default option; an empty range gives 0 and 0
113 if (startBin < 0) startBin = 0;
114 if (endBin > (int)signal.size()) endBin = signal.size();
115 if (endBin <= startBin) return true;
116
117 std::vector<T> v(signal.begin() + startBin, signal.begin() + endBin);
118
119 baseLine = TMath::Median((int)v.size(), v.data());
120
121 std::sort(v.begin(), v.end());
122 const size_t q1 = static_cast<size_t>(0.25 * (v.size() - 1));
123 const size_t q3 = static_cast<size_t>(0.75 * (v.size() - 1));
124 baseLineSigma = (v[q3] - v[q1]) / 1.349; // IQR/1.349 = sigma for gaussian data
125 return true;
126}

128template <typename T>
129bool BaselineSigmaExcludeOutliers(const std::vector<T>& signal, int startBin, int endBin, double& baseLine,
130 double& baseLineSigma) {
131 baseLine = 0;
132 baseLineSigma = 0;
133
134 // [startBin, endBin) clamped to the signal, as the default option; an empty range gives 0 and 0
135 if (startBin < 0) startBin = 0;
136 if (endBin > (int)signal.size()) endBin = signal.size();
137 if (endBin <= startBin) return true;
138
139 std::vector<double> v;
140 for (int i = startBin; i < endBin; ++i) v.emplace_back(signal[i]);
141
142 std::sort(v.begin(), v.end());
143 const size_t q1 = static_cast<size_t>(0.25 * (v.size() - 1));
144 const size_t q3 = static_cast<size_t>(0.75 * (v.size() - 1));
145
146 std::vector<double> filteredData;
147 for (const auto& value : v) {
148 if (value >= v[q1] && value <= v[q3]) filteredData.emplace_back(value);
149 }
150
151 if (filteredData.empty()) return false;
152
153 baseLine = TMath::Median(filteredData.size(), filteredData.data());
154
155 double mean = std::accumulate(filteredData.begin(), filteredData.end(), 0.0) / filteredData.size();
156 double variance = 0.0;
157 for (const auto& value : filteredData) variance += std::pow(value - mean, 2);
158 baseLineSigma = std::sqrt(variance / filteredData.size());
159
160 return true;
161}

265template <typename T>
266std::vector<double> TRestPulseShapeAnalysis::GetMovingAverage(const std::vector<T>& values, int halfWindow) {
267 const int nPoints = values.size();
268 halfWindow = std::max(0, halfWindow);
269
270 // Prefix sums, so that the averages do not accumulate rounding errors
271 std::vector<double> sum(nPoints + 1, 0);
272 for (int i = 0; i < nPoints; i++) sum[i + 1] = sum[i] + values[i];
273
274 std::vector<double> average(nPoints, 0);
275 for (int i = 0; i < nPoints; i++) {
276 const int startBin = std::max(0, i - halfWindow);
277 const int endBin = std::min(nPoints - 1, i + halfWindow);
278 average[i] = (sum[endBin + 1] - sum[startBin]) / (endBin - startBin + 1);
279 }
280
281 return average;
282}

298std::vector<float> TRestPulseShapeAnalysis::GetSignalSmoothed(const std::vector<T>& signal,
299 int averagingPoints) {
300 const int pulseDepth = signal.size();
301 if (pulseDepth == 0) return {};
302
303 averagingPoints = std::max(1, averagingPoints);
304 if (averagingPoints > pulseDepth) averagingPoints = pulseDepth;
305 averagingPoints = (averagingPoints / 2) * 2 + 1; // make it odd >= averagingPoints
306 if (averagingPoints > pulseDepth) averagingPoints -= 1;
307 if (averagingPoints <= 0) averagingPoints = 1;
308
309 const int halfWindow = averagingPoints / 2;
310 const auto average = GetMovingAverage(signal, halfWindow);
311
312 // The points closer than halfWindow to the edges take the average of the closest full window
313 const int first = std::min(halfWindow, pulseDepth - 1);
314 const int last = std::max(first, pulseDepth - halfWindow - 1);
315 std::vector<float> smoothed(pulseDepth);
316 for (int i = 0; i < pulseDepth; i++) smoothed[i] = average[std::clamp(i, first, last)];
317
318 return smoothed;
319}

409template <typename T>
410std::vector<std::pair<int, float>> TRestPulseShapeAnalysis::GetPointsOverThreshold(
411 const std::vector<T>& signal, std::pair<int, int>& range, const std::pair<double, double>& thrPar,
412 int nPointsOver, int nPointsFlat, double baseLineSigma) {
413 if (signal.empty()) return {};
414
415 if (range.first < 0) range.first = 0;
416 if (range.second <= 0 || range.second > (int)signal.size()) range.second = signal.size();
417
418 const int start = std::max(0, (int)range.first);
419 const int stop = std::min((int)signal.size(), (int)range.second);
420 if (start >= stop) return {};
421
422 std::vector<std::pair<int, float>> pointsOverThreshold;
423
424 double pointTh = thrPar.first;
425 double signalTh = thrPar.second;
426
427 double threshold = pointTh * baseLineSigma;
428
429 for (int i = start; i < stop; i++) {
430 // Filling a pulse with consecutive points that are over threshold
431 if (signal[i] > threshold) {
432 int pos = i;
433 std::vector<double> pulse;
434 pulse.push_back(signal[i]);
435 i++;
436
437 // If the pulse ends in a flat end above the threshold, the parameter
438 // nPointsFlat will serve to artificially end the pulse.
439 // If nPointsFlat is big enough, this parameter will not affect the
440 // decision to cut this anomalous behaviour. And all points over threshold
441 // will be added to the pulse vector.
442 int flatN = 0;
443 while (i < stop && signal[i] > threshold) {
444 if (TMath::Abs(signal[i] - signal[i - 1]) > threshold) {
445 flatN = 0;
446 } else {
447 flatN++;
448 }
449
450 if (flatN < nPointsFlat) {
451 pulse.push_back(signal[i]);
452 i++;
453 } else {
454 break;
455 }
456 }
457
458 if (pulse.size() >= (unsigned int)nPointsOver) {
459 double mean = std::accumulate(pulse.begin(), pulse.end(), 0.0) / pulse.size();
460 double sq_sum = std::inner_product(pulse.begin(), pulse.end(), pulse.begin(), 0.0);
461 double stdev = std::sqrt(sq_sum / pulse.size() - mean * mean);
462
463 if (stdev > signalTh * baseLineSigma)
464 for (unsigned int j = 0; j < pulse.size(); j++)
465 pointsOverThreshold.push_back(std::make_pair(pos + j, pulse[j]));
466 }
467 }
468 }
469
470 return pointsOverThreshold;
471}

573std::vector<std::pair<double, double>> TRestPulseShapeAnalysis::GetPeaks(
574 const std::vector<std::pair<double, double>>& points, double threshold, double distance, int halfWindow,
575 int maxGreaterEqual, bool refineMaximum, int amplitudeHalfWindow) {
576 std::vector<std::pair<double, double>> peaks;
577 const int nPoints = points.size();
578 if (nPoints == 0) return peaks;
579
580 halfWindow = std::max(1, halfWindow);
581 amplitudeHalfWindow = std::max(0, amplitudeHalfWindow);
582
583 std::vector<double> values(nPoints);
584 for (int i = 0; i < nPoints; i++) values[i] = points[i].second;
585 const auto smoothed = GetMovingAverage(values, halfWindow);
586
587 int lastPeak = -1;
588 for (int i = halfWindow; i < nPoints - halfWindow; i++) {
589 if (smoothed[i] <= threshold) continue;
590 if (lastPeak >= 0 && points[i].first - points[lastPeak].first < distance) continue;
591
592 int nGreaterEqual = 0;
593 for (int j = i - halfWindow; j <= i + halfWindow && nGreaterEqual <= maxGreaterEqual; j++) {
594 if (j != i && smoothed[i] <= smoothed[j]) nGreaterEqual++;
595 }
596 if (nGreaterEqual > maxGreaterEqual) continue;
597
598 // On flat tops the candidate is at the left of the actual maximum
599 int maxBin = i;
600 for (int j = i + 1; refineMaximum && j < nPoints && points[j].first - points[i].first <= distance;
601 j++) {
602 if (smoothed[j] > smoothed[maxBin]) maxBin = j;
603 }
604
605 const int startBin = std::max(0, maxBin - amplitudeHalfWindow);
606 const int endBin = std::min(nPoints - 1, maxBin + amplitudeHalfWindow);
607 double amplitude = 0;
608 for (int j = startBin; j <= endBin; j++) amplitude += points[j].second;
609 amplitude /= endBin - startBin + 1;
610
611 peaks.emplace_back(points[maxBin].first, amplitude);
612 lastPeak = maxBin;
613 }
614
615 return peaks;
616}

650std::unique_ptr<TF1> TRestPulseShapeAnalysis::FitAget(TGraph& signal, bool withSine,
651 std::optional<double> fixedBaseline, bool verbose) {
652 const int n = signal.GetN();
653 if (n < 4) return nullptr;
654
655 const double peakU = withSine ? kAgetPeakU : 1.0;
656 const double baseline0 = fixedBaseline.value_or(0.0);
657 const int maxBin = TMath::LocMax(n, signal.GetY());
658 const double maxTime = signal.GetPointX(maxBin);
659 const double maxAbove = std::max(1.0, signal.GetPointY(maxBin) - baseline0);
660 const double xMin = signal.GetPointX(0), xMax = signal.GetPointX(n - 1);
661
662 // The half maximum of the rise is ~0.4 widths before the maximum
663 double width = (xMax - xMin) * 0.2;
664 for (int i = 0; i < maxBin; i++) {
665 if (signal.GetPointY(i) - baseline0 > maxAbove / 2) {
666 if (maxTime > signal.GetPointX(i)) width = (maxTime - signal.GetPointX(i)) / 0.4;
667 break;
668 }
669 }
670 width = std::max(width, 1e-3);
671
672 auto f = std::make_unique<TF1>(
673 "agetFit",
674 [withSine, peakU](const double* x, const double* p) {
675 const double u = (x[0] - p[2]) / p[3] + peakU;
676 if (u <= 0) return p[0];
677 return p[0] + p[1] * AgetShape(u, withSine) / AgetShape(peakU, withSine);
678 },
679 xMin, xMax, 4);
680 f->SetParNames("Baseline", "Amplitude", "Time", "Width");
681 if (fixedBaseline) {

703std::pair<double, double> TRestPulseShapeAnalysis::GetMaxAget(TGraph& signal) {
704 if (signal.GetN() == 0) return {-1, -1};
705 const double maxTime = signal.GetPointX(TMath::LocMax(signal.GetN(), signal.GetY()));

709 const double x = signal.GetPointX(i);

714 if (!fit) {

728std::vector<std::pair<double, double>> TRestPulseShapeAnalysis::GetIntWindow(TGraph& signal,
729 double intWindow) {
730 if (!(intWindow > 0) || !std::isfinite(intWindow)) {
731 throw std::invalid_argument("GetIntWindow: the integration window must be positive (" +
732 std::to_string(intWindow) + ")");
733 }
734 const int nPoints = signal.GetN();
735
736 std::map<int, std::pair<int, double>> windowMap;
737 for (int j = 0; j < nPoints; j++) {
738 const int index = static_cast<int>(std::floor(signal.GetPointX(j) / intWindow));
739 auto it = windowMap.find(index);
740 if (it != windowMap.end()) {
741 it->second.first++;
742 it->second.second += signal.GetPointY(j);
743 } else {
744 windowMap[index] = std::make_pair(1, signal.GetPointY(j));
745 }
746 }
747
748 std::vector<std::pair<double, double>> result;
749
750 for (const auto& [index, pair] : windowMap) {
751 double hitTime = index * intWindow + intWindow / 2.;
752 double energy = pair.second / pair.first;
753 result.push_back(std::make_pair(hitTime, energy));
754 }
755
756 return result;
757}

825 if (signal.GetN() == 0) return {-1, -1};

827 double eAvg = 0;
828 double hitTimeAvg = 0;
829 for (const auto& [hitTime, energy] : tripleMax) {
830 hitTimeAvg += hitTime * energy;
831 eAvg += energy;
832 }
833

950template <typename T>
951void TRestPulseShapeAnalysis::GetSinNoise(std::vector<T>& noise, double maxAmplitude, double period,
952 double phase, TRandom* random) {
953 if (!random) random = gRandom;
954 const size_t size = noise.size();
955 if (size == 0 || period <= 0 || maxAmplitude <= 0.0) {
956 if (size > 0) std::fill(noise.begin(), noise.end(), static_cast<T>(0));
957 return;
958 }
959
960 double uBase = random->Uniform(0.0, 1.0);
961 double sinBase = std::sin(uBase * TMath::Pi() / 2.0);
962 double baseAmplitude = maxAmplitude * (sinBase * sinBase);
963
964 double slowFreq = period * random->Uniform(4.0, 8.0);
965 double slowPhase = random->Uniform(0.0, 2.0 * TMath::Pi());
966
967 for (size_t i = 0; i < size; ++i) {
968 double angle = (2.0 * TMath::Pi() * i / period) + phase;
969
970 double slowAngle = (2.0 * TMath::Pi() * i / slowFreq) + slowPhase;
971 double dynamicAmp = baseAmplitude * std::abs(std::sin(slowAngle));
972
973 if (dynamicAmp > maxAmplitude) dynamicAmp = maxAmplitude;
974
975 noise[i] = static_cast<T>(dynamicAmp * std::sin(angle));
976 }
977}

1002std::vector<double> TRestPulseShapeAnalysis::BuildResponse(const std::string& type, double shapingTime,
1003 double gain, int& center) {
1004 center = 0;
1005 if (!(shapingTime > 0)) return {};
1006 if (type != "gaus" && type != "exponential" && type != "shaper" && type != "shaperSin") return {};
1007
1008 std::vector<double> r;
1009 if (type == "gaus") {
1010 const int cBin = (int)(shapingTime * 3.5);
1011 r.resize(2 * cBin);
1012 center = cBin;
1013 for (int i = 0; i < 2 * cBin; ++i) r[i] = ShapingFunction(type, (i - cBin) / shapingTime);
1014 } else {
1015 r.resize((int)(5 * shapingTime));
1016 for (size_t i = 0; i < r.size(); ++i) r[i] = ShapingFunction(type, (double)i / shapingTime);
1017 }
1018
1019 const double sum = std::accumulate(r.begin(), r.end(), 0.0);
1020 if (!(sum > 0)) return {};
1021 for (double& v : r) v *= gain / sum;
1022 return r;
1023}

1025std::vector<double> TRestPulseShapeAnalysis::Convolve(const std::vector<double>& in,
1026 const std::vector<double>& response, int center,
1027 bool ignoreNegative) {
1028 const int nBins = (int)in.size();
1029 const int nr = (int)response.size();
1030 std::vector<double> out(nBins, 0.0);
1031
1032 for (int m = 0; m < nBins; ++m) {
1033 const double v = in[m];
1034 if (v == 0 || (ignoreNegative && v < 0)) continue;
1035
1036 const int kMin = std::max(0, center - m);
1037 const int kMax = std::min(nr, nBins - m + center);
1038 for (int k = kMin; k < kMax; ++k) out[m + k - center] += response[k] * v;
1039 }
1040 return out;
1041}
