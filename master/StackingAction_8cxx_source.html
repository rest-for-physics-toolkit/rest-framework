28G4ClassificationOfNewTrack StackingAction::ClassifyNewTrack(const G4Track* track) {
29 const bool isFullChain = fSimulationManager->GetRestMetadata()->isFullChainActivated();
30 const G4ClassificationOfNewTrack decayClassification = isFullChain ? fWaiting : fKill;
31 // Volume biasing: a secondary created after the crossing point of a biasing volume, replayed by the
32 // next stage
33 if (SimulationManager::GetOutputManager()->TakeBiasingKilledSecondary(track)) {
34 return fKill;
35 }
36 if (track->GetParentID() <= 0) {
37 // always process the first track regardless
38 return fUrgent;
39 }
40
41 auto particle = track->GetParticleDefinition();
42 if (fParticlesToIgnore.find(particle) != fParticlesToIgnore.end()) {
43 // ignore this track
44 return fKill;
45 }
46
47 if (isFullChain) {
48 const string particleName = particle->GetParticleName();
49 // the particle may be an excited state of an isotope we want to remove
50 // strip the excited state which is assumed to always be inside some "[]"
51 regex pattern("\\[.*\\]");
52 std::string particleNameStripped = regex_replace(particleName, pattern, "");
53 if (fSimulationManager->GetRestMetadata()->IsIsotopeFullChainStop(particleNameStripped)) {
54 return fKill;
55 }
56 }
57
58 if (track->GetCreatorProcess()->GetProcessType() != G4ProcessType::fDecay) {
59 return fUrgent;
60 }
61
62 if (particle->GetParticleType() == "nucleus" && !particle->GetPDGStable()) {
63 // unstable nucleus
64 if (particle->GetPDGLifeTime() > fMaxAllowedLifetime) {
65 G4String energy = G4BestUnit(track->GetKineticEnergy(), "Energy");
66 G4String lifeTime = G4BestUnit(particle->GetPDGLifeTime(), "Time");
67 return decayClassification;
68 }
69 }
70
71 return fUrgent;
72}
