81 fStageOutputs.clear();

85 for (const auto& element : fNode) {

96 std::unique_ptr<TRestEventProcess> processPtr(proc);

114 fPipelineConnections.emplace_back(inputEvent, outputEvent);
115 fPipelineOutputClasses.emplace_back(outputEvent);

124void TRestEventProcessManager::Run() {
125 if (fRunInfo == nullptr) {
126 throw std::runtime_error("TRestEventProcessManager::Run - run context is null");
127 }
128
129 if (fProcessChain.empty()) {
130 return;
131 }
132
133 // Without inputAnalysisStorage the output analysis tree does not copy the observables of the input
134 fRunInfo->SetInputAnalysisStorage(fInputAnalysisStorage);
135 if (!fRunInfo->HasOutputFileOpen()) {
136 fRunInfo->OpenOutputFile();
137 }
138
139 for (auto& proc : fProcessChain) {
140 proc->SetRunInfo(fRunInfo);
141 proc->SetManager(fManager);
142 TRestLogScope logScope(proc->GetLogContext());
143 proc->Initialize();
144 // After the initialization: the values taken from other metadata (readout, medium, drift velocity...)
145 proc->PrintMetadata();
146 }
147 // Once all the observables are registered
148 for (auto& proc : fProcessChain) proc->InitCuts();
149
150 fStageInputs.assign(fProcessChain.size(), nullptr);
151 fStageOutputs.clear();
152 fStageOutputs.resize(fProcessChain.size());
153 // Output of the latest stage of each event name, and the names in order of first appearance
154 std::map<std::string, TRestEvent*> latestOutputs;
155 std::vector<std::string> outputNames;
156
157 for (size_t i = 0; i < fProcessChain.size(); ++i) {
158 const std::string& inputName = fPipelineConnections[i].first;
159 const std::string& outputName = fPipelineConnections[i].second;
160 const std::string procClassName = fProcessChain[i]->GetClassName();
161
162 // A previous stage output takes precedence over the input file, otherwise chained processes
163 // on the same event type would all read the unmodified file event
164 if (inputName != "None" && !inputName.empty()) {
165 if (latestOutputs.find(inputName) != latestOutputs.end()) {
166 fStageInputs[i] = latestOutputs[inputName];
167 } else if (fRunInfo->HasEvent(inputName)) {
168 fStageInputs[i] = fRunInfo->GetEvent(inputName);
169 } else {
170 throw std::runtime_error("TRestEventProcessManager::Run - input event '" + inputName +
171 "' of process '" + procClassName +
172 "' is neither in the input file nor produced by a previous process");
173 }
174 }
175
176 // Processes that change the event type declare it in GetOutputEvent
177 std::string outputClassName = fPipelineOutputClasses[i];
178 if (outputClassName.empty()) {
179 if (!fStageInputs[i]) {
180 throw std::runtime_error(
181 "TRestEventProcessManager::Run - outputClass must be specified for generator process '" +
182 procClassName + "'");
183 }
184 outputClassName = fStageInputs[i]->GetClassName();
185 }
186 fStageOutputs[i] = EventRegistry::Instance().Create(outputClassName, outputName);
187 if (latestOutputs.find(outputName) == latestOutputs.end()) outputNames.push_back(outputName);
188 latestOutputs[outputName] = fStageOutputs[i].get();
189 }
190
191 // One tree per event name, filled with the output of its last stage. Registered once per name in
192 // order of first appearance (the first one binds the event info of the analysis tree). Without
193 // outputEventStorage only the final event (output of the last process) is stored
194 std::set<std::string> storedNames;
195 if (fOutputEventStorage) {
196 for (const auto& outputName : outputNames) {
197 fRunInfo->RegisterEvent<TRestEvent>(outputName, *latestOutputs[outputName]);
198 storedNames.insert(outputName);
199 }
200 } else if (!fStageOutputs.empty()) {
201 fRunInfo->RegisterEvent<TRestEvent>(fPipelineConnections.back().second, *fStageOutputs.back());
202 storedNames.insert(fPipelineConnections.back().second);
203 }
204 // With inputEventStorage the events of the input file are stored as read, unless a stored process
205 // output has the same name (it takes precedence, a single tree per name)
206 if (fInputEventStorage) {
207 for (const auto& [inputName, inputEvent] : fRunInfo->GetEventMap()) {
208 if (!inputEvent || storedNames.count(inputName) > 0) continue;
209 fRunInfo->RegisterEvent<TRestEvent>(inputName, *inputEvent);
210 }
211 }
212
213 long long totalEntries = fRunInfo->GetEntries();
214 if (totalEntries <= 0) {
215 for (const auto& process : fProcessChain) {
216 const long long processEntries = process->GetInputEventCount();
217 if (processEntries >= 0) {
218 totalEntries = processEntries;
219 break;
220 }
221 }
222 }
223
224 // A process reading an external file whose number of events is unknown (e.g. a DAQ binary file): the
225 // loop goes on until it has no more events (InputFinished), the progress is the fraction of file read
226 TRestEventProcess* fileSource = nullptr;
227 if (totalEntries <= 0) {
228 for (const auto& process : fProcessChain) {
229 if (process->GetInputFraction() >= 0) {
230 fileSource = process.get();
231 break;
232 }
233 }
234 }
235 const bool readUntilEnd = fileSource != nullptr;
236
237 long long loopStart = 0;
238 long long loopEnd = readUntilEnd ? std::numeric_limits<long long>::max() : totalEntries;
239
240 if (fEntryStart != -1) loopStart = fEntryStart;
241 if (fEntryEnd != -1) loopEnd = fEntryEnd;
242
243 if (fEntryEnd == -1 && fEventsToProcess > 0) loopEnd = loopStart + fEventsToProcess;
244 // Neither eventsToProcess nor an explicit entry range go beyond the entries of the input
245 if (totalEntries > 0) loopEnd = std::min(loopEnd, totalEntries);
246
247 if (readUntilEnd && loopStart > 0) {
248 throw std::runtime_error("TRestEventProcessManager::Run - process '" + fileSource->GetName() +
249 "' reads its file sequentially, it cannot start at entry " +
250 std::to_string(loopStart));
251 }
252
253 const bool showProgress = (fEntryStart <= 0);
254 // Progress by the fraction of file read if the number of events is not known
255 const bool progressByFile = readUntilEnd && loopEnd == std::numeric_limits<long long>::max();
256
257 if (progressByFile) {
258 RESTInfo << "TRestEventProcessManager: Starting event loop until the end of the input of '"
259 << fileSource->GetName() << "'" << RESTendl;
260 } else {
261 RESTInfo << "TRestEventProcessManager: Starting event loop. Entries [" << loopStart << ", " << loopEnd
262 << ")" << RESTendl;
263 }
264
265 if (showProgress) {
266 if (progressByFile)
267 RESTProgress.Reset(100, 30, "% of the file");
268 else
269 RESTProgress.Reset(loopEnd - loopStart);
270 }
271
272 long long acceptedCount = 0;
273 long long rejectedCount = 0;
274 fProcessTimes.assign(fProcessChain.size(), 0);
275 fProcessCalls.assign(fProcessChain.size(), 0);
276 fReadTime = 0;
277 fWriteTime = 0;
278 fReadCalls = 0;
279 fWriteCalls = 0;
280 using Clock = std::chrono::steady_clock;
281
282 const bool hasInputTree = (totalEntries > 0);
283
284 for (long long entry = loopStart; entry < loopEnd; ++entry) {
285 if (TRestManager::StopRequested()) {
286 RESTWarning << "TRestEventProcessManager: stop requested, breaking event loop at entry " << entry
287 << RESTendl;
288 break;
289 }
290
291 // No more entries in the input event trees
292 const auto readStart = Clock::now();
293 const bool entryRead = !hasInputTree || fRunInfo->GetEntry(entry);
294 fReadTime += SecondsSince(readStart);
295 fReadCalls++;
296 if (hasInputTree && !entryRead && !fRunInfo->fInputEventTrees.empty()) {
297 RESTWarning << "TRestEventProcessManager: entry " << entry
298 << " could not be read from the input file, breaking event loop" << RESTendl;
299 break;
300 }
301
302 bool eventAccepted = true;
303 bool inputFinished = false;
304 for (size_t i = 0; i < fProcessChain.size(); ++i) {
305 TRestEvent& outputEvent = *fStageOutputs[i];
306 const TRestEvent& inputEvent = fStageInputs[i] ? *fStageInputs[i] : outputEvent;
307
308 // The messages of the events and tools used by the process take its verbose
309 TRestLogScope logScope(fProcessChain[i]->GetLogContext());
310 const auto processStart = Clock::now();
311 const bool processed = fProcessChain[i]->ProcessEvent(inputEvent, outputEvent);
312 fProcessTimes[i] += SecondsSince(processStart);
313 fProcessCalls[i]++;
314 if (!processed) {
315 // End of the external file of the process: not a rejected event
316 inputFinished = fProcessChain[i]->InputFinished();
317 eventAccepted = false;
318 break;
319 }
320 if (!fProcessChain[i]->PassCuts()) {
321 eventAccepted = false;
322 break;
323 }
324 // Events created by a generator (e.g. from DAQ files) take the run and subrun of this run,
325 // the others keep the ones of their input
326 if (!fStageInputs[i]) outputEvent.InitializeReferences(fRunInfo);
327 }
328
329 if (inputFinished) {
330 if (showProgress && progressByFile) RESTProgress.Update(100);
331 break;
332 }
333
334 if (eventAccepted) {
335 acceptedCount++;
336 const auto writeStart = Clock::now();
337 fRunInfo->Fill();
338 fWriteTime += SecondsSince(writeStart);
339 fWriteCalls++;
340 } else {
341 rejectedCount++;
342 }
343
344 if (showProgress && ((entry - loopStart) % 10 == 0 || entry == loopEnd - 1)) {
345 if (progressByFile)
346 RESTProgress.Update(static_cast<int>(100 * fileSource->GetInputFraction()));
347 else
348 RESTProgress.Update(entry - loopStart + 1);
349 }
350 }
351
352 std::cout << "\n";
353
354 for (auto& proc : fProcessChain) {
355 TRestLogScope logScope(proc->GetLogContext());
356 proc->EndProcess();
357 }
358
359 RESTInfo << "TRestEventProcessManager: Pipeline finished. Accepted: " << acceptedCount
360 << " events, Rejected: " << rejectedCount << " events" << RESTendl;
361 PrintProcessTimes();
362 RESTInfo << "Output file name: " << fRunInfo->GetOutputFileName() << RESTendl;

365void TRestEventProcessManager::PrintProcessTimes() const {
366 if (fProcessTimes.empty()) return;
367 double total = fReadTime + fWriteTime;
368 for (const double time : fProcessTimes) total += time;
369 if (total <= 0) return;
370
371 // name, total (s), per call (ms), fraction of the loop (%)
372 std::vector<std::string> names;
373 std::vector<double> times;
374 std::vector<long long> calls;
375 for (size_t i = 0; i < fProcessChain.size(); i++) {
376 names.push_back(fProcessChain[i]->GetName() + " (" + fProcessChain[i]->GetClassName() + ")");
377 times.push_back(fProcessTimes[i]);
378 calls.push_back(fProcessCalls[i]);
379 }
380 names.push_back("reading the input");
381 times.push_back(fReadTime);
382 calls.push_back(fReadCalls);
383 names.push_back("writing the output (accepted events)");
384 times.push_back(fWriteTime);
385 calls.push_back(fWriteCalls);
386
387 size_t width = 0;
388 for (const auto& name : names) width = std::max(width, name.size());
389 RESTInfo << "TRestEventProcessManager: time of the event loop, " << total << " s:" << RESTendl;
390 char line[512];
391 for (size_t i = 0; i < names.size(); i++) {
392 const double perCall = calls[i] > 0 ? 1e3 * times[i] / calls[i] : 0;
393 std::snprintf(line, sizeof(line), " %-*s %10.3f s %10.3f ms/event %6.1f %%", static_cast<int>(width),
394 names[i].c_str(), times[i], perCall, 100 * times[i] / total);
395 RESTInfo << line << RESTendl;
396 }
