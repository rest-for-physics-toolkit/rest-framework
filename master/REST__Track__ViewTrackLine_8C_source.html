106// Branches of the line observables (prefix_observable) with the prefix `prefix`, or if empty of the first
107// prefix with all of them; empty if they are not in the tree
108std::vector<std::string> GetLineBranches(TTree* tree, const std::string& prefix,
109 const char* const observables[6]) {
110 if (!prefix.empty()) {
111 std::vector<std::string> names;
112 for (int i = 0; i < 6; i++) {
113 const char* observable = observables[i];
114 const std::string name = prefix + "_" + observable;
115 if (!tree->GetBranch(name.c_str())) return {};
116 names.push_back(name);
117 }
118 return names;
119 }
120 const std::string suffix = std::string("_") + observables[0];
121 TObjArray* branches = tree->GetListOfBranches();
122 for (int i = 0; i < branches->GetEntries(); i++) {
123 const std::string name = branches->At(i)->GetName();
124 if (name.size() <= suffix.size()) continue;
125 if (name.compare(name.size() - suffix.size(), suffix.size(), suffix) != 0) continue;
126 const std::vector<std::string> names =
127 GetLineBranches(tree, name.substr(0, name.size() - suffix.size()), observables);
128 if (!names.empty()) return names;
129 }
130 return {};
131}

163// Signals of `event` in one pad of `pad` per readout axis: X strips, Y strips and pixels (X and Y). The
164// channels not in the readout (no X nor Y) are not drawn
165template <typename Event>
166void DrawSplitSignals(const Event& event, const TRestDetectorReadout& readout, TPad* pad, const char* xTitle,
167 const char* yTitle) {
168 const char* names[3] = {"X strips", "Y strips", "pixels"};
169 std::vector<int> groups[3];
170 for (int n = 0; n < event.GetNumberOfSignals(); n++) {
171 const auto position = readout.GetPositionFromChannel(event.GetSignal(n).GetSignalID());
172 const bool x = !std::isnan(position.X()), y = !std::isnan(position.Y());
173 if (!x && !y) continue;
174 groups[x && !y ? 0 : y && !x ? 1 : 2].push_back(n);
175 }
176
177 int nPads = 0;
178 for (const auto& group : groups) nPads += !group.empty();
179 if (nPads == 0) {
180 std::cout << "TrackLineViewer: no signal of the event is in the readout" << std::endl;
181 return;
182 }
183 pad->Divide(nPads, 1);
184
185 int padIndex = 1;
186 for (int g = 0; g < 3; g++) {
187 if (groups[g].empty()) continue;
188 pad->cd(padIndex++);
189 TRestEvent::SetPadMargins(gPad);
190 auto* graphs = new TMultiGraph();
191 graphs->SetTitle(Form("Event ID %d, %s", event.GetID(), names[g]));
192 for (int n : groups[g]) {
193 const auto& signal = event.GetSignal(n);
194 auto* graph = new TGraph(signal.GetGraph());
195 graph->SetLineColor((signal.GetSignalID() % 72) + 1);
196 graphs->Add(graph, "L");
197 }
198 TRestEvent::DrawOwned(graphs, "A", xTitle, yTitle);
199 }
200}

203bool TrackLineViewer::ReadLine(const std::string& prefix, const char* const observables[6], double origin[3],
204 double end[3]) const {
205 TRestRun* run = fEvent ? fEvent->GetRestRun() : nullptr;
206 TTree* tree = run ? run->fAnalysisTree : nullptr;
207 if (!tree) return false;
208
209 // Values of the current entry (TRestRun::GetEntry, called by the browser)
210 const std::vector<std::string> names = GetLineBranches(tree, prefix, observables);
211 if (names.empty()) return false;
212 for (int i = 0; i < 3; i++) {
213 origin[i] = run->GetObservable<double>(names[i]);
214 end[i] = run->GetObservable<double>(names[i + 3]);
215 }
216 return true;
217}

219void TrackLineViewer::DrawLine(TVirtualPad* pad, const double origin[3], const double end[3], Color_t color,
220 int freeAxis) const {
221 TH1* frame = nullptr;
222 for (auto* object : *pad->GetListOfPrimitives()) {
223 if (auto* subPad = dynamic_cast<TVirtualPad*>(object)) {
224 DrawLine(subPad, origin, end, color, freeAxis);
225 } else if (!frame) {
226 frame = dynamic_cast<TH1*>(object);
227 }
228 }
229 if (!frame || (frame->GetDimension() == 3 && freeAxis >= 0)) return;
230
231 TVirtualPad* safePad = gPad;
232 pad->cd();
233 if (frame->GetDimension() == 3) {
234 for (int a = 0; a < 3; a++) {
235 if (std::isnan(origin[a]) || std::isnan(end[a])) {
236 if (safePad) safePad->cd();
237 return;
238 }
239 }
240 auto* line = new TPolyLine3D(2);
241 line->SetPoint(0, origin[0], origin[1], origin[2]);
242 line->SetPoint(1, end[0], end[1], end[2]);
243 SetLineStyle(line, color);
244 line->Draw();
245 for (int i = 0; i < 2; i++) {
246 const double* p = i == 0 ? origin : end;
247 auto* marker = new TPolyMarker3D(1, i == 0 ? kOriginMarker : kEndMarker);
248 marker->SetPoint(0, p[0], p[1], p[2]);
249 SetMarkerStyle(marker, color);
250 marker->Draw();
251 }
252 } else {
253 // Views of the energy against a coordinate have no second axis
254 const int u = GetAxis(frame->GetXaxis()->GetTitle());
255 const int v = GetAxis(frame->GetYaxis()->GetTitle());
256 const bool free = freeAxis >= 0 && (u == freeAxis || v == freeAxis);
257 if (free && u >= 0 && v >= 0) {
258 // Coordinate of the origin and end along the other axis, across the range of the free one
259 const bool vertical = u != freeAxis;
260 const int a = vertical ? u : v;
261 const TAxis* range = vertical ? frame->GetYaxis() : frame->GetXaxis();
262 const double low = range->GetBinLowEdge(range->GetFirst());
263 const double high = range->GetBinUpEdge(range->GetLast());
264 for (int i = 0; i < 2; i++) {
265 const double c = i == 0 ? origin[a] : end[a];
266 if (std::isnan(c)) continue;
267 auto* line = vertical ? new TLine(c, low, c, high) : new TLine(low, c, high, c);
268 SetLineStyle(line, color);
269 line->SetLineStyle(3);
270 line->Draw();
271 auto* marker = vertical ? new TMarker(c, high, i == 0 ? kOriginMarker : kEndMarker)
272 : new TMarker(high, c, i == 0 ? kOriginMarker : kEndMarker);
273 SetMarkerStyle(marker, color);
274 marker->Draw();
275 }
276 } else if (u >= 0 && v >= 0 && !std::isnan(origin[u]) && !std::isnan(origin[v]) &&
277 !std::isnan(end[u]) && !std::isnan(end[v])) {
278 auto* line = new TLine(origin[u], origin[v], end[u], end[v]);
279 SetLineStyle(line, color);
280 line->Draw();
281 auto* originMarker = new TMarker(origin[u], origin[v], kOriginMarker);
282 SetMarkerStyle(originMarker, color);
283 originMarker->Draw();
284 auto* endMarker = new TMarker(end[u], end[v], kEndMarker);
285 SetMarkerStyle(endMarker, color);
286 endMarker->Draw();
287 }
288 }
289 pad->Modified();
290 if (safePad) safePad->cd();
291}

306bool TrackLineViewer::DrawSignals() {
307 const auto* rawEvent = dynamic_cast<const TRestRawSignalEvent*>(fEvent);
308 const auto* signalEvent = dynamic_cast<const TRestDetectorSignalEvent*>(fEvent);
309 if (!rawEvent && !signalEvent) return false;
310 if ((rawEvent ? rawEvent->GetNumberOfSignals() : signalEvent->GetNumberOfSignals()) == 0) return false;
311 const TRestDetectorReadout* readout = GetReadout();
312 if (!readout) {
313 std::cout << "TrackLineViewer: no readout in the file, the signals are not split by axis"
314 << std::endl;
315 return false;
316 }
317
318 fCanvas->cd();
319 fSignalsPad = new TPad("signalsPad", "Signals", 0, 0, 1, 1);
320 fSignalsPad->Draw();
321 if (rawEvent) {
322 DrawSplitSignals(*rawEvent, *readout, fSignalsPad, "Time Bin", "ADC Channels");
323 } else {
324 DrawSplitSignals(*signalEvent, *readout, fSignalsPad, "Time", "Amplitude / Charge");
325 }

330void TrackLineViewer::Plot(const char* option) {
331 // Only this plot in the canvas: the pads of the events are removed, not deleted (each event owns its
332 // pad), the signals pad is deleted
333 ClearCanvasSelection();
334 fCanvas->cd();
335 fCanvas->GetListOfPrimitives()->Clear("nodelete");
336 delete fSignalsPad;
337 fSignalsPad = nullptr;
338 fPad = nullptr;
339 if (!fEvent) {
340 fCanvas->Update();
341 return;
342 }
343
344 const std::string drawOption = option ? option : "";
345 if (drawOption.empty() && DrawSignals()) {
346 fCanvas->cd();
347 fCanvas->Modified();
348 fCanvas->Update();
349 return;
350 }
351
352 const std::string className = fEvent->GetClassName();
353 const std::string eventOption =
354 drawOption.empty() && className == "TRestDetectorHitsEvent" ? "markers(colz)" : drawOption;
355 fCanvas->cd();
356 fPad = fEvent->DrawEvent(eventOption.c_str());
357 if (!fPad) {
358 fCanvas->Update();
359 return;
360 }
361 // Some events draw their pad in DrawEvent, others only build it
362 fCanvas->cd();
363 if (!fCanvas->GetListOfPrimitives()->FindObject(fPad)) fPad->Draw();
364
365 if (className == "TRestTrackEvent" || className == "TRestDetectorHitsEvent") {
366 double origin[3], end[3];
367 bool drawn = false;
368 if (ReadLine(fG4Prefix, kGeant4Observables, origin, end)) {
369 // The coordinate along the drift (Z) of the hits is not the Geant4 one
370 DrawLine(fPad, origin, end, kRed, 2);
371 std::cout << "Geant4 entry (" << origin[0] << ", " << origin[1] << ", " << origin[2] << "), end ("
372 << end[0] << ", " << end[1] << ", " << end[2] << ")" << std::endl;
373 drawn = true;
374 }
375 if (ReadLine(fPrefix, kLineObservables, origin, end)) {
376 DrawLine(fPad, origin, end, kBlack);
377 std::cout << "Line origin (" << origin[0] << ", " << origin[1] << ", " << origin[2] << "), end ("
378 << end[0] << ", " << end[1] << ", " << end[2] << ")" << std::endl;
379 drawn = true;
380 }
381 if (!drawn) {
382 std::cout << "TrackLineViewer: no line observables (TRestTrackLineAnalysisProcess or "
383 "TRestGeant4AnalysisProcess Entry/End) for this entry, drawing only the event"
384 << std::endl;
385 }
386 }
387 fPad->Modified();
388 fCanvas->cd();
389 fCanvas->Modified();
390 fCanvas->Update();
391}
