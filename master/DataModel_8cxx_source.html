27void TRestGeant4Event::UpdatePrimaryData(const G4Event* event) {
28 if (!event) return;
29
30 auto& d = fEventData;
31 d.primaryParticleIDs.clear();
32 d.primaryEnergies.clear();
33 d.primaryDirectionX.clear();
34 d.primaryDirectionY.clear();
35 d.primaryDirectionZ.clear();
36
37 // The primary generator gives a vertex per particle (each with its own origin): the particles of all
38 // the vertices are stored, the primary position is the one of the first vertex
39 const auto* firstVertex = event->GetPrimaryVertex();
40 if (!firstVertex) return;
41
42 const auto& position = firstVertex->GetPosition();
43 d.primaryPositionX = position.x() / CLHEP::mm;
44 d.primaryPositionY = position.y() / CLHEP::mm;
45 d.primaryPositionZ = position.z() / CLHEP::mm;
46
47 const auto* metadata = GetGeant4Metadata();

49 const auto* primaryVertex = event->GetPrimaryVertex(v);
50 if (!primaryVertex) continue;

54

60 metadata->fGeant4PhysicsInfo.InsertParticleName(definition->GetPDGEncoding(),
61 definition->GetParticleName());
62 }
63
64 d.primaryEnergies.emplace_back(primaryParticle->GetKineticEnergy() / CLHEP::keV);
65
66 const auto& direction = primaryParticle->GetMomentumDirection();
67 d.primaryDirectionX.push_back(direction.x());

359void OutputManager::RemoveUnwantedTracks() {
360 const auto& metadata = fSimulationManager->GetRestMetadata();
361 auto& eventData = fEvent->fEventData;
362
363 std::set<int> trackIDsToKeep;
364
365 const auto tracks = fEvent->GetTracks();
366
367 for (const auto& track : tracks) {
368 if (trackIDsToKeep.count(track.GetTrackID()) > 0) continue;
369
370 const auto hits = track.GetHits();
371
372 for (std::size_t i = 0; i < track.GetNumberOfHits(); ++i) {
373 if (!metadata->GetRemoveUnwantedTracksKeepZeroEnergyTracks() &&
374 hits.GetEnergy(static_cast<int>(i)) <= 0) {
375 continue;
376 }
377
378 const auto volume = metadata->GetGeant4GeometryInfo().GetVolumeFromID(track.GetHitVolumeID(i));
379
380 if (!metadata->IsKeepTracksVolume(volume)) continue;
381
382 trackIDsToKeep.insert(track.GetTrackID());
383
384 int parentID = track.GetParentID();
385
386 while (parentID >= 0) {
387 const auto parentTrackIt = fEvent->GetTrackIDToTrackIndex().find(parentID);
388
389 if (parentTrackIt == fEvent->GetTrackIDToTrackIndex().end()) {
390 break;
391 }
392
393 trackIDsToKeep.insert(parentID);
394
395 parentID = tracks[parentTrackIt->second].GetParentID();
396 }
397
398 break;
399 }
400 }
401
402 if (metadata->GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) {
403 cout << "Event " << fEvent->GetID() << " tracks to keep " << trackIDsToKeep.size() << "/"
404 << fEvent->GetNumberOfTracks() << endl;
405 }
406
407 RemoveTracks(trackIDsToKeep);
408
409 // A track whose parent was removed hangs from the primary generator
410 for (std::size_t k = 0; k < eventData.trackIDs.size(); ++k) {
411 const int parentID = eventData.parentIDs[k];
412
413 if (parentID > 0 &&
414 fEvent->GetTrackIDToTrackIndex().find(parentID) == fEvent->GetTrackIDToTrackIndex().end()) {
415 eventData.parentIDs[k] = 0;
416 }
417 }
418}

422 const auto& geometryInfo = metadata->GetGeant4GeometryInfo();
423
424 std::set<int> trackIDsToKeep;
