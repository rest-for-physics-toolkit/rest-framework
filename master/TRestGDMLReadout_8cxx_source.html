93void TRestGDMLReadout::BuildGeometry() {
94 if (!fNode) {
95 throw std::runtime_error("TRestGDMLReadout: fNode is not initialized!");
96 }
97
98 ReleaseGeometry();
99
100 fPathToPhysicalIDMap.clear();
101 fPhysicalIDToReadoutPath.clear();
102 fReadoutPositions.clear();
103 if (!LoadDecoding(fDecodingFile)) {
104 throw std::runtime_error("TRestGDMLReadout::BuildGeometry - cannot load the decoding file '" +
105 fDecodingFile + "' (not found or without channels)");
106 }
107
108 RESTInfo << "Importing external GDML layout file: " << fReadoutParams.fGDMLFileName << RESTendl;
109
110 gGeoManager = nullptr;
111 // Entities and parameters of the configuration resolved, as restG4 reads it
112 fGeoManager = TRestGDMLParser::Import(fReadoutParams.fGDMLFileName);
113
114 if (!fGeoManager) {
115 throw std::runtime_error("TRestGDMLReadout::BuildGeometry - ROOT failed to parse GDML input: " +
116 fReadoutParams.fGDMLFileName);
117 }
118 TGeoManager::SetVerboseLevel(0);
119
120 int parsedChannelsCount = 0;
121 std::smatch match;
122 // Channels of each readoutInfo entry (path, physical ID), to find their readout volumes
123 std::vector<std::vector<std::pair<std::string, int>>> channelsOfInfo(fReadoutParams.fReadoutInfo.size());
124
125 std::vector<int> currentOffsets;
126 // Compiled once (not for each node); an invalid expression is an error of the configuration
127 std::vector<std::regex> channelRegexes;
128 for (const auto& info : fReadoutParams.fReadoutInfo) {
129 currentOffsets.push_back(info.fOffset);
130 try {
131 channelRegexes.emplace_back(info.fVolume);
132 } catch (const std::regex_error& e) {
133 throw std::runtime_error("TRestGDMLReadout: invalid regular expression volume '" + info.fVolume +
134 "': " + e.what());
135 }
136 }
137
138 TGeoIterator it(fGeoManager->GetTopVolume());
139 TGeoNode* node = nullptr;
140
141 while ((node = it())) {
142 if (!node->GetVolume() || node->GetVolume()->IsAssembly()) continue;
143
144 TString tPath;
145 it.GetPath(tPath);
146 std::string nodePath = tPath.Data();
147
148 for (size_t i = 0; i < fReadoutParams.fReadoutInfo.size(); ++i) {
149 const auto& info = fReadoutParams.fReadoutInfo[i];
150 if (std::regex_search(nodePath, match, channelRegexes[i])) {
151 int physicalID = 0;
152
153 if (match.size() > 2) {
154 try {
155 int firstCapturedNum = std::stoi(match.str(1));
156 int secondCapturedNum = std::stoi(match.str(2));
157 physicalID =
158 currentOffsets[i] + (firstCapturedNum * info.fMultiplier) + secondCapturedNum;
159 } catch (const std::exception& e) {
160 continue;
161 }
162 } else if (match.size() > 1) {
163 try {
164 physicalID = currentOffsets[i] + std::stoi(match.str(1));
165 } catch (const std::exception& e) {
166 continue;
167 }
168 } else {
169 physicalID = currentOffsets[i];
170 }
171
172 if (!fGeoManager->cd(nodePath.c_str())) continue;
173 std::string canonicalPath = fGeoManager->GetPath();
174
175 if (fPathToPhysicalIDMap.count(canonicalPath)) break;
176
177 fPathToPhysicalIDMap[canonicalPath] = physicalID;
178 channelsOfInfo[i].emplace_back(canonicalPath, physicalID);
179 parsedChannelsCount++;
180
181 if (match.size() <= 1) currentOffsets[i]++;
182 break;
183 }
184 }
185 }
186
187 // Readout volume of each channel: the one matching readoutVolume in the same parent volume (the copy
188 // of the assembly holding the sensitive volume), so its position follows the rotations of each copy
189 auto parentOf = [](const std::string& path) { return path.substr(0, path.rfind('/')); };
190 for (size_t i = 0; i < fReadoutParams.fReadoutInfo.size(); ++i) {
191 const auto& info = fReadoutParams.fReadoutInfo[i];
192 if (info.fReadoutVolume.empty()) continue;
193 std::regex readoutRegex;
194 try {
195 readoutRegex = std::regex(info.fReadoutVolume);
196 } catch (const std::regex_error& e) {
197 throw std::runtime_error("TRestGDMLReadout: invalid regular expression readoutVolume '" +
198 info.fReadoutVolume + "': " + e.what());
199 }
200 std::map<std::string, std::vector<std::string>> readoutsOfParent;
201 TGeoIterator readoutIt(fGeoManager->GetTopVolume());
202 while ((node = readoutIt())) {
203 if (!node->GetVolume() || node->GetVolume()->IsAssembly()) continue;
204 TString tPath;
205 readoutIt.GetPath(tPath);
206 const std::string nodePath = tPath.Data();
207 if (!std::regex_search(nodePath, readoutRegex) || !fGeoManager->cd(nodePath.c_str())) continue;
208 const std::string canonicalPath = fGeoManager->GetPath();
209 readoutsOfParent[parentOf(canonicalPath)].push_back(canonicalPath);
210 }
211 for (const auto& [path, physicalID] : channelsOfInfo[i]) {
212 const auto it = readoutsOfParent.find(parentOf(path));
213 const size_t found = it == readoutsOfParent.end() ? 0 : it->second.size();
214 if (found != 1) {
215 throw std::runtime_error("TRestGDMLReadout: the channel " + path + " has " +
216 std::to_string(found) + " volumes matching readoutVolume '" +
217 info.fReadoutVolume + "' in its parent volume (exactly one needed)");
218 }
219 fPhysicalIDToReadoutPath[physicalID] = it->second.front();
220 }
221 }
222
223 TGeoIterator maskIt(fGeoManager->GetTopVolume());
224 while ((node = maskIt())) {
225 TString tPath;
226 maskIt.GetPath(tPath);
227 std::string nodePath = tPath.Data();
228 TGeoVolume* vol = node->GetVolume();
229 if (!vol) continue;
230
231 bool isReadoutNode = false;
232 for (const auto& [canonicalPath, id] : fPathToPhysicalIDMap) {
233 if (canonicalPath.find(nodePath) != std::string::npos ||
234 nodePath.find(canonicalPath) != std::string::npos) {
235 isReadoutNode = true;
236 break;
237 }
238 }
239 // The readout volumes (e.g. light guides) are shown too, in another colour
240 bool isSignalReadout = false;
241 for (const auto& [id, readoutPath] : fPhysicalIDToReadoutPath) {
242 if (readoutPath.find(nodePath) != std::string::npos ||
243 nodePath.find(readoutPath) != std::string::npos) {
244 isReadoutNode = true;
245 isSignalReadout = nodePath.find(readoutPath) != std::string::npos;
246 break;
247 }
248 }
249
250 if (!isReadoutNode) {
251 if (!vol->IsAssembly() && vol->GetNdaughters() == 0) {
252 vol->SetVisibility(kFALSE);
253 node->SetVisibility(kFALSE);
254 } else {
255 vol->SetVisibility(kFALSE);
256 node->SetVisibility(kTRUE);
257 }
258 } else {
259 vol->SetVisibility(kTRUE);
260 node->SetVisibility(kTRUE);
261 vol->SetLineColor(isSignalReadout ? kAzure + 1 : kGray);
262 }
263 }
264
265 fGeoManager->CloseGeometry();
266 UpdateYAMLFromParams<TRestGDMLReadout>(fNode);
267
268 RESTInfo << "Successfully indexed " << parsedChannelsCount << " channels ("
269 << fPhysicalIDToReadoutPath.size()
270 << " with readout volume) conserving native geometry structures." << RESTendl;
271}

273ROOT::Math::XYZVector TRestGDMLReadout::GetReadoutPosition(int daqID) const {
274 if (fReadoutPositions.empty() && fGeoManager) {
275 // Origin of each readout volume in the global frame
276 for (const auto& [physicalID, path] : fPhysicalIDToReadoutPath) {
277 if (!fGeoManager->cd(path.c_str())) continue;
278 const TGeoMatrix* matrix = fGeoManager->GetCurrentMatrix();
279 if (!matrix) continue;
280 const double local[3] = {0, 0, 0};
281 double global[3] = {0, 0, 0};
282 matrix->LocalToMaster(local, global);
283 fReadoutPositions[physicalID] = {global[0], global[1], global[2]};
284 }
285 }
286 const auto it = fReadoutPositions.find(GetPhysicalID(daqID));
287 if (it != fReadoutPositions.end()) return it->second;
288 const double nan = std::numeric_limits<double>::quiet_NaN();
289 return {nan, nan, nan};
290}

293 const auto readoutPosition = GetReadoutPosition(daqID);
294 if (std::isnan(readoutPosition.X())) return std::numeric_limits<double>::quiet_NaN();
