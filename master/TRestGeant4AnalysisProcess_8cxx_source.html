168std::string ParticleObservableName(const std::string& particle) {
169 std::string name;
170 for (const char c : particle) {
171 if (c == '-') {
172 name += "Minus";
173 } else if (c == '+') {
174 name += "Plus";
175 } else {
176 name += std::isalnum(static_cast<unsigned char>(c)) ? c : '_';
177 }
178 }
179 return name;
180}
181
184std::string VolumeObservableName(const std::string& volume) {
185 std::string name;
186 for (const char c : volume) name += std::isalnum(static_cast<unsigned char>(c)) ? c : '_';
187 const auto first = name.find_first_not_of('_');
188 if (first == std::string::npos) return "sensitive";
189 return name.substr(first, name.find_last_not_of('_') - first + 1);
190}

252void TRestGeant4AnalysisProcess::InitProcess() {
253 // The metadata of the run or, without one (e.g. no input file), the one given
254 const auto* runMetadata =
255 fRunInfo ? dynamic_cast<TRestGeant4Metadata*>(fRunInfo->GetMetadataClass("TRestGeant4Metadata"))
256 : nullptr;
257 if (runMetadata) fGeant4Metadata = runMetadata;
258 const auto* metadata = fGeant4Metadata;
259 if (!metadata) throw std::runtime_error("TRestGeant4AnalysisProcess: no TRestGeant4Metadata found");
260 fGeometryInfo = &metadata->GetGeant4GeometryInfo();
261 const auto& physicsInfo = metadata->GetGeant4PhysicsInfo();
262 const auto knownParticles = physicsInfo.GetAllParticles();
263 const auto knownProcesses = physicsInfo.GetAllProcesses();
264
265 // Volume IDs of a list of expressions, an error if none of them matches
266 auto resolveVolumes = [this](const std::vector<std::string>& expressions, const std::string& what) {
267 std::set<int> ids;
268 for (const auto& expression : expressions) {
269 const auto matched = fGeometryInfo->GetVolumeIDsMatchingExpression(expression);
270 if (matched.empty()) {
271 RESTWarning << "TRestGeant4AnalysisProcess: " << what << ", no volume matches '" << expression
272 << "'" << RESTendl;
273 }
274 ids.insert(matched.begin(), matched.end());
275 }
276 if (!expressions.empty() && ids.empty()) {
277 throw std::runtime_error("TRestGeant4AnalysisProcess: no volume of " + what +
278 " found in the geometry");
279 }
280 return ids;
281 };
282 // Volume IDs of the sensitive volumes, the default of a selection without volumes
283 auto sensitiveIDs = [this, metadata](const std::string& what) {
284 const auto& sensitive = metadata->GetSensitiveVolumes();
285 const auto ids = fGeometryInfo->GetVolumeIDsFromNames({sensitive.begin(), sensitive.end()});
286 if (ids.empty()) {
287 throw std::runtime_error("TRestGeant4AnalysisProcess: " + what +
288 " without 'volumes' and no sensitive volume found");
289 }
290 return ids;
291 };
292 // Names checked against the physics information, an error if one is not there
293 auto checkNames = [](const std::vector<std::string>& values, const std::set<std::string>& known,
294 const std::string& what) {
295 for (const auto& value : values) {
296 if (!known.count(value)) {
297 throw std::runtime_error("TRestGeant4AnalysisProcess: " + what + " '" + value +
298 "' not in the physics information of the simulation metadata");
299 }
300 }
301 return std::set<std::string>(values.begin(), values.end());
302 };
303 auto checkName = [](const std::string& name, std::set<std::string>& names) {
304 if (name.empty()) throw std::runtime_error("TRestGeant4AnalysisProcess: selection without name");
305 if (!names.insert(name).second) {
306 throw std::runtime_error("TRestGeant4AnalysisProcess: duplicated selection '" + name + "'");
307 }
308 };
309
310 // The selections are resolved before registering, so the vector does not change afterwards and the
311 // registered references stay valid
312 // Without selections, one per sensitive volume of activeVolumes, named as it, with the entering particle
313 auto selections = fSelections;
314 if (selections.empty()) {
315 for (const auto& expression : metadata->GetSensitiveVolumeExpressions()) {
316 TRestGeant4SelectionInfo sensitive;
317 sensitive.fSelectionName = VolumeObservableName(expression);
318 sensitive.fVolumes = {expression};
319 sensitive.fEntering = true;
320 selections.push_back(sensitive);
321 }
322 }
323
324 fSelectionObs.clear();
325 std::set<std::string> names;
326 for (const auto& info : selections) {
327 checkName(info.fSelectionName, names);
328 const auto what = "selection '" + info.fSelectionName + "'";
329 Selection selection;
330 selection.name = info.fSelectionName;
331 selection.volumeIDs =
332 info.fVolumes.empty() ? sensitiveIDs(what) : resolveVolumes(info.fVolumes, what);
333 selection.particles = checkNames(info.fParticles, knownParticles, what + ", particle");
334 // Processes sharing type and subtype (e.g. eIoni and ionIoni) are registered once
335 for (const auto& process : checkNames(info.fProcesses, knownProcesses, what + ", process")) {
336 selection.processIDs.insert(physicsInfo.GetProcessID(process));
337 }
338 for (const auto& particle : checkNames(info.fSplitByParticle, knownParticles, what + ", particle")) {
339 if (!selection.particles.empty() && !selection.particles.count(particle)) {
340 throw std::runtime_error("TRestGeant4AnalysisProcess: " + what + ", splitByParticle '" +
341 particle + "' not in its particles");
342 }
343 selection.splitParticles.push_back(particle);
344 }
345 selection.splitEnergy.assign(selection.splitParticles.size(), 0);
346 selection.entering = info.fEntering;
347 if (info.fPerVolume) {
348 selection.perVolumeIDs.assign(selection.volumeIDs.begin(), selection.volumeIDs.end());
349 selection.volumeEnergy.assign(selection.perVolumeIDs.size(), 0);
350 }
351 selection.interactions = info.fInteractions;
352 if (!info.fTrackEdepVolumes.empty() && !info.fInteractions) {
353 throw std::runtime_error("TRestGeant4AnalysisProcess: " + what +
354 ", trackEdepVolumes without interactions");
355 }
356 selection.trackEdepVolumeIDs = resolveVolumes(info.fTrackEdepVolumes, what + ", trackEdepVolumes");
357 fSelectionObs.push_back(selection);
358 }
359
360 RegisterObservable("totalEdep", fObs.totalEdep);
361 RegisterObservable("sensitiveVolumeEnergy", fObs.sensitiveVolumeEnergy);
362 RegisterObservable("nTracks", fObs.nTracks);
363 RegisterObservable("boundingSize", fObs.boundingSize);
364 RegisterObservable("energyPrimary", fObs.energyPrimary);
365 RegisterObservable("xOriginPrimary", fObs.xOriginPrimary);
366 RegisterObservable("yOriginPrimary", fObs.yOriginPrimary);
367 RegisterObservable("zOriginPrimary", fObs.zOriginPrimary);
368 RegisterObservable("xDirectionPrimary", fObs.xDirectionPrimary);
369 RegisterObservable("yDirectionPrimary", fObs.yDirectionPrimary);
370 RegisterObservable("zDirectionPrimary", fObs.zDirectionPrimary);
371 RegisterObservable("thetaPrimary", fObs.thetaPrimary);
372 RegisterObservable("phiPrimary", fObs.phiPrimary);
373 RegisterObservable("eventPrimaryParticleName", fObs.eventPrimaryParticleName);
374 RegisterObservable("subEventPrimaryParticleName", fObs.subEventPrimaryParticleName);
375
376 for (auto& selection : fSelectionObs) {
377 const auto prefix = selection.name + "_";
378 RegisterObservable(prefix + "Energy", selection.energy);
379 RegisterObservable(prefix + "MeanPosX", selection.meanPosX);
380 RegisterObservable(prefix + "MeanPosY", selection.meanPosY);
381 RegisterObservable(prefix + "MeanPosZ", selection.meanPosZ);
382 RegisterObservable(prefix + "FirstHitTime", selection.firstHitTime);
383 RegisterObservable(prefix + "NTracks", selection.nTracks);
384 for (size_t p = 0; p < selection.splitParticles.size(); p++) {
385 RegisterObservable(prefix + ParticleObservableName(selection.splitParticles[p]) + "_Energy",
386 selection.splitEnergy[p]);
387 }
388 if (!selection.perVolumeIDs.empty()) {
389 // Volume names as observable names (two equal names, of volumes or particles, are an error of
390 // RegisterObservable)
391 for (size_t v = 0; v < selection.perVolumeIDs.size(); v++) {
392 const auto volume =
393 VolumeObservableName(fGeometryInfo->GetVolumeFromID(selection.perVolumeIDs[v]));
394 RegisterObservable(prefix + volume + "_Energy", selection.volumeEnergy[v]);
395 }
396 RegisterObservable(prefix + "MaxVolumeEnergy", selection.maxVolumeEnergy);
397 }
398 if (selection.interactions) {
399 RegisterObservable(prefix + "NInteractions", selection.nInteractions);
400 RegisterObservable(prefix + "InteractionX", selection.interactionX);
401 RegisterObservable(prefix + "InteractionY", selection.interactionY);
402 RegisterObservable(prefix + "InteractionZ", selection.interactionZ);
403 RegisterObservable(prefix + "InteractionTime", selection.interactionTime);
404 RegisterObservable(prefix + "InteractionKineticEnergy", selection.interactionKineticEnergy);
405 RegisterObservable(prefix + "InteractionSecondariesEnergy",
406 selection.interactionSecondariesEnergy);
407 RegisterObservable(prefix + "InteractionTrackEdep", selection.interactionTrackEdep);
408 RegisterObservable(prefix + "InteractionTrackMaxVolumeEdep",
409 selection.interactionTrackMaxVolumeEdep);
410 }
411 if (!selection.entering) continue;
412 auto& entering = selection.enteringObs;
413 RegisterObservable(prefix + "NEntering", entering.nEntering);
414 RegisterObservable(prefix + "EntryParticle", entering.particle);
415 RegisterObservable(prefix + "EntryParentParticle", entering.parentParticle);
416 RegisterObservable(prefix + "EntryCreatorProcess", entering.creatorProcess);
417 RegisterObservable(prefix + "EntryEnergy", entering.entryEnergy);
418 RegisterObservable(prefix + "EntryTime", entering.entryTime);
419 RegisterObservable(prefix + "EntryX", entering.entryX);
420 RegisterObservable(prefix + "EntryY", entering.entryY);
421 RegisterObservable(prefix + "EntryZ", entering.entryZ);
422 RegisterObservable(prefix + "EndX", entering.endX);
423 RegisterObservable(prefix + "EndY", entering.endY);
424 RegisterObservable(prefix + "EndZ", entering.endZ);
425 RegisterObservable(prefix + "InitialEnergy", entering.initialEnergy);
426 RegisterObservable(prefix + "InitialX", entering.initialX);
427 RegisterObservable(prefix + "InitialY", entering.initialY);
428 RegisterObservable(prefix + "InitialZ", entering.initialZ);
429 }

432bool TRestGeant4AnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
433 output.CopyFrom(&input);
434 const auto* event = dynamic_cast<const TRestGeant4Event*>(&output);
435 if (!event) throw std::runtime_error("TRestGeant4AnalysisProcess: output is not a TRestGeant4Event");
436
437 const auto& data = event->fEventData;
438 const double nan = std::numeric_limits<double>::quiet_NaN();
439
440 // Assignments keep the addresses of the registered members
441 fObs.totalEdep = event->GetTotalDepositedEnergy();
442 fObs.sensitiveVolumeEnergy = event->GetSensitiveVolumeEnergy();
443 fObs.nTracks = static_cast<int>(event->GetNumberOfTracks());
444
445 fObs.energyPrimary = data.primaryEnergies.empty() ? nan : data.primaryEnergies[0];
446 const auto origin = event->GetPrimaryPosition();
447 fObs.xOriginPrimary = origin.X();
448 fObs.yOriginPrimary = origin.Y();
449 fObs.zOriginPrimary = origin.Z();
450 const ROOT::Math::XYZVector direction = event->GetNumberOfPrimaries() == 0
451 ? ROOT::Math::XYZVector(nan, nan, nan)
452 : event->GetPrimaryDirection(0).Unit();
453 fObs.xDirectionPrimary = direction.X();
454 fObs.yDirectionPrimary = direction.Y();
455 fObs.zDirectionPrimary = direction.Z();
456 fObs.thetaPrimary = direction.Theta();
457 fObs.phiPrimary = direction.Phi();
458 fObs.eventPrimaryParticleName =
459 event->GetNumberOfPrimaries() == 0 ? "" : event->GetPrimaryParticleName(0);
460 fObs.subEventPrimaryParticleName = event->IsSubEvent() ? event->GetSubEventParticleName() : "generator";
461
462 fTimeOrigin = event->GetFirstDepositTime();
463 ProcessSelections(*event);
464 for (auto& selection : fSelectionObs) {
465 if (selection.entering) ProcessEntering(*event, selection);
466 }
467 ProcessInteractions(*event);
468
469 return true;
470}

473void TRestGeant4AnalysisProcess::ProcessSelections(const TRestGeant4Event& event) {
474 const auto& data = event.fEventData;
475 const auto& hits = data.hitsStorage;
476 const double nan = std::numeric_limits<double>::quiet_NaN();
477
478 for (auto& selection : fSelectionObs) {
479 selection.energy = 0;
480 selection.meanPosX = selection.meanPosY = selection.meanPosZ = 0;
481 selection.firstHitTime = std::numeric_limits<double>::infinity();
482 selection.nTracks = 0;
483 std::fill(selection.splitEnergy.begin(), selection.splitEnergy.end(), 0);
484 std::fill(selection.volumeEnergy.begin(), selection.volumeEnergy.end(), 0);
485 selection.maxVolumeEnergy = 0;
486 selection.trackIndices.clear();
487 }
488
489 double min[3] = {std::numeric_limits<double>::max(), std::numeric_limits<double>::max(),
490 std::numeric_limits<double>::max()};
491 double max[3] = {std::numeric_limits<double>::lowest(), std::numeric_limits<double>::lowest(),
492 std::numeric_limits<double>::lowest()};
493 bool anyHit = false;
494
495 for (size_t t = 0; t < data.trackIDs.size(); t++) {
496 const auto particle = event.GetParticleName(data.trackParticleIDs[t]);
497 const size_t start = data.trackStartIndices[t];
498 const size_t end = start + data.trackNHits[t];
499
500 for (size_t h = start; h < end; h++) {
501 if (hits.energy[h] <= 0) continue;
502 const double position[3] = {hits.x[h], hits.y[h], hits.z[h]};
503 for (int a = 0; a < 3; a++) {
504 min[a] = std::min(min[a], position[a]);
505 max[a] = std::max(max[a], position[a]);
506 }
507 anyHit = true;
508 }
509
510 for (auto& selection : fSelectionObs) {
511 if (!selection.particles.empty() && !selection.particles.count(particle)) continue;
512 double trackEnergy = 0;
513 bool any = false;
514 for (size_t h = start; h < end; h++) {
515 const double energy = hits.energy[h];
516 if (energy <= 0) continue;
517 if (!selection.volumeIDs.count(data.hitVolumeID[h])) continue;
518 if (!selection.processIDs.empty() && !selection.processIDs.count(data.hitProcessID[h]))
519 continue;
520 trackEnergy += energy;
521 if (!selection.perVolumeIDs.empty()) {
522 const auto& ids = selection.perVolumeIDs;
523 const auto v =
524 std::lower_bound(ids.begin(), ids.end(), data.hitVolumeID[h]) - ids.begin();
525 selection.volumeEnergy[v] += energy;
526 }
527 selection.meanPosX += energy * hits.x[h];
528 selection.meanPosY += energy * hits.y[h];
529 selection.meanPosZ += energy * hits.z[h];
530 selection.firstHitTime = std::min(selection.firstHitTime, hits.time[h] - fTimeOrigin);
531 any = true;
532 }
533 if (!any) continue;
534
535 selection.energy += trackEnergy;
536 selection.nTracks++;
537 for (size_t p = 0; p < selection.splitParticles.size(); p++) {
538 if (selection.splitParticles[p] == particle) selection.splitEnergy[p] += trackEnergy;
539 }
540 selection.trackIndices.push_back(t);
541 }
542 }
543
544 fObs.boundingSize =
545 anyHit ? std::sqrt((max[0] - min[0]) * (max[0] - min[0]) + (max[1] - min[1]) * (max[1] - min[1]) +
546 (max[2] - min[2]) * (max[2] - min[2]))
547 : 0;
548
549 for (auto& selection : fSelectionObs) {
550 for (const double energy : selection.volumeEnergy) {
551 selection.maxVolumeEnergy = std::max(selection.maxVolumeEnergy, energy);
552 }
553 if (selection.energy > 0) {
554 selection.meanPosX /= selection.energy;
555 selection.meanPosY /= selection.energy;
556 selection.meanPosZ /= selection.energy;
557 } else {
558 selection.meanPosX = selection.meanPosY = selection.meanPosZ = nan;
559 selection.firstHitTime = nan;
560 }
561 RESTDebug << "TRestGeant4AnalysisProcess: " << selection.name << " energy " << selection.energy
562 << ", tracks " << selection.nTracks << RESTendl;
563 }

567void TRestGeant4AnalysisProcess::ProcessEntering(const TRestGeant4Event& event, Selection& selection) {
568 const auto& data = event.fEventData;
569 const auto& hits = data.hitsStorage;
570
571 auto& entering = selection.enteringObs;
572 entering.clear();
573
574 auto inVolumes = [&selection, &data](size_t h) {
575 return selection.volumeIDs.count(data.hitVolumeID[h]) > 0;
576 };
577 const auto& indexOfID = event.GetTrackIDToTrackIndex();
578
579 // Indices of the entering tracks and of the primaries created in the volumes, by track ID
580 std::map<int, size_t> enteringTracks;
581 std::map<int, size_t> insidePrimaries;
582 for (const size_t t : selection.trackIndices) {
583 // The track itself or the first ancestor whose first hit is out of the volumes
584 size_t current = t;
585 while (true) {
586 if (data.trackNHits[current] > 0 && !inVolumes(data.trackStartIndices[current])) {
587 enteringTracks.emplace(data.trackIDs[current], current);
588 break;
589 }
590 const auto parent = indexOfID.find(data.parentIDs[current]);
591 // The product of a primary at rest (an ion that decays, e.g. the alpha of Po210): as a primary
592 if (parent != indexOfID.end() && data.parentIDs[parent->second] == 0 &&
593 data.trackInitialEnergies[parent->second] == 0) {
594 if (data.trackNHits[current] > 0) insidePrimaries.emplace(data.trackIDs[current], current);
595 break;
596 }
597 if (data.parentIDs[current] == 0 || parent == indexOfID.end()) {
598 // A primary starting in the volumes (e.g. a source in the gas), or a track whose parent is
599 // not in the event (removed without hits, registerEmptyTracks: false): it starts there
600 const bool primary = data.parentIDs[current] == 0 && data.trackInitialEnergies[current] != 0;
601 if ((primary || data.parentIDs[current] != 0) && data.trackNHits[current] > 0) {
602 insidePrimaries.emplace(data.trackIDs[current], current);
603 }
604 break;
605 }
606 current = static_cast<size_t>(parent->second);
607 }
608 }
609
610 entering.nEntering = static_cast<int>(enteringTracks.size());
611 const auto& tracks = enteringTracks.empty() ? insidePrimaries : enteringTracks;
612 if (tracks.empty()) return;
613
614 const size_t t = tracks.begin()->second;
615 entering.particle = event.GetParticleName(data.trackParticleIDs[t]);
616 const auto parent = indexOfID.find(data.parentIDs[t]);
617 if (data.parentIDs[t] != 0 && parent != indexOfID.end()) {
618 entering.parentParticle = event.GetParticleName(data.trackParticleIDs[parent->second]);
619 }
620 entering.creatorProcess = event.GetProcessName(data.trackCreatorProcessIDs[t]);
621 entering.initialEnergy = data.trackInitialEnergies[t];
622 entering.initialX = data.trackInitialPositionX[t];
623 entering.initialY = data.trackInitialPositionY[t];
624 entering.initialZ = data.trackInitialPositionZ[t];
625
626 // First and last hits in the volumes, the track entered them or starts in them so there is at least one
627 const size_t start = data.trackStartIndices[t];
628 bool entered = false;
629 for (size_t h = start; h < start + data.trackNHits[t]; h++) {
630 if (!inVolumes(h)) continue;
631 if (!entered) {
632 entering.entryEnergy = data.hitKineticEnergy[h];
633 entering.entryTime = hits.time[h] - fTimeOrigin;
634 entering.entryX = hits.x[h];
635 entering.entryY = hits.y[h];
636 entering.entryZ = hits.z[h];
637 entered = true;
638 }
639 entering.endX = hits.x[h];
640 entering.endY = hits.y[h];
641 entering.endZ = hits.z[h];
642 }
643
644 RESTDebug << "TRestGeant4AnalysisProcess: " << selection.name << ", " << entering.nEntering
645 << " entering tracks, first " << entering.particle << RESTendl;
646}

651void TRestGeant4AnalysisProcess::ProcessInteractions(const TRestGeant4Event& event) {
652 bool any = false;
653 for (auto& selection : fSelectionObs) {
654 if (!selection.interactions) continue;
655 any = true;
656 selection.nInteractions = 0;
657 selection.interactionX.clear();
658 selection.interactionY.clear();
659 selection.interactionZ.clear();
660 selection.interactionTime.clear();
661 selection.interactionKineticEnergy.clear();
662 selection.interactionSecondariesEnergy.clear();
663 selection.interactionTrackEdep.clear();
664 selection.interactionTrackMaxVolumeEdep.clear();
665 }
666 if (!any) return;
667
668 const auto& data = event.fEventData;
669 const auto& hits = data.hitsStorage;
670
671 // Track indices of the secondaries of each track ID
672 std::map<int, std::vector<size_t>> secondaries;
673 for (size_t t = 0; t < data.trackIDs.size(); t++) secondaries[data.parentIDs[t]].push_back(t);
674
675 // Energy deposited in the volumes (all if empty) by a track and all its descendants, in total and in
676 // the volume with the most
677 auto descendantsEdep = [&](size_t track, const std::set<int>& volumeIDs, double& maxVolumeEnergy) {
678 double energy = 0;
679 std::map<int, double> volumeEnergy;
680 std::vector<size_t> pending = {track};
681 while (!pending.empty()) {
682 const size_t t = pending.back();
683 pending.pop_back();
684 const size_t start = data.trackStartIndices[t];
685 for (size_t h = start; h < start + data.trackNHits[t]; h++) {
686 if (hits.energy[h] <= 0) continue;
687 if (!volumeIDs.empty() && !volumeIDs.count(data.hitVolumeID[h])) continue;
688 energy += hits.energy[h];
689 volumeEnergy[data.hitVolumeID[h]] += hits.energy[h];
690 }
691 const auto children = secondaries.find(data.trackIDs[t]);
692 if (children != secondaries.end()) {
693 pending.insert(pending.end(), children->second.begin(), children->second.end());
694 }
695 }
696 maxVolumeEnergy = 0;
697 for (const auto& [volume, e] : volumeEnergy) maxVolumeEnergy = std::max(maxVolumeEnergy, e);
698 return energy;
699 };
700
701 for (size_t t = 0; t < data.trackIDs.size(); t++) {
702 const auto particle = event.GetParticleName(data.trackParticleIDs[t]);
703 const size_t start = data.trackStartIndices[t];
704 const size_t end = start + data.trackNHits[t];
705
706 for (auto& selection : fSelectionObs) {
707 if (!selection.interactions) continue;
708 if (!selection.particles.empty() && !selection.particles.count(particle)) continue;
709
710 // Selected hits of this track: their index in the vectors and their hit index
711 std::vector<std::pair<size_t, size_t>> selected;
712 for (size_t h = start; h < end; h++) {
713 if (!selection.volumeIDs.count(data.hitVolumeID[h])) continue;
714 if (!selection.processIDs.empty() && !selection.processIDs.count(data.hitProcessID[h]))
715 continue;
716 selected.emplace_back(selection.interactionX.size(), h);
717 selection.interactionX.push_back(hits.x[h]);
718 selection.interactionY.push_back(hits.y[h]);
719 selection.interactionZ.push_back(hits.z[h]);
720 selection.interactionTime.push_back(hits.time[h] - fTimeOrigin);
721 selection.interactionKineticEnergy.push_back(data.hitKineticEnergy[h]);
722 selection.interactionSecondariesEnergy.push_back(0);
723 }
724 selection.nInteractions += static_cast<int>(selected.size());
725 if (selected.empty()) continue;
726
727 // The same for all the interactions of the track
728 double maxVolumeEdep = 0;
729 const double trackEdep = descendantsEdep(t, selection.trackEdepVolumeIDs, maxVolumeEdep);
730 selection.interactionTrackEdep.insert(selection.interactionTrackEdep.end(), selected.size(),
731 trackEdep);
732 selection.interactionTrackMaxVolumeEdep.insert(selection.interactionTrackMaxVolumeEdep.end(),
733 selected.size(), maxVolumeEdep);
734
735 const auto children = secondaries.find(data.trackIDs[t]);
736 if (children == secondaries.end()) continue;
737 for (const size_t c : children->second) {
738 // The closest selected hit with the creator process of the secondary
739 double minDistance2 = std::numeric_limits<double>::max();
740 size_t closest = selected.size();
741 for (size_t i = 0; i < selected.size(); i++) {
742 const size_t h = selected[i].second;
743 if (data.hitProcessID[h] != data.trackCreatorProcessIDs[c]) continue;
744 const double dx = hits.x[h] - data.trackInitialPositionX[c];
745 const double dy = hits.y[h] - data.trackInitialPositionY[c];
746 const double dz = hits.z[h] - data.trackInitialPositionZ[c];
747 const double distance2 = dx * dx + dy * dy + dz * dz;
748 if (distance2 < minDistance2) {
749 minDistance2 = distance2;
750 closest = i;
751 }
752 }
753 if (closest < selected.size()) {
754 selection.interactionSecondariesEnergy[selected[closest].first] +=
755 data.trackInitialEnergies[c];
756 }
757 }
758 }
759 }
