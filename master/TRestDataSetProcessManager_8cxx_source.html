118void TRestDataSetProcessManager::LoadConfig() {
119 UpdateParamsFromYAML<TRestDataSetProcessManager>(fNode);
120 fProcesses.clear();
121 for (const auto& element : fNode) {
122 const auto key = element.first.as<std::string>();
123 auto value = element.second;
124 if (!value || !value.IsMap()) continue;
125 auto meta = CreateChildMetadata(key, value);
126 if (!meta) continue;
127 if (auto* process = dynamic_cast<TRestDataSetProcess*>(meta.get())) {
128 // The vector grows first, so that the ownership is only transferred once nothing can throw

165 const std::string context = "TRestDataSetProcessManager '" + fName + "'";

173 // A process already in the input (same name) is replaced if it has the same class; another class would
174 // replace a different metadata (the configurations are stored by name)

181 if (storedClass != process->GetClassName()) {

191 if (inputFile.empty()) dataSet.fOutputFileName = fOutputFileName;
192 for (const auto& process : fProcesses) {

194 << RESTendl;

196 process->Apply(dataSet);
197 }

204 // again (replaced by Save)

211 : nullptr;
212 TList* classKeys = inputStore ? inputStore->GetListOfKeys() : nullptr;
213 for (int c = 0; classKeys && c < classKeys->GetSize(); c++) {
214 const TObject* classKey = classKeys->At(c);
215 TDirectory* classFolder = inputStore->GetDirectory(classKey->GetName());
216 if (!classFolder) continue;
217 for (auto* key : *classFolder->GetListOfKeys()) {
218 TDirectory* folder = classFolder->GetDirectory(key->GetName());
219 if (!folder || names.count(key->GetName())) continue;
220 const std::string path =
221 TRestDataSetProcess::kStoreDirectory + "/" + classKey->GetName() + "/" + key->GetName();

223 }
224 }
225 }
226
227 // This pipeline (the processes are written apart, with their results), the processes and their cuts
228 YAML::Node config = TRestYAMLUtils::Detach(fNode);
229 for (const auto& name : names) config.remove(name);
230 TRestMetadata::WriteMetadata(file.get(), fName, config);
231 std::set<std::string> cuts;
232 for (const auto& process : fProcesses) {
233 process->Save(file.get());
234 cuts.insert(process->fCutNames.begin(), process->fCutNames.end());
235 }
236 if (fManager) {
237 for (auto* cut : fManager->GetMetadataOfType<TRestCut>())
238 if (cuts.count(cut->GetName())) cut->WriteMetadata(file.get());
239 }

245std::map<std::string, std::string> TRestDataSetProcessManager::ParseInputFormat() const {
246 std::map<std::string, std::string> tags;
247 // A dataset generated in memory has no file name: only the keys of its runs
248 if (fInputFormat.empty() || fInputFileName.empty()) return tags;
249
250 // The pattern as a regex: each [tag] a part without "_", the rest literal
251 std::string pattern = std::filesystem::path(fInputFormat).filename().string();
252 if (pattern.size() > 5 && pattern.substr(pattern.size() - 5) == ".root")
253 pattern.resize(pattern.size() - 5);
254 const std::regex tagRegex(R"(\[([\w.]+)\])");
255 const std::regex special(R"([.^$|()\[\]{}*+?\\])");
256 std::vector<std::string> names;
257 std::string regex;
258 std::smatch match;
259 std::string rest = pattern;
260 while (std::regex_search(rest, match, tagRegex)) {
261 regex += std::regex_replace(match.prefix().str(), special, R"(\$&)") + "([^_]+)";
262 names.push_back(match[1].str());
263 rest = match.suffix().str();
264 }
265 regex += std::regex_replace(rest, special, R"(\$&)");
266
267 const std::string name = std::filesystem::path(fInputFileName).stem().string();
268 std::smatch values;
269 if (!std::regex_match(name, values, std::regex(regex))) {
270 throw std::runtime_error("TRestDataSetProcessManager '" + fName + "': the input " + fInputFileName +
271 " does not match the inputFormat \"" + fInputFormat + "\"");
272 }
273 for (size_t i = 0; i < names.size(); i++) tags[names[i]] = values[i + 1].str();
274 return tags;
275}

277std::string TRestDataSetProcessManager::ResolveOutputFileName(const TRestDataSet& dataSet) const {
278 const auto tags = ParseInputFormat();
279 return TRestTools::ReplaceDelimited(fOutputFileName, '[', ']',
280 [&](const std::string& key) -> std::optional<std::string> {
281 // [key|default]: the default is for the keys of the runs
282 const auto tag = tags.find(key.substr(0, key.find('|')));
283 if (tag != tags.end()) return tag->second;
284 return dataSet.ResolveFilePattern("[" + key + "]");
285 });
