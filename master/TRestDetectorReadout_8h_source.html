310 ROOT::Math::XYZVector GetReadoutNormal(const ROOT::Math::XYZVector& position) const {
311 const double distance = GetDistanceToReadout(position);
312 if (std::isnan(distance)) return {0, 0, 0};
313 using V = ROOT::Math::XYZVector;
314 const V normal(GetDistanceToReadout(position + V(1, 0, 0)) - distance,
315 GetDistanceToReadout(position + V(0, 1, 0)) - distance,
316 GetDistanceToReadout(position + V(0, 0, 1)) - distance);
317 return normal.R() > 0 ? normal.Unit() : normal;
318 }
