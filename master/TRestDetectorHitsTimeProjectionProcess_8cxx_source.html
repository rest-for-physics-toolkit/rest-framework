89void TRestDetectorHitsTimeProjectionProcess::InitProcess() {
90 const std::string user = "TRestDetectorHitsTimeProjectionProcess";
91 if (!fManager) throw std::runtime_error(user + ": no TRestManager");
92
93 auto* readouts =
94 dynamic_cast<TRestDetectorReadoutManager*>(fManager->GetMetadataClass("TRestDetectorReadoutManager"));
95 if (!readouts) throw std::runtime_error(user + ": no TRestDetectorReadoutManager found");
96 fReadout = readouts->SelectReadout(fReadoutName, user);
97
98 if (fDriftVelocity <= 0) {
99 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "charge", user);
100 if (!medium) {
101 throw std::runtime_error(user +
102 ": set 'driftVelocity' or define a TRestDetectorMedium with charge");
103 }
104 fDriftVelocity = medium->GetChargeResponse(user).GetDriftVelocity();
105 if (fDriftVelocity <= 0) {
106 throw std::runtime_error(user + ": the medium '" + medium->GetName() +
107 "' gives no drift velocity");
108 }
109 fMediumName = medium->GetName();

112 UpdateYAMLFromParams<TRestDetectorHitsTimeProjectionProcess>(fNode);

117bool TRestDetectorHitsTimeProjectionProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
118 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
119 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
120 if (!inputEvent || !outputEvent) {
121 throw std::runtime_error(
122 "TRestDetectorHitsTimeProjectionProcess: input/output is not a TRestDetectorHitsEvent");
123 }
124
125 outputEvent->Initialize();
126 outputEvent->SetEventInfo(inputEvent);
127 fNHitsWithoutDrift = 0;
128
129 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
130 XYZVector position = inputEvent->GetPosition(n);
131 const double time = inputEvent->GetTime(n);
132 // The distance to the readout is a projection on its normal: the undefined coordinates (NaN) are
133 // evaluated at 0, they do not change it along an axis normal
134 const XYZVector defined(std::isnan(position.X()) ? 0 : position.X(),
135 std::isnan(position.Y()) ? 0 : position.Y(),
136 std::isnan(position.Z()) ? 0 : position.Z());
137 const double distance = fReadout->GetDistanceToReadout(defined);
138 const XYZVector normal = fReadout->GetReadoutNormal(defined);
139 if (std::isnan(distance) || normal.R() == 0) {
140 fNHitsWithoutDrift++;
141 outputEvent->AddHit(position, inputEvent->GetEnergy(n), time, inputEvent->GetType(n),
142 inputEvent->GetParticleID(n));
143 continue;
144 }
145 // Axis of the normal (the readout normals are along an axis), moved by the missing distance
146 const double nx = std::abs(normal.X()), ny = std::abs(normal.Y()), nz = std::abs(normal.Z());
147 const double shift = fDriftVelocity * time - distance;
148 if (nx >= ny && nx >= nz) {
149 position.SetX(defined.X() + shift / normal.X());
150 } else if (ny >= nz) {
151 position.SetY(defined.Y() + shift / normal.Y());
152 } else {
153 position.SetZ(defined.Z() + shift / normal.Z());
154 }
155 outputEvent->AddHit(position, inputEvent->GetEnergy(n), time,
156 TRestHitsData::GetHitType(position.X(), position.Y(), position.Z()),
157 inputEvent->GetParticleID(n));
158 }
159
160 return outputEvent->GetNumberOfHits() > 0;
161}
