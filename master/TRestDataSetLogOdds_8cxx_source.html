74double LogOdds(double probability) {
75 if (probability <= 0) return 1000.;
76 if (probability >= 1) return -1000.;
77 return std::log(1. - probability) - std::log(probability);

101double TRestDataSetLogOddsObservable::GetProbability(double value) const {
102 if (fPDF.empty() || !(value >= fRange.first && value < fRange.second)) return 0;
103 const auto bin = static_cast<size_t>((value - fRange.first) / (fRange.second - fRange.first) * fNBins);
104 return bin < fPDF.size() ? fPDF[bin] : 0;
105}

128void TRestDataSetLogOdds::ComputeParameters(const TRestDataSet&, ROOT::RDF::RNode frame) {
129 // All the histograms booked before the event loop
130 std::vector<ROOT::RDF::RResultPtr<TH1D>> histograms;
131 for (const auto& observable : fObservables) {
132 histograms.push_back(frame.Histo1D({("pdf_" + observable.fObservable).c_str(), "", observable.fNBins,
133 observable.fRange.first, observable.fRange.second},
134 observable.fObservable));
135 }
136 for (size_t i = 0; i < fObservables.size(); i++) {
137 auto& observable = fObservables[i];
138 const TH1D& histogram = *histograms[i];
139 const double integral = histogram.Integral();
140 if (integral <= 0) {
141 throw std::runtime_error("TRestDataSetLogOdds '" + fName + "': no entries of " +
142 observable.fObservable + " in its range");
143 }
144 observable.fPDF.assign(observable.fNBins, 0);
145 for (int bin = 0; bin < observable.fNBins; bin++)
146 observable.fPDF[bin] = histogram.GetBinContent(bin + 1) / integral;
147 RESTDebug << "TRestDataSetLogOdds '" << fName << "': PDF of " << observable.fObservable << " from "
148 << integral << " entries" << RESTendl;
149 }
150 UpdateYAMLFromParams<TRestDataSetLogOdds>(fNode);
151}

153void TRestDataSetLogOdds::Apply(TRestDataSet& dataSet) const {
154 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
155 std::vector<std::string> columns;
156 std::string total;
157 for (const auto& observable : fObservables) {
158 const std::string column = GetColumn(observable.fObservable);
159 // The observable copied: the data frame can be used after this process
160 auto logOdds = [observable](double value) { return LogOdds(observable.GetProbability(value)); };
161 const std::string input = AsDouble(frame, observable.fObservable);
162 frame = DefineColumn(frame, column, logOdds, {input});
163 columns.push_back(column);
164 if (!total.empty()) total += " + ";
165 total += column;
166 }
167 frame = DefineColumn(frame, GetTotalColumn(), total);
168 columns.push_back(GetTotalColumn());
169 dataSet.SetDataFrame(frame, columns);
170}
