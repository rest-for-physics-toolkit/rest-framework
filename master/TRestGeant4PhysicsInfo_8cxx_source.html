53void TRestGeant4PhysicsInfo::InsertProcessName(int id, const std::string& processName,
54 const std::string& processType) {
55 {
56 std::shared_lock<std::shared_mutex> lock(insertMutex);
57 if (fProcessNamesMap.count(id) > 0) return;

59 std::unique_lock<std::shared_mutex> lock(insertMutex);
