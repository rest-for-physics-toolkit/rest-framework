81struct EventKey {
82 int run;
83 int event;
84 int subEvent;
85 bool operator==(const EventKey& other) const {
86 return run == other.run && event == other.event && subEvent == other.subEvent;
87 }
88};
89struct EventKeyHash {
90 size_t operator()(const EventKey& key) const {
91 size_t hash = std::hash<int>()(key.run);
92 hash = hash * 1000003u ^ std::hash<int>()(key.event);
93 return hash * 1000003u ^ std::hash<int>()(key.subEvent);
94 }

127std::string NormalizeType(std::string type) {
128 type.erase(std::remove(type.begin(), type.end(), ' '), type.end());
129 // The unsigned ones first (UInt_t contains Int_t)
130 static const std::vector<std::pair<std::string, std::string>> aliases = {
131 {"UInt_t", "unsignedint"},
132 {"ULong64_t", "unsignedlonglong"},
133 {"Double_t", "double"},
134 {"Float_t", "float"},
135 {"Int_t", "int"},
136 {"Long64_t", "longlong"},
137 {"Bool_t", "bool"},
138 {"std::", ""},
139 {"ROOT::VecOps::RVec", "vector"},
140 {"ROOT::RVec", "vector"},
141 {"RVec", "vector"}};
142 for (const auto& [alias, name] : aliases) {
143 for (size_t at = type.find(alias); at != std::string::npos; at = type.find(alias, at + name.size())) {
144 type.replace(at, alias.size(), name);
145 }
146 }
147 return type;
148}

151std::unique_ptr<CombinedColumn> BookColumn(ROOT::RDF::RNode& frame, const std::string& column,
152 const std::string& type) {
153 const double nan = std::numeric_limits<double>::quiet_NaN();
154 const std::string t = NormalizeType(type);
155 if (t == "double") return std::make_unique<TypedColumn<double>>(frame, column, nan);
156 if (t == "float") return std::make_unique<TypedColumn<float>>(frame, column, static_cast<float>(nan));
157 if (t == "int") return std::make_unique<TypedColumn<int>>(frame, column, 0);
158 if (t == "unsignedint") return std::make_unique<TypedColumn<unsigned int>>(frame, column, 0u);
159 if (t == "longlong") return std::make_unique<TypedColumn<long long>>(frame, column, 0LL);
160 if (t == "unsignedlonglong")
161 return std::make_unique<TypedColumn<unsigned long long>>(frame, column, 0ULL);
162 if (t == "bool") return std::make_unique<TypedColumn<bool>>(frame, column, false);
163 if (t == "string") return std::make_unique<TypedColumn<std::string>>(frame, column, std::string());
164 if (t == "vector<double>")
165 return std::make_unique<TypedColumn<ROOT::RVec<double>>>(frame, column, ROOT::RVec<double>());
166 if (t == "vector<float>")
167 return std::make_unique<TypedColumn<ROOT::RVec<float>>>(frame, column, ROOT::RVec<float>());
168 if (t == "vector<int>")
169 return std::make_unique<TypedColumn<ROOT::RVec<int>>>(frame, column, ROOT::RVec<int>());
170 return nullptr;
171}

190void TRestDataSetCombine::Apply(TRestDataSet& dataSet) const {
191 const std::string context = "TRestDataSetCombine '" + fName + "'";
192 TRestDataSet other(fCombineFileName);
193 ROOT::RDF::RNode otherFrame = other.GetDataFrame();
194
195 // The columns combined
196 std::vector<std::string> columns;
197 for (const auto& column : other.GetColumns()) {
198 if (std::find(kKeyColumns.begin(), kKeyColumns.end(), column) != kKeyColumns.end()) continue;
199 bool match = fObservables.empty();
200 for (const auto& pattern : fObservables) {
201 if (std::regex_match(column, std::regex(TRestTools::PatternToRegex(pattern)))) match = true;
202 }
203 if (match) columns.push_back(column);
204 }
205 if (columns.empty()) {
206 throw std::runtime_error(context + ": no column of " + fCombineFileName + " matches the observables");
207 }
208
209 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
210 const auto existing = frame.GetColumnNames();
211 auto exists = [&existing](const std::string& name) {
212 return std::find(existing.begin(), existing.end(), name) != existing.end();
213 };
214 const std::string entryColumn = fName + "_entry";
215 const std::string matchedColumn = fName + "_matched";
216 for (const auto& name : {entryColumn, matchedColumn}) {
217 if (exists(name)) throw std::runtime_error(context + ": the dataset already has a column " + name);
218 }
219
220 // Reading of the keys and the columns of combineFileName, booked for a single event loop
221 auto runs = otherFrame.Take<int>("runIndex");
222 auto events = otherFrame.Take<int>("eventID");
223 auto subEvents = otherFrame.Take<int>("subEventID");
224 std::vector<std::pair<std::string, std::unique_ptr<CombinedColumn>>> combined;
225 for (const auto& column : columns) {
226 const std::string name = fPrefix + column;
227 if (exists(name)) {
228 throw std::runtime_error(context + ": the dataset already has a column " + name +
229 " (give a prefix or select the observables)");
230 }
231 const std::string type = otherFrame.GetColumnType(column);
232 auto booked = BookColumn(otherFrame, column, type);
233 if (!booked) {
234 throw std::runtime_error(context + ": the column " + column + " of " + fCombineFileName +
235 " has a type that cannot be combined (" + type + ")");
236 }
237 combined.emplace_back(name, std::move(booked));
238 }
239
240 // The runs of the dataset in combineFileName (runNumber and subRunNumber), -1 if not there
241 auto runMap = std::make_shared<std::vector<int>>();
242 int missingRuns = 0;
243 for (const auto& run : dataSet.GetRuns()) {
244 int index = -1;
245 for (size_t r = 0; r < other.GetRuns().size(); r++) {
246 const auto& candidate = *other.GetRuns()[r].run;
247 if (candidate.GetRunNumber() == run.run->GetRunNumber() &&
248 candidate.GetSubRunNumber() == run.run->GetSubRunNumber()) {
249 index = static_cast<int>(r);
250 break;
251 }
252 }
253 if (index < 0) missingRuns++;
254 runMap->push_back(index);
255 }
256 if (missingRuns > 0) {
257 RESTWarning << context << ": " << missingRuns << " runs of the dataset are not in "
258 << fCombineFileName << " (their entries are unmatched)" << RESTendl;
259 }
260
261 // Entry of each event of combineFileName (the event loop runs here)
262 auto entries = std::make_shared<std::unordered_map<EventKey, long long, EventKeyHash>>();
263 for (size_t n = 0; n < runs->size(); n++) {
264 const EventKey key{(*runs)[n], (*events)[n], (*subEvents)[n]};
265 if (!entries->emplace(key, static_cast<long long>(n)).second) {
266 throw std::runtime_error(context + ": the event " + std::to_string(key.event) + " (subevent " +
267 std::to_string(key.subEvent) + ") of the run index " +
268 std::to_string(key.run) + " is twice in " + fCombineFileName);
269 }
270 }
271 RESTInfo << context << ": " << columns.size() << " columns of " << entries->size() << " entries of "
272 << fCombineFileName << RESTendl;
273
274 frame = frame.Define(entryColumn,
275 [runMap, entries](int run, int event, int subEvent) -> long long {
276 if (run < 0 || run >= static_cast<int>(runMap->size()) || (*runMap)[run] < 0)
277 return -1;
278 const auto found = entries->find({(*runMap)[run], event, subEvent});
279 return found == entries->end() ? -1 : found->second;
280 },
281 {"runIndex", "eventID", "subEventID"});
282 frame = frame.Define(matchedColumn, [](long long entry) { return entry >= 0 ? 1 : 0; }, {entryColumn});
283
284 std::vector<std::string> newColumns;
285 for (auto& [name, column] : combined) {
286 frame = column->Define(frame, name, entryColumn);
287 newColumns.push_back(name);
288 }
289 newColumns.push_back(matchedColumn);
290 if (!fKeepUnmatched)
291 frame = frame.Filter([](long long entry) { return entry >= 0; }, {entryColumn}, fName);
292 dataSet.SetDataFrame(frame, newColumns);
293}
