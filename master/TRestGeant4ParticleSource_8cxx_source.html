66vector<vector<TRestGeant4ParticleState>> ReadDecay0(const string& fileName) {
67 const string error = "TRestGeant4ParticleSource: Decay0 file '" + fileName + "': ";
68 ifstream input(TRestTools::GetFullPath(fileName));
69 if (!input.is_open()) throw runtime_error(error + "cannot be opened");
70
71 // Geant3 particle IDs: mass (MeV), name and charge
72 struct Particle {
73 double mass;
74 string name;
75 int charge;
76 };
77 const map<int, Particle> particles = {{1, {0, "gamma", 0}}, {2, {0.511, "e+", 1}},
78 {3, {0.511, "e-", -1}}, {5, {105.658, "mu+", 1}},
79 {6, {105.658, "mu-", -1}}, {14, {938.272, "proton", 1}},
80 {47, {3727.379, "alpha", 2}}};
81 auto readParticle = [&](int id, double px, double py, double pz) {
82 const auto particle = particles.find(id);
83 if (particle == particles.end()) {
84 throw runtime_error(error + "particle ID " + to_string(id) + " not supported");
85 }
86 const double mass = particle->second.mass;
87 TRestGeant4ParticleState state;
88 state.SetParticleName(particle->second.name);
89 state.SetParticleCharge(particle->second.charge);
90 state.SetEnergy(1000.0 * (sqrt(px * px + py * py + pz * pz + mass * mass) - mass));
91 state.SetDirection(XYZVector(px, py, pz));
92 return state;
93 };
94
95 // The header: the number of events of either format
96 int nEvents = 0;
97 bool oldFormat = false;
98 string line;
99 for (int i = 0; i < 30 && getline(input, line); ++i) {
100 const auto position = line.find("@nevents=");
101 if (position != string::npos) {
102 nEvents = stoi(line.substr(position + 9));
103 break;
104 }
105 if (line.find("First event and full number of events:") != string::npos) {
106 int firstEvent = 0;
107 input >> firstEvent >> nEvents;
108 oldFormat = true;
109 break;
110 }
111 }
112 if (nEvents <= 0) {
113 throw runtime_error(error +
114 "no '@nevents=' (bxdecay0) nor 'First event and full number of events:' "
115 "(old Decay0) in its header");
116 }
117
118 vector<vector<TRestGeant4ParticleState>> events;
119 for (int event = 0; event < nEvents; ++event) {
120 int nParticles = 0;
121 if (oldFormat) {
122 int eventNumber = 0;
123 double time = 0;
124 input >> eventNumber >> time >> nParticles;
125 } else {
126 while (getline(input, line) && line.find("@event_start") == string::npos) {
127 }
128 getline(input, line);
129 input >> nParticles;
130 }
131 if (!input) throw runtime_error(error + "less events than given in its header");
132 vector<TRestGeant4ParticleState> eventParticles;
133 for (int i = 0; i < nParticles; ++i) {
134 int id = 0;
135 double time = 0, px = 0, py = 0, pz = 0;
136 if (oldFormat) {
137 input >> id >> px >> py >> pz >> time;
138 } else {
139 string name;
140 input >> id >> time >> px >> py >> pz >> name;
141 }
142 eventParticles.push_back(readParticle(id, px, py, pz));
143 }
144 events.push_back(std::move(eventParticles));
145 }
146 return events;
147}

179void TRestGeant4ParticleSource::CheckJoint() {
180 const auto* energy = dynamic_cast<const TRestGeant4EnergyDistributionJoint*>(fEnergy.get());
181 const auto* angular = dynamic_cast<const TRestGeant4AngularDistributionJoint*>(fAngular.get());
182 if (!energy || !angular || energy->fType != angular->fType) {
183 throw runtime_error(GetClassName() +
184 ": formula2, TH2D and cosmic must be the type of both energy and angular");
185 }
186 fJointType = energy->fType;
187
188 if (fJointType == "formula2") {
189 fJointFormula = Pick(energy->fFormula, angular->fFormula, "formula");
190 // An unknown formula name throws
191 StringToEnergyAndAngularDistributionFormulas(fJointFormula);
192 return;
193 }
194
195 fJointFile = Pick(energy->fFilename, angular->fFilename, "filename");
196 fJointName = Pick(energy->fNameInFile, angular->fNameInFile, "nameInFile");
197 if (fJointFile.empty() || fJointName.empty()) {
198 throw runtime_error(GetClassName() + ": 'filename' and 'nameInFile' are needed for " + fJointType);
199 }
200 if (fJointType == "cosmic") {
201 // As legacy, nameInFile is the particle key, the histogram <key>_energy_zenith
202 const auto particle = kCosmicParticles.find(fJointName);
203 if (particle == kCosmicParticles.end()) {
204 throw runtime_error(GetClassName() + ": unknown cosmic particle key '" + fJointName + "'");
205 }
206 if (fParticleName.empty()) fParticleName = particle->second;
207 if (fParticleName != particle->second) {
208 throw runtime_error(GetClassName() + ": the particle of the cosmic key '" + fJointName + "' is " +
209 particle->second + ", not " + fParticleName);
210 }
211 fJointName += "_energy_zenith";
212 }

243void TRestGeant4ParticleSource::BuildJointCumulative() {
244 const int nx = fJointFunction->GetNpx(), ny = fJointFunction->GetNpy();
245 const double xMin = fJointFunction->GetXmin(), yMin = fJointFunction->GetYmin();
246 const double dx = (fJointFunction->GetXmax() - xMin) / nx, dy = (fJointFunction->GetYmax() - yMin) / ny;
247 // Integral of each cell with 2 x 2 Gauss-Legendre points (exact for cubic polynomials)
248 const double node = 0.5 / std::sqrt(3.0);
249 auto cumulative = make_shared<vector<double>>(static_cast<size_t>(nx) * ny + 1, 0.0);
250 size_t cell = 0;
251 for (int j = 0; j < ny; j++) {
252 const double yCenter = yMin + (j + 0.5) * dy;
253 for (int i = 0; i < nx; i++) {
254 const double xCenter = xMin + (i + 0.5) * dx;
255 double integral = 0;
256 for (const double u : {-node, node})
257 for (const double v : {-node, node})
258 integral += fJointFunction->Eval(xCenter + u * dx, yCenter + v * dy);
259 // Negative values taken as positive, as TF2::GetRandom2
260 (*cumulative)[cell + 1] = (*cumulative)[cell] + std::abs(integral) * 0.25 * dx * dy;
261 cell++;
262 }
263 }
264 const double total = cumulative->back();
265 if (!(total > 0)) throw runtime_error(GetClassName() + ": the integral of the formula is zero");
266 fFlux = total; // The formulas (CosmicMuons) are fluxes: counts / s / cm2 / keV / rad
267 for (auto& value : *cumulative) value /= total;
268 fJointCumulative = cumulative;
269}

271void TRestGeant4ParticleSource::SampleJointFunction(TRandom& random, double& x, double& y) const {
272 const auto& cumulative = *fJointCumulative;
273 const int nx = fJointFunction->GetNpx();
274 const double xMin = fJointFunction->GetXmin(), yMin = fJointFunction->GetYmin();
275 const double dx = (fJointFunction->GetXmax() - xMin) / nx;
276 const double dy = (fJointFunction->GetYmax() - yMin) / fJointFunction->GetNpy();
277 const double r = random.Rndm();
278 // The cell of r, as TF2::GetRandom2: x linear in the cell with r, y uniform
279 size_t cell = std::upper_bound(cumulative.begin(), cumulative.end(), r) - cumulative.begin();
280 cell = std::min(std::max<size_t>(cell, 1), cumulative.size() - 1) - 1;
281 const double width = cumulative[cell + 1] - cumulative[cell];
282 const double xInCell = width > 0 ? (r - cumulative[cell]) / width : 0.5;
283 x = xMin + dx * (cell % nx + xInCell);
284 y = yMin + dy * (cell / nx + random.Rndm());
285}

287void TRestGeant4ParticleSource::LoadJointDistribution() {
288 const auto* energy = static_cast<const TRestGeant4EnergyDistributionJoint*>(fEnergy.get());
289 const auto& energyRange = energy->fRange;
290 const auto* angular = static_cast<const TRestGeant4AngularDistributionJoint*>(fAngular.get());
291 const auto& angleRange = angular->fRange;
292
293 if (fJointType == "formula2") {
294 // Energy in keV and angle in rad
295 fJointFunction = make_unique<TF2>(EnergyAndAngularDistributionFormulasToRootFormula(
296 StringToEnergyAndAngularDistributionFormulas(fJointFormula)));
297 const double energyMin = std::max<double>(energyRange.first, fJointFunction->GetXmin());
298 const double energyMax = std::min<double>(energyRange.second, fJointFunction->GetXmax());
299 const double angleMin = std::max<double>(angleRange.first, fJointFunction->GetYmin());
300 const double angleMax = std::min<double>(angleRange.second, fJointFunction->GetYmax());
301 if (energyMin >= energyMax || angleMin >= angleMax) {
302 throw runtime_error(GetClassName() + ": 'range' is outside the formula one");
303 }
304 fJointFunction->SetRange(energyMin, angleMin, energyMax, angleMax);
305 fJointFunction->SetNpx(energy->fFormulaNPoints);
306 fJointFunction->SetNpy(angular->fFormulaNPoints);
307 if (!fJointCumulative) BuildJointCumulative();
308 fJointEnergyUnit = 1;
309 fJointAngleUnit = 1;
310 return;
311 }
312
313 // As legacy, the histograms have the energy in MeV and the angle in deg; the units of the content from
314 // the title of the z axis: a content per energy is multiplied by the width of its bin (the counts of the
315 // bin), to sample it and for the flux
316 fJointHistogram = ReadHistogram2D(fJointFile, fJointName); // Any 2D histogram, as a TH2D
317 fJointEnergyUnit = 1000;
318 fJointAngleUnit = TMath::DegToRad();
319 const std::string title = fJointHistogram->GetZaxis()->GetTitle();
320 const auto units = ParseHistogramContentUnits(title);
321 for (int i = 1; i <= fJointHistogram->GetNbinsX(); i++) {
322 const double energy = fJointHistogram->GetXaxis()->GetBinCenter(i) * fJointEnergyUnit;
323 const double binWidth = fJointHistogram->GetXaxis()->GetBinWidth(i) * fJointEnergyUnit;
324 const double width = units.perEnergy > 0 ? binWidth / units.perEnergy : 1;
325 for (int j = 1; j <= fJointHistogram->GetNbinsY(); j++) {
326 const double angle = fJointHistogram->GetYaxis()->GetBinCenter(j) * fJointAngleUnit;
327 double content = fJointHistogram->GetBinContent(i, j) * width;
328 if (energy < energyRange.first || energy > energyRange.second || angle < angleRange.first ||
329 angle > angleRange.second) {
330 content = 0;
331 } else if (fJointType == "cosmic" && std::abs(std::cos(angle)) > 1e-6) {
332 // As legacy, the flux through a horizontal surface to the one through the disk normal to the
333 // direction of the cosmic generator
334 content /= std::abs(std::cos(angle));
335 }
336 fJointHistogram->SetBinContent(i, j, content);
337 }
338 }
339 const double integral = fJointHistogram->Integral();
340 if (integral <= 0) throw runtime_error(GetClassName() + ": the histogram is empty in 'range'");
341 fJointHistogram->ComputeIntegral();
342 // Only a flux normalizes to a time (the exposure); otherwise it is not computed
343 fFlux = units.flux ? integral : 0;
344 if (!units.flux) {
345 RESTWarning << GetClassName() << ": the content of '" << fJointName << "' ('" << title
346 << "', title of its z axis) is not a flux (counts / s / cm2, optionally per eV, keV, MeV "
347 << "or GeV): its flux and the exposure of the simulation are not computed" << RESTendl;
348 }
349}

351vector<TRestGeant4ParticleState> TRestGeant4ParticleSource::Generate(TRandom& random, const XYZVector& origin,
352 const XYZVector& center, bool inward) {
353 if (IsDecay0()) {
354 if (!fDecay0Events) throw runtime_error(GetClassName() + ": the Decay0 file is not loaded");
355 auto particles = (*fDecay0Events)[random.Integer(static_cast<unsigned int>(fDecay0Events->size()))];
356 for (auto& particle : particles) particle.SetOrigin(origin);
357 return particles;
358 }
359
360 XYZVector axis = fAngular->GetDirection();
361 if (inward && (center - origin).R() > 0) axis = (center - origin).Unit();
362
363 double energy = 0;
364 XYZVector direction;
365 if (!fJointType.empty()) {
366 double x = 0, angle = 0;
367 if (fJointFunction) {
368 SampleJointFunction(random, x, angle);
369 } else if (fJointHistogram) {
370 fJointHistogram->GetRandom2(x, angle, &random);
371 } else {
372 throw runtime_error(GetClassName() + ": the distribution is not loaded");
373 }
374 energy = x * fJointEnergyUnit;
375 direction = DirectionFromAxis(random, axis, angle * fJointAngleUnit);
376 } else {
377 energy = fEnergy->Sample(random);
378 direction = fAngular->Sample(random, axis);
379 }
380
381 TRestGeant4ParticleState particle;
382 particle.SetParticleName(fParticleName);
383 particle.SetExcitationLevel(fExcitedLevel);
384 particle.SetParticleCharge(fCharge);
385 particle.SetEnergy(energy);
386 particle.SetDirection(direction);
387 particle.SetOrigin(origin);
388 return {particle};
389}
