93void TRestGeant4CRY::LoadConfig() {
94 UpdateParamsFromYAML<TRestGeant4CRY>(fNode);
95
96 const std::string error = GetClassName() + " '" + GetName() + "': ";
97 if (fSubBoxLength <= 0) {
98 throw std::runtime_error(error +
99 "subBoxLength is needed: the side of the square where the particles "
100 "start, about the size of the detector");
101 }
102 const std::set<std::string> known = {"muons", "neutrons", "protons", "gammas",
103 "electrons", "pions", "kaons"};
104 if (fParticles.empty()) throw std::runtime_error(error + "particles is empty (all by default)");
105 for (const auto& particle : fParticles) {
106 if (!known.count(particle)) {
107 throw std::runtime_error(error + "unknown particles '" + particle +
108 "' (muons, neutrons, protons, gammas, electrons, pions or kaons)");
109 }
110 }
111 if (fNParticlesMin < 1 || fNParticlesMax < fNParticlesMin) {
112 throw std::runtime_error(error + "it must be 1 <= nParticlesMin <= nParticlesMax");
113 }
114 if (std::abs(fLatitude) > 90) throw std::runtime_error(error + "latitude must be in [-90, 90]");
115 const double altitude = fAltitude / 1000.0; // m
116 if (altitude != 0 && altitude != 2100 && altitude != 11300) {
117 throw std::runtime_error(error + "altitude must be 0, 2100 m or 11300 m (the CRY data tables)");
118 }
119 if (!std::regex_match(fDate, std::regex(R"(\d{1,2}-\d{1,2}-\d{4})"))) {
120 throw std::runtime_error(error + "date must be month-day-year, e.g. 7-1-2012");
121 }
122 if (fMaxExposureTime < 0) throw std::runtime_error(error + "maxExposureTime must not be negative");
123
124 UpdateYAMLFromParams<TRestGeant4CRY>(fNode);
125}

132 << " returnMuons " << Returns("muons") << " returnPions " << Returns("pions") << " returnKaons "
133 << Returns("kaons") << " nParticlesMin " << fNParticlesMin << " nParticlesMax " << fNParticlesMax
