126int Binomial(TRandom& random, int n, double p) {
127 if (n <= 0 || !(p > 0)) return 0;
128 if (p >= 1) return n;
129 if (p > 0.5) return n - Binomial(random, n, 1 - p);
130 const double q = 1 - p;
131 const double mean = n * p;
132 if (mean < 10) {
133 // The first k whose cumulated probability exceeds the uniform (P(0) = q^n >= exp(-20))
134 double u = random.Rndm();
135 double probability = std::pow(q, n);
136 int k = 0;
137 while (u > probability && k < n) {
138 u -= probability;
139 probability *= (n - k) * p / ((k + 1) * q);
140 k++;
141 }
142 return k;
143 }
144 const double pi = std::acos(-1.0);
145 const double logFactorialN = std::lgamma(n + 1.0);
146 const double logP = std::log(p);
147 const double logQ = std::log(q);
148 const double width = std::sqrt(2 * mean * q);
149 while (true) {
150 double y, k;
151 do {
152 y = std::tan(pi * random.Rndm());
153 k = width * y + mean;
154 } while (k < 0 || k >= n + 1);
155 k = std::floor(k);
156 // Probability of k over the envelope (at most 1)
157 const double ratio =
158 1.2 * width * (1 + y * y) *
159 std::exp(logFactorialN - std::lgamma(k + 1) - std::lgamma(n - k + 1) + k * logP + (n - k) * logQ);
160 if (random.Rndm() <= ratio) return static_cast<int>(k);
161 }
162}

189std::function<double(double)> MakeCaptureFunction(const std::string& text, double maxDistance,
190 const std::string& prefix) {
191 const auto separator = text.rfind(':');
192 const auto extension = text.find(".root");
193 if (separator != std::string::npos && extension != std::string::npos && separator > extension) {
194 const std::string fileName = text.substr(0, separator);
195 const std::string objectName = text.substr(separator + 1);
196 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str()));
197 if (!file || file->IsZombie()) throw std::runtime_error(prefix + "cannot open " + fileName);
198 TObject* object = file->Get(objectName.c_str());
199 if (auto* graph = dynamic_cast<TGraph*>(object)) {
200 std::shared_ptr<TGraph> copy(static_cast<TGraph*>(graph->Clone()));
201 return [copy](double r) { return copy->Eval(r); };
202 }
203 if (auto* histogram = dynamic_cast<TH1*>(object)) {
204 std::shared_ptr<TH1> copy(static_cast<TH1*>(histogram->Clone()));
205 copy->SetDirectory(nullptr);
206 return [copy](double r) { return copy->Interpolate(r); };
207 }
208 throw std::runtime_error(prefix + "no graph or histogram '" + objectName + "' in " + fileName);
209 }
210 // Not added to the list of functions of ROOT (several processes may have the same formula)
211 auto function =
212 std::make_shared<TF1>("lightCaptureProbability", text.c_str(), 0, maxDistance, TF1::EAddToList::kNo);
213 if (!function->IsValid()) {
214 throw std::runtime_error(prefix + "captureProbability '" + text + "' is not a valid formula of x");
215 }
216 return [function](double r) { return function->Eval(r); };
217}

299void TRestDetectorLightCollectionProcess::AddPhotoelectrons(const XYZVector& position, int photoelectrons,
300 double time, int particleID,
301 TRestDetectorHitsEvent& output) {
302 if (photoelectrons <= 0) return;
303 const auto* profile = fLight ? fLight->GetScintillationProfile(particleID) : nullptr;
304 for (int p = 0; p < photoelectrons; p++) {
305 double t = time;
306 if (profile) t += profile->SampleTime(fRandom);
307 if (fWLS) t += fWLS->GetReemissionTime(fRandom);
308 output.AddHit(position, 1.0, t, TRestDetectorHitsEvent::REST_HitType::XYZ, particleID);
309 }
310 fNPhotoelectrons += photoelectrons;
311}

319void TRestDetectorLightCollectionProcess::CaptureLocally(const XYZVector& position, int photons, double time,
320 int particleID, TRestDetectorHitsEvent& output) {
321 const auto channels = fReadout->GetChannelsInRange(position, fMaxDistance);
322 // Multinomial: each channel takes a binomial of the photons left, with its probability relative to
323 // what is left of the total
324 int left = photons;
325 double probabilityLeft = 1;
326 bool collected = false;
327 for (const auto& channel : channels) {
328 // With WLS fibres the capture is by the fibre, half of the light in each direction: a channel
329 // takes the halves of the paths reaching its photosensor
330 std::vector<TRestDetectorReadout::GuidedLightPath> paths;
331 double probability = GetCaptureProbability(channel.distance);
332 if (fWLS && probability > 0) {
333 paths = fReadout->GetGuidedLightPaths(channel.daqID, position);
334 if (paths.empty()) {
335 throw std::runtime_error("TRestDetectorLightCollectionProcess: the channel " +
336 std::to_string(channel.daqID) + " of readout " +
337 fReadout->GetName() +
338 " has no distance to its readout (needed by wlsMedium)");
339 }
340 probability *= std::min(1.0, paths.size() / 2.0);
341 }
342 if (probability <= 0) continue;
343 collected = true;
344 if (probability > probabilityLeft + 1e-9) {
345 throw std::runtime_error(
346 "TRestDetectorLightCollectionProcess: the capture probabilities of the channels around a "
347 "position add up to more than 1 (captureProbability too large for the distance between "
348 "channels)");
349 }
350 if (left <= 0) {
351 probabilityLeft -= probability;
352 continue;
353 }
354 const int captured =
355 probability >= probabilityLeft ? left : Binomial(fRandom, left, probability / probabilityLeft);
356 probabilityLeft -= probability;
357 left -= captured;
358 if (!fWLS) {
359 AddPhotoelectrons(channel.position, Detect(captured), time, particleID, output);
360 continue;
361 }
362 // Along the fibre to its read end: the captured photons shared equally among the paths
363 int share = captured;
364 for (size_t k = 0; k < paths.size(); k++) {
365 const int onPath =
366 k + 1 == paths.size() ? share : Binomial(fRandom, share, 1.0 / (paths.size() - k));
367 share -= onPath;
368 double survival = paths[k].reflected ? fWLS->GetEndReflectivity() : 1.0;
369 const double attenuationLength = fWLS->GetAttenuationLength();
370 if (attenuationLength > 0) survival *= std::exp(-paths[k].distance / attenuationLength);
371 const double lightSpeed = fWLS->GetLightSpeed();
372 const double arrival = lightSpeed > 0 ? time + paths[k].distance / lightSpeed : time;
373 AddPhotoelectrons(channel.position, Detect(Binomial(fRandom, onPath, survival)), arrival,
374 particleID, output);
375 }
376 }
377 if (!collected) fNHitsWithoutChannel++;
378}

380void TRestDetectorLightCollectionProcess::Attenuate(const XYZVector& position, int photons, double time,
381 int particleID, TRestDetectorHitsEvent& output) {
382 const int channel = fReadout->GetChannelFromPosition(position.X(), position.Y(), position.Z());
383 if (channel < 0) {
384 fNHitsWithoutChannel++;
385 return;
386 }
387 const double distance = fReadout->GetDistanceToReadout(channel, position);
388 if (std::isnan(distance)) {
389 throw std::runtime_error("TRestDetectorLightCollectionProcess: the channel " +
390 std::to_string(channel) + " of readout " + fReadout->GetName() +
391 " has no distance to its readout (e.g. no readoutVolume)");
392 }
393 const double attenuationLength = fLight->GetAttenuationLength();
394 const double lightSpeed = fLight->GetEffectiveLightSpeed();
395 const int arriving =
396 attenuationLength > 0 ? Binomial(fRandom, photons, std::exp(-distance / attenuationLength)) : photons;
397 AddPhotoelectrons(position, Detect(arriving), lightSpeed > 0 ? time + distance / lightSpeed : time,
398 particleID, output);
399}

401bool TRestDetectorLightCollectionProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
402 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
403 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
404 if (!inputEvent || !outputEvent) {
405 throw std::runtime_error(
406 "TRestDetectorLightCollectionProcess: input/output is not a TRestDetectorHitsEvent");
407 }
408
409 outputEvent->Initialize();
410 outputEvent->SetEventInfo(inputEvent);
411 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
412 fNPhotoelectrons = 0;
413 fNHitsWithoutChannel = 0;
414
415 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
416 const int photons = static_cast<int>(std::lround(inputEvent->GetEnergy(n)));
417 if (photons <= 0) continue;
418 if (fModel == "localCapture") {
419 CaptureLocally(inputEvent->GetPosition(n), photons, inputEvent->GetTime(n),
420 inputEvent->GetParticleID(n), *outputEvent);
421 } else {
422 Attenuate(inputEvent->GetPosition(n), photons, inputEvent->GetTime(n),
423 inputEvent->GetParticleID(n), *outputEvent);
424 }
425 }
426
427 RESTDebug << "TRestDetectorLightCollectionProcess: event " << inputEvent->GetID() << ", "
428 << fNPhotoelectrons << " photoelectrons, " << fNHitsWithoutChannel << " hits without channel"
429 << RESTendl;
430 return outputEvent->GetNumberOfHits() > 0;
431}
