108int Bin(double value, double min, double max, int n) {
109 if (!(max > min)) return 0;
110 const int bin = static_cast<int>(std::floor((value - min) / (max - min) * n));
111 return std::clamp(bin, 0, n - 1);
112}

115double Sample(const std::vector<int>& counts, double min, double max, TRandom& random) {
116 long total = 0;
117 for (const int count : counts) total += count;
118 if (total <= 0 || !(max > min)) return min;
119 long target = static_cast<long>(random.Uniform() * total);
120 size_t bin = 0;
121 for (; bin + 1 < counts.size(); bin++) {
122 if (target < counts[bin]) break;
123 target -= counts[bin];
124 }
125 const double width = (max - min) / counts.size();
126 return min + (bin + random.Uniform()) * width;
127}

130void WriteHistogram(const std::string& name, const std::string& title, const std::vector<int>& counts,

132 TH1D histogram(name.c_str(), title.c_str(), counts.size(), min, max > min ? max : min + 1);
133 histogram.SetDirectory(nullptr);

171void TRestDetectorAmplificationResponse::SetResult(const std::vector<Avalanche>& avalanches,
172 const Losses& losses) {
173 fPrimaries = static_cast<int>(avalanches.size());
174 fCollected = 0;
175 fLosses = losses;
176 fSizeCounts.assign(fSizeBins, 0);
177 fTimeCounts.assign(fTimeBins, 0);
178 fMeanSize = fSizeVariance = fSizeMax = fTMin = fTMax = 0;
179 fAnodeElectrons = fSharedElectrons = 0;
180 const bool map = fCellBins > 0 && fCellHigh[0] > fCellLow[0] && fCellHigh[1] > fCellLow[1];
181 const size_t cells = map ? static_cast<size_t>(fCellBins) * fCellBins : 0;
182 fCellPrimaries.assign(cells, 0);
183 fCellCollected.assign(cells, 0);
184 fCellSizeSum.assign(cells, 0);
185
186 double sum = 0, sum2 = 0;
187 for (const auto& avalanche : avalanches) {
188 const int cell = map ? GetCellBin((avalanche.fX - fCellLow[0]) / (fCellHigh[0] - fCellLow[0]),
189 (avalanche.fY - fCellLow[1]) / (fCellHigh[1] - fCellLow[1]))
190 : -1;
191 if (map) fCellPrimaries[cell]++;
192 if (!(avalanche.fSize > 0)) continue;
193 if (map) {
194 fCellCollected[cell]++;
195 fCellSizeSum[cell] += avalanche.fSize;
196 }
197 fTMin = fCollected == 0 ? avalanche.fTime : std::min(fTMin, avalanche.fTime);
198 fTMax = fCollected == 0 ? avalanche.fTime : std::max(fTMax, avalanche.fTime);
199 fCollected++;
200 sum += avalanche.fSize;
201 sum2 += avalanche.fSize * avalanche.fSize;
202 fAnodeElectrons += avalanche.fSize;
203 fSharedElectrons += avalanche.fShared;
204 fSizeMax = std::max(fSizeMax, avalanche.fSize);
205 }
206 if (fCollected == 0) return;
207 const double n = fCollected;
208 fMeanSize = sum / n;
209 fSizeVariance = n > 1 ? std::max(0.0, (sum2 - n * fMeanSize * fMeanSize) / (n - 1)) : 0;
210 for (const auto& avalanche : avalanches) {
211 if (!(avalanche.fSize > 0)) continue;
212 fSizeCounts[Bin(avalanche.fSize, 0, fSizeMax, fSizeBins)]++;
213 fTimeCounts[Bin(avalanche.fTime, fTMin, fTMax, fTimeBins)]++;
214 }

238double TRestDetectorAmplificationResponse::GetCellTransparency(int bin) const {
239 if (bin < 0 || static_cast<size_t>(bin) >= fCellPrimaries.size() || fCellPrimaries[bin] <= 0) {
240 return GetTransparency();
241 }
242 return static_cast<double>(fCellCollected[bin]) / fCellPrimaries[bin];
243}

245double TRestDetectorAmplificationResponse::GetCellGainFactor(int bin) const {
246 if (bin < 0 || static_cast<size_t>(bin) >= fCellCollected.size() || fCellCollected[bin] <= 0 ||
247 !(fMeanSize > 0)) {
248 return 1;
249 }
250 return fCellSizeSum[bin] / fCellCollected[bin] / fMeanSize;
251}

253double TRestDetectorAmplificationResponse::GetSharingSigma() const {
254 const double sizeX = fCellHigh[0] - fCellLow[0], sizeY = fCellHigh[1] - fCellLow[1];
255 if (!(sizeX > 0) || !(sizeY > 0) || !(fAnodeElectrons > 0)) return 0;
256 // From a point uniform over a length L, a Gaussian displacement of sigma << L crosses one of its ends
257 // with probability sigma / (sqrt(2 pi) L): two edges across each axis
258 const double fraction = fSharedElectrons / fAnodeElectrons;
259 return fraction * std::sqrt(2 * M_PI) / (2 / sizeX + 2 / sizeY);
260}

324TRestDetectorAmplificationResponse TRestDetectorAmplificationResponse::Read(TFile* file) {

335 response.fPenning = metadata["penning"].as<double>(0);

341 response.fLosses.fDielectric = losses["dielectric"].as<int>(0);
342 response.fLosses.fAttached = losses["attached"].as<int>(0);
343 response.fLosses.fOther = losses["other"].as<int>(0);
344 }
345 response.fMeanSize = metadata["meanSize"].as<double>(0);
346 response.fSizeVariance = metadata["sizeVariance"].as<double>(0);
347 response.fSizeMax = metadata["sizeMax"].as<double>(0);
348 response.fSizeCounts = metadata["sizeCounts"].as<std::vector<int>>(std::vector<int>());
349 response.fTMin = metadata["tMin"].as<double>(0);
350 response.fTMax = metadata["tMax"].as<double>(0);
351 response.fTimeCounts = metadata["timeCounts"].as<std::vector<int>>(std::vector<int>());
352 // The map of the cell (none in the responses computed before it)
353 const auto low = metadata["cellLow"].as<std::vector<double>>(std::vector<double>());
354 const auto high = metadata["cellHigh"].as<std::vector<double>>(std::vector<double>());
355 if (low.size() == 2 && high.size() == 2) {
356 response.fCellLow = {low[0], low[1]};
357 response.fCellHigh = {high[0], high[1]};
358 }
359 response.fCellPrimaries = metadata["cellPrimaries"].as<std::vector<int>>(std::vector<int>());
360 response.fCellCollected = metadata["cellCollected"].as<std::vector<int>>(std::vector<int>());
361 response.fCellSizeSum = metadata["cellSizeSum"].as<std::vector<double>>(std::vector<double>());
362 response.fAnodeElectrons = metadata["anodeElectrons"].as<double>(0);
363 response.fSharedElectrons = metadata["sharedElectrons"].as<double>(0);
364 const size_t cells =
365 response.fCellPrimaries.empty() ? 0 : static_cast<size_t>(response.fCellBins) * response.fCellBins;
366 if (response.fSizeCounts.size() != static_cast<size_t>(response.fSizeBins) ||
367 response.fTimeCounts.size() != static_cast<size_t>(response.fTimeBins) ||
368 response.fCollected > response.fPrimaries || response.fCellPrimaries.size() != cells ||
369 response.fCellCollected.size() != cells || response.fCellSizeSum.size() != cells) {
370 throw std::runtime_error(error + "the response is not consistent (bins or counts)");
371 }
372 return response;
373}
