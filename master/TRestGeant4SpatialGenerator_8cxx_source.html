144XYZVector TRestGeant4SpatialGeneratorVolume::Sample(TRandom& random) {
145 if (IsGdml()) throw runtime_error(GetClassName() + ": a GDML volume needs the Geant4 geometry");
146 const double phi = random.Uniform(0, 2 * M_PI);
147 XYZVector local;
148 if (fShape == "box") {
149 local = {fSize[0] * (random.Uniform() - 0.5), fSize[1] * (random.Uniform() - 0.5),
150 fSize[2] * (random.Uniform() - 0.5)};
151 } else if (fShape == "sphere") {
152 local = fSize[0] * cbrt(random.Uniform()) * IsotropicDirection(random);
153 } else {
154 const double r = fSize[0] * sqrt(random.Uniform());
155 local = {r * cos(phi), r * sin(phi), fSize[1] * (random.Uniform() - 0.5)};
156 }
157 return ToWorld(local);
158}

228XYZVector TRestGeant4SpatialGeneratorCosmic::SampleFromDirection(TRandom& random,
229 const XYZVector& direction) const {
230 if (fRadius <= 0) throw runtime_error(GetClassName() + ": the radius is not set");
231 // Two unit vectors normal to the direction
232 const XYZVector reference = abs(direction.Z()) < 0.9 ? XYZVector(0, 0, 1) : XYZVector(1, 0, 0);
233 const XYZVector v = direction.Cross(reference).Unit();
234 const XYZVector w = direction.Cross(v);
235
236 const double r = fRadius * sqrt(random.Uniform());
237 const double phi = random.Uniform(0, 2 * M_PI);
238 const XYZVector onDisk = r * (cos(phi) * v + sin(phi) * w);
239 return onDisk - sqrt(fRadius * fRadius - r * r) * direction;
240}

275bool MoveToBox(TRestGeant4ParticleState& particle, const XYZVector& half) {
276 const XYZVector origin = particle.GetOrigin();
277 const XYZVector direction = particle.GetDirection().Unit();
278 const double o[3] = {origin.X(), origin.Y(), origin.Z()};
279 const double d[3] = {direction.X(), direction.Y(), direction.Z()};
280 const double h[3] = {half.X(), half.Y(), half.Z()};
281 double enter = -numeric_limits<double>::infinity(), leave = numeric_limits<double>::infinity();
282 for (int i = 0; i < 3; i++) {
283 if (d[i] == 0) {
284 if (abs(o[i]) >= h[i]) return false;
285 continue;
286 }
287 const double t1 = (-h[i] - o[i]) / d[i], t2 = (h[i] - o[i]) / d[i];
288 enter = max(enter, min(t1, t2));
289 leave = min(leave, max(t1, t2));
290 }
291 if (leave <= max(enter, 0.0)) return false;
292 if (enter <= 0) return true; // inside the world
293 // 1 um inside the world, not on its surface
294 const double distance = enter + min(1e-3, (leave - enter) / 2);
295 const double mass = CRYParticleMass(particle.GetParticleName());
296 const double kinetic = particle.GetEnergy();
297 const double beta = mass > 0 ? sqrt(kinetic * (kinetic + 2 * mass)) / (kinetic + mass) : 1;
298 constexpr double lightSpeed = 299792.458; // mm/us
299 particle.SetOrigin(origin + distance * direction);
300 particle.SetTime(particle.GetTime() + distance / (beta * lightSpeed));
301 return true;
302}

379bool TRestGeant4SpatialGeneratorCRY::NextShower(vector<TRestGeant4ParticleState>& particles) {
380 if (!fState) throw runtime_error(GetClassName() + ": CRY is not started");
381 if (!fEngine) {
382 // The CRY of this copy, created in its thread (the random generator of CRY is one per thread)
383 lock_guard<mutex> lock(fState->mutex);
384 fEngine = make_shared<Engine>();
385 const long seed = fState->seed + static_cast<long>(fState->engines.size());
386 fEngine->engine = CRYEngineFactory()(fState->setup, seed);
387 fState->engines.push_back(fEngine);
388 }
389 particles.clear();
390 const XYZVector center = GetCenter();
391 const bool world = fWorldHalfSize.R() > 0;
392 // The showers without a particle crossing the world are skipped (they count in the exposure)
393 while (particles.empty()) {
394 if (fMaxExposureTime > 0 && fState->exposure.load() >= fMaxExposureTime) return false;
395 vector<TRestGeant4ParticleState> shower = fEngine->engine->Generate();
396 fState->showers++;
397 const double exposure = fEngine->engine->GetTimeSimulated();
398 AtomicAdd(fState->exposure, exposure - fEngine->exposure);
399 fEngine->exposure = exposure;
400 for (auto& particle : shower) {
401 particle.SetOrigin(RotateToWorld(particle.GetOrigin()) + center);
402 particle.SetDirection(RotateToWorld(particle.GetDirection()));
403 if (!world || MoveToBox(particle, fWorldHalfSize)) particles.push_back(particle);
404 }
405 // Without the world, a shower is used as it is (an empty one too: CRY returns at least a particle)
406 if (!world) break;
407 }
408 return true;
409}
