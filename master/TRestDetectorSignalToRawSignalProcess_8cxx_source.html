134void TRestDetectorSignalToRawSignalProcess::InitProcess() {
135 const std::string prefix = "TRestDetectorSignalToRawSignalProcess: ";
136 if (fSampling <= 0 || fNPoints <= 0) {
137 throw std::runtime_error(prefix + "sampling and nPoints must be > 0");
138 }
139 if (fTriggerMode == "integralThreshold" && fIntegralThreshold <= 0) {
140 throw std::runtime_error(prefix + "triggerMode integralThreshold needs integralThreshold > 0");
141 }
142 if (fIntegralWindow < 0) throw std::runtime_error(prefix + "integralWindow must be >= 0");
143 fTriggerObservableValue = nullptr;
144 if (fTriggerMode == "observable") {
145 if (fTriggerObservable.empty()) {
146 throw std::runtime_error(prefix + "triggerMode observable needs triggerObservable");
147 }
148 if (!fRunInfo) throw std::runtime_error(prefix + "triggerMode observable needs a TRestRun");
149 fTriggerObservableValue = &fRunInfo->GetObservable<double>(fTriggerObservable);
150 } else if (!fTriggerObservable.empty()) {
151 throw std::runtime_error(prefix + "triggerObservable is only for triggerMode observable");
152 }
153 if (fMaxValue <= kMinADC || fMaxValue > kMaxADC) {
154 throw std::runtime_error(prefix + "maxValue must be in (-32768, 32767]");
155 }
156
157 fADCGain = fGain;
158 fADCOffset = fOffset;
159 const auto& [minEnergy, maxEnergy] = fCalibrationEnergy;
160 const auto& [minRange, maxRange] = fCalibrationRange;
161 if (minEnergy != maxEnergy || minRange != maxRange) {
162 if (minEnergy == maxEnergy || minRange == maxRange) {
163 throw std::runtime_error(prefix +
164 "calibrationEnergy and calibrationRange need two different values");
165 }
166 // ADC(E) = min + range * (minRange + (E - minEnergy) * (maxRange - minRange) / (maxEnergy -
167 // minEnergy))
168 const double range = kMaxADC - kMinADC;
169 fADCGain = range * (maxRange - minRange) / (maxEnergy - minEnergy);
170 fADCOffset = kMinADC + range * minRange - fADCGain * minEnergy;
171 }
172
173 RegisterObservable("TriggerTime", fTriggerTime);
174 RegisterObservable("NSaturatedSignals", fNSaturatedSignals);
175}

177bool TRestDetectorSignalToRawSignalProcess::GetTriggerTime(const TRestDetectorSignalEvent& event,
178 double& time) const {
179 if (fTriggerMode == "fixed") {
180 time = fTriggerFixedStartTime;
181 return true;
182 }
183 if (fTriggerMode == "observable") {
184 time = *fTriggerObservableValue;
185 return std::isfinite(time);
186 }
187
188 // Points with positive charge of all the signals, sorted in time
189 std::vector<std::pair<double, double>> points;
190 for (int s = 0; s < event.GetNumberOfSignals(); s++) {
191 for (const auto& point : event.GetSignal(s).GetPointsVector()) {
192 if (point.second > 0) points.push_back(point);
193 }
194 }
195 if (points.empty()) return false;
196 std::sort(points.begin(), points.end());
197
198 if (fTriggerMode == "firstDeposit") {
199 time = points.front().first;
200 return true;
201 }
202
203 // integralThreshold: charge in (t - window, t] at each point
204 const double window = fIntegralWindow > 0 ? (double)fIntegralWindow : fNPoints * fSampling / 2.;
205 double charge = 0;
206 size_t first = 0;
207 for (const auto& [t, q] : points) {
208 charge += q;
209 while (points[first].first <= t - window) charge -= points[first++].second;
210 if (charge >= fIntegralThreshold) {
211 time = t;
212 return true;
213 }
214 }
215 return false;
216}

218bool TRestDetectorSignalToRawSignalProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
219 const auto* signalEvent = dynamic_cast<const TRestDetectorSignalEvent*>(&input);
220 auto* rawEvent = dynamic_cast<TRestRawSignalEvent*>(&output);
221 if (!signalEvent || !rawEvent) {
222 throw std::runtime_error(
223 "TRestDetectorSignalToRawSignalProcess: input/output is not a TRestDetectorSignalEvent/"
224 "TRestRawSignalEvent");
225 }
226
227 rawEvent->Initialize();
228 rawEvent->SetEventInfo(signalEvent);
229 fNSaturatedSignals = 0;
230 fTriggerTime = 0;
231
232 if (signalEvent->GetNumberOfSignals() <= 0) return false;
233 if (!GetTriggerTime(*signalEvent, fTriggerTime)) {
234 RESTDebug << "TRestDetectorSignalToRawSignalProcess: event " << signalEvent->GetID()
235 << " without trigger" << RESTendl;
236 return false;
237 }
238 const double startTime = fTriggerTime - fTriggerDelay;
239
240 for (int s = 0; s < signalEvent->GetNumberOfSignals(); s++) {
241 const auto& signal = signalEvent->GetSignal(s);
242 auto points = signal.GetPointsVector();
243 std::sort(points.begin(), points.end());
244
245 std::vector<double> charges(fNPoints, 0);
246 if (fMode == "integrate") {
247 for (const auto& [t, q] : points) {
248 const double bin = std::floor((t - startTime) / fSampling);
249 if (bin >= 0 && bin < fNPoints) charges[(int)bin] += q;
250 }
251 } else if (!points.empty()) {
252 size_t i = 0;
253 for (int b = 0; b < fNPoints; b++) {
254 const double t = startTime + b * fSampling;
255 if (t < points.front().first || t > points.back().first) continue;
256 while (i + 1 < points.size() && points[i + 1].first < t) i++;
257 if (i + 1 >= points.size()) {
258 charges[b] = points[i].second;
259 continue;
260 }
261 const auto& [t0, q0] = points[i];
262 const auto& [t1, q1] = points[i + 1];
263 charges[b] = t1 > t0 ? q0 + (t - t0) / (t1 - t0) * (q1 - q0) : q0;
264 }
265 }
266
267 std::vector<short> samples(fNPoints);
268 bool saturated = false;
269 for (int b = 0; b < fNPoints; b++) {
270 double value = std::round(fADCOffset + fADCGain * charges[b]);
271 if (value < kMinADC || value > fMaxValue) {
272 saturated = true;
273 value = std::clamp(value, kMinADC, (double)fMaxValue);
274 }
275 samples[b] = (short)value;
276 }
277 if (saturated) fNSaturatedSignals++;
278 rawEvent->AddSignal(signal.GetSignalID(), samples);
279 }
280
281 return rawEvent->GetNumberOfSignals() > 0;
282}
