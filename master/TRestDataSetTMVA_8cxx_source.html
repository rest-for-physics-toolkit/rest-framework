136double Separation(const TH1D& signal, const TH1D& background) {
137 const double integralS = signal.Integral(), integralB = background.Integral();
138 if (integralS <= 0 || integralB <= 0) return 0;
139 double separation = 0;
140 for (int bin = 1; bin <= signal.GetNbinsX(); bin++) {
141 const double s = signal.GetBinContent(bin) / integralS, b = background.GetBinContent(bin) / integralB;
142 if (s + b > 0) separation += (s - b) * (s - b) / (s + b);
143 }
144 return separation / 2;
145}

148std::pair<double, double> Range(const std::vector<double>& a, const std::vector<double>& b) {
149 double min = std::numeric_limits<double>::max(), max = std::numeric_limits<double>::lowest();
150 for (const auto* values : {&a, &b}) {
151 for (double value : *values) {
152 min = std::min(min, value);
153 max = std::max(max, value);
154 }
155 }
156 if (min > max) return {0, 1};
157 const double margin = max > min ? (max - min) * 1e-6 : 0.5;
158 return {min, max + margin};
159}

171std::unique_ptr<TTree> MakeTree(const std::string& name, const std::vector<std::string>& observables,
172 const std::vector<std::vector<double>>& values) {
173 TDirectory::TContext context(nullptr);
174 auto tree = std::make_unique<TTree>(name.c_str(), name.c_str());
175 std::vector<double> row(observables.size());
176 for (size_t i = 0; i < observables.size(); i++)
177 tree->Branch(observables[i].c_str(), &row[i], (observables[i] + "/D").c_str());
178 const size_t entries = values.empty() ? 0 : values.front().size();
179 for (size_t entry = 0; entry < entries; entry++) {
180 for (size_t i = 0; i < observables.size(); i++) row[i] = values[i][entry];
181 tree->Fill();
182 }
183 tree->ResetBranchAddresses();
184 return tree;

196std::pair<std::vector<double>, std::vector<double>> ReadScores(TTree* tree, const std::string& method,
197 int signalClass) {
198 std::pair<std::vector<double>, std::vector<double>> scores;
199 if (!tree) return scores;
200 Int_t classID = 0;
201 Float_t score = 0;
202 tree->SetBranchAddress("classID", &classID);
203 tree->SetBranchAddress(method.c_str(), &score);
204 for (Long64_t entry = 0; entry < tree->GetEntries(); entry++) {
205 tree->GetEntry(entry);
206 (classID == signalClass ? scores.first : scores.second).push_back(score);
207 }
208 tree->ResetBranchAddresses();

366void TRestDataSetTMVA::Apply(TRestDataSet& dataSet) const {
367 const std::string context = "TRestDataSetTMVA '" + fName + "'";
368 if (fWeights.empty()) throw std::runtime_error(context + ": no weights (not trained)");
369
370 // A reader per slot of the data frame (TMVA::Reader is not thread safe), with the values of its entry
371 struct SlotReader {
372 std::unique_ptr<TMVA::Reader> reader;
373 std::vector<float> values;
374 TMVA::MethodBase* method = nullptr;
375 };
376 const unsigned nSlots = std::max(1u, ROOT::GetThreadPoolSize());
377 const bool debug = fVerboseLevel >= TRestLogManager::REST_Verbose_Level::REST_Debug;
378 const auto methodType = TMVA::Types::Instance().GetMethodType(fMethod.c_str());
379 auto readers = std::make_shared<std::vector<SlotReader>>(nSlots);
380 for (auto& slot : *readers) {
381 slot.reader = std::make_unique<TMVA::Reader>(debug ? "!Color:!Silent" : "!Color:Silent");
382 slot.values.resize(fObservables.size());
383 for (size_t i = 0; i < fObservables.size(); i++)
384 slot.reader->AddVariable(fObservables[i].c_str(), &slot.values[i]);
385 slot.method = dynamic_cast<TMVA::MethodBase*>(slot.reader->BookMVA(methodType, fWeights.c_str()));
386 if (!slot.method)
387 throw std::runtime_error(context + ": the weights of " + fMethod + " cannot be read");
388 }
389
390 // The observables in a vector (hidden column), then the score
391 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
392 std::string inputs = "ROOT::RVecD{";
393 for (size_t i = 0; i < fObservables.size(); i++)
394 inputs += (i ? ", " : "") + std::string("static_cast<double>(") + fObservables[i] + ")";
395 inputs += "}";
396 const std::string inputColumn = fName + "_inputs";
397 frame = DefineColumn(frame, inputColumn, inputs);
398 auto score = [readers](unsigned slot, const ROOT::RVecD& values) {
399 auto& reader = (*readers)[slot];
400 for (size_t i = 0; i < reader.values.size(); i++) reader.values[i] = values[i];
401 return reader.reader->EvaluateMVA(reader.method);
402 };
403 const std::string column = GetColumn();
404 frame = frame.HasColumn(column) ? frame.RedefineSlot(column, score, {inputColumn})
405 : frame.DefineSlot(column, score, {inputColumn});
406 dataSet.SetDataFrame(frame, {column});
407}
