77bool TRestRawSignalShapingProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
78 const auto* inEvent = dynamic_cast<const TRestRawSignalEvent*>(&input);
79 auto* outEvent = dynamic_cast<TRestRawSignalEvent*>(&output);
80 if (!inEvent || !outEvent) {
81 throw std::runtime_error("TRestRawSignalShapingProcess: input/output is not a TRestRawSignalEvent");
82 }
83
84 // An event without signals passes (empty), as in the other signal processes
85 const int nSignals = inEvent->GetNumberOfSignals();
86 outEvent->CopyFrom(&input);
87
88 constexpr long kMin = std::numeric_limits<short>::min();
89 constexpr long kMax = std::numeric_limits<short>::max();
90
91 const bool useBaseline = fBaselineRange.second > fBaselineRange.first;
92
93 for (int s = 0; s < nSignals; ++s) {
94 const std::vector<short> data = inEvent->GetSignal(s).GetData();
95
96 double baseline = 0, sigma = 0;
97 std::vector<double> centered(data.begin(), data.end());
98 if (useBaseline) {
99 TRestPulseShapeAnalysis::GetBaselineSigma(data, fBaselineRange.first, fBaselineRange.second,
100 baseline, sigma);
101 for (double& v : centered) v -= baseline;
102 }
103
104 std::vector<double> shaped =
105 TRestPulseShapeAnalysis::Convolve(centered, fResponse, fResponseCenter, !useBaseline);
106 if (useBaseline) {
107 for (double& v : shaped) v += baseline;
108 }
109
110 auto& outSig = outEvent->GetSignal(s);
111 const int nPoints = std::min((int)shaped.size(), outSig.GetNPoints());
112 for (int b = 0; b < nPoints; ++b) {
113 const short newSample = static_cast<short>(std::clamp(std::lround(shaped[b]), kMin, kMax));
114 outSig.IncreaseBinBy(b, newSample - outSig.GetPoint(b));
115 }
116 }
117
118 return true;
119}
