301vector<TRestGeant4ParticleState> TRestGeant4PrimaryGenerator::Generate() {
302 // The particles and positions of a new shower of CRY
303 if (auto* cry = dynamic_cast<TRestGeant4SpatialGeneratorCRY*>(fSpatial.get())) {
304 vector<TRestGeant4ParticleState> particles;
305 cry->NextShower(particles);
306 return particles;
307 }
308
309 const auto* cosmic = dynamic_cast<const TRestGeant4SpatialGeneratorCosmic*>(fSpatial.get());
310 // The cosmic origin depends on the direction, it is set below
311 const XYZVector position = cosmic ? XYZVector() : SamplePosition();
312
313 vector<TRestGeant4ParticleState> particles;
314 for (auto& source : fSources) {
315 const XYZVector center = fSpatial->GetCenter();
316 auto sourceParticles = source.Generate(fRandom, position, center, fSpatial->IsInward());
317 for (auto& particle : sourceParticles) {
318 if (cosmic) particle.SetOrigin(cosmic->SampleFromDirection(fRandom, particle.GetDirection()));
319 particles.push_back(particle);
320 }
321 }
322 return particles;
323}

329XYZVector TRestGeant4PrimaryGenerator::SamplePosition() {
330 const auto sample = [this]() {
331 return fSpatial->IsGdml() ? SampleGdmlPosition() : fSpatial->Sample(fRandom);
332 };
333 if (!fDensity) return sample();
334
335 // The density accepts each position with its probability
336 constexpr int maxAttempts = 10000000;
337 for (int attempt = 0; attempt < maxAttempts; attempt++) {
338 const XYZVector position = sample();
339 const double density = fDensity->Eval(position.X(), position.Y(), position.Z());
340 if (density > 1) {
341 throw runtime_error("TRestGeant4PrimaryGenerator: spatialDensityFunction > 1 at (" +
342 to_string(position.X()) + ", " + to_string(position.Y()) + ", " +
343 to_string(position.Z()) + ")");
344 }
345 if (fRandom.Uniform() < density) return position;
346 }
347 throw runtime_error("TRestGeant4PrimaryGenerator: no position accepted by spatialDensityFunction");
348}
