81std::optional<double> ParseValueWithUnits(const std::string& text) {
82 // The longest number at the start: "10eV" is 10 and "eV" (an incomplete exponent is not part of the
83 // number), which a stream (>> double) fails to read. Read without the locale (classic). Not
84 // std::from_chars: the libc++ of Xcode 16 has no floating point from_chars
85 const char* begin = text.data();
86 const char* end = text.data() + text.size();
87 while (begin != end && std::isspace(static_cast<unsigned char>(*begin))) begin++;
88 const auto digits = [end](const char* p) {
89 while (p != end && std::isdigit(static_cast<unsigned char>(*p))) p++;
90 return p;
91 };
92 const char* rest = begin;
93 if (rest != end && (*rest == '+' || *rest == '-')) rest++;
94 const char* mantissa = rest;
95 rest = digits(rest);
96 bool hasDigits = rest != mantissa;
97 if (rest != end && *rest == '.') {
98 const char* fraction = rest + 1;
99 rest = digits(fraction);
100 hasDigits = hasDigits || rest != fraction;
101 }
102 if (!hasDigits) return std::nullopt;
103 if (rest != end && (*rest == 'e' || *rest == 'E')) {
104 const char* exponent = rest + 1;
105 if (exponent != end && (*exponent == '+' || *exponent == '-')) exponent++;
106 const char* exponentEnd = digits(exponent);
107 if (exponentEnd != exponent) rest = exponentEnd;
108 }
109 std::istringstream number(std::string(begin, rest));
110 number.imbue(std::locale::classic());
111 double value = 0;
112 if (!(number >> value)) return std::nullopt;
113 const std::string units = TRestTools::Trim(std::string(rest, end));
114 if (!units.empty()) {
115 try {
116 value *= ParseUnit(units);
117 } catch (const std::invalid_argument&) {
118 return std::nullopt;
119 }
120 }
121 return value;
