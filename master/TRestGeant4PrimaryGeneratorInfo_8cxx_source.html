36void TRestGeant4PrimaryGeneratorInfo::CopyGenerator(const TRestGeant4PrimaryGeneratorInfo& other) {
37 // Its own section: writing the copy must not modify the section of the original (nor its sources)
38 fNode = YAML::Clone(other.fNode);
39 fSpatialDensityFunction = other.fSpatialDensityFunction;
40 fSources.clear();
41 for (const auto& source : other.fSources) fSources.push_back(source.Clone());
42 // Not rebuilt from the section: SetGdmlPlacement and SetCosmicRadius modify it
43 fSpatialGenerator = other.fSpatialGenerator ? other.fSpatialGenerator->Clone() : nullptr;
44 // It writes to the section of the copy (e.g. the cry exposure), not to the one of the original
45 if (fSpatialGenerator) fSpatialGenerator->SetYAMLNode(fNode);
46}

48void TRestGeant4PrimaryGeneratorInfo::LoadConfig() {
49 // The sources (`source`, one or a list) are a registered field
50 fSources.clear();
51 UpdateParamsFromYAML<TRestGeant4PrimaryGeneratorInfo>(fNode);
52
53 // The spatial generator reads its fields (type, from, shape...) from the generator section itself
54 fSpatialGenerator = TRestGeant4SpatialGenerator::Create("generator", fNode);
55
56 if (IsCRY()) {
57 if (!fSources.empty()) {
58 throw runtime_error(GetClassName() + ": the cry generator has no 'source' (CRY gives them)");
59 }
60 } else if (fSources.empty()) {
61 throw runtime_error(GetClassName() + ": no 'source' in the generator");
62 }
63
64 if (!fSpatialDensityFunction.empty() &&
65 !dynamic_cast<const TRestGeant4SpatialGeneratorShape*>(fSpatialGenerator.get())) {
66 throw runtime_error(GetClassName() +
67 ": 'spatialDensityFunction' needs a volume or surface generator");
68 }
69 if (fActivity < 0) throw runtime_error(GetClassName() + ": 'activity' must be positive");
70 if (fActivity > 0 && (IsCosmic() || IsCRY())) {
71 throw runtime_error(GetClassName() +
72 ": 'activity' is not used by the cosmic and cry generators (their exposure is "
73 "computed from their flux)");
74 }
75 if (fActivity > 0 && GetSurfaceRate() > 0) {
76 throw runtime_error(GetClassName() +
77 ": 'activity' and 'flux' given, only one of them can set the rate");
78 }
79 if (IsCosmic() && (fSources.size() != 1 || fSources[0].IsDecay0())) {
80 throw runtime_error(GetClassName() +
81 ": the cosmic generator needs a single source, not a Decay0 one");
82 }
83
84 UpdateYAMLFromParams<TRestGeant4PrimaryGeneratorInfo>(fNode);
85}

91void TRestGeant4PrimaryGeneratorInfo::LoadDistributions(long seed) {
92 for (auto& source : fSources) source.LoadDistributions();
93 // The cosmic generator (a single source): the flux of its source, to normalize to a time
94 if (auto* cosmic = dynamic_cast<TRestGeant4SpatialGeneratorCosmic*>(fSpatialGenerator.get())) {
95 if (fSources.size() == 1) cosmic->SetFlux(fSources[0].GetFlux());
96 }
97 if (auto* cry = dynamic_cast<TRestGeant4SpatialGeneratorCRY*>(fSpatialGenerator.get())) cry->Start(seed);
98}

195void TRestGeant4PrimaryGeneratorInfo::Print() const {
196 if (!fSpatialGenerator) return;
197 RESTMetadata << "Generator: " << fSpatialGenerator->GetClassName() << RESTendl;
198 if (IsGdml()) RESTMetadata << "Generator volume: " << GetGdmlVolume() << RESTendl;
199 if (IsCosmic()) {
200 RESTMetadata << "Cosmic generation surface: " << GetCosmicSurfaceCm2() << " cm2, flux "
201 << GetCosmicFlux() << " counts/s/cm2" << RESTendl;
202 }
203 if (GetSurfaceRate() > 0) {
204 RESTMetadata << "Surface flux: " << GetSurfaceRate() << " primaries/s" << RESTendl;
205 }
206 if (IsCRY()) {
207 const auto* cry = dynamic_cast<const TRestGeant4SpatialGeneratorCRY*>(fSpatialGenerator.get());
208 RESTMetadata << "CRY: " << cry->GetCRYSetup() << RESTendl;
209 }
210 if (!fSpatialDensityFunction.empty()) {
211 RESTMetadata << "Density: " << fSpatialDensityFunction << RESTendl;
212 }
213 for (const auto& source : fSources) {
214 if (source.IsDecay0()) {
215 RESTMetadata << "Source: Decay0 file " << source.fUse << RESTendl;
216 } else {
217 RESTMetadata << "Source: " << source.fParticleName << ", "
218 << source.GetEnergyDistribution()->GetClassName() << ", "
219 << source.GetAngularDistribution()->GetClassName() << RESTendl;
220 }
