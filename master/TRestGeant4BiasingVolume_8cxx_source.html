52double TRestGeant4BiasingVolume::GetEntryFraction(const ROOT::Math::XYZVector& from,
53 const ROOT::Math::XYZVector& to) const {
54 if (isInside(from.X(), from.Y(), from.Z())) return -1;
55
56 const ROOT::Math::XYZVector p = from - GetBiasingVolumePosition();
57 const ROOT::Math::XYZVector d = to - from;
58
59 if (fVolumeType == "virtualBox") {
60 // Slab method: the segment enters the box at the largest entry of the three slabs
61 const double half = fVolumeSize / 2.0;
62 const double origin[3] = {p.X(), p.Y(), p.Z()};
63 const double direction[3] = {d.X(), d.Y(), d.Z()};
64 double tEnter = -std::numeric_limits<double>::infinity();
65 double tExit = std::numeric_limits<double>::infinity();
66 for (int i = 0; i < 3; i++) {
67 if (direction[i] == 0) {
68 if (origin[i] <= -half || origin[i] >= half) return -1; // parallel, outside the slab
69 continue;
70 }
71 double t1 = (-half - origin[i]) / direction[i];
72 double t2 = (half - origin[i]) / direction[i];
73 if (t1 > t2) std::swap(t1, t2);
74 tEnter = std::max(tEnter, t1);
75 tExit = std::min(tExit, t2);
76 }
77 if (tEnter >= tExit || tExit <= 0 || tEnter > 1) return -1;
78 return std::max(tEnter, 0.0);
79 } else if (fVolumeType == "virtualSphere") {
80 // |p + t d|^2 = r^2, the first root is the entry (p is outside)
81 const double a = d.Mag2();
82 if (a == 0) return -1;
83 const double b = 2 * p.Dot(d);
84 const double c = p.Mag2() - fVolumeSize * fVolumeSize;
85 const double discriminant = b * b - 4 * a * c;
86 if (discriminant <= 0) return -1;
87 const double t = (-b - std::sqrt(discriminant)) / (2 * a);
88 if (t < 0 || t > 1) return -1;
89 return t;
90 }
91 return -1;
92}

94bool TRestGeant4BiasingVolume::Contains(const TRestGeant4BiasingVolume& inner) const {
95 const ROOT::Math::XYZVector d = inner.GetBiasingVolumePosition() - GetBiasingVolumePosition();
96 const double offset[3] = {std::abs(d.X()), std::abs(d.Y()), std::abs(d.Z())};
97 // Extent of the inner volume along each axis from its center (half side or radius)
98 const bool innerBox = inner.fVolumeType == "virtualBox";
99 const double innerExtent = innerBox ? inner.fVolumeSize / 2.0 : double(inner.fVolumeSize);
100
101 if (fVolumeType == "virtualBox") {
102 const double half = fVolumeSize / 2.0;
103 for (int i = 0; i < 3; i++) {
104 if (offset[i] + innerExtent >= half) return false;
105 }
106 return true;
107 } else if (fVolumeType == "virtualSphere") {
108 if (innerBox) {
109 // The farthest corner of the box
110 double r2 = 0;
111 for (int i = 0; i < 3; i++) r2 += (offset[i] + innerExtent) * (offset[i] + innerExtent);
112 return std::sqrt(r2) < fVolumeSize;
113 }
114 return d.R() + innerExtent < fVolumeSize;
115 }
116 return false;
117}
