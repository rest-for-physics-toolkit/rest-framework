149bool TRestRawToDetectorSignalProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
150 const auto* rawEvent = dynamic_cast<const TRestRawSignalEvent*>(&input);
151 auto* signalEvent = dynamic_cast<TRestDetectorSignalEvent*>(&output);
152 if (!rawEvent || !signalEvent) {
153 throw std::runtime_error(
154 "TRestRawToDetectorSignalProcess: input/output is not a TRestRawSignalEvent/"
155 "TRestDetectorSignalEvent");
156 }
157
158 signalEvent->Initialize();
159 signalEvent->SetEventInfo(&input);
160
161 fNSignalsRejected = 0;
162 fSignalsID.clear();
163 fSignalsPhysicalID.clear();
164 fSignalsMaxAmplitude.clear();
165 const auto thrParams = std::make_pair(fPointsThreshold, fSignalThreshold);
166
167 for (int s = 0; s < rawEvent->GetNumberOfSignals(); s++) {
168 const auto& rawSignal = rawEvent->GetSignal(s);
169 const auto data = rawSignal.GetData();
170
171 std::vector<double> times, amplitudes;
172 if (fZeroSuppression) {
173 double baseline, baselineSigma;
174 auto corrected = CalculateBaselineAndSigma(data, fBaselineRange.first, fBaselineRange.second,
175 baseline, baselineSigma, fBaselineOption);
176 auto range = fIntegralRange; // clamped to the signal size by GetPointsOverThreshold
177 for (const auto& [bin, value] : GetPointsOverThreshold(
178 corrected, range, thrParams, fPointsOverThreshold, fNPointsFlat, baselineSigma)) {
179 times.push_back(fTriggerStarts + fSampling * bin);
180 amplitudes.push_back(fGain * value);
181 }
182 } else {
183 for (size_t p = 0; p < data.size(); p++) {
184 if (data[p] > fThreshold) {
185 times.push_back(fTriggerStarts + fSampling * p);
186 amplitudes.push_back(fGain * data[p]);
187 }
188 }
189 }
190
191 if (times.empty()) {
192 fNSignalsRejected++;
193 continue;
194 }
195 signalEvent->AddSignal(rawSignal.GetSignalID(), times, amplitudes);
196 fSignalsID.push_back(rawSignal.GetSignalID());
197 fSignalsMaxAmplitude.push_back(*std::max_element(amplitudes.begin(), amplitudes.end()));
198 if (fReadout) fSignalsPhysicalID.push_back(fReadout->GetPhysicalID(rawSignal.GetSignalID()));
199 }
200
201 return fKeepEmptyEvents || signalEvent->GetNumberOfSignals() > 0;
202}
