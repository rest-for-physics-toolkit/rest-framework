105int GetSecondsFromTimeExpression(const string& expression) {
106 // expression is of the form "20h", "10m", "30s" etc.
107 const regex timeRegex("^(\\d+)([hms])$");
108 smatch matches;
109 if (regex_match(expression, matches, timeRegex)) {
110 const int time = stoi(matches[1].str());
111 const string modifier = matches[2].str();
112
113 if (modifier == "h") {
114 return time * 60 * 60;
115 } else if (modifier == "m") {
116 return time * 60;
117 } else if (modifier == "s") {
118 return time;
119 }
120 }
121
122 return 0;
123}

125int GetSecondsFromFullTimeExpression(const string& expression) {
126 // expression is of the form "1h20m30s", "1h", "20m30s", "10s" etc.
127 int seconds = 0;
128 const regex fullTimeRegex("^(\\d+h)?(\\d+m)?(\\d+s)?$");
129 smatch matches;
130 if (regex_match(expression, matches, fullTimeRegex)) {
131 for (int i = 1; i <= 3; i++) {
132 if (matches[i].matched) seconds += GetSecondsFromTimeExpression(matches[i].str());
133 }
134 }
135
136 return seconds;
137}

139int GetNumberFromStringScientific(string& s) {
140 // Allows to parse inputs of the form 1E5, 1.5E10 etc.
141 regex rgx("^([0-9]+.?([0-9]+)?)E([0-9]+)");
142 std::smatch matches;
143 if (std::regex_search(s, matches, rgx)) {
144 return int(stod(matches[1].str()) * TMath::Power(10, stoi(matches[3].str())));
145 } else {
146 return stoi(s);
147 }

150Options ProcessCommandLineOptions(int argc, char* const argv[]) {
151 Options options;
152 options.argc = argc;
153 options.argv = const_cast<char**>(argv);
154
155 if (argc < 2) {
156 // Invoked without parameter
157 ShowUsage();
158 exit(0);
159 }
160
161 // See https://cplusplus.com/articles/DEN36Up4/
162 for (int i = 1; i < argc; i++) {
163 string arg = argv[i];
164 if ((arg == "-h") || (arg == "--help")) {
165 ShowUsage();
166 exit(0);
167 } else if (arg == "--version") {
168 cout << REST_Version::GetSummary() << endl;
169 exit(0);
170 } else if ((arg == "-c") || (arg == "--config")) {
171 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
172 options.configFile =
173 argv[++i]; // Increment 'i' so we don't get the argument as the next argv[i].
174 } else {
175 cerr << "--config option requires one argument" << endl;
176 exit(1);
177 }
178 } else if ((arg == "-o") || (arg == "--output")) {
179 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
180 options.outputFile =
181 argv[++i]; // Increment 'i' so we don't get the argument as the next argv[i].
182 } else {
183 cerr << "--output option requires one argument" << endl;
184 exit(1);
185 }
186 } else if ((arg == "-g") || (arg == "--geometry")) {
187 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
188 options.geometryFile =
189 argv[++i]; // Increment 'i' so we don't get the argument as the next argv[i].
190 } else {
191 cerr << "--geometry option requires one argument" << endl;
192 exit(1);
193 }
194 } else if ((arg == "-i") || (arg == "--interactive")) {
195 options.interactive = true;
196 // TODO: not yet implemented
197 cout << "--interactive option not yet implemented" << endl;
198 exit(1);
199 } else if ((arg == "-j") || (arg == "--threads") || (arg == "-t")) {
200 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
201 options.nThreads =
202 stoi(argv[++i]); // Increment 'i' so we don't get the argument as the next argv[i].
203 if (options.nThreads < 0) {
204 cout << "--threads option error: number of threads must be >= 0" << endl;
205 exit(1);
206 }
207 } else {
208 cerr << "--threads option requires one argument." << endl;
209 exit(1);
210 }
211 } else if ((arg == "-n") || (arg == "--events")) {
212 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
213 string s = argv[++i]; // Increment 'i' so we don't get the argument as the next argv[i].
214 options.nEvents = GetNumberFromStringScientific(s);
215 if (options.nEvents <= 0) {
216 cout << "--events option error: number of events must be > 0 (input: " << options.nEvents
217 << " from " << s << ")" << endl;
218 exit(1);
219 }
220 } else {
221 cerr << "--events option requires one argument." << endl;
222 exit(1);
223 }
224 } else if ((arg == "-e") || (arg == "--entries")) {
225 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
226 string s = argv[++i]; // Increment 'i' so we don't get the argument as the next argv[i].
227 options.nRequestedEntries = GetNumberFromStringScientific(s);
228 if (options.nRequestedEntries <= 0) {
229 cout << "--entries option error: number of entries must be > 0 (input: "
230 << options.nRequestedEntries << " from " << s << ")" << endl;
231 exit(1);
232 }
233 } else {
234 cerr << "--entries option requires one argument." << endl;
235 exit(1);
236 }
237 } else if ((arg == "-s") || (arg == "--seed")) {
238 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
239 options.seed =
240 stoi(argv[++i]); // Increment 'i' so we don't get the argument as the next argv[i].
241 if (options.seed <= 0) {
242 cout << "--seed option error: seed must be positive number" << endl;
243 exit(1);
244 }
245 } else {
246 cerr << "--seed option requires one argument." << endl;
247 exit(1);
248 }
249 } else if (arg == "--time") {
250 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
251 options.timeLimitSeconds = GetSecondsFromFullTimeExpression(
252 argv[++i]); // Increment 'i' so we don't get the argument as the next argv[i].
253 if (options.timeLimitSeconds <= 0) {
254 cout << "--time option error: time limit must be of the format 1h20m30s, 10m20s, 1h, etc."
255 << endl;
256 exit(1);
257 }
258 } else {
259 cerr << "--time option requires one argument." << endl;
260 exit(1);
261 }
262 } else if (arg == "--set") {
263 if (i + 1 < argc) {
264 try {
265 TRestYAMLUtils::AddParameterOverride(options.parameters, argv[++i]);
266 } catch (const std::exception& error) {
267 cerr << error.what() << endl;
268 exit(1);
269 }
270 } else {
271 cerr << "--set option requires one argument (name=value)." << endl;
272 exit(1);
273 }
274 } else if (arg == "--update-gdml") {
275 options.updateGdml = true;
276 } else if (arg == "--runNumber") {
277 if (i + 1 < argc) { // Make sure we aren't at the end of argv!
278 options.runNumber =
279 stoi(argv[++i]); // Increment 'i' so we don't get the argument as the next argv[i].
280 if (options.runNumber < 0) {
281 cout << "--runNumber option error: runNumber cannot be negative" << endl;
282 exit(1);
283 }
284 } else {
285 cerr << "--runNumber option requires one argument." << endl;
286 exit(1);
287 }
288 } else {
289 const string argument = argv[i];
290 if (argument[0] == '-') {
291 cerr << "Bad CLI option '" << argument << "'" << endl << endl;
292 ShowUsage();
293 exit(1);
294 }
295 if (!options.configFile.empty()) {
296 // more than one configuration file specified
297 cerr << "Attempting to set configuration file to '" << argument
298 << "' but it is already defined as '" << options.configFile
299 << "'. It can only be specified once" << endl;
300 exit(1);
301 }
302 options.configFile = argv[i]; // invoked as restG4 <configFile>, restG4 -j 4 <configFile>, etc.
303 }
304 }
305
306 if (options.configFile.empty()) {
307 cerr << "Input configuration file not specified" << endl;
308 exit(1);
309 }
310
311 return options;
312}

322void CheckSourceParticles(const TRestGeant4Metadata& metadata) {
323 for (const auto& source : metadata.GetGeant4PrimaryGeneratorInfo().fSources) {
324 if (source.IsDecay0()) continue;
325 const double excitation = source.fExcitedLevel;
326 if (PrimaryGeneratorAction::FindGeant4Particle(source.fParticleName, excitation) == nullptr) {
327 cerr << "Source particle '" << source.fParticleName
328 << "' not found in Geant4: a particle (e.g. alpha, e-, gamma) or an ion by its element and "
329 "mass number (e.g. Po210, Rn222, not 210Po)"
330 << endl;
331 exit(1);
332 }
333 }

672void Application::RunBiasingStages(int64_t nEvents) {
673 auto* metadata = fSimulationManager.GetRestMetadata();
674 const size_t nVolumes = metadata->GetNumberOfBiasingVolumes();
675 G4UImanager* UI = G4UImanager::GetUIpointer();
676
677 int64_t eventsInStage = nEvents;
678 for (size_t stage = 0; stage <= nVolumes; stage++) {
679 cout << "Biasing stage " << stage << " of " << nVolumes << ": " << eventsInStage << " events"
680 << (stage == 0 ? " of the generator" : " replaying the particles stored") << endl;
681 UI->ApplyCommand("/run/beamOn " + to_string(eventsInStage));
682
683 if (stage == 0 && fSimulationManager.GetBiasingPrimariesInside() > 0) {
684 cout << "WARNING: " << fSimulationManager.GetBiasingPrimariesInside()
685 << " primaries started inside the biasing volume 0 (never stopped by it): the generator "
686 "should be outside the outermost biasing volume"
687 << endl;
688 }
689 if (stage == nVolumes) break;
690
691 if (fSimulationManager.GetAbortFlag()) {
692 // The maxExposureTime of CRY is the end of the primaries, not an abort
693 if (stage == 0 && fSimulationManager.IsGeneratorExhausted()) {
694 fSimulationManager.ResetAbortFlag();
695 } else {
696 cerr << "ERROR: the simulation was stopped (time limit or CTRL-C) in the biasing stage "
697 << stage << " of " << nVolumes << ", before the last one: the output has no events"
698 << endl;
699 return;
700 }
701 }
702
703 const int64_t factor = static_cast<int64_t>(metadata->GetBiasingVolume(stage).GetBiasingFactor());
704 const int64_t particles =
705 static_cast<int64_t>(fSimulationManager.GetNumberOfStoredBiasingParticles());
706 if (particles == 0) {
707 cerr << "ERROR: no particle entered the biasing volume " << stage
708 << " in its energy range in the biasing stage " << stage
709 << ": the next stages are not simulated and the output has no events" << endl;
710 return;
711 }
712 if (particles > numeric_limits<int>::max() / factor) {
713 cerr << "ERROR: the biasing stage " << stage + 1 << " would simulate " << particles << " x "
714 << factor << " events, above the Geant4 limit (" << numeric_limits<int>::max()
715 << "): reduce nEvents or the biasingFactor of the volume " << stage << endl;
716 return;
717 }
718 eventsInStage = fSimulationManager.StartNextBiasingStage();
719 }
720}

722void Application::ValidateOutputFile(const string& filename) {
723 bool error = false;
724
725 auto run = TRestRun(filename);
726
727 TRestEvent* restEvent = run.GetEvent();
728
729 if (restEvent == nullptr) {
730 error = true;
731 cerr << "Error: No input event could be retrieved from the file." << endl;
732 } else {
733 TRestGeant4Event* g4Event = dynamic_cast<TRestGeant4Event*>(restEvent);
734
735 if (g4Event == nullptr) {
736 error = true;
737 cerr << "Error: The event was found, but it is NOT a TRestGeant4Event." << endl;
738 } else {
739 g4Event->SetRestRun(&run);
740 const auto* metadata = g4Event->GetGeant4Metadata();
741 if (!metadata) {
742 std::cerr << "'TRestGeant4Metadata' not found in file\n";
743 error = true;
744 } else {
745 if (!metadata->fGeant4GeometryInfo.GetGeometry()) {
746 std::cerr << "'Geometry' not found in file\n";
747 error = true;
748 }
749 }
750 }
751 }
752
753 const auto analysisTree = run.fAnalysisTree;
754 if (analysisTree == nullptr) {
755 error = true;
756 cerr << "'AnalysisTree' not found in output file" << endl;
757 }
758 if (!run.GetMetadataClass("TRestGeant4PhysicsLists")) {
759 error = true;
760 cerr << "'PhysicsList' metadata not found in output file" << endl;
761 }
762
763 if (error) {
764 run.fInputFile->ls();
765 exit(1);
766 }
