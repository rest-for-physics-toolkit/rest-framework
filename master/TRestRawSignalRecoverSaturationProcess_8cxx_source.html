111bool TRestRawSignalRecoverSaturationProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
112 output.CopyFrom(&input);
113
114 auto* sigEvent = dynamic_cast<TRestRawSignalEvent*>(&output);
115 if (!sigEvent) {
116 throw std::runtime_error(
117 "TRestRawSignalRecoverSaturationProcess: output is not a TRestRawSignalEvent");
118 }
119
120 fObs.clear();
121
122 const auto thrParams = std::make_pair(fPointsThreshold, fSignalThreshold);
123 const bool debug = GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug;
124 auto skip = [&](const TRestRawSignal& sig, const std::string& reason) {
125 if (debug)
126 std::cout << "recSat: event " << sigEvent->GetID() << " signal " << sig.GetSignalID()
127 << " not recovered: " << reason << std::endl;
128 };
129
130 for (int s = 0; s < sigEvent->GetNumberOfSignals(); s++) {
131 auto& signal = sigEvent->GetSignal(s);
132 const std::vector<short> data = signal.GetData();
133 const int nPoints = (int)data.size();
134 if (nPoints == 0) continue;
135
136 const int maxPeakBin = (int)(std::max_element(data.begin(), data.end()) - data.begin());
137 short maxValue = data[maxPeakBin];
138
139 // Saturated bins: consecutive bins equal to the maximum, starting at its first occurrence
140 std::vector<int> saturatedBins;
141 for (int i = maxPeakBin; i < nPoints && data[i] == maxValue; i++) saturatedBins.push_back(i);
142
143 if (maxValue < fMinSaturationValue && !fProcessAllSignals) continue;
144
145 if ((int)saturatedBins.size() < fMinSaturatedBins && !fProcessAllSignals) {
146 skip(signal, "only " + std::to_string(saturatedBins.size()) + " consecutive bins at maximum " +
147 std::to_string(maxValue));
148 continue;
149 }
150
151 fObs.SignalsSaturated++;
152
153 if (fProcessAllSignals && (int)saturatedBins.size() < fMinSaturatedBins) {
154 saturatedBins.clear();
155 const int from = std::max(0, maxPeakBin - fNBinsIfNotSaturated / 2);
156 const int to = std::min(nPoints, maxPeakBin + fNBinsIfNotSaturated / 2);
157 for (int i = from; i < to; i++) saturatedBins.push_back(i);
158 if (saturatedBins.empty()) continue;
159 maxValue = data[saturatedBins.front()];
160 }
161
162 double baseline = 0, sigma = 0;
163 if (!GetBaselineSigma(data, fBaselineRange.first, fBaselineRange.second, baseline, sigma,
164 fBaselineOption)) {
165 skip(signal, "baseline calculation failed");
166 continue;
167 }
168
169 std::vector<float> subtracted(nPoints);
170 for (int i = 0; i < nPoints; i++) subtracted[i] = data[i] - baseline;
171 std::pair<int, int> range = {0, nPoints};
172 const auto pointsOverThr =
173 GetPointsOverThreshold(subtracted, range, thrParams, fPointsOverThreshold, fNPointsFlat, sigma);
174
175 if (pointsOverThr.empty()) {
176 skip(signal, "no points over threshold (baseline " + std::to_string(baseline) + ", sigma " +
177 std::to_string(sigma) + ")");
178 continue;
179 }
180
181 // Graph with the pulse range (points over threshold plus a margin) without the saturated bins
182 std::vector<bool> isSaturated(nPoints, false);
183 for (int b : saturatedBins) isSaturated[b] = true;
184 TGraph g;
185 const int fitStart = std::max(0, pointsOverThr.front().first - fFitMargin);
186 const int fitEnd = std::min(nPoints - 1, pointsOverThr.back().first + fFitMargin);
187 for (int i = fitStart; i <= fitEnd; i++) {
188 if (!isSaturated[i]) g.AddPoint(i, data[i]);
189 }
190
191 const auto fit = FitAget(g, false, baseline);
192 if (!fit) {
193 skip(signal, "fit failed or invalid (" + std::to_string(g.GetN()) + " points, " +
194 std::to_string(saturatedBins.size()) + " saturated bins)");
195 continue;
196 }
197
198 // A saturated pulse is at or above the saturation value in all the saturated bins, a fit that is
199 // below it (beyond the baseline fluctuations) in any of them does not describe the pulse
200 const bool isSaturatedSignal = (int)saturatedBins.size() >= fMinSaturatedBins;
201 if (isSaturatedSignal) {
202 auto below = std::find_if(saturatedBins.begin(), saturatedBins.end(),
203 [&](int b) { return fit->Eval(b) < maxValue - sigma; });
204 if (below != saturatedBins.end()) {
205 std::string pars;
206 for (int i = 0; i < fit->GetNpar(); i++)
207 pars +=
208 std::string(" ") + fit->GetParName(i) + "=" + std::to_string(fit->GetParameter(i));
209 skip(signal, "bad fit, below saturation at bin " + std::to_string(*below) + " (fit " +
210 std::to_string(fit->Eval(*below)) + ", saturation " +
211 std::to_string(maxValue) + ", saturated bins " +
212 std::to_string(saturatedBins.front()) + "-" +
213 std::to_string(saturatedBins.back()) + ";" + pars + ")");
214 continue;
215 }
216 }
217
218 bool anyBinRecovered = false;
219 double signalAddedAmplitude = 0, signalAddedIntegral = 0;
220 std::vector<int> deltas(nPoints, 0);
221 for (int b : saturatedBins) {
222 // Difference to each bin value (the window of processAllSignals is not flat), keeping the
223 // recovered value within the range of a short sample
224 const double value =
225 std::clamp(fit->Eval(b) - data[b], (double)SHRT_MIN - data[b], (double)SHRT_MAX - data[b]);
226 if (value > 0 || fProcessAllSignals) {
227 deltas[b] = (int)value;
228 anyBinRecovered = true;
229 signalAddedIntegral += value;
230 signalAddedAmplitude = std::max(signalAddedAmplitude, value);
231 }
232 }
233
234 if (!anyBinRecovered) {
235 skip(signal, "fit does not exceed saturation " + std::to_string(maxValue));
236 continue;
237 }
238
239 for (int b : saturatedBins) {
240 if (deltas[b] != 0) signal.IncreaseBinByClamped(b, deltas[b]);
241 }
242 fObs.SignalsIDsRecovered.push_back(signal.GetSignalID());
243 fObs.SignalsRecovered++;
244 fObs.AddedAmplitude += signalAddedAmplitude;
245 fObs.AddedIntegral += signalAddedIntegral;
246 }
247
248 return true;
249}
