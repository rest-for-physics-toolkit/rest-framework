89void TRestDetectorHitsSmearingProcess::InitProcess() {
90 if (fEnergyReference <= 0 || fResolutionReference < 0) {
91 throw std::runtime_error(
92 "TRestDetectorHitsSmearingProcess: energyReference must be > 0 and resolutionReference >= 0");
93 }
94
95 fCharge = nullptr;
96 if (fSubtractFano) {
97 if (!fManager) throw std::runtime_error("TRestDetectorHitsSmearingProcess: no TRestManager");
98 const std::string user = "TRestDetectorHitsSmearingProcess";
99 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "charge", user);
100 if (!medium) {
101 throw std::runtime_error(user + ": subtractFano needs a TRestDetectorMedium with charge");
102 }
103 fCharge = &medium->GetChargeResponse(user);
104 if (fCharge->GetWValue() <= 0) {
105 throw std::runtime_error(user + ": subtractFano needs the wValue of the charge of '" +
106 medium->GetName() + "'");
107 }
108 fMediumName = medium->GetName();
109 UpdateYAMLFromParams<TRestDetectorHitsSmearingProcess>(fNode);
110 }
111
112 fNBelowFano = 0;
113 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
114 fRandom.SetSeed(fSeed);
115}

117bool TRestDetectorHitsSmearingProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
118 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
119 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
120 if (!inputEvent || !outputEvent) {
121 throw std::runtime_error(
122 "TRestDetectorHitsSmearingProcess: input/output is not a TRestDetectorHitsEvent");
123 }
124
125 outputEvent->Initialize();
126 outputEvent->SetEventInfo(inputEvent);
127 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
128
129 double energy = 0;
130 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
131 if (inputEvent->GetEnergy(n) > 0) energy += inputEvent->GetEnergy(n);
132 }
133
134 double gain = 1;
135 if (energy > 0) {
136 const double sigma = fResolutionReference / 100. / kFWHM * std::sqrt(fEnergyReference / energy);
137 double variance = sigma * sigma;
138 if (fCharge) variance -= fCharge->GetFanoFactor() * fCharge->GetWValue() / energy;
139 if (variance > 0) {
140 gain = fRandom.Gaus(1, std::sqrt(variance));
141 } else if (fCharge) {
142 fNBelowFano++;
143 }
144 }
145
146 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
147 outputEvent->AddHit(inputEvent->GetPosition(n), inputEvent->GetEnergy(n) * gain,
148 inputEvent->GetTime(n), inputEvent->GetType(n), inputEvent->GetParticleID(n));
149 }
150
151 return outputEvent->GetNumberOfHits() > 0;
152}
