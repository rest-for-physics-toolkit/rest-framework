96void FEMDAQ::OpenFiles(const std::vector<std::string>& flags) {
97 if (runConfig.isTCM) {
98 // The TCM only writes its log, in the current directory
99 CloseLogFiles();
100 baseFileName.clear();
101 OpenFileLogs();
102 return;
103 }
104
105 const std::string dataPath = runConfig.run->fMainDataPath;
106 if (!dataPath.empty() && !std::filesystem::exists(dataPath)) {
107 throw std::runtime_error("Data path: " + dataPath + " does not exist");
108 }
109
110 fileNameRoot.clear();
111 baseFileName.clear();
112
113 CloseLogFiles();
114 CloseRootFile();
115
116 bool openRootFile = false;
117 bool openLogFile = false;
118
119 if (flags.empty()) {
120 openRootFile = true;
121 openLogFile = true;
122 } else {
123 for (const auto& flag : flags) {
124 if (flag == "all") {
125 openRootFile = true;
126 openLogFile = true;
127 } else if (flag == "root") {
128 openRootFile = true;
129 } else if (flag == "log") {
130 openLogFile = true;
131 } else {
132 throw std::runtime_error("Invalid fopen flag: " + flag +
133 "; valid flags are: all, root or log");
134 }
135 }
136 }
137
138 if (openRootFile) {
139 runConfig.NewRun();
140 baseFileName = runConfig.GetOutputFileName();
141 prometheus.setRunNumber(runConfig.run->GetRunNumber());
142 OpenRootFile();
143 }
144 if (openLogFile) OpenFileLogs();
145}

147void FEMDAQ::OpenRootFile() {
148 auto& run = *runConfig.run;
149
150 // The pattern: FormOutputFile resolves it and adds mainDataPath (once)
151 run.SetOutputFileName(runConfig.GetOutputPattern());
152 run.SetStartTimeStamp(0);
153 run.SetEndTimeStamp(0);
154 run.SetEntriesSaved(0);
155 run.FormOutputFile();
156 fileNameRoot = run.GetOutputFileName();
157 std::cout << "New file " << fileNameRoot << " " << std::endl;
158 run.fOutputFile->SetCompressionAlgorithm(ROOT::RCompressionSetting::EAlgorithm::kZLIB);
159 run.fOutputFile->SetCompressionLevel(1);
160 run.RegisterEvent<TRestRawDAQSignalEvent>(kEventTree, sEvent);
161 run.fOutputEventTrees[kEventTree]->SetAutoFlush(-30000000); // 30 MB
162
163 // Written now so the file is a valid REST file while it is taken, and again when it is closed
164 run.AddMetadata(&run);
165 run.AddMetadata(runConfig.daq.get());
166
167 const std::string yamlDump = runConfig.Dump();
168 prometheus.setRunName(fileNameRoot);
169 prometheus.setMetadata(yamlDump);
170}

172void FEMDAQ::OpenFileLogs() {
173 const std::string tmstmp = TRestTools::GetTimeStampFromUnixTime(TRestTools::GetCurrentTime());
174 // The log files are named as the first file of the run, otherwise with the time
175 std::string base;
176 if (!baseFileName.empty()) {
177 base = baseFileName;
178 if (base.size() > 5 && base.substr(base.size() - 5) == ".root") base.erase(base.size() - 5);
179 } else {
180 // No spaces in the file name: YYYY-MM-DD_HH:MM:SS
181 std::string time = tmstmp;
182 std::replace(time.begin(), time.end(), ' ', '_');
183 base = runConfig.run ? runConfig.run->PrefixMainDataPath(time) : time;
184 }
185
186 for (auto& FEM : FEMArray) {
187 std::string fileName;
188 if (runConfig.isTCM)
189 fileName = base + "_TCM.log";
190 else
191 fileName = base + "_FEM" + std::to_string(FEM.femID) + ".log";
192 std::cout << fileName << std::endl;
193 std::lock_guard<std::mutex> lock(FEM.logMutex);
194 FEM.logFile = fopen(fileName.c_str(), "a");
195 if (!FEM.logFile) throw std::runtime_error("Cannot open the log file " + fileName);
196 fprintf(FEM.logFile, "\n--- LOG FILE INITIALIZED AT %s ---\n", tmstmp.c_str());
197 fflush(FEM.logFile);
198 DumpExecFileToFEMLog(FEM);
199 }

319void FEMDAQ::UpdateThread() {
320 double prevRateTime = TRestTools::GetCurrentTime();
321 int prevEvCount = storedEvents.load();
322 const int updateTime = runConfig.updateRateTime;
323
324 while (!abrt && !stopRun) {
325 std::unique_lock<std::mutex> lk(cv_m);
326 cv.wait_for(lk, std::chrono::seconds(updateTime), [this] { return abrt || stopRun; });
327
328 double rateTime = TRestTools::GetCurrentTime();
329 const double elapsed = rateTime - prevRateTime;
330 const double runElapsedTime = rateTime - runStartTime;
331 auto tmpstm = TRestTools::GetTimeStampFromUnixTime(rateTime);
332 const int evCount = storedEvents.load();
333 const double rate = (evCount - prevEvCount) / elapsed;
334
335 prometheus.setEvents(evCount);
336 prometheus.setDesyncEvents(nOutOfSync);
337 prometheus.setRate(rate);
338
339 if (runConfig.verboseLevel >= RunConfig::Verbosity::REST_Info && (!abrt && !stopRun))
340 std::cout << tmpstm << " Total events: " << evCount << " Rate: " << rate << " Hz "
341 << "Run time " << TRestTools::ToTimeStringLong(runElapsedTime) << std::endl;
342 prevEvCount = evCount;
343 prevRateTime = rateTime;
344 if (runConfig.nEvents > 0 && evCount >= runConfig.nEvents) {
345 stopRun = true;
346 break;
347 } else if (runConfig.maxTimeSeconds > 0 && runElapsedTime >= runConfig.maxTimeSeconds) {
348 stopRun = true;
349 break;
350 }
351 }
