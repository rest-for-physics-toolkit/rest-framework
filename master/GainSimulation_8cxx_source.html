32std::vector<double> SimulateAvalanches(Garfield::MediumMagboltz& gas, double gap, double field, int n) {
33 const double half = kLateralSize * gap;
34 Garfield::SolidBox box(0, 0, gap / 2, half, half, gap / 2);
35 Garfield::GeometrySimple geometry;
36 geometry.AddSolid(&box, &gas);
37 Garfield::ComponentConstant component;
38 component.SetGeometry(&geometry);
39 // Field along +z: the electrons drift to -z, from the mesh to the anode
40 component.SetElectricField(0, 0, field);
41 Garfield::Sensor sensor;
42 sensor.AddComponent(&component);
43 sensor.SetArea(-half, -half, 0, half, half, gap);
44
45 Garfield::AvalancheMicroscopic avalanche;
46 avalanche.SetSensor(&sensor);
47
48 std::vector<double> sizes;
49 const double tolerance = 1e-6 * gap;
50 for (int i = 0; i < n; i++) {
51 avalanche.AvalancheElectron(0, 0, gap - tolerance, 0, kInitialEnergy);
52 int collected = 0;
53 for (unsigned int e = 0; e < avalanche.GetNumberOfElectronEndpoints(); e++) {
54 double x0, y0, z0, t0, e0, x1, y1, z1, t1, e1;
55 int status;
56 avalanche.GetElectronEndpoint(e, x0, y0, z0, t0, e0, x1, y1, z1, t1, e1, status);
57 if (z1 <= tolerance) collected++;
58 }
59 sizes.push_back(collected);
60 }
61 return sizes;
62}

117 const double n = sizes.back().size();
118 const double mean = sum / n;
119 const double variance = (sum2 - n * mean * mean) / (n - 1);
120 table.fFields.push_back(field);
121 table.fGain.push_back(mean);
122 table.fTheta.push_back(variance > 0 ? mean * mean / variance - 1 : 0);
123 if (verbose) {
