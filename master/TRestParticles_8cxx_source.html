59int IonID(const std::string& name) {
60 size_t n = 0;
61 while (n < name.size() && std::isalpha(static_cast<unsigned char>(name[n]))) n++;
62 const std::string symbol = name.substr(0, n);
63 size_t m = n;
64 while (m < name.size() && std::isdigit(static_cast<unsigned char>(name[m]))) m++;
65 if (n == 0 || m == n || m - n > 3) return 0;
66 int level = 0;
67 if (m < name.size()) {
68 // Geant4 excitation energy in keV: "[0.000]" is the ground state, any other an excited one
69 if (name[m] != '[' || name.back() != ']') return 0;
70 try {
71 size_t used = 0;
72 const std::string energy = name.substr(m + 1, name.size() - m - 2);
73 const double value = std::stod(energy, &used);
74 if (used != energy.size() || value < 0) return 0;
75 if (value > 0) level = 9;
76 } catch (const std::exception&) {
77 return 0;
78 }
79 }
80 const int a = std::stoi(name.substr(n, m - n));
81 for (int z = 1; z <= kNElements; z++) {
82 if (symbol == kElements[z - 1]) {
83 if (a < z) return 0;
84 return 1000000000 + z * 10000 + a * 10 + level;
85 }
86 }
87 return 0;
88}

94int GetParticleID(const std::string& name) {
95 const auto& table = ParticleTable();
96 const auto it = table.find(name);
97 if (it != table.end()) return it->second;
98 if (const int ion = IonID(name)) return ion;
99 // A PDG encoding given as a number
100 try {
101 size_t used = 0;
102 const int id = std::stoi(name, &used);
103 if (used == name.size()) return id;
104 } catch (const std::exception&) {
105 }
106 return 0;
107}

109std::string GetParticleName(int id) {
110 if (id == 0) return "";
111 for (const auto& [name, code] : ParticleTable()) {
112 if (code == id) return name;
113 }
114 const int z = GetIonZ(id);
115 if (z >= 1 && z <= kNElements) {
116 const std::string name = kElements[z - 1] + std::to_string(GetIonA(id));
117 return id % 10 == 0 ? name : name + "*";
118 }
119 return std::to_string(id);
120}

122bool Matches(int id, const std::string& name) {
123 if (name == "ion") return IsIon(id);
124 const int wanted = GetParticleID(name);
125 if (wanted == 0) throw std::invalid_argument("REST_Particles: unknown particle '" + name + "'");
126 if (IsIon(wanted)) return IsIon(id) && GetIonZ(id) == GetIonZ(wanted) && GetIonA(id) == GetIonA(wanted);
127 return id == wanted;
128}
