100bool TRestDetectorHitsToSignalProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
101 const auto* hitsEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
102 auto* signalEvent = dynamic_cast<TRestDetectorSignalEvent*>(&output);
103 if (!hitsEvent || !signalEvent) {
104 throw std::runtime_error(
105 "TRestDetectorHitsToSignalProcess: input/output is not a TRestDetectorHitsEvent/"
106 "TRestDetectorSignalEvent");
107 }
108
109 signalEvent->Initialize();
110 signalEvent->SetEventInfo(hitsEvent);
111
112 // Charge of each channel versus the time, sorted
113 std::map<int, std::map<double, double>> charges;
114 fNHitsRejected = 0;
115 // Consecutive hits at the same position (e.g. the photoelectrons of a channel) share its channel
116 XYZVector lastPosition;
117 int lastChannel = -1;
118 bool hasLast = false;
119 for (size_t n = 0; n < hitsEvent->GetNumberOfHits(); n++) {
120 const XYZVector position = hitsEvent->GetPosition(n);
121 double time = hitsEvent->GetTime(n);
122 if (!hasLast || position != lastPosition) {
123 lastChannel = fReadout->GetChannelFromPosition(position.X(), position.Y(), position.Z());
124 lastPosition = position;
125 hasLast = true;
126 }
127 const int daqID = lastChannel;
128 if (daqID < 0) {
129 fNHitsRejected++;
130 continue;
131 }
132 if (fSampling > 0) time = std::floor(time / fSampling) * fSampling;
133 charges[daqID][time] += hitsEvent->GetEnergy(n);
134 }
135
136 for (const auto& [daqID, points] : charges) {
137 signalEvent->AddSignal(daqID, std::vector<std::pair<double, double>>(points.begin(), points.end()));
138 }
139
140 RESTDebug << "TRestDetectorHitsToSignalProcess: " << signalEvent->GetNumberOfSignals() << " signals, "
141 << fNHitsRejected << " hits rejected" << RESTendl;
142 return signalEvent->GetNumberOfSignals() > 0;
143}
