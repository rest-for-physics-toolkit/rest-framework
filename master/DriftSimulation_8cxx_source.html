80bool ReadNode(FILE* pipe, TRestDetectorDriftResponse::Node& node, size_t positionSize, size_t timeSize) {
81 int counts[2];
82 double range[6];
83 if (fread(counts, sizeof(int), 2, pipe) != 2 || fread(range, sizeof(double), 6, pipe) != 6) return false;
84 node.fPrimaries = counts[0];
85 node.fArrived = counts[1];
86 node.fUMin = range[0];
87 node.fUMax = range[1];
88 node.fVMin = range[2];
89 node.fVMax = range[3];
90 node.fTMin = range[4];
91 node.fTMax = range[5];
92 node.fPositionCounts.resize(positionSize);
93 node.fTimeCounts.resize(timeSize);
94 return fread(node.fPositionCounts.data(), sizeof(int), positionSize, pipe) == positionSize &&
95 fread(node.fTimeCounts.data(), sizeof(int), timeSize, pipe) == timeSize;
96}

99void SetGridOnReadout(TRestDetectorDriftResponse& response, const TRestDetectorReadout& readout) {
100 if (response.HasRange(0) && response.HasRange(1) && response.HasRange(2)) return;
101 const std::string error = "restGarfield: drift response " + response.fResponseFile + ": ";
102 const auto [low, high] = readout.GetChannelsBoundingBox();
103 if (std::isnan(low.X()) || std::isnan(high.X())) {
104 throw std::runtime_error(error + "the readout has no channels: give xRange, yRange and zRange");
105 }
106 const XYZVector normal = readout.GetReadoutNormal((low + high) / 2);
107 if (normal.R() == 0) throw std::runtime_error(error + "the readout has no normal");
108 for (int axis = 0; axis < 3; axis++) {
109 if (response.HasRange(axis)) continue;
110 if (std::abs(Component(normal, axis)) > 1e-6) {
111 throw std::runtime_error(error + "the range along the readout normal (" + "xyz"[axis] +
112 "Range) must be given, and the normal must be along x, y or z");
113 }
114 response.SetRange(axis, Component(low, axis) - response.fMargin,
115 Component(high, axis) + response.fMargin);
116 }

119void ComputeDriftResponse(const TRestDetectorMedium& medium, const TRestDetectorTransportTable& table,
120 const TRestDetectorReadout& readout, TRestDetectorDriftResponse& response,
121 bool verbose, int jobs) {
122 const std::string error = "restGarfield: drift response " + response.fResponseFile + ": ";
123 if (!response.HasRange(0) || !response.HasRange(1) || !response.HasRange(2)) {
124 throw std::runtime_error(error + "the ranges of the grid are not set (SetGridOnReadout)");
125 }
126 auto gas = CreateGasFromTable(medium, table, medium.GetPressure());
127 const std::array<double, 3> offset = {response.fFieldMapOffset[0], response.fFieldMapOffset[1],
128 response.fFieldMapOffset[2]};
129 auto map = LoadCSTFieldMap(response.fFieldMap, offset);
130 map->SetMedium(gas.get());
131 double low[3], high[3]; // cm
132 GetMeshLimits(*map, low, high, 0);
133
134 // The readout normal at the centre of the grid, along an axis: the drift region is cut at the boundary
135 const XYZVector centre((response.fXRange.first + response.fXRange.second) / 2,
136 (response.fYRange.first + response.fYRange.second) / 2,
137 (response.fZRange.first + response.fZRange.second) / 2);
138 const XYZVector normal = readout.GetReadoutNormal(centre);
139 if (normal.R() == 0) throw std::runtime_error(error + "the readout has no distance to it");
140 int axis = -1;
141 for (int a = 0; a < 3; a++) {
142 if (std::abs(Component(normal, a)) > 1 - 1e-6) axis = a;
143 }
144 if (axis < 0) throw std::runtime_error(error + "the normal of the readout must be along x, y or z");
145 const double boundary = response.fBoundary;
146 const XYZVector onBoundary = centre - (readout.GetDistanceToReadout(centre) - boundary) * normal;
147 const double plane = Component(onBoundary, axis) * kCmPerMm;
148 if (Component(normal, axis) > 0) {
149 low[axis] = std::max(low[axis], plane);
150 } else {
151 high[axis] = std::min(high[axis], plane);
152 }
153 if (!(low[axis] < high[axis])) {
154 throw std::runtime_error(error + "the field map does not reach the drift region above the boundary");
155 }
156
157 Garfield::Sensor sensor;
158 sensor.AddComponent(map.get());
159 sensor.SetArea(low[0], low[1], low[2], high[0], high[1], high[2]);
160 Garfield::AvalancheMC drift;
161 drift.SetSensor(&sensor);
162 drift.SetDistanceSteps(response.fStepSize * kCmPerMm);
163 // Without attachment in the table (e.g. Ar + isobutane), AvalancheMC does not integrate it along each
164 // drift line: faster, and without its message (ComputeAlphaEta, status -5) for the lines whose steps
165 // cross a cell out of the gas near a wall
166 const auto electron = table.fCarriers.find("electron");
167 if (electron == table.fCarriers.end() ||
168 std::all_of(electron->second.fAttachment.begin(), electron->second.fAttachment.end(),
169 [](double attachment) { return attachment <= 0; })) {
170 drift.DisableAttachment();
171 }
172
173 response.fMedium = medium.GetName();
174 response.fPressure = medium.GetPressure();
175 response.fTemperature = medium.GetTemperature();
176 response.fTransportTable = table.GetKey();
177 // Not a structured binding: captured by the lambdas below (C++17)
178 const auto axes = TRestDetectorDriftResponse::PlaneAxes(normal);
179 const XYZVector u = axes.first, v = axes.second;
180 const size_t nodes = response.GetNumberOfNodes();
181 response.fNodes.assign(nodes, TRestDetectorDriftResponse::Node());
182
183 // The response of a node: its electrons drifted
184 auto computeNode = [&](size_t index) {
185 const XYZVector start = response.GetNodePosition(index);
186 const double distance = readout.GetDistanceToReadout(start);
187 const XYZVector startOnBoundary = start - (distance - boundary) * normal;
188 std::vector<std::array<double, 3>> arrivals;
189 const double x[3] = {start.X() * kCmPerMm, start.Y() * kCmPerMm, start.Z() * kCmPerMm};
190 // Start points out of the gas of the map (e.g. the corners of a grid around a cylinder, nodes with
191 // the field 0) are not drifted: survival 0, without the message of AvalancheMC for each electron
192 const bool inArea = x[0] >= low[0] && x[0] <= high[0] && x[1] >= low[1] && x[1] <= high[1] &&
193 x[2] >= low[2] && x[2] <= high[2] && map->GetMedium(x[0], x[1], x[2]) != nullptr;
194 if (distance <= boundary + kArrivalTolerance) {
195 // On or below the boundary plane: the drift belongs to the amplification
196 arrivals.assign(response.fElectrons, std::array<double, 3>{0, 0, 0});
197 } else if (inArea) {
198 for (int e = 0; e < response.fElectrons; e++) {
199 if (!drift.DriftElectron(x[0], x[1], x[2], 0)) continue;
200 if (drift.GetNumberOfElectronEndpoints() == 0) continue;
201 double x0, y0, z0, t0, x1, y1, z1, t1;
202 int status = 0;
203 drift.GetElectronEndpoint(0, x0, y0, z0, t0, x1, y1, z1, t1, status);
204 const XYZVector end(x1 / kCmPerMm, y1 / kCmPerMm, z1 / kCmPerMm);
205 if (std::abs(readout.GetDistanceToReadout(end) - boundary) > kArrivalTolerance) continue;
206 const XYZVector displacement = end - startOnBoundary;
207 arrivals.push_back({displacement.Dot(u), displacement.Dot(v), t1 * kUsPerNs});
208 }
209 }
210 response.SetNode(index, response.fElectrons, arrivals);
211 };
212 // The nodes of a process (part of parts: every parts-th node, the drift lengths mixed), with its progress
213 auto computePart = [&](size_t part, size_t parts) {
214 const size_t partNodes = (nodes - part + parts - 1) / parts;
215 size_t done = 0, reported = 0;
216 for (size_t index = part; index < nodes; index += parts) {
217 computeNode(index);
218 done++;
219 if (verbose && done * 10 / partNodes > reported) {
220 reported = done * 10 / partNodes;
221 RESTInfo << "restGarfield: drift response, " << done << " of " << partNodes << " nodes"
222 << (parts > 1 ? " (process " + std::to_string(part + 1) + ")" : "") << RESTendl;
223 }
224 }
225 };
226
227 const size_t parts = std::max<size_t>(1, std::min<size_t>(std::max(jobs, 1), nodes));
228 if (parts == 1) {
229 computePart(0, 1);
230 return;
231 }
232 // In parallel (Garfield++ is not thread safe): forked processes, each with its own random seed (from the
233 // generator of Garfield++, drawn once here), send their nodes through a pipe
234 const auto seed = static_cast<unsigned int>(Garfield::RndmUniform() * 1e9) + 1;
235 struct Child {
236 pid_t pid;
237 FILE* pipe;
238 };
239 std::vector<Child> children;
240 for (size_t part = 0; part < parts; part++) {
241 int fd[2];
242 if (pipe(fd) != 0) throw std::runtime_error(error + "cannot create a pipe");
243 std::cout.flush();
244 std::fflush(nullptr);
245 const pid_t pid = fork();
246 if (pid < 0) throw std::runtime_error(error + "cannot create a process");
247 if (pid == 0) {
248 close(fd[0]);
249 int status = 1;
250 try {
251 Garfield::randomEngine.Seed(seed + part);
252 computePart(part, parts);
253 FILE* output = fdopen(fd[1], "wb");
254 bool ok = output != nullptr;
255 for (size_t index = part; ok && index < nodes; index += parts) {
256 ok = WriteNode(output, response.fNodes[index]);
257 }
258 if (output && fclose(output) == 0 && ok) status = 0;
259 } catch (const std::exception& exception) {
260 std::cerr << "restGarfield: " << exception.what() << std::endl;
261 }
262 std::cout.flush();
263 // Without the exit handlers of the parent (ROOT)
264 _exit(status);
265 }
266 close(fd[1]);
267 children.push_back({pid, fdopen(fd[0], "rb")});
268 }
269 RESTInfo << "restGarfield: " << nodes << " nodes of the drift response in " << parts << " processes"
270 << RESTendl;
271 bool ok = true;
272 const size_t positionSize = static_cast<size_t>(response.fPositionBins) * response.fPositionBins;
273 for (size_t part = 0; part < parts; part++) {
274 auto& child = children[part];
275 for (size_t index = part; ok && index < nodes; index += parts) {
276 ok = child.pipe && ReadNode(child.pipe, response.fNodes[index], positionSize, response.fTimeBins);
277 }
278 if (child.pipe) fclose(child.pipe);
279 int status = 0;
280 waitpid(child.pid, &status, 0);
281 ok &= WIFEXITED(status) && WEXITSTATUS(status) == 0;
282 }
283 if (!ok) throw std::runtime_error(error + "a process computing the nodes failed");
284}

286void RunDriftResponse(const TRestDetectorMedium& medium, TRestDetectorDriftResponse& response, bool verbose,
287 int jobs) {
288 const std::string error = "restGarfield: drift response " + response.fResponseFile + ": ";
289 const std::string fileName = response.fResponseFile;
290 if (!response.fOverwrite && std::filesystem::exists(fileName)) {
291 RESTInfo << "restGarfield: drift response " << fileName
292 << " already exists (set overwrite: true to compute it again)" << RESTendl;
293 return;
294 }
295 if (!medium.IsGas()) throw std::runtime_error(error + "the medium must be a gas");
296
297 // The readout of the readouts file
298 if (response.fReadoutsFile.empty()) throw std::runtime_error(error + "readouts (a YAML file) is needed");
299 const YAML::Node config = TRestYAMLUtils::OpenConfigFile(response.fReadoutsFile);
300 const auto section = TRestYAMLUtils::GetMetadataClass(config, "TRestDetectorReadoutManager");
301 if (!section.second) {
302 throw std::runtime_error(error + "no TRestDetectorReadoutManager in " + response.fReadoutsFile);
303 }
304 TRestDetectorReadoutManager readouts(section.first, section.second);
305 readouts.LoadReadout();
306 std::string readoutName = response.fReadoutName;
307 const TRestDetectorReadout* readout = readouts.SelectReadout(readoutName, "restGarfield drift response");
308
309 // The transport table of the medium file
310 const auto mediumFile = medium.GetMediumFile();
311 std::unique_ptr<TRestDetectorTransportTable> table;
312 {
313 const MediumFileLock lock(mediumFile);
314 std::unique_ptr<TFile> file(TFile::Open(mediumFile.string().c_str(), "READ"));
315 if (!file || file->IsZombie()) throw std::runtime_error(error + "cannot open " + mediumFile.string());
316 std::string key = response.fTransportTable;
317 if (key.empty()) {
318 const YAML::Node tables = TRestDetectorTransportTable::GetTables(file.get(), medium.GetName());
319 if (tables.size() != 1) {
320 throw std::runtime_error(error + "the medium file has " + std::to_string(tables.size()) +
321 " transport tables of " + medium.GetName() +
322 ": give transportTable (1 needed)");
323 }
324 key = tables[0]["key"].as<std::string>();
325 }
326 table = std::make_unique<TRestDetectorTransportTable>(
327 TRestDetectorTransportTable::Read(file.get(), medium.GetName(), key));
328 }
329
330 SetGridOnReadout(response, *readout);
331 RESTInfo << "restGarfield: computing the drift response " << fileName << " ("
332 << response.GetNumberOfNodes() << " nodes, " << response.fElectrons << " electrons each)"
333 << RESTendl;
334 const std::pair<TRestWithUnits, TRestWithUnits>* ranges[3] = {&response.fXRange, &response.fYRange,
335 &response.fZRange};
336 for (int axis = 0; axis < 3; axis++) {
337 RESTInfo << " " << "xyz"[axis] << ": [" << ranges[axis]->first << ", " << ranges[axis]->second
338 << "] mm, " << response.fGridNodes[axis] << " nodes" << RESTendl;
339 }
340 const auto start = std::chrono::steady_clock::now();
341 ComputeDriftResponse(medium, *table, *readout, response, verbose, jobs);
342 const double seconds = std::chrono::duration<double>(std::chrono::steady_clock::now() - start).count();
343 // The time of an electron in one process (the start points out of the gas included), to plan a grid
344 const double electrons = static_cast<double>(response.GetNumberOfNodes()) * response.fElectrons;
345 RESTInfo << "restGarfield: drift response computed in " << seconds << " s with " << std::max(jobs, 1)
346 << " processes, " << seconds * std::max(jobs, 1) / electrons * 1e3 << " ms per electron"
347 << RESTendl;
348
349 YAML::Node generator;
350 generator["program"] = "Garfield++ AvalancheMC (field map)";
351 const std::time_t now = std::time(nullptr);
352 char date[32];
353 std::strftime(date, sizeof(date), "%Y-%m-%d %H:%M:%S", std::localtime(&now));
354 generator["date"] = date;
355 generator["code"] = REST_Version::GetSummary();
356 const std::filesystem::path path(fileName);
357 if (path.has_parent_path()) std::filesystem::create_directories(path.parent_path());
358 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "RECREATE"));
359 if (!file || file->IsZombie()) throw std::runtime_error(error + "cannot create the file");
360 response.Write(file.get(), generator);
361 RESTInfo << "restGarfield: drift response written to " << fileName << RESTendl;
362}
