114bool TRestDetectorSignalPeaksFinderProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
115 output.CopyFrom(&input);
116
117 const auto* signalEvent = dynamic_cast<const TRestDetectorSignalEvent*>(&input);
118 if (!signalEvent) {
119 throw std::runtime_error(
120 "TRestDetectorSignalPeaksFinderProcess: input is not a TRestDetectorSignalEvent");
121 }
122
123 fObs.clear();
124
125 // (time, signal ID, amplitude) of all the peaks of the event
126 std::vector<std::tuple<double, int, double>> peaks;
127 std::set<int> peakSignals;
128 for (int s = 0; s < signalEvent->GetNumberOfSignals(); s++) {
129 const auto& signal = signalEvent->GetSignal(s);
130 for (const auto& [time, amplitude] :
131 TRestPulseShapeAnalysis::GetPeaks(signal.GetPointsVector(), fThreshold, fDistance, fHalfWindow,
132 fMaxGreaterEqual, fRefineMaximum, fAmplitudeHalfWindow)) {
133 peaks.emplace_back(time, signal.GetSignalID(), amplitude);
134 peakSignals.insert(signal.GetSignalID());
135 }
136 }
137 std::sort(peaks.begin(), peaks.end());
138
139 for (const auto& [time, signalID, amplitude] : peaks) {
140 // Peaks are sorted by time, so a peak opens a new window if it is too far from the current one
141 if (fObs.WindowTime.empty() || time - fObs.WindowTime.back() > fWindow) {
142 fObs.WindowTime.push_back(time);
143 fObs.WindowMultiplicity.push_back(0);
144 fObs.WindowAmplitude.push_back(0);
145 }
146 fObs.WindowMultiplicity.back()++;
147 fObs.WindowAmplitude.back() += amplitude;
148
149 fObs.PeaksSignalID.push_back(signalID);
150 fObs.PeaksTime.push_back(time);
151 fObs.PeaksAmplitude.push_back(amplitude);
152 fObs.PeaksWindowIndex.push_back(fObs.WindowTime.size() - 1);
153 fObs.PeaksAmplitudeSum += amplitude;
154 }
155
156 fObs.NPeaks = peaks.size();
157 fObs.NPeaksSignals = peakSignals.size();
158 fObs.NWindows = fObs.WindowTime.size();
159
160 return true;
161}
