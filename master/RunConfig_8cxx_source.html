110 auto readField = [&](const std::string& prompt) -> std::string {
111 if (CommandFetcher::g_shutdown.load()) {
112 throw std::runtime_error("AddMetadata aborted (CTRL-C)");
113 }
114
115 char* buffer = readline(prompt.c_str());
116
117 if (CommandFetcher::g_shutdown.load()) {
118 if (buffer) free(buffer);
119 throw std::runtime_error("AddMetadata aborted (CTRL-C)");
120 }
121
122 if (buffer == nullptr) {
123 throw std::runtime_error("AddMetadata aborted (CTRL-C)");
124 }
125
126 std::string str(buffer);
127 free(buffer);
128 return str;
129 };

143
144 input = readField(">>>> Run Type (" + run->GetRunType() + "): ");

147 changes.emplace_back(section + ".runType", input);

149

152 if (!input.empty() && input != value) {

154 changes.emplace_back(section + ".runInfo." + field, value);
