92void TRestDataSetSignalSelection::ComputeParameters(const TRestDataSet&, ROOT::RDF::RNode frame) {
93 const std::string context = "TRestDataSetSignalSelection '" + fName + "'";
94 std::vector<double> values = *frame.Take<double>(AsDouble(frame, fObservable));
95 if (values.empty()) throw std::runtime_error(context + ": no entries in the reference");
96 std::sort(values.begin(), values.end());
97 const size_t n = values.size();
98 constexpr double infinity = std::numeric_limits<double>::infinity();
99
100 // The lowest value above the lower fraction of the entries, the highest below the upper one
101 fRange = {-infinity, infinity};
102 if (fQuantiles.first > 0) {
103 const auto index = static_cast<size_t>(std::floor(fQuantiles.first * n));
104 fRange.first = values[std::min(index, n - 1)];
105 }
106 if (fQuantiles.second < 1) {
107 const auto index = static_cast<size_t>(std::ceil(fQuantiles.second * n));
108 fRange.second = values[index > 0 ? index - 1 : 0];
109 }
110 const auto first = std::lower_bound(values.begin(), values.end(), fRange.first);
111 const auto last = std::upper_bound(values.begin(), values.end(), fRange.second);
112 fEfficiency = last > first ? static_cast<double>(last - first) / n : 0;
113
114 RESTInfo << context << ": " << fObservable << " in [" << fRange.first << ", " << fRange.second
115 << "], efficiency " << fEfficiency << " of " << n << " reference entries" << RESTendl;
116 UpdateYAMLFromParams<TRestDataSetSignalSelection>(fNode);
117}
