87void TRestDetectorScintillationProcess::InitProcess() {
88 const std::string user = "TRestDetectorScintillationProcess";
89 if (!fManager) throw std::runtime_error(user + ": no TRestManager");
90
91 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "light", user);
92 if (!medium) throw std::runtime_error(user + ": needs a TRestDetectorMedium with light");
93 fLight = &medium->GetLightResponse(user);
94 if (fLight->GetLightYield() <= 0) {
95 throw std::runtime_error(user + ": the light of the medium '" + medium->GetName() +
96 "' needs lightYield > 0");
97 }
98 fMediumName = medium->GetName();
99 // The stored metadata keeps the medium used
100 UpdateYAMLFromParams<TRestDetectorScintillationProcess>(fNode);
101
102 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
103 fRandom.SetSeed(fSeed);
104
105 RegisterObservable("NPhotons", fNPhotons);
106}

108bool TRestDetectorScintillationProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
109 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
110 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
111 if (!inputEvent || !outputEvent) {
112 throw std::runtime_error(
113 "TRestDetectorScintillationProcess: input/output is not a TRestDetectorHitsEvent");
114 }
115
116 outputEvent->Initialize();
117 outputEvent->SetEventInfo(inputEvent);
118 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
119 fNPhotons = 0;
120
121 const double lightYield = fLight->GetLightYield();
122 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
123 const double energy = inputEvent->GetEnergy(n);
124 if (energy <= 0) continue;
125 const int photons = fRandom.Poisson(energy * lightYield);
126 if (photons <= 0) continue;
127 outputEvent->AddHit(inputEvent->GetPosition(n), photons, inputEvent->GetTime(n),
128 inputEvent->GetType(n), inputEvent->GetParticleID(n));
129 fNPhotons += photons;
130 }
131
132 RESTDebug << "TRestDetectorScintillationProcess: event " << inputEvent->GetID() << ", " << fNPhotons
133 << " photons" << RESTendl;
134 return outputEvent->GetNumberOfHits() > 0;
