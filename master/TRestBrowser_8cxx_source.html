134void TRestBrowser::SetViewer(const std::string& viewerName, double geomScale) {
135 if (CountString((string)viewerName, "Viewer") > 0) {
136 TClass* cl = TClass::GetClass(viewerName.c_str());
137 if (!cl) {
138 throw std::runtime_error("TRestBrowser::SetViewer: the class '" + viewerName +
139 "' is not registered in ROOT (is its library loaded?)");
140 }
141 if (!cl->InheritsFrom("TRestEventViewer")) {
142 throw std::invalid_argument("TRestBrowser::SetViewer: the class '" + viewerName +
143 "' is not a TRestEventViewer");
144 }
145
146 std::unique_ptr<TRestEventViewer> viewer(static_cast<TRestEventViewer*>(cl->New()));
147 if (viewer != nullptr) {
148 viewer->SetGeomScale(geomScale);
149 SetViewer(viewer.release(), geomScale);
150 } else {
151 RESTError << viewerName << " not recognized! Did you install the corresponding library?"
152 << RESTendl;
153 RESTError << "Also check EVE feature is turned on in REST for 3d event viewing." << RESTendl;
154 RESTWarning << "Using default event viewer" << RESTendl;
155 }
156 } else {
157 throw std::invalid_argument("TRestBrowser::SetViewer: illegal viewer '" + viewerName +
158 "' (its class name must contain \"Viewer\")");
159 }

281bool TRestBrowser::LoadEventEntry(int n) {
282 if (fRestRun->fInputFile == nullptr || fRestRun->fInputFile->IsZombie()) {
283 RESTWarning << "TRestBrowser::LoadEventEntry. No File..." << RESTendl;
284 return kFALSE;
285 }
286 if (pureAnalysis) {
287 RESTWarning << "TRestBrowser::LoadEventEntry. This is a pure analysis file..." << RESTendl;
288 return kFALSE;
289 }
290
291 if (fRestRun->fAnalysisTree != nullptr && n < fRestRun->fAnalysisTree->GetEntries() && n >= 0) {
292 fRestRun->GetEntry(n);
293 TRestEvent* ev = fRestRun->GetEvent();
294 if (!ev) {
295 RESTError << "internal error!" << RESTendl;
296 return kFALSE;
297 } else {
298 fEventRow = fRestRun->GetCurrentEntry();
299 fEventId = ev->GetID();
300 fEventSubId = ev->GetSubID();
301
302 fEventRowNumberBox->SetIntNumber(fEventRow);
303 fEventIdNumberBox->SetIntNumber(fEventId);
304 fEventSubIdNumberBox->SetIntNumber(fEventSubId);
305 fRestRun->PrintObservables();
306 }
307 } else {
308 RESTWarning << "TRestBrowser::LoadEventEntry. Event out of limits" << RESTendl;
309 return kFALSE;
310 }
311
312 if (fEventViewer != nullptr) {
313 TRestEvent* inputEvent = (fRestRun->GetEvent());
314 inputEvent->SetRestRun(fRestRun);
315 fEventViewer->AddEvent(inputEvent);
316 fEventViewer->Plot(fPlotOptionTextBox->GetText());
317 cout << endl;
318 }
319
320 fCanDefault->cd();
321 return kTRUE;
322}

368bool TRestBrowser::OpenFile(const std::string& filename) {
369 if (filename.find("http") != std::string::npos || TRestTools::fileExists(filename)) {
370 // Browser created without file (default constructor): the control panels are created now
371 if (frmMain == nullptr) Initialize();
372 // A file already open: a new run (deleting the events of the previous one), the viewer without
373 // event and the event types of the new file
374 if (fRestRun->GetInputFile() != nullptr) {
375 if (fEventViewer != nullptr) fEventViewer->SetEvent(nullptr);
376 delete fRestRun;
377 fRestRun = new TRestRun();
378 if (fEventTypeComboBox != nullptr) fEventTypeComboBox->RemoveAll();
379 fEventRow = 0;
380 if (fEventRowNumberBox != nullptr) fEventRowNumberBox->SetIntNumber(0);
381 pureAnalysis = kTRUE;
382 if (fEventViewer != nullptr) fEventViewer->Plot("");
383 }
384
385 try {
386 fRestRun->OpenInputFile(filename);
387 } catch (const std::exception& error) {
388 RESTError << "TRestBrowser: cannot open " << filename << ": " << error.what() << RESTendl;
389 return false;
390 }
391 if (!fRestRun->IsOpen()) {
392 RESTError << "TRestBrowser: cannot open " << filename << RESTendl;
393 return false;
394 }
395 fInputFileName = filename;
396 fRestRun->fInputFile->cd();
397
398 TGeoManager* geometry = gGeoManager;
399
400 auto eventMap = fRestRun->GetEventMap();
401 if (!eventMap.empty()) {
402 // Selected: the event of the run, the last saved event type of the file
403 // (TRestRun::GetOutputEvent) if it has one
404 for (auto& [eventType, eventObj] : eventMap) {
405 fEventTypeComboBox->AddEntry(eventType.c_str(), fEventTypeComboBox->GetNumberOfEntries());
406 if (eventObj == fRestRun->GetEvent())
407 fEventTypeComboBox->Select(fEventTypeComboBox->GetNumberOfEntries() - 1, false);
408 }
409
410 // init viewer
411 pureAnalysis = kFALSE;
412 if (fEventViewer == nullptr) SetViewer("TRestEventViewer");
413 if (geometry != nullptr && fEventViewer != nullptr) fEventViewer->SetGeometry(geometry);
414 RowValueChangedAction(0);
415 } else {
416 pureAnalysis = kTRUE;
417 }
418
419 TRestEvent* ev = fRestRun->GetEvent();
420 if (!ev) {
421 RESTError << "internal error!" << RESTendl;
422 } else {
423 fEventRowNumberBox->SetIntNumber(fRestRun->GetCurrentEntry());
424 fEventIdNumberBox->SetIntNumber(ev->GetID());
425 fEventSubIdNumberBox->SetIntNumber(ev->GetSubID());
426 }
427 return true;
428 } else {
429 RESTError << "file: " << filename << " does not exist!" << RESTendl;
430 }
431 return false;
432}

523long long FindSelectedEntry(TTree* tree, TTreeFormula& formula, long long start, int step) {
524 const long long entries = tree->GetEntries();
525 int treeNumber = -1;
526 for (long long row = start; row >= 0 && row < entries; row += step) {
527 if (tree->LoadTree(row) < 0) return -1;
528 if (tree->GetTreeNumber() != treeNumber) {
529 treeNumber = tree->GetTreeNumber();
530 formula.UpdateFormulaLeaves();
531 }
532 const int ndata = formula.GetNdata();
533 for (int i = 0; i < ndata; ++i) {
534 if (formula.EvalInstance(i) != 0) return row;
535 }
536 }
537 return -1;
538}

541void TRestBrowser::SelectEvent(int step) {
542 const std::string sel = fSelectionTextBox->GetText();
543 if (sel.empty()) {
544 fEventRow += step;
545 LoadEventEntry(fEventRow);
546 return;
547 }
548 TTree* tree = fRestRun->fAnalysisTree;
549 if (tree == nullptr) {
550 RESTWarning << "TRestBrowser: no analysis tree for the selection" << RESTendl;
551 return;
552 }
553 TTreeFormula formula("Selection", sel.c_str(), tree);
554 if (formula.GetNdim() <= 0) {
555 cout << "invalid selection!" << endl;
556 return;
557 }
558 const long long row = FindSelectedEntry(tree, formula, fEventRow + step, step);
559 if (row < 0) {
560 RESTWarning << "TRestBrowser: no " << (step > 0 ? "next" : "previous") << " entry passing \"" << sel
561 << "\"" << RESTendl;
562 return;
563 }
564 fEventRow = row;
565 LoadEventEntry(fEventRow);
566}
