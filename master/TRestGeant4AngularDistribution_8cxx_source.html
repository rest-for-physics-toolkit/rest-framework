46void CheckAngleRange(const pair<TRestWithUnits, TRestWithUnits>& range, const string& className) {

56unique_ptr<TRestGeant4AngularDistribution> TRestGeant4AngularDistribution::Create(const string& name,
57 const YAML::Node& node) {
58 if (!node || !node["type"]) {
59 throw runtime_error("TRestGeant4AngularDistribution: the angular 'type' is missing");
60 }
61 const auto type = ToLower(node["type"].as<string>());
62 if (type == "flux") return make_unique<TRestGeant4AngularDistributionFlux>(name, node);
63 if (type == "isotropic") return make_unique<TRestGeant4AngularDistributionIsotropic>(name, node);
64 if (type == "formula") return make_unique<TRestGeant4AngularDistributionFormula>(name, node);
65 if (type == "th1d") return make_unique<TRestGeant4AngularDistributionTH1D>(name, node);
66 if (type == "formula2" || type == "th2d" || type == "cosmic") {
67 return make_unique<TRestGeant4AngularDistributionJoint>(name, node);
68 }
69 throw runtime_error("TRestGeant4AngularDistribution: unknown angular type '" + type +
70 "', it must be flux, isotropic, formula, TH1D, formula2, TH2D or cosmic");
71}

114XYZVector TRestGeant4AngularDistributionIsotropic::Sample(TRandom& random, const XYZVector& axis) {
115 if (fIsotropicConeHalfAngle <= 0) return IsotropicDirection(random);
116 // Uniform in the solid angle of the cone
117 const double cosTheta = random.Uniform(cos(static_cast<double>(fIsotropicConeHalfAngle)), 1);
118 return DirectionFromAxis(random, axis, acos(cosTheta));

144void TRestGeant4AngularDistributionFormula::LoadDistribution() {
145 fFunction = make_shared<TF1>(
146 AngularDistributionFormulasToRootFormula(StringToAngularDistributionFormulas(fFormula)));
147 // The range cannot be larger than the formula one
148 const double min = std::max<double>(fRange.first, fFunction->GetXmin());
149 const double max = std::min<double>(fRange.second, fFunction->GetXmax());
150 if (min >= max) throw runtime_error(GetClassName() + ": 'range' is outside the formula one");
151 fFunction->SetRange(min, max);
152 fFunction->SetNpx(fFormulaNPoints);
153}

184void TRestGeant4AngularDistributionTH1D::LoadDistribution() {
185 fHistogram = ReadHistogram1D(fFilename, fNameInFile); // Any 1D histogram, as a TH1D
186 for (int i = 1; i <= fHistogram->GetNbinsX(); i++) {
187 const double angle = fHistogram->GetBinCenter(i);
188 if (angle < fRange.first || angle > fRange.second) fHistogram->SetBinContent(i, 0);
189 }
190 if (fHistogram->Integral() <= 0) {
191 throw runtime_error(GetClassName() + ": the histogram is empty in 'range'");
192 }
193 fHistogram->ComputeIntegral();
194}
