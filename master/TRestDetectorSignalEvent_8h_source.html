255 inline void RemoveSignal(int index) {
256 if (index >= (int)fSignalData.signalIDs.size()) return;
257
258 int start = fSignalData.offsets[index];
259 int end = (index + 1 < (int)fSignalData.offsets.size()) ? fSignalData.offsets[index + 1]
260 : (int)fSignalData.allSignalsCharge.size();
261 int sizeToRemove = end - start;
262
263 fSignalData.allSignalsCharge.erase(fSignalData.allSignalsCharge.begin() + start,
264 fSignalData.allSignalsCharge.begin() + end);
265 fSignalData.allSignalsTime.erase(fSignalData.allSignalsTime.begin() + start,
266 fSignalData.allSignalsTime.begin() + end);
267
268 fSignalData.signalIDs.erase(fSignalData.signalIDs.begin() + index);
269 fSignalData.offsets.erase(fSignalData.offsets.begin() + index);
270
271 for (size_t i = index; i < fSignalData.offsets.size(); ++i) {
272 fSignalData.offsets[i] -= sizeToRemove;
273 }
274 fSignalsViews.clear();
275 }
