102void TRestEveEventViewer::SetGeometry(TGeoManager* geo) {
103 // Already drawn: not added twice to the global scene
104 if (geo == fGeometry) return;
105
106 DeleteGeometry();
107 TRestEventViewer::SetGeometry(geo);
108
109 if (fGeometry == nullptr) {
110 return;
111 }
112
113 // Placements of each logical volume: the volumes placed many times (e.g. thousands of fibres) are hidden,
114 // drawing them in every viewer makes the display very slow
115 std::map<TGeoVolume*, size_t> copies;
116 TGeoIterator next(fGeometry->GetTopVolume());
117 while (TGeoNode* node = next()) copies[node->GetVolume()]++;
118
119 const unsigned int nVolumes = fGeometry->GetListOfVolumes()->GetEntries();
120 double transparencyLevel = 85;
121 if (nVolumes > 30) {
122 transparencyLevel = 90;
123 }
124 for (unsigned int i = 0; i < nVolumes; i++) {
125 auto volume = fGeometry->GetVolume(i);
126 auto material = volume->GetMaterial();
127 bool visible = true;
128 if (material->GetDensity() <= 0.01) {
129 volume->SetTransparency(99);
130 // We consider this vacuum for display purposes
131 if (material->GetDensity() <= 0.001) visible = false;
132 } else {
133 volume->SetTransparency(transparencyLevel);
134 }
135 const size_t nCopies = copies.count(volume) ? copies.at(volume) : 0;
136 if (visible && fMaxVolumeCopies > 0 && nCopies > fMaxVolumeCopies) {
137 RESTWarning << "TRestEveEventViewer: volume '" << volume->GetName() << "' not drawn, placed "
138 << nCopies << " times (more than " << fMaxVolumeCopies
139 << ", SetMaxVolumeCopies(0) draws all)" << RESTendl;
140 visible = false;
141 }
142 volume->SetVisibility(visible);
143 }
144
145 // From the world (top node, mm as the hits: TGeoManager in Geant4 units) down to the deepest volume
146 TEveGeoTopNode* eveGeoTopNode = new TEveGeoTopNode(fGeometry, fGeometry->GetTopNode());
147 eveGeoTopNode->SetVisLevel(std::max(5, fGeometry->GetMaxLevel()));
148
149 gEve->AddGlobalElement(eveGeoTopNode);
150
151 if (!fGlobalSceneInViewer3D) {
152 viewer3D->AddScene(gEve->GetGlobalScene());
153 fGlobalSceneInViewer3D = true;
154 }
155
156 // The cameras were reset with the empty scenes (Initialize): fitted now to the world
157 gEve->FullRedraw3D(kTRUE);
158}
