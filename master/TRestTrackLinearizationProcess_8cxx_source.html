128std::vector<TRestHitsAnalysis::Node> TRestTrackLinearizationProcess::Linearize(const TRestHits& hits) const {
129 if (fMaxNodes < 2 || hits.GetNumberOfHits() == 0) return {};
130 for (size_t h = 1; h < hits.GetNumberOfHits(); h++) {
131 if (hits.GetType(h) != hits.GetType(0)) return {};
132 }
133
134 const auto points = hits.GetPoints();
135 const auto energies = hits.GetEnergies();
136 TRestHitsAnalysis::XYZVector first, last;
137 if (!TRestHitsAnalysis::GetLine(points, energies, first, last)) return {};
138
139 int nodes = fMaxNodes;
140 if (fNodeDistance > 0) {
141 nodes = std::clamp((int)std::lround((last - first).R() / fNodeDistance) + 1, 2, fMaxNodes);
142 }
143 std::vector<TRestHitsAnalysis::XYZVector> line(nodes);
144 for (int n = 0; n < nodes; n++) line[n] = first + (double)n / (nodes - 1) * (last - first);
145
146 return TRestHitsAnalysis::KMeansClustering(points, energies, line, fMaxIterations, fFixBoundaries);
147}

149bool TRestTrackLinearizationProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
150 const auto* inputEvent = dynamic_cast<const TRestTrackEvent*>(&input);
151 auto* outputEvent = dynamic_cast<TRestTrackEvent*>(&output);
152 if (!inputEvent || !outputEvent) {
153 throw std::runtime_error("TRestTrackLinearizationProcess: input/output is not a TRestTrackEvent");
154 }
155
156 outputEvent->Initialize();
157 outputEvent->SetEventInfo(inputEvent);
158
159 const double nan = std::numeric_limits<double>::quiet_NaN();
160 for (int t = 0; t < inputEvent->GetNumberOfTracks(); t++) {
161 const TRestHits hits = inputEvent->GetTrack(t).GetHits();
162 const int trackID = inputEvent->GetTrackID(t);
163 const auto nodes = Linearize(hits);
164 fNTracks++;
165
166 if (nodes.empty()) {
167 fNotLinearized++;
168 RESTDebug << "TRestTrackLinearizationProcess: event " << inputEvent->GetID() << ", track "
169 << trackID << " not linearized (no energy or hits of different types), kept as it is"
170 << RESTendl;
171 std::vector<TRestHitsAnalysis::XYZVector> sigmas;
172 for (size_t h = 0; h < hits.GetNumberOfHits(); h++)
173 sigmas.push_back(inputEvent->GetSigma(t, (int)h));
174 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, trackID, hits, sigmas);
175 continue;
176 }
177
178 const auto type = hits.GetType(0);
179 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, trackID);
180 for (const auto& node : nodes) {
181 // Undefined coordinate back to NaN, as in the hits
182 const TRestHitsAnalysis::XYZVector position(
183 TRestHitsData::IsDefined(type, 0) ? node.position.X() : nan,
184 TRestHitsData::IsDefined(type, 1) ? node.position.Y() : nan,
185 TRestHitsData::IsDefined(type, 2) ? node.position.Z() : nan);
186 outputEvent->AddHitToTrack(position, node.energy, 0, type, node.sigma);
187 }
188 RESTDebug << "TRestTrackLinearizationProcess: track " << trackID << " with " << hits.GetNumberOfHits()
189 << " hits reduced to " << nodes.size() << " nodes" << RESTendl;
190 }
191
192 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) outputEvent->PrintOnlyTracks();
193 return true;
194}
