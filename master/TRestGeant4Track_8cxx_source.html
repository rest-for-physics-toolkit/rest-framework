223ROOT::Math::XYZVector TRestGeant4Track::GetMeanPositionInVolume(int volID) const {
224 if (!IsValid()) return {};

327double TRestGeant4Track::GetEnergyInVolume(const std::string& volumeName, bool children) const {
328 const auto* metadata = GetGeant4Metadata();
329 if (metadata == nullptr) return 0;
330
331 const auto volumeId = metadata->GetGeant4GeometryInfo().GetIDFromVolume(volumeName);
332 if (!children) return GetEnergyInVolume(volumeId);
333
334 double energy = 0;
335 std::vector<TRestGeant4Track> tracks{*this};
336
337 while (!tracks.empty()) {
338 TRestGeant4Track track = tracks.back();
339 tracks.pop_back();
340 energy += track.GetEnergyInVolume(volumeId);
341 for (const auto& secondaryTrack : track.GetSecondaryTracks()) tracks.push_back(secondaryTrack);
342 }
343 return energy;
344}
