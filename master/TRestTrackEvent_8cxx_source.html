264void TRestTrackEvent::MergeTrackHits(int n, int i, int j) {
265 const int start = GetTrackStartIndex(n);
266 const int gi = start + i, gj = start + j;
267
268 const double enI = fData.hits.energy[gi];
269 const double enJ = fData.hits.energy[gj];
270 const double totalEnergy = enI + enJ;
271
272 if (totalEnergy > 0) {
273 fData.hits.x[gi] = (fData.hits.x[gi] * enI + fData.hits.x[gj] * enJ) / totalEnergy;
274 fData.hits.y[gi] = (fData.hits.y[gi] * enI + fData.hits.y[gj] * enJ) / totalEnergy;
275 fData.hits.z[gi] = (fData.hits.z[gi] * enI + fData.hits.z[gj] * enJ) / totalEnergy;
276 fData.hits.time[gi] = (fData.hits.time[gi] * enI + fData.hits.time[gj] * enJ) / totalEnergy;
277
278 fData.sigmaX[gi] = (fData.sigmaX[gi] * enI + fData.sigmaX[gj] * enJ) / totalEnergy;
279 fData.sigmaY[gi] = (fData.sigmaY[gi] * enI + fData.sigmaY[gj] * enJ) / totalEnergy;
280 fData.sigmaZ[gi] = (fData.sigmaZ[gi] * enI + fData.sigmaZ[gj] * enJ) / totalEnergy;
281
282 fData.hits.energy[gi] = (float)totalEnergy;
283 }
284
285 RemoveTrackHit(n, j);
286}

292double TRestTrackEvent::GetMaxTrackRelativeZ() {
293 TRestTrack tckX = GetMaxEnergyTrackInX();
294 TRestTrack tckY = GetMaxEnergyTrackInY();
295
296 if (tckX.GetEvent() == nullptr || tckY.GetEvent() == nullptr) {
297 RESTWarning << "Track is empty, skipping" << RESTendl;
298 return -1;
299 }
300
301 TRestHits hitsX = tckX.GetHits();
302 TRestHits hitsY = tckY.GetHits();
303
304 std::vector<std::pair<double, double>> zEn;
305 double totEn = 0;
306
307 for (unsigned int i = 0; i < hitsX.GetNumberOfHits(); i++) {
308 zEn.emplace_back(hitsX.GetZ(i), hitsX.GetEnergy(i));
309 totEn += hitsX.GetEnergy(i);
310 }
311 for (unsigned int i = 0; i < hitsY.GetNumberOfHits(); i++) {
312 zEn.emplace_back(hitsY.GetZ(i), hitsY.GetEnergy(i));
313 totEn += hitsY.GetEnergy(i);
314 }
315
316 std::sort(zEn.begin(), zEn.end());
317
318 double integ = 0;
319 size_t pos = 0;
320 for (pos = 0; pos < zEn.size(); pos++) {
321 integ += zEn[pos].second;
322 if (integ >= totEn / 2.) break;
323 }
324
325 double length = zEn.front().first - zEn.back().first;
326 double diff = zEn.front().first - zEn[pos].first;
327
328 return length == 0 ? 0 : diff / length;
329}

360int TRestTrackEvent::KMeansClustering(int trackIndex, int nodes, TRestTrackEvent& output, int maxIt,
361 bool fixBoundaries) const {
362 const TRestHits& hits = fTrackHits[trackIndex];
363 const auto points = hits.GetPoints();
364 if (nodes < 2 || points.empty()) return -1;
365
366 std::vector<XYZVector> centroids(nodes);
367 for (int n = 0; n < nodes; n++) centroids[n] = points[(n * (points.size() - 1)) / (nodes - 1)];
368 const auto result =
369 TRestHitsAnalysis::KMeansClustering(points, hits.GetEnergies(), centroids, maxIt, fixBoundaries);
370 if (result.empty()) return -1;
371
372 const REST_HitType type = hits.GetType(0);
373 int newTrackIndex = output.AddTrack(output.GetNumberOfTracks() + 1, GetTrackID(trackIndex));
374 const double nan = std::numeric_limits<double>::quiet_NaN();
375 for (const auto& node : result) {
376 // Undefined coordinates back to NaN, as in the hits (GetPoints sets them to 0)
377 const XYZVector position(TRestHitsData::IsDefined(type, 0) ? node.position.X() : nan,
378 TRestHitsData::IsDefined(type, 1) ? node.position.Y() : nan,
379 TRestHitsData::IsDefined(type, 2) ? node.position.Z() : nan);
380 output.AddHitToTrack(position, node.energy, 0, type, node.sigma);
381 }
382 return newTrackIndex;
383}

430double GetHitMarkerSize(const TRestHits& hits, int n) {
431 const double minSize = 0.4, maxSize = 2.;
432 if (hits.GetNumberOfHits() == 1) return 2;
433 if (hits.GetNumberOfHits() > 100) return 0.8;
434
435 const double maxEnergy = hits.GetMaximumHitEnergy(), meanEnergy = hits.GetMeanHitEnergy();
436 if (maxEnergy <= meanEnergy) return maxSize;
437 const double slope = maxSize / (maxEnergy - meanEnergy);
438 return std::max(0.1, slope * (hits.GetEnergy(n) - meanEnergy) + maxSize - minSize);
439}

490void TRestTrackEvent::DrawTracks(TVirtualPad* pad) const {
491 std::vector<int> tracks(GetNumberOfTracks());
492 std::iota(tracks.begin(), tracks.end(), 0);
493
494 std::vector<double> values[3];
495 double maxEnergy = 0;
496 bool has3D = false;
497 for (int t : tracks) {
498 const auto& hits = fTrackHits[t];
499 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
500 for (int a = 0; a < 3; a++) {
501 if (TRestHitsData::IsDefined(hits.GetType(n), a))
502 values[a].push_back(hits.GetCoordinate(n, a));
503 }
504 maxEnergy = std::max(maxEnergy, hits.GetEnergy(n));
505 if (hits.GetType(n) % REST_HitType::XYZ == 0) has3D = true;
506 }
507 }
508
509 // Ranges as TRestDetectorHitsEvent (margin around the hits)
510 double min[3], max[3];
511 for (int a = 0; a < 3; a++) {
512 const TRestHits::Binning binning = TRestHits::GetBinning(values[a]);
513 min[a] = binning.min;
514 max[a] = binning.max;
515 }
516
517 // {u, v} coordinates of each view: the projections with hits defining both or, if there are none
518 // (1D hits), each coordinate against the energy (v = -1)
519 auto definesBoth = [](int type, int u, int v) {
520 return TRestHitsData::IsDefined(type, u) && (v < 0 || TRestHitsData::IsDefined(type, v));
521 };
522 std::vector<std::pair<int, int>> views;
523 for (const auto& plane : TRestHitsData::kPlanes) {
524 bool hasHits = false;
525 for (int t : tracks) {
526 for (size_t n = 0; n < fTrackHits[t].GetNumberOfHits() && !hasHits; n++) {
527 hasHits = definesBoth(fTrackHits[t].GetType(n), plane[0], plane[1]);
528 }
529 }
530 if (hasHits) views.emplace_back(plane[0], plane[1]);
531 }
532 if (views.empty()) {
533 for (int a = 0; a < 3; a++) {
534 if (!values[a].empty()) views.emplace_back(a, -1);
535 }
536 }
537
538 const int nPads = views.size() + (has3D ? 1 : 0);
539 if (nPads == 0) return;
540 pad->Divide(nPads, 1);
541
542 // A colour per track: ROOT colours 2 (red, the first one in legacy) to 9
543 std::vector<Color_t> colors;
544 for (int t : tracks) colors.push_back(2 + t % 8);
545
546 const std::string title = Form("Event ID %d", GetID());
547 for (size_t i = 0; i < views.size(); i++) {
548 const auto [u, v] = views[i];
549 pad->cd(i + 1);
550 SetPadMargins(gPad);
551
552 auto* frame = new TH1F("frame", title.c_str(), 1, min[u], max[u]);
553 frame->SetMinimum(v >= 0 ? min[v] : 0);
554 frame->SetMaximum(v >= 0 ? max[v] : 1.1 * maxEnergy);
555 DrawOwned(frame, "AXIS", TRestHitsData::kAxisTitles[u],
556 v >= 0 ? TRestHitsData::kAxisTitles[v] : "Energy");
557
558 // First the hits of every track, then the track lines over them (as legacy)
559 std::vector<TGraph*> lines;
560 for (size_t k = 0; k < tracks.size(); k++) {
561 const int t = tracks[k];
562 const auto& hits = fTrackHits[t];
563
