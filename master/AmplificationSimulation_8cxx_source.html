54bool WriteResult(FILE* pipe, const Losses& losses, const std::vector<Avalanche>& avalanches) {
55 const int count = avalanches.size();
56 if (fwrite(&count, sizeof(int), 1, pipe) != 1) return false;
57 const auto lost = LossCounts(losses);
58 if (fwrite(lost.data(), sizeof(int), lost.size(), pipe) != lost.size()) return false;
59 for (const auto& avalanche : avalanches) {
60 const double values[5] = {avalanche.fX, avalanche.fY, avalanche.fSize, avalanche.fTime,
61 avalanche.fShared};
62 if (fwrite(values, sizeof(double), 5, pipe) != 5) return false;
63 }
64 return true;
65}

68bool ReadResult(FILE* pipe, Losses& losses, std::vector<Avalanche>& avalanches) {
69 int count = 0;
70 if (fread(&count, sizeof(int), 1, pipe) != 1) return false;
71 std::array<int, 5> lost;
72 if (fread(lost.data(), sizeof(int), lost.size(), pipe) != lost.size()) return false;
73 losses.fOut += lost[0];
74 losses.fConductor += lost[1];
75 losses.fDielectric += lost[2];
76 losses.fAttached += lost[3];
77 losses.fOther += lost[4];
78 for (int i = 0; i < count; i++) {
79 double values[5];
80 if (fread(values, sizeof(double), 5, pipe) != 5) return false;
81 avalanches.push_back({values[0], values[1], values[2], values[3], values[4]});
82 }
83 return true;
84}

105 response.fFieldMapOffset[2]};

109 for (const char axis : response.fPeriodicity) periodic[axis - 'x'] = true;
110 if (periodic[0]) response.fMirror ? map->EnableMirrorPeriodicityX() : map->EnablePeriodicityX();
111 if (periodic[1]) response.fMirror ? map->EnableMirrorPeriodicityY() : map->EnablePeriodicityY();
112 if (periodic[2]) response.fMirror ? map->EnableMirrorPeriodicityZ() : map->EnablePeriodicityZ();

114 map->GetElementaryCell(low[0], low[1], low[2], high[0], high[1], high[2]);
115 const double start = response.fStart * kCmPerMm, anode = response.fAnode * kCmPerMm;
116 if (start <= low[2] || start > high[2] || anode < low[2] || anode >= start) {
117 throw std::runtime_error(error + "start and anode must be inside the field maps along z (" +
118 std::to_string(low[2] / kCmPerMm) + " to " +
119 std::to_string(high[2] / kCmPerMm) + " mm), start above anode");
120 }

122 Garfield::Sensor sensor;
123 sensor.AddComponent(map.get());

125 for (int a = 0; a < 3; a++) {
126 const double margin = periodic[a] ? kPeriodicCells * (high[a] - low[a]) : 0;
127 areaLow[a] = low[a] - margin;
128 areaHigh[a] = high[a] + margin;
129 }
130 sensor.SetArea(areaLow[0], areaLow[1], areaLow[2], areaHigh[0], areaHigh[1], areaHigh[2]);

132 avalanche.SetSensor(&sensor);

134 response.fMedium = medium.GetName();
135 response.fPressure = medium.GetPressure();
136 response.fTemperature = medium.GetTemperature();
137 response.fPenning = penning;
138 // The cell of the map of the response, x and y in mm
139 response.fCellLow = {low[0] / kCmPerMm, low[1] / kCmPerMm};
140 response.fCellHigh = {high[0] / kCmPerMm, high[1] / kCmPerMm};

142 // The avalanches of a process (part of parts), with their progress: start point, size, time and
143 // electrons in another cell of each primary, and where the lost ones ended
144 auto computePart = [&](size_t part, size_t parts, Losses& losses, std::vector<Avalanche>& avalanches) {
145 const int electrons = response.fElectrons;
146 const int partElectrons = (electrons - static_cast<int>(part) + static_cast<int>(parts) - 1) / parts;
147 int reported = 0, primaries = 0;
148 for (int e = static_cast<int>(part); e < electrons; e += parts) {
149 const double x = low[0] + Garfield::RndmUniform() * (high[0] - low[0]);
150 const double y = low[1] + Garfield::RndmUniform() * (high[1] - low[1]);
151 avalanche.AvalancheElectron(x, y, start, 0, kInitialEnergy);
152 primaries++;
153 int size = 0, shared = 0;
154 double time = 0;
155 for (unsigned int i = 0; i < avalanche.GetNumberOfElectronEndpoints(); i++) {
156 double x0, y0, z0, t0, e0, x1, y1, z1, t1, e1;
157 int status = 0;
158 avalanche.GetElectronEndpoint(i, x0, y0, z0, t0, e0, x1, y1, z1, t1, e1, status);
159 if (z1 > anode) continue;
160 size++;
161 time += t1;
162 // On the pad of another cell (periodic copies)
163 if (x1 < low[0] || x1 > high[0] || y1 < low[1] || y1 > high[1]) shared++;
164 }
165 avalanches.push_back({x / kCmPerMm, y / kCmPerMm, static_cast<double>(size),
166 size > 0 ? time / size * kUsPerNs : 0, static_cast<double>(shared)});
167 if (size > 0) {
168 // Collected: nothing else
169 } else if (avalanche.GetNumberOfElectronEndpoints() > 0) {
170 // Lost: where its first electron that stopped ended
171 double x0, y0, z0, t0, e0, x1, y1, z1, t1, e1;
172 int status = 0;
173 avalanche.GetElectronEndpoint(0, x0, y0, z0, t0, e0, x1, y1, z1, t1, e1, status);
174 const CSTRegion region = map->GetRegionNear(x1, y1, z1);
175 if (status == Garfield::StatusAttached) {
176 losses.fAttached++;
177 } else if (status == Garfield::StatusLeftDriftArea || region == CSTRegion::kOutside) {
178 losses.fOut++;
179 } else if (region == CSTRegion::kDielectric) {
180 losses.fDielectric++;
181 } else if (region == CSTRegion::kConductor) {
182 losses.fConductor++;
183 } else {
184 losses.fOther++;
185 }
186 } else {
187 losses.fOther++;
188 }
189 if (verbose && primaries * 10 / std::max(1, partElectrons) > reported) {
190 reported = primaries * 10 / std::max(1, partElectrons);
191 RESTInfo << "restGarfield: amplification response, " << primaries << " of " << partElectrons
192 << " avalanches" << (parts > 1 ? " (process " + std::to_string(part + 1) + ")" : "")
193 << RESTendl;
194 }
195 }
196 };

198 Losses losses;
199 std::vector<Avalanche> avalanches;
200 const size_t parts = std::max<size_t>(1, std::min<size_t>(std::max(jobs, 1), response.fElectrons));
201 if (parts == 1) {
202 computePart(0, 1, losses, avalanches);
203 response.SetResult(avalanches, losses);
204 return;
205 }
206 // In parallel (Garfield++ is not thread safe): forked processes, each with its own random seed (from the
207 // generator of Garfield++, drawn once here), send their avalanches through a pipe
208 const auto seed = static_cast<unsigned int>(Garfield::RndmUniform() * 1e9) + 1;
209 struct Child {
210 pid_t pid;
211 FILE* pipe;
212 };
213 std::vector<Child> children;
214 for (size_t part = 0; part < parts; part++) {
215 int fd[2];
216 if (pipe(fd) != 0) throw std::runtime_error(error + "cannot create a pipe");
217 std::cout.flush();
218 std::fflush(nullptr);
219 const pid_t pid = fork();
220 if (pid < 0) throw std::runtime_error(error + "cannot create a process");
221 if (pid == 0) {
222 close(fd[0]);
223 int status = 1;
224 try {
225 Garfield::randomEngine.Seed(seed + part);
226 Losses partLosses;
227 std::vector<Avalanche> partAvalanches;
228 computePart(part, parts, partLosses, partAvalanches);
229 FILE* output = fdopen(fd[1], "wb");
230 const bool ok = output != nullptr && WriteResult(output, partLosses, partAvalanches);
231 if (output && fclose(output) == 0 && ok) status = 0;
232 } catch (const std::exception& exception) {
233 std::cerr << "restGarfield: " << exception.what() << std::endl;
234 }
235 std::cout.flush();
236 // Without the exit handlers of the parent (ROOT)
237 _exit(status);
238 }
239 close(fd[1]);
240 children.push_back({pid, fdopen(fd[0], "rb")});
241 }
242 RESTInfo << "restGarfield: " << response.fElectrons << " avalanches of the amplification response in "
243 << parts << " processes" << RESTendl;
244 bool ok = true;
245 for (auto& child : children) {
246 ok = ok && child.pipe && ReadResult(child.pipe, losses, avalanches);
247 if (child.pipe) fclose(child.pipe);
248 int status = 0;
249 waitpid(child.pid, &status, 0);
250 ok &= WIFEXITED(status) && WEXITSTATUS(status) == 0;
251 }
252 if (!ok) throw std::runtime_error(error + "a process computing the avalanches failed");
253 response.SetResult(avalanches, losses);

256void RunAmplificationResponse(const TRestDetectorMedium& medium, TRestDetectorAmplificationResponse& response,
257 bool verbose, int jobs) {
258 const std::string error = "restGarfield: amplification response " + response.fResponseFile + ": ";
259 const std::string fileName = response.fResponseFile;
260 if (!response.fOverwrite && std::filesystem::exists(fileName)) {
261 RESTInfo << "restGarfield: amplification response " << fileName
262 << " already exists (set overwrite: true to compute it again)" << RESTendl;
263 return;
264 }
265 RESTInfo << "restGarfield: computing the amplification response " << fileName << " ("
266 << response.fElectrons << " avalanches)" << RESTendl;
267 const auto start = std::chrono::steady_clock::now();
268 ComputeAmplificationResponse(medium, response, verbose, jobs);
269 const double seconds = std::chrono::duration<double>(std::chrono::steady_clock::now() - start).count();
270 RESTInfo << "restGarfield: amplification response computed in " << seconds << " s with "
271 << std::max(jobs, 1) << " processes, " << seconds * std::max(jobs, 1) / response.fElectrons
272 << " s per avalanche; transparency " << response.GetTransparency() << ", mean gain "
273 << response.fMeanSize << ", theta " << response.GetTheta() << RESTendl;
274 const auto& lost = response.fLosses;
275 RESTInfo << "restGarfield: primaries lost " << response.fPrimaries - response.fCollected
276 << ": out of the cell " << lost.fOut << ", on a conductor " << lost.fConductor
277 << ", on a dielectric " << lost.fDielectric << ", attached " << lost.fAttached << ", other "
278 << lost.fOther << RESTendl;
279 if (response.fAnodeElectrons > 0) {
280 RESTInfo << "restGarfield: electrons at the anode out of the cell of their primary (charge sharing) "
281 << response.fSharedElectrons / response.fAnodeElectrons << ", sigma of the displacement "
282 << response.GetSharingSigma() * 1000 << " um" << RESTendl;
283 }
284 if (response.HasCellMap()) {
285 double minimum = 1, maximum = 0;
286 for (size_t bin = 0; bin < response.fCellPrimaries.size(); bin++) {
287 if (response.fCellPrimaries[bin] <= 0) continue;
288 minimum = std::min(minimum, response.GetCellTransparency(bin));
289 maximum = std::max(maximum, response.GetCellTransparency(bin));
290 }
291 RESTInfo << "restGarfield: map of the cell, " << response.fCellBins << " x " << response.fCellBins
292 << " bins, " << static_cast<double>(response.fPrimaries) / response.fCellPrimaries.size()
293 << " primaries per bin, transparency from " << minimum << " to " << maximum << RESTendl;
294 // Spread of the gain factors of the bins and the one expected from the statistics alone (relative
295 // variance of the mean size of n avalanches 1 / ((1 + theta) n)): a gain depending on the position
296 // gives a spread above the statistical one; a statistical spread of a few % is applied as a false
297 // non-uniformity, more avalanches reduce it
298 double spread = 0, expected = 0;
299 int bins = 0;
300 for (size_t bin = 0; bin < response.fCellCollected.size(); bin++) {
301 if (response.fCellCollected[bin] <= 0) continue;
302 const double factor = response.GetCellGainFactor(static_cast<int>(bin));
303 spread += (factor - 1) * (factor - 1);
304 expected += 1 / ((1 + response.GetTheta()) * response.fCellCollected[bin]);
305 bins++;
306 }
307 if (bins > 0) {
308 RESTInfo << "restGarfield: gain factors of the bins, rms " << std::sqrt(spread / bins)
309 << ", statistical " << std::sqrt(expected / bins) << RESTendl;
310 }
311 }
312
313 YAML::Node generator;
314 generator["program"] = "Garfield++ AvalancheMicroscopic (field maps of a unit cell)";
315 const std::time_t now = std::time(nullptr);
316 char date[32];
317 std::strftime(date, sizeof(date), "%Y-%m-%d %H:%M:%S", std::localtime(&now));
318 generator["date"] = date;
319 generator["code"] = REST_Version::GetSummary();
320 const std::filesystem::path path(fileName);
321 if (path.has_parent_path()) std::filesystem::create_directories(path.parent_path());
322 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "RECREATE"));
323 if (!file || file->IsZombie()) throw std::runtime_error(error + "cannot create the file");
324 response.Write(file.get(), generator);
325 RESTInfo << "restGarfield: amplification response written to " << fileName << RESTendl;
326}
