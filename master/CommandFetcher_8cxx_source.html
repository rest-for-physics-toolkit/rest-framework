29void CommandFetcher::handleCommand(const std::string& line) {
30 if (line.empty()) return;
31
32 std::istringstream iss(line);
33 std::string cmd;
34 iss >> cmd;
35
36 std::vector<std::string> args;
37 std::string arg;
38 while (iss >> arg) args.push_back(arg);
39
40 if (cmd == "exit" || cmd == "quit") {
41 requestShutdown();
42 return;
43 } else if (cmd == "exec" && !args.empty()) {
44 execFile(args[0]);
45 return;
46 } else if (cmd == "fem" && !args.empty()) {
47 daq->setActiveFEM(args[0]);
48 return;
49 } else if (cmd == "addMetadata") {
50 // Write Metadata
51 if (!daq->isReadOnly()) daq->UpdateRunConfigInfo();
52 } else if (cmd == "fopen") {
53 // Open log and root files
54 if (!daq->isReadOnly()) daq->OpenFiles(args);
55 // Flag options, empty/all --> Logs and root files, log --> only logs,
56 // root --> only root
57 } else if (cmd == "fclose") {
58 // Close log and root files
59 if (!daq->isReadOnly()) daq->CloseFiles();
60 } else if (cmd == "credits") {
61 handleCreditsCommand(args);
62 } else if (cmd == "startDAQ") {
63 // Start acquisiton loop
64 daq->startDAQ(args);
65 } else if (cmd == "stopDAQ") {
66 // Stop acquisiton loop
67 daq->stopDAQ();
68 } else if (cmd == "Pedestals") {
69 // Start Pedestal loop (DCC)
70 daq->Pedestals(args);
71 } else if (cmd == "verbose" && !args.empty()) {
72 // Setting verbose (override run config)
73 runConfig.SetVerboseLevel(args[0]);
74 } else if (cmd == "sleep" && !args.empty()) {
75 int sleepTime = 1;
76 auto result = std::from_chars(args[0].data(), args[0].data() + args[0].size(), sleepTime);
77 if (result.ec == std::errc()) {
78 std::this_thread::sleep_for(std::chrono::seconds(sleepTime));
79 } else {
80 std::cout << "Cannot decode sleep time " << args[0] << std::endl;
81 }
82 } else {
83 // Send cmd to to DAQ
84 daq->SendCommand(line.c_str());
85 }
86}

88void CommandFetcher::handleCreditsCommand(const std::vector<std::string>& args) {
89 if (args.empty()) {
90 std::cout << "Usage: credits show | credits restore [frameThr reqFrames]" << std::endl;
91 return;
92 }
93
94 const std::string& subCmd = args[0];
95
96 if (subCmd == "show") {
97 daq->CreditsShow();
98 } else if (subCmd == "restore") {
99 if (args.size() == 1) {
100 daq->CreditsRestore();
101 } else if (args.size() >= 3) {
102 uint32_t frameThr = 0;
103 uint32_t reqFrames = 0;
104
105 auto res1 = std::from_chars(args[1].data(), args[1].data() + args[1].size(), frameThr);
106 auto res2 = std::from_chars(args[2].data(), args[2].data() + args[2].size(), reqFrames);
107
108 if (res1.ec != std::errc() || res2.ec != std::errc()) {
109 std::cout << "Error: Invalid integer arguments for credits restore." << std::endl;
110 } else if (frameThr == 0 || frameThr > reqFrames) {
111 // frameThr 0 would send requests without end, above reqFrames none
112 std::cout << "Error: credits restore needs 0 < frameThr <= reqFrames." << std::endl;
113 } else {
114 daq->CreditsRestore(reqFrames, frameThr);
115 }
116 } else {
117 std::cout << "Usage: credits restore (to use defaults) OR credits "
118 "restore <frameThr> <reqFrames>"
119 << std::endl;
120 }
121 } else {
122 std::cout << "Unknown credits sub-command: " << subCmd << std::endl;
123 }

178void CommandFetcher::runCommand(const std::string& line) {
179 // A failed command (wrong arguments, FEM timeout...) does not end the session
180 try {
181 handleCommand(line);
182 } catch (const std::exception& e) {
183 // CTRL-C during a command (e.g. addMetadata) ends the program, as in FEMDAQ
184 if (shutdownRequested()) throw;
185 std::cout << "[!] Command '" << line << "' failed: " << e.what() << std::endl;
186 }
