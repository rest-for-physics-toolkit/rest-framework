168void TRestDataSetCalibration::ComputeParameters(const TRestDataSet&, ROOT::RDF::RNode frame) {
169 auto histogram = frame.Histo1D({"spectrum", "", fNBins, fRange.first, fRange.second}, fObservable);
170 fSpectrum.reset(static_cast<TH1D*>(histogram->Clone("spectrum")));
171 fSpectrum->SetDirectory(nullptr);
172 fSpectrum->SetTitle((";" + fObservable + ";Counts").c_str());
173 if (fSpectrum->Integral() <= 0) {
174 throw std::runtime_error("TRestDataSetCalibration '" + fName + "': the spectrum of " + fObservable +
175 " is empty");
176 }
177
178 // The position of the peaks from the maximum, the first peak
179 const double maximumEnergy = fPeaks.front().fEnergy;
180 const auto model = TRestPeakFit::ModelFromString(fPeakModel);
181 const auto background = TRestPeakFit::BackgroundFromString(fBackground);
182
183 fLinearFit = std::make_unique<TGraph>();
184 fLinearFit->SetName("linearFit");
185 fLinearFit->SetTitle((";" + fObservable + ";Energy (keV)").c_str());
186 for (size_t i = 0; i < fPeaks.size(); i++) {
187 auto& peak = fPeaks[i];
188 const double energy = peak.fEnergy;
189 const auto zone = peak.HasRange()
190 ? peak.fRange
191 : TRestPeakFit::EstimateWindow(*fSpectrum, maximumEnergy, energy, fWindow);
192 auto window = zone;
193 if (fPeakSearch) {
194 if (const auto found = TRestPeakFit::FindPeakWindow(*fSpectrum, zone, fWindow)) {
195 window = *found;
196 } else {
197 RESTWarning << "TRestDataSetCalibration '" << fName << "': no peak found for " << energy
198 << " keV"
199 << " in (" << zone.first << ", " << zone.second << "), fitted in it" << RESTendl;
200 }
201 }
202 const auto [start, end] = window;
203 const auto fit =
204 TRestPeakFit::FitPeak(*fSpectrum, start, end, "g" + std::to_string(i), model, background);
205 if (!fit.valid) {
206 RESTWarning << "TRestDataSetCalibration '" << fName << "': the fit of the peak " << energy
207 << " keV in (" << start << ", " << end << ") did not converge" << RESTendl;
208 }
209 peak.fPosition = fit.mean;
210 peak.fFWHM = fit.GetRelativeFWHM();
211 peak.fFitRange = window;
212 peak.fFitStatus = fit.status;
213 peak.fFitParameters = YAML::Node(YAML::NodeType::Map);
214 for (const auto& [parameter, value] : fit.parameters)
215 peak.fFitParameters[parameter] = std::vector<double>{value.first, value.second};
216 // A failed fit (mean not finite) is not used for the linear fit
217 if (!std::isfinite(fit.mean)) {
218 RESTWarning << "TRestDataSetCalibration '" << fName << "': the position of the peak " << energy
219 << " keV is not finite (" << fit.mean << "), not used for the linear fit" << RESTendl;
220 continue;
221 }
222 fLinearFit->SetPoint(fLinearFit->GetN(), fit.mean, energy);
223 RESTDebug << " peak " << energy << " keV fitted in (" << start << ", " << end << "): " << fit.mean
224 << RESTendl;
225 }
226 if (fLinearFit->GetN() == 0) {
227 throw std::runtime_error("TRestDataSetCalibration '" + fName +
228 "': no peak could be fitted, no linear fit is possible");
229 }
230 if (fZeroPoint) fLinearFit->SetPoint(fLinearFit->GetN(), 0, 0);
231 try {
232 std::tie(fIntercept, fSlope) = TRestPeakFit::FitLine(*fLinearFit);
233 } catch (const std::exception& error) {
234 throw std::runtime_error("TRestDataSetCalibration '" + fName + "': " + error.what());
235 }
236 UpdateYAMLFromParams<TRestDataSetCalibration>(fNode);
237
238 RESTInfo << "TRestDataSetCalibration '" << fName << "': energy = " << fSlope << " * " << fObservable
239 << " + " << fIntercept << " keV" << RESTendl;
240}
