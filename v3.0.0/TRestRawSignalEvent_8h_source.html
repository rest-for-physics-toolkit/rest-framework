85 bool IsADCSaturation(int nFlat = 3) const {
86 if (nFlat <= 0) return false;
87 const int nPoints = GetNPoints();
88 if (nPoints <= 0) return false;

90 const int maxBin = std::distance(samples, std::max_element(samples, samples + nPoints));
91 if (maxBin + nFlat > nPoints) return false;

93 if (samples[i] != samples[maxBin]) return false;
