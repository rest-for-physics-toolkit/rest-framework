287vector<TRestGeant4ParticleState> TRestGeant4PrimaryGenerator::Generate() {
288 // The particles and positions of a new shower of CRY
289 if (auto* cry = dynamic_cast<TRestGeant4SpatialGeneratorCRY*>(fSpatial.get())) {
290 vector<TRestGeant4ParticleState> particles;
291 cry->NextShower(particles);
292 return particles;
293 }
294
295 const auto* cosmic = dynamic_cast<const TRestGeant4SpatialGeneratorCosmic*>(fSpatial.get());
296 // The cosmic origin depends on the direction, it is set below
297 const XYZVector position = cosmic ? XYZVector() : SamplePosition();
298
299 vector<TRestGeant4ParticleState> particles;
300 for (auto& source : fSources) {
301 const XYZVector center = fSpatial->GetCenter();
302 auto sourceParticles = source.Generate(fRandom, position, center, fSpatial->IsInward());
303 for (auto& particle : sourceParticles) {
304 if (cosmic) particle.SetOrigin(cosmic->SampleFromDirection(fRandom, particle.GetDirection()));
305 particles.push_back(particle);
306 }
307 }
308 return particles;
309}

315XYZVector TRestGeant4PrimaryGenerator::SamplePosition() {
316 const auto sample = [this]() {
317 return fSpatial->IsGdml() ? SampleGdmlPosition() : fSpatial->Sample(fRandom);
318 };
319 if (!fDensity) return sample();
320
321 // The density accepts each position with its probability
322 constexpr int maxAttempts = 10000000;
323 for (int attempt = 0; attempt < maxAttempts; attempt++) {
324 const XYZVector position = sample();
325 const double density = fDensity->Eval(position.X(), position.Y(), position.Z());
326 if (density > 1) {
327 throw runtime_error("TRestGeant4PrimaryGenerator: spatialDensityFunction > 1 at (" +
328 to_string(position.X()) + ", " + to_string(position.Y()) + ", " +
329 to_string(position.Z()) + ")");
330 }
331 if (fRandom.Uniform() < density) return position;
332 }
333 throw runtime_error("TRestGeant4PrimaryGenerator: no position accepted by spatialDensityFunction");
334}
