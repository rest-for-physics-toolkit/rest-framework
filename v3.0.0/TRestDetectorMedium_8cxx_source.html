62void TRestDetectorMedium::LoadConfig() {
63 UpdateParamsFromYAML<TRestDetectorMedium>(fNode);
64 const std::string prefix = "TRestDetectorMedium '" + fName + "': ";
65 fState = TRestTools::NormalizeOption("state", fState, kStates);
66 for (const auto& [component, fraction] : fComponents) {
67 if (fraction <= 0) throw std::runtime_error(prefix + "the fraction of " + component + " must be > 0");
68 }
69 if (fPressure <= 0 || fTemperature <= 0) {
70 throw std::runtime_error(prefix + "pressure and temperature must be > 0");
71 }
72
73 // Medium file: the definition (from the file, or checked against it), given to the responses
74 std::unique_ptr<TFile> file;
75 if (!fMediumFile.empty()) {
76 // A URL (e.g. a versioned table of rest-common) is downloaded and kept as the medium file
77 const bool url = TRestTools::IsURL(fMediumFile);
78 const std::string path = url ? TRestTools::DownloadFile(fMediumFile) : GetMediumFile().string();
79 if (std::filesystem::is_regular_file(path)) {
80 file.reset(TFile::Open(path.c_str(), "READ"));
81 if (!file || file->IsZombie()) throw std::runtime_error(prefix + "cannot open " + path);
82 if (!url) fMediumFile = path;
83 const YAML::Node metadata = TRestMetadata::ReadMetadata(file.get(), fName);
84 if (!metadata || !metadata.IsMap()) {
85 throw std::runtime_error(prefix + "no medium '" + fName + "' in " + path);
86 }
87 if (metadata["components"]) {
88 const auto components = metadata["components"].as<std::map<std::string, double>>();
89 const std::string state = TRestYAMLUtils::GetParam<std::string>(metadata, "state", fState);
90 if (fNode["components"] && (components != fComponents || state != fState)) {
91 throw std::runtime_error(prefix + "state or components different from those of " + path);
92 }
93 fComponents = components;
94 fState = state;
95 }
96 } else {
97 RESTWarning << prefix << "medium file " << fMediumFile
98 << " not found: no tables, the fixed parameters of the responses are used"
99 << RESTendl;
100 }
101 }
102 // The pressure only applies to gases. UpdateYAMLFromParams writes the default one (1 bar) also for the
103 // other states, so it is accepted (re-building the medium from its own node or from a file)
104 if (!IsGas() && fNode["pressure"] && std::abs(fPressure - 1.) > 1e-9) {
105 throw std::runtime_error(prefix + "pressure is only for gases (state " + fState + ")");
106 }
107
108 fCharge.reset();
109 if (fChargeNode && !fChargeNode.IsNull()) {
110 if (!fChargeNode.IsMap()) throw std::runtime_error(prefix + "charge must be a map");
111 fCharge = std::make_shared<TRestDetectorChargeResponse>("charge", fChargeNode);
112 fCharge->CheckConfig();
113 fCharge->SetMedium(*this, file.get());
114 }
115 fLight.reset();
116 if (fLightNode && !fLightNode.IsNull()) {
117 if (!fLightNode.IsMap()) throw std::runtime_error(prefix + "light must be a map");
118 fLight = std::make_shared<TRestDetectorLightResponse>("light", fLightNode);
119 fLight->CheckConfig();
120 }
121 fWLS.reset();
122 if (fWLSNode && !fWLSNode.IsNull()) {
123 if (!fWLSNode.IsMap()) throw std::runtime_error(prefix + "wls must be a map");
124 fWLS = std::make_shared<TRestDetectorWLSResponse>("wls", fWLSNode);
125 fWLS->CheckConfig();
126 }
127 UpdateYAMLFromParams<TRestDetectorMedium>(fNode);
128}

163 if (!name.empty()) {

176 }
177 found = medium;
178 }
179 return found;

198YAML::Node TRestDetectorMedium::GetFileMetadata(TFile* file, const std::string& name,
199 const YAML::Node& definition) {
200 // ReadMetadata gives a null node (defined for yaml-cpp) if the file has no medium yet
201 YAML::Node metadata = TRestMetadata::ReadMetadata(file, name);
202 if (!metadata || !metadata.IsMap()) metadata = YAML::Node(YAML::NodeType::Map);
203 if (metadata["components"]) {
204 if (definition["components"] && definition["components"].as<std::map<std::string, double>>() !=
205 metadata["components"].as<std::map<std::string, double>>()) {
206 throw std::runtime_error("TRestDetectorMedium: " + std::string(file->GetName()) +
207 " belongs to another medium (different components)");
208 }
209 } else {
210 // First table, or a file written without the definition: the definition is stored
211 if (definition["state"]) metadata["state"] = YAML::Clone(definition["state"]);
212 if (definition["components"]) metadata["components"] = YAML::Clone(definition["components"]);
213 }
214 metadata["name"] = name;
215 return metadata;

294double TRestDetectorScintillationProfile::SampleTime(TRandom& random) const {
295 const TRestDetectorScintillationComponent* component = &fComponents.back();
296 double u = random.Uniform();
297 for (const auto& c : fComponents) {
298 if (u < c.fFraction) {
299 component = &c;
300 break;
301 }
302 u -= c.fFraction;
303 }
304 double time = 0;
305 if (component->fDecayTime > 0) time += random.Exp(component->fDecayTime);
306 if (component->fRiseTime > 0) time += random.Exp(component->fRiseTime);
307 return time;
308}

471void TRestDetectorChargeResponse::LoadTables(TFile* file) {
472 const std::string prefix = "TRestDetectorChargeResponse of '" + fMediumName + "': ";
473 // Transport tables: at pressure and temperature, else (gases) the closest pressure
474 const YAML::Node transport = TRestDetectorTransportTable::GetTables(file, fMediumName);
475 std::vector<std::string> keys;
476 if (!fTransportTable.empty()) {
477 keys.push_back(fTransportTable);
478 } else {
479 double closest = -1;
480 for (size_t i = 0; i < transport.size(); i++) {
481 const double pressure = transport[i]["pressure"].as<double>();
482 const double temperature = transport[i]["temperature"].as<double>();
483 const bool scaling = TRestYAMLUtils::GetParam<bool>(transport[i], "pressureScaling", true);
484 if (!SameTemperature(temperature, fTemperature) ||
485 (!scaling && !SamePressure(pressure, fPressure)))
486 continue;
487 if (closest < 0 ||
488 std::abs(std::log(pressure / fPressure)) < std::abs(std::log(closest / fPressure)) - 1e-9) {
489 closest = pressure;
490 }
491 }
492 for (size_t i = 0; i < transport.size(); i++) {
493 if (closest > 0 && SamePressure(transport[i]["pressure"].as<double>(), closest) &&
494 SameTemperature(transport[i]["temperature"].as<double>(), fTemperature)) {
495 keys.push_back(transport[i]["key"].as<std::string>());
496 }
497 }
498 if (closest > 0 && !SamePressure(closest, fPressure)) {
499 RESTDebug << prefix << "transport tables at " << closest << " bar scaled with E/p to "
500 << fPressure << " bar" << RESTendl;
501 }
502 }
503 for (const auto& key : keys) {
504 auto table = std::make_shared<const TRestDetectorTransportTable>(
505 TRestDetectorTransportTable::Read(file, fMediumName, key));
506 if (!table->HasCarrier("electron")) {
507 throw std::runtime_error(prefix + "transport table " + key + " has no electrons");
508 }
509 if (!SameTemperature(table->fTemperature, fTemperature)) {
510 RESTWarning << prefix << "temperature " << fTemperature << " K but transport table " << key
511 << " is at " << table->fTemperature << " K" << RESTendl;
512 }
513 fTransportTables.push_back(table);
514 fLoadedTables.push_back(key);
515 }
516
517 // Gain tables: only at pressure and temperature
518 const YAML::Node gain = TRestDetectorGainTable::GetTables(file, fMediumName);
519 for (size_t i = 0; i < gain.size(); i++) {
520 if (!SamePressure(gain[i]["pressure"].as<double>(), fPressure) ||
521 !SameTemperature(gain[i]["temperature"].as<double>(), fTemperature)) {
522 continue;
523 }
524 const auto key = gain[i]["key"].as<std::string>();
525 fGainTables.push_back(std::make_shared<const TRestDetectorGainTable>(
526 TRestDetectorGainTable::Read(file, fMediumName, key)));
527 fLoadedTables.push_back(key);
528 }
529 if (fLoadedTables.empty()) {
530 RESTWarning << prefix << "no tables at " << fPressure << " bar and " << fTemperature << " K in "
531 << file->GetName() << ", the fixed transport parameters are used" << RESTendl;
532 }
533}

535const TRestDetectorTransportTable* TRestDetectorChargeResponse::GetTransportTable(double field) const {
536 const TRestDetectorTransportTable* best = nullptr;
537 double bestWidth = 0;
538 for (const auto& table : fTransportTables) {
539 const auto [minField, maxField] = table->GetFieldRange(fPressure);
540 if (field < minField || field > maxField) continue;
541 const double width = maxField / minField;
542 if (!best || width < bestWidth) {
543 best = table.get();
544 bestWidth = width;
545 }
546 }
547 if (best || fTransportTables.empty()) return best;
548
549 // Out of all of them: the closest range
550 double bestDistance = std::numeric_limits<double>::max();
551 for (const auto& table : fTransportTables) {
552 const auto [minField, maxField] = table->GetFieldRange(fPressure);
553 const double distance = std::abs(std::log(field < minField ? minField / field : field / maxField));
554 if (distance < bestDistance) {
555 best = table.get();
556 bestDistance = distance;
557 }
558 }
559 if (!fRangeWarned->exchange(true)) {
560 RESTWarning << "TRestDetectorChargeResponse of '" << fMediumName << "': field " << field * 10
561 << " V/cm out of the range of the transport tables, using the closest one ("
562 << best->GetKey() << ", constant out of its range)" << RESTendl;
563 }
564 return best;
565}

631const TRestDetectorGainTable& TRestDetectorChargeResponse::GetGainTable(double gap,
632 const std::string& key) const {
633 std::vector<const TRestDetectorGainTable*> candidates;
634 std::string available;
635 for (const auto& table : fGainTables) {
636 available += (available.empty() ? "" : ", ") + table->GetKey();
637 if (std::abs(table->fTableGap - gap) > 1e-6 * gap) continue;
638 if (!key.empty() && table->GetKey() != key) continue;
639 candidates.push_back(table.get());
640 }
641 const std::string prefix = "TRestDetectorChargeResponse of '" + fMediumName + "': ";
642 if (candidates.empty()) {
643 throw std::runtime_error(prefix + "no gain table with gap " + std::to_string(gap * 1000) + " um" +
644 (key.empty() ? "" : " and key " + key) + " at " + std::to_string(fPressure) +
645 " bar and " + std::to_string(fTemperature) +
646 " K (available: " + (available.empty() ? "none" : available) +
647 "; generate it with restGarfield)");
648 }
649 if (candidates.size() > 1) {
650 std::string keys;
651 for (const auto* table : candidates) keys += (keys.empty() ? "" : ", ") + table->GetKey();
652 throw std::runtime_error(prefix + "several gain tables with that gap (" + keys +
653 "): choose one with the key");
654 }
655 return *candidates.front();
656}

658bool TRestDetectorChargeResponse::DiffuseElectron(XYZVector& position, const XYZVector& normal,
659 double distance, double longDiff, double transDiff,
660 double attachment, TRandom& random) {
661 // Coefficients in sqrt(cm) and 1/cm, distances in mm
662 const double distanceCm = distance / 10.;
663 if (attachment > 0 && random.Uniform() > std::exp(-attachment * distanceCm)) return false;
664
665 const double sigmaL = 10. * longDiff * std::sqrt(distanceCm);
666 const double sigmaT = 10. * transDiff * std::sqrt(distanceCm);
667 // Transverse basis of the normal
668 const XYZVector u =
669 (std::abs(normal.X()) < 0.9 ? XYZVector(1, 0, 0) : XYZVector(0, 1, 0)).Cross(normal).Unit();
670 const XYZVector v = normal.Cross(u);
671 // One statement per draw: the order of evaluation of the operands of an expression depends on the
672 // compiler, and with it the coordinate each random number goes to
673 const double du = random.Gaus(0, sigmaT);
674 const double dv = random.Gaus(0, sigmaT);
675 const double dn = random.Gaus(0, sigmaL);
676 position += du * u + dv * v + dn * normal;
677 return true;
678}
