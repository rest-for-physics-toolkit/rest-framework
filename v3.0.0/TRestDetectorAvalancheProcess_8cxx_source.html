41double Gamma(TRandom& random, double shape, double scale) {
42 if (!(shape > 0) || !(scale > 0)) return 0;
43 if (shape < 1) {
44 // Gamma(shape + 1) U^(1 / shape)
45 const double gamma = Gamma(random, shape + 1, scale);
46 return gamma * std::pow(random.Rndm(), 1 / shape);
47 }
48 const double d = shape - 1. / 3;
49 const double c = 1 / std::sqrt(9 * d);
50 while (true) {
51 const double x = random.Gaus(0, 1);
52 double v = 1 + c * x;
53 if (v <= 0) continue;
54 v = v * v * v;
55 const double u = random.Rndm();
56 if (u < 1 - 0.0331 * x * x * x * x || std::log(u) < 0.5 * x * x + d * (1 - v + std::log(v))) {
57 return d * v * scale;
58 }
59 }
60}

112void TRestDetectorAvalancheProcess::InitProcess() {
113 const std::string user = "TRestDetectorAvalancheProcess";
114 if (fAmplificationVoltage <= 0 || fGap <= 0) {
115 throw std::runtime_error(user + ": amplificationVoltage and gap must be > 0");
116 }
117 if (!fManager) throw std::runtime_error(user + ": no TRestManager");
118 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "charge", user);
119 if (!medium) throw std::runtime_error(user + ": needs a TRestDetectorMedium with charge");
120 fCharge = &medium->GetChargeResponse(user);
121 if (fCharge->GetWValue() <= 0) {
122 throw std::runtime_error(user + ": the charge of the medium '" + medium->GetName() +
123 "' needs wValue > 0");
124 }
125
126 const auto& table = fCharge->GetGainTable(fGap, fTable);
127 const double field = fAmplificationVoltage / fGap; // V/mm
128 fGain = table.GetGain(field);
129 fTheta = table.GetTheta(field);
130 if (fGain <= 0 || fTheta <= -1) {
131 throw std::runtime_error(user + ": table " + table.GetKey() + " gives gain " + std::to_string(fGain) +
132 " and theta " + std::to_string(fTheta) +
133 " (gain > 0 and theta > -1 needed)");
134 }
135 fMediumName = medium->GetName();
136 fTable = table.GetKey();
137 // The stored metadata keeps the medium, the table and the gain used
138 UpdateYAMLFromParams<TRestDetectorAvalancheProcess>(fNode);
139
140 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
141 fRandom.SetSeed(fSeed);
142}

144bool TRestDetectorAvalancheProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
145 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
146 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
147 if (!inputEvent || !outputEvent) {
148 throw std::runtime_error(
149 "TRestDetectorAvalancheProcess: input/output is not a TRestDetectorHitsEvent");
150 }
151
152 outputEvent->Initialize();
153 outputEvent->SetEventInfo(inputEvent);
154 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
155
156 const double wValue = fCharge->GetWValue();
157 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
158 const double energy = inputEvent->GetEnergy(n);
159 if (energy <= 0) continue;
160 // Sum of the Polya gains of the electrons of the hit
161 const double electrons = energy / wValue;
162 const double gain = Gamma(fRandom, electrons * (fTheta + 1), fGain / (fTheta + 1));
163 outputEvent->AddHit(inputEvent->GetPosition(n), energy * gain / electrons, inputEvent->GetTime(n),
164 inputEvent->GetType(n), inputEvent->GetParticleID(n));
165 }
166
167 return outputEvent->GetNumberOfHits() > 0;
168}
