206void TRestFibreReadout::BuildGrids() {
207 fGrids.clear();
208 int firstID = 0;
209 for (const auto& family : fReadoutParams.fFibres) {
210 Grid grid;
211 grid.axis = family.GetAxis();
212 grid.u = (grid.axis + 1) % 3;
213 grid.v = (grid.axis + 2) % 3;
214 const int transverse[2] = {grid.u, grid.v};
215 int count[2] = {0, 0};
216 for (int t = 0; t < 2; t++) {
217 const double offset = family.fOffset[t];
218 const double size = fReadoutParams.fSize[transverse[t]];
219 if (offset >= 0 && offset <= size) {
220 // Fibres at offset + n pitch inside the box (the small tolerance keeps one at the far edge)
221 count[t] = static_cast<int>(std::floor((size - offset) / family.fPitch + 1e-9)) + 1;
222 }
223 }
224 grid.nU = count[0];
225 grid.nV = count[1];
226 grid.ends = family.GetReadoutEnds();
227 const int alternateAxis = family.GetAlternateAxis();
228 grid.alternate = alternateAxis < 0 ? -1 : alternateAxis == grid.u ? 0 : 1;
229 grid.firstID = firstID;
230 firstID += grid.nU * grid.nV * static_cast<int>(grid.ends.size());
231 fGrids.push_back(grid);
232 }
233}

265bool TRestFibreReadout::GetFibre(int physicalID, size_t& family, int& i, int& j, int& end) const {
266 for (size_t f = 0; f < fGrids.size(); f++) {
267 const auto& grid = fGrids[f];
268 const int fibres = grid.nU * grid.nV;
269 const int local = physicalID - grid.firstID;
270 if (local < 0 || local >= fibres * static_cast<int>(grid.ends.size())) continue;
271 family = f;
272 i = (local % fibres) / grid.nV;
273 j = (local % fibres) % grid.nV;
274 end = GetFibreEnd(grid, local / fibres, i, j);
275 return true;
276 }
277 return false;

281 if (!fNode || fNode.IsNull()) {
282 throw std::runtime_error("TRestFibreReadout: the YAML node is not initialized");
283 }
284 InitializeReadout();
285
286 // Decoding: the file if given (not empty nor the default of TRestDetectorReadout), otherwise DAQ ID =
287 // physical ID
288 if (!fDecodingFile.empty() && fDecodingFile != "default_decoding.dec") {
289 if (!LoadDecoding(fDecodingFile)) {
290 throw std::runtime_error("TRestFibreReadout::BuildGeometry - cannot load the decoding file '" +
291 fDecodingFile + "' (not found or without channels)");
292 }
293 } else {
294 std::map<int, int> identity;
295 for (int id = 0; id < GetNumberOfChannels(); id++) identity[id] = id;
296 SetDecodingMap(std::move(identity));
297 }

300 if (!fibreMedium) {

302 fibreMedium = new TGeoMedium("fibre_medium", 1, polystyrene);
303 }
304

306 const unsigned int kNoPhysicalID = static_cast<unsigned int>(-1);
307 int nodeCounter = 0;

310 const auto& family = fReadoutParams.fFibres[f];
311 // A thin tube to draw the fibre (the real one is in the GDML), along local Z rotated to the axis
312 const double radius = family.fPitch / 10.0;
313 const double halfLength = fReadoutParams.fSize[grid.axis] / 2.0;
314 auto* tube = new TGeoTube(("fibre_shape_" + std::to_string(f)).c_str(), 0, radius, halfLength);
315 auto* volume = new TGeoVolume(("FIBRE_" + std::to_string(f)).c_str(), tube, fibreMedium);
316 volume->SetLineColor(f % 2 == 0 ? kBlue : kGreen + 2);
317 auto* rotation = new TGeoRotation(("fibre_rotation_" + std::to_string(f)).c_str());
318 if (grid.axis == 0) rotation->RotateY(90);
319 if (grid.axis == 1) rotation->RotateX(90);
320
321 // One assembly per row of fibres: a single volume with all of them overflows the voxel finder
322 for (int i = 0; i < grid.nU; i++) {
323 auto* row =
324 new TGeoVolumeAssembly(("FIBRE_ROW_" + std::to_string(f) + "_" + std::to_string(i)).c_str());
325 for (int j = 0; j < grid.nV; j++) {
326 double centre[3];
327 centre[grid.axis] = fReadoutParams.fPosition[grid.axis];
328 centre[grid.u] = GetFibreCoordinate(f, 0, i);
329 centre[grid.v] = GetFibreCoordinate(f, 1, j);
330 // The physical ID of the first read end of the fibre
331 const int physicalID = grid.firstID + i * grid.nV + j;
332 TGeoNode* node = row->AddNode(volume, physicalID,
333 new TGeoCombiTrans(centre[0], centre[1], centre[2], rotation));
334 node->SetUniqueID(physicalID);
335 nodeCounter++;
336 }
337 TGeoNode* rowNode = top->AddNode(row, static_cast<int>(f * 100000 + i));
338 rowNode->SetUniqueID(kNoPhysicalID);
339 }
340 }
341 fGeoManager->CloseGeometry();
342 std::cout << "[+] Geometry built: " << nodeCounter << " fibres in " << fGrids.size() << " families."
343 << std::endl;

346int TRestFibreReadout::GetChannelFromPosition(double x, double y, double z) const {
347 const double position[3] = {x, y, z};
348 for (int axis = 0; axis < 3; axis++) {
349 const double minimum = GetMinimum(axis);
350 if (!(position[axis] >= minimum && position[axis] <= minimum + fReadoutParams.fSize[axis])) return -1;
351 }
352
353 int closestID = -1;
354 double closest = std::numeric_limits<double>::infinity();
355 const XYZVector point(x, y, z);
356 for (size_t f = 0; f < fGrids.size(); f++) {
357 const auto& grid = fGrids[f];
358 if (grid.nU == 0 || grid.nV == 0) continue;
359 const double pitch = fReadoutParams.fFibres[f].fPitch;
360 const int counts[2] = {grid.nU, grid.nV};
361 const int axes[2] = {grid.u, grid.v};
362 int index[2];
363 double distance2 = 0;
364 for (int t = 0; t < 2; t++) {
365 const double first = GetFibreCoordinate(f, t, 0);
366 index[t] = static_cast<int>(std::lround((position[axes[t]] - first) / pitch));
367 index[t] = std::clamp(index[t], 0, counts[t] - 1);
368 const double delta = position[axes[t]] - GetFibreCoordinate(f, t, index[t]);
369 distance2 += delta * delta;
370 }
371 if (distance2 < closest) {
372 closest = distance2;
373 // Read at both ends: the end closer to the position (the upper one from the centre)
374 int e = 0;
375 if (grid.ends.size() > 1 && Component(point, grid.axis) >= fReadoutParams.fPosition[grid.axis]) {
376 e = 1;
377 }
378 closestID = grid.firstID + e * grid.nU * grid.nV + index[0] * grid.nV + index[1];
379 }
380 }
381 return closestID < 0 ? -1 : GetDAQID(closestID);
382}

413std::vector<TRestDetectorReadout::ChannelInRange> TRestFibreReadout::GetChannelsInRange(
414 const XYZVector& position, double maxDistance) const {
415 std::vector<ChannelInRange> channels;
416 if (!(maxDistance >= 0)) return channels;
417
418 for (size_t f = 0; f < fGrids.size(); f++) {
419 const auto& grid = fGrids[f];
420 // The fibres span the box along their axis
421 const double along = Component(position, grid.axis);
422 const double minimum = GetMinimum(grid.axis);
423 if (!(along >= minimum && along <= minimum + fReadoutParams.fSize[grid.axis])) continue;
424
425 const double pitch = fReadoutParams.fFibres[f].fPitch;
426 const double pu = Component(position, grid.u), pv = Component(position, grid.v);
427 const double firstU = GetFibreCoordinate(f, 0, 0), firstV = GetFibreCoordinate(f, 1, 0);
428 const int iMin = std::max(0, static_cast<int>(std::ceil((pu - maxDistance - firstU) / pitch)));
429 const int iMax =
430 std::min(grid.nU - 1, static_cast<int>(std::floor((pu + maxDistance - firstU) / pitch)));
431 const int jMin = std::max(0, static_cast<int>(std::ceil((pv - maxDistance - firstV) / pitch)));
432 const int jMax =
433 std::min(grid.nV - 1, static_cast<int>(std::floor((pv + maxDistance - firstV) / pitch)));
434 for (int i = iMin; i <= iMax; i++) {
435 const double u = firstU + i * pitch;
436 for (int j = jMin; j <= jMax; j++) {
437 const double v = firstV + j * pitch;
438 const double distance = std::hypot(pu - u, pv - v);
439 if (distance > maxDistance) continue;
440 for (size_t e = 0; e < grid.ends.size(); e++) {
441 const int daqID =
442 GetDAQID(grid.firstID + static_cast<int>(e) * grid.nU * grid.nV + i * grid.nV + j);
443 if (daqID < 0) continue;
444 // The read end of the fibre
445 double end[3];
446 end[grid.axis] = GetEndCoordinate(f, GetFibreEnd(grid, e, i, j));
447 end[grid.u] = u;
448 end[grid.v] = v;
449 channels.push_back({daqID, distance, XYZVector(end[0], end[1], end[2])});
450 }
451 }
452 }
453 }
454 return channels;
455}
