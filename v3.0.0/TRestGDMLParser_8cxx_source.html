56std::string ReplaceMatches(const std::string& text, const std::regex& re, Function replacement) {
57 std::string out;
58 auto last = text.cbegin();
59 for (std::sregex_iterator it(text.begin(), text.end(), re), end; it != end; ++it) {
60 out.append(last, (*it)[0].first);
61 out += replacement(*it);
62 last = (*it)[0].second;
63 }
64 out.append(last, text.cend());
65 return out;
66}

81std::string ApplyConstants(const std::string& text, const std::map<std::string, std::string>& parameters,
82 std::set<std::string>* applied) {
83 return ReplaceMatches(text, kConstant, [&](const std::smatch& tag) {
84 const std::string element = tag[0].str();
85 std::smatch name;
86 if (!std::regex_search(element, name, kName)) return element;
87 const auto parameter = parameters.find(name[1].str());
88 if (parameter == parameters.end()) return element;
89 if (applied) applied->insert(parameter->first);
90 const std::string value = "value=\"" + ConstantValue(parameter->first, parameter->second) + "\"";
91 return std::regex_replace(element, kValue, value, std::regex_constants::format_first_only);

102std::string Expand(const std::string& location, std::map<std::string, Entity>& entities,
103 std::vector<std::string>& includes, const std::map<std::string, std::string>& parameters,
104 std::set<std::string>* applied) {
105 std::string text = ReadLocation(location);
106 // An included file: its content without its own XML declaration
107 if (!includes.empty()) text = std::regex_replace(text, kXmlDeclaration, "");
108 includes.push_back(location);
109
110 std::smatch doctype;
111 if (std::regex_search(text, doctype, kDoctype)) {
112 const std::string subset = doctype[1].str();
113 for (std::sregex_iterator it(subset.begin(), subset.end(), kEntity), end; it != end; ++it) {
114 const std::string name = (*it)[1].str();
115 const std::string value = (*it)[3].matched ? (*it)[3].str() : (*it)[4].str();
116 // As XML, the first declaration of an entity is the one used
117 if (entities.count(name)) continue;
118 Entity entity;
119 entity.external = (*it)[2].matched;
120 if (entity.external) {
121 entity.value = EntityLocation(value, location);
122 } else {
123 const auto parameter = parameters.find(name);
124 entity.value = parameter == parameters.end() ? value : parameter->second;
125 if (parameter != parameters.end() && applied) applied->insert(name);
126 }
127 entities[name] = entity;
128 }
129 text.erase(doctype.position(0), doctype.length(0));
130 }
131
132 text = ReplaceMatches(text, kReference, [&](const std::smatch& reference) {
133 const auto entity = entities.find(reference[1].str());
134 // Not declared (e.g. &lt; or &amp;): kept
135 if (entity == entities.end()) return reference[0].str();
136 if (!entity->second.external) return entity->second.value;
137 for (const auto& include : includes) {
138 if (include == entity->second.value) {
139 throw std::runtime_error("GDML: " + entity->second.value + " includes itself");
140 }
141 }
142 return Expand(entity->second.value, entities, includes, parameters, applied);
143 });
144 includes.pop_back();
145 return text;
146}

149void UpdateFile(const std::string& fileName, const std::map<std::string, std::string>& parameters,
150 std::set<std::string>* applied, std::set<std::string>& visited) {
151 if (!visited.insert(fs::weakly_canonical(fileName).string()).second) return;
152 const std::string text = ReadText(fileName);
153 std::vector<std::string> includes;
154 std::string updated = ReplaceMatches(text, kEntity, [&](const std::smatch& declaration) {
155 const std::string name = declaration[1].str();
156 const std::string value = declaration[3].matched ? declaration[3].str() : declaration[4].str();
157 if (declaration[2].matched) {
158 const std::string location = EntityLocation(value, fileName);
159 if (!IsURL(location)) includes.push_back(location);
160 return declaration[0].str();
161 }
162 const auto parameter = parameters.find(name);
163 if (parameter == parameters.end()) return declaration[0].str();
164 if (parameter->second.find('"') != std::string::npos) {
165 throw std::runtime_error("GDML: parameter '" + name + "' cannot contain '\"'");
166 }
167 if (applied) applied->insert(name);
168 // In an entity value % (parameter entity) and & are not allowed: character references
169 std::string escaped;
170 for (const char character : parameter->second) {
171 if (character == '%') {
172 escaped += "&#37;";
173 } else if (character == '&') {
174 escaped += "&#38;";
175 } else {
176 escaped += character;
177 }
178 }
179 return "<!ENTITY " + name + " \"" + escaped + "\">";
180 });
181 updated = ApplyConstants(updated, parameters, applied);
182 if (updated != text) {
183 std::ofstream output(fileName);
184 if (!output) throw std::runtime_error("GDML: cannot write " + fileName);
185 output << updated;
186 std::cout << "GDML: parameters written in " << fileName << std::endl;
187 }
188 for (const auto& include : includes) UpdateFile(include, parameters, applied, visited);
189}
