42bool FindWorldPlacement(const G4VPhysicalVolume* mother, const G4VPhysicalVolume* target,
43 const G4RotationMatrix& motherRotation, const G4ThreeVector& motherTranslation,
44 G4RotationMatrix& rotation, G4ThreeVector& translation) {
45 if (mother == target) {
46 rotation = motherRotation;
47 translation = motherTranslation;
48 return true;
49 }
50 const auto* logicalVolume = mother->GetLogicalVolume();
51 for (size_t i = 0; i < logicalVolume->GetNoDaughters(); i++) {
52 const auto* daughter = logicalVolume->GetDaughter(i);
53 const G4RotationMatrix daughterRotation = motherRotation * daughter->GetObjectRotationValue();
54 const G4ThreeVector daughterTranslation =
55 motherRotation * daughter->GetObjectTranslation() + motherTranslation;
56 if (FindWorldPlacement(daughter, target, daughterRotation, daughterTranslation, rotation,
57 translation)) {
58 return true;
59 }
60 }
61 return false;
62}

314void DetectorConstruction::ConstructSDandField() {
315 const TRestGeant4Metadata& metadata = *fSimulationManager->GetRestMetadata();
316
317 set<G4LogicalVolume*> logicalVolumesSelected;
318 for (const auto& userSensitiveVolume : metadata.GetSensitiveVolumes()) {
319 // Each sensitive detector declaration in the configuration should correspond to at least one logical
320 // volume
321 G4LogicalVolume* logicalVolume;
322 // Check if user selected a Geant4 physical volume by name
323 G4VPhysicalVolume* physicalVolume =
324 G4PhysicalVolumeStore::GetInstance()->GetVolume(userSensitiveVolume, false);
325 if (physicalVolume == nullptr) {
326 const G4String geant4VolumeName =
327 metadata.GetGeant4GeometryInfo().GetGeant4PhysicalNameFromAlternativeName(
328 userSensitiveVolume);
329 // Check if user selected a Geant4 physical volume by REST name (only relevant for assemblies)
330 physicalVolume = G4PhysicalVolumeStore::GetInstance()->GetVolume(geant4VolumeName, false);
331 }
332 if (physicalVolume == nullptr) {
333 // Check if user selected a Geant4 logical volume by name
334 logicalVolume = G4LogicalVolumeStore::GetInstance()->GetVolume(userSensitiveVolume, false);
335 } else {
336 // Sensitive detector already found
337 logicalVolume = physicalVolume->GetLogicalVolume();
338 }
339 if (logicalVolume != nullptr) {
340 logicalVolumesSelected.insert(logicalVolume);
341 } else {
342 // Check if the user string matches a logical volume by expanding the input as a regex
343 // This can have multiple hits
344 const auto logicalVolumesMatchingExpression =
345 metadata.GetGeant4GeometryInfo().GetAllLogicalVolumesMatchingExpression(userSensitiveVolume);
346 if (logicalVolumesMatchingExpression.empty()) {
347 cerr << "Detector construction error: could not find matching logical volume(s) for '"
348 << userSensitiveVolume << "'" << endl;
349 exit(1);
350 } else {
351 for (const auto& logicalVolumeName : logicalVolumesMatchingExpression) {
352 logicalVolumesSelected.insert(
353 G4LogicalVolumeStore::GetInstance()->GetVolume(logicalVolumeName, false));
354 }
355 }
356 }
357 }
358
359 G4SDManager* SDManager = G4SDManager::GetSDMpointer();
360
361 for (G4LogicalVolume* logicalVolume : logicalVolumesSelected) {
362 auto name = logicalVolume->GetName();
363 G4VSensitiveDetector* sensitiveDetector = new SensitiveDetector(fSimulationManager, name);
364 SDManager->AddNewDetector(sensitiveDetector);
365 logicalVolume->SetSensitiveDetector(sensitiveDetector);
366
367 auto region = new G4Region(name);
368 logicalVolume->SetRegion(region);
369 }
370}

372bool DetectorConstruction::IsPointInsideAnyDaughterVolume(const G4LogicalVolume* logVol,
373 const G4ThreeVector& point) const {
374 if (!logVol) {
375 G4cout << "DetectorConstruction::IsPointInsideAnyDaughterVolume : logVol is nullptr" << G4endl;
376 return false;
377 }
378
379 for (size_t i = 0; i < logVol->GetNoDaughters(); ++i) {
380 G4VPhysicalVolume* daughter = logVol->GetDaughter(i);
381 G4LogicalVolume* daughterLogVol = daughter->GetLogicalVolume();
382 G4VSolid* daughterSolid = daughterLogVol->GetSolid();
383
384 // localPoint will be the point in the daughter volume reference system
385 // which is the one used by the solid to determine if the point is inside or not
386 // GetRotation is the frame rotation (the inverse of the object one), which takes the point from
387 // the mother frame to the daughter one
388 G4ThreeVector localPoint = point - daughter->GetTranslation();
389 if (daughter->GetRotation()) {
390 localPoint = (*daughter->GetRotation()) * localPoint;
391 }
392
393 if (daughterSolid->Inside(localPoint) == kInside) {
394 return true;
395 }
396 }
397 return false;
398}
