81// Number of steps between consecutive points longer than factor times the median step
82int CountJumps(const std::vector<ROOT::Math::XYZVector>& points, double factor) {
83 if (points.size() < 3) return 0;
84 std::vector<double> steps;
85 for (size_t n = 1; n < points.size(); n++) steps.push_back(std::sqrt((points[n] - points[n - 1]).Mag2()));
86 std::vector<double> sorted = steps;
87 const size_t half = sorted.size() / 2;
88 std::nth_element(sorted.begin(), sorted.begin() + half, sorted.end());
89 double median = sorted[half];
90 if (sorted.size() % 2 == 0)
91 median = (median + *std::max_element(sorted.begin(), sorted.begin() + half)) / 2;
92 return (int)std::count_if(steps.begin(), steps.end(), [&](double s) { return s > factor * median; });
93}

145bool TRestTrackPathMinimizationProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
146 const auto* inputEvent = dynamic_cast<const TRestTrackEvent*>(&input);
147 auto* outputEvent = dynamic_cast<TRestTrackEvent*>(&output);
148 if (!inputEvent || !outputEvent) {
149 throw std::runtime_error("TRestTrackPathMinimizationProcess: input/output is not a TRestTrackEvent");
150 }
151
152 outputEvent->Initialize();
153 outputEvent->SetEventInfo(inputEvent);
154 fBranches = 0;
155
156 for (int t = 0; t < inputEvent->GetNumberOfTracks(); t++) {
157 const TRestHits hits = inputEvent->GetTrack(t).GetHits();
158 const int trackID = inputEvent->GetTrackID(t);
159 const size_t nHits = hits.GetNumberOfHits();
160 fNTracks++;
161
162 bool sameType = true;
163 for (size_t h = 1; h < nHits; h++) sameType &= hits.GetType(h) == hits.GetType(0);
164
165 std::vector<size_t> path;
166 if (sameType) {
167 path = TRestHitsAnalysis::GetShortestPath(hits.GetPoints(), fCyclic, fMaxExactNodes);
168 } else {
169 fNotOrdered++;
170 RESTDebug << "TRestTrackPathMinimizationProcess: event " << inputEvent->GetID() << ", track "
171 << trackID << " not ordered (hits of different types), kept as it is" << RESTendl;
172 for (size_t h = 0; h < nHits; h++) path.push_back(h);
173 }
174
175 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, trackID);
176 for (size_t h : path) {
177 outputEvent->AddHitToTrack(hits.GetPosition(h), hits.GetEnergy(h), hits.GetTime(h),
178 hits.GetType(h), inputEvent->GetSigma(t, (int)h));
179 }
180
181 if (sameType) {
182 const auto points = hits.GetPoints();
183 std::vector<ROOT::Math::XYZVector> ordered;
184 for (size_t h : path) ordered.push_back(points[h]);
185 fBranches = std::max(fBranches, CountJumps(ordered, fBranchFactor));
186 }
187 }
188
189 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) outputEvent->PrintOnlyTracks();
190 return true;
191}
