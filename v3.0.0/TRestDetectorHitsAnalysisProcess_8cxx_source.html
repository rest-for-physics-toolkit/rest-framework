115// Energy weighted sigma of one coordinate of the hits of a type (NaN without hits or energy)
116double GetSigma(const TRestHits& hits, int type, int axis) {
117 double sum = 0, sum2 = 0, energy = 0;
118 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
119 if (hits.GetType(n) != type) continue;
120 const double e = hits.GetEnergy(n);
121 const double c = hits.GetCoordinate(n, axis);
122 sum += e * c;
123 sum2 += e * c * c;
124 energy += e;
125 }
126 if (energy <= 0) return kNaN;
127 const double mean = sum / energy;
128 return std::sqrt(std::max(sum2 / energy - mean * mean, 0.));
129}

171void TRestDetectorHitsAnalysisProcess::InitProcess() {
172 // Projections and their pairs; the vectors do not change after this, so the registered references
173 // stay valid
174 const std::map<std::string, int> hitTypes = {{"X", TRestHitsData::X}, {"Y", TRestHitsData::Y},
175 {"Z", TRestHitsData::Z}, {"XY", TRestHitsData::XY},
176 {"XZ", TRestHitsData::XZ}, {"YZ", TRestHitsData::YZ},
177 {"XYZ", TRestHitsData::XYZ}};
178 fProjectionObs.clear();
179 for (const auto& name : fProjections) {
180 auto it = hitTypes.find(name);
181 if (it == hitTypes.end()) {
182 throw std::runtime_error("TRestDetectorHitsAnalysisProcess: unknown projection '" + name +
183 "' (X, Y, Z, XY, XZ, YZ or XYZ)");
184 }
185 ProjectionObservables projection;
186 projection.name = name;
187 projection.type = it->second;
188 fProjectionObs.push_back(projection);
189 }
190
191 fBalances.clear();
192 for (size_t i = 0; i < fProjectionObs.size(); i++) {
193 for (size_t j = i + 1; j < fProjectionObs.size(); j++) {
194 BalanceObservables balance;
195 balance.first = i;
196 balance.second = j;
197 // The coordinate only each projection defines, if there is exactly one
198 int nFirst = 0, nSecond = 0;
199 for (int a = 0; a < 3; a++) {
200 const bool inFirst = TRestHitsData::IsDefined(fProjectionObs[i].type, a);
201 const bool inSecond = TRestHitsData::IsDefined(fProjectionObs[j].type, a);
202 if (inFirst && !inSecond) {
203 balance.firstAxis = a;
204 nFirst++;
205 }
206 if (inSecond && !inFirst) {
207 balance.secondAxis = a;
208 nSecond++;
209 }
210 }
211 if (nFirst != 1 || nSecond != 1) balance.firstAxis = balance.secondAxis = -1;
212 fBalances.push_back(balance);
213 }
214 }
215
216 RegisterObservable("NHits", fNHits);
217 RegisterObservable("Energy", fEnergy);
218 RegisterObservable("MaxHitEnergy", fMaxHitEnergy);
219 RegisterObservable("MinHitEnergy", fMinHitEnergy);
220 RegisterObservable("MeanHitEnergy", fMeanHitEnergy);
221 RegisterObservable("MeanX", fMeanX);
222 RegisterObservable("MeanY", fMeanY);
223 RegisterObservable("MeanZ", fMeanZ);
224 RegisterObservable("SigmaX", fSigmaX);
225 RegisterObservable("SigmaY", fSigmaY);
226 RegisterObservable("SigmaZ", fSigmaZ);
227 RegisterObservable("SkewXY", fSkewXY);
228 RegisterObservable("SkewZ", fSkewZ);
229 // Only when computed, so that they are not in the tree without their parameters
230 if (fGaussFit) {
231 RegisterObservable("GaussSigmaX", fGaussSigmaX);
232 RegisterObservable("GaussSigmaY", fGaussSigmaY);
233 RegisterObservable("GaussSigmaZ", fGaussSigmaZ);
234 }
235 if (fFiducialRadius > 0) RegisterObservable("FiducialEnergy", fFiducialEnergy);
236
237 for (auto& projection : fProjectionObs) {
238 RegisterObservable("NHits_" + projection.name, projection.nHits);
239 RegisterObservable("Energy_" + projection.name, projection.energy);
240 }
241 for (auto& balance : fBalances) {
242 const std::string pair =
243 fProjectionObs[balance.first].name + "_" + fProjectionObs[balance.second].name;
244 RegisterObservable("NHitsBalance_" + pair, balance.nHits);
245 RegisterObservable("EnergyBalance_" + pair, balance.energy);
246 RegisterObservable("SigmaBalance_" + pair, balance.sigma);
247 }
248}

250bool TRestDetectorHitsAnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
251 output.CopyFrom(&input);
252 const auto* event = dynamic_cast<const TRestDetectorHitsEvent*>(&output);
253 if (!event) {
254 throw std::runtime_error("TRestDetectorHitsAnalysisProcess: output is not a TRestDetectorHitsEvent");
255 }
256
257 const TRestHits& hits = *event->GetHits();
258 // Assignments keep the addresses of the registered members
259 fNHits = hits.GetNumberOfHits();
260 if (fNHits == 0) return false;
261
262 fEnergy = hits.GetTotalEnergy();
263 fMaxHitEnergy = hits.GetMaximumHitEnergy();
264 fMinHitEnergy = hits.GetMinimumHitEnergy();
265 fMeanHitEnergy = hits.GetMeanHitEnergy();
266
267 // Coordinates any hit defines
268 bool defined[3] = {false, false, false};
269 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
270 for (int a = 0; a < 3; a++) defined[a] |= TRestHitsData::IsDefined(hits.GetType(n), a);
271 }
272 fMeanX = defined[0] ? hits.GetMeanPositionX() : kNaN;
273 fMeanY = defined[1] ? hits.GetMeanPositionY() : kNaN;
274 fMeanZ = defined[2] ? hits.GetMeanPositionZ() : kNaN;
275 fSigmaX = defined[0] ? hits.GetSigmaX() : kNaN;
276 fSigmaY = defined[1] ? hits.GetSigmaY() : kNaN;
277 fSigmaZ = defined[2] ? hits.GetSigmaZ() : kNaN;
278 fSkewXY = defined[0] || defined[1] ? hits.GetSkewXY() : kNaN;
279 fSkewZ = defined[2] ? hits.GetSkewZ() : kNaN;
280
281 double* gaussSigmas[3] = {&fGaussSigmaX, &fGaussSigmaY, &fGaussSigmaZ};
282 for (int a = 0; a < 3; a++) {
283 if (!defined[a]) {
284 *gaussSigmas[a] = kNaN;
285 } else if (fGaussFit) {
286 std::vector<double> values, energies;
287 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
288 if (!TRestHitsData::IsDefined(hits.GetType(n), a)) continue;
289 values.push_back(hits.GetCoordinate(n, a));
290 energies.push_back(hits.GetEnergy(n));
291 }
292 *gaussSigmas[a] = TRestHitsAnalysis::GetGaussSigma(values, energies);
293 }
294 }
295
296 fFiducialEnergy = 0;
297 if (fFiducialRadius > 0) {
298 const auto& axis = fFiducialAxis;
299 const double norm = std::sqrt(axis[0] * axis[0] + axis[1] * axis[1] + axis[2] * axis[2]);
300 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
301 // Position relative to the center, 0 along the coordinates the hit does not define
302 double d[3];
303 for (int a = 0; a < 3; a++) {
304 d[a] = TRestHitsData::IsDefined(hits.GetType(n), a)
305 ? hits.GetCoordinate(n, a) - fFiducialCenter[a]
306 : 0;
307 }
308 const double along = (d[0] * axis[0] + d[1] * axis[1] + d[2] * axis[2]) / norm;
309 const double radial2 = d[0] * d[0] + d[1] * d[1] + d[2] * d[2] - along * along;
310 if (radial2 > fFiducialRadius * fFiducialRadius) continue;
311 if (fFiducialLength > 0 && std::abs(along) > fFiducialLength / 2) continue;
312 fFiducialEnergy += hits.GetEnergy(n);
313 }
314 }
315
316 for (auto& projection : fProjectionObs) {
317 projection.nHits = 0;
318 projection.energy = 0;
319 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
320 if (hits.GetType(n) != projection.type) continue;
321 projection.nHits++;
322 projection.energy += hits.GetEnergy(n);
323 }
324 for (int a = 0; a < 3; a++) {
325 projection.sigma[a] =
326 TRestHitsData::IsDefined(projection.type, a) ? GetSigma(hits, projection.type, a) : kNaN;
327 }
328 }
329 for (auto& balance : fBalances) {
330 const auto& first = fProjectionObs[balance.first];
331 const auto& second = fProjectionObs[balance.second];
332 balance.nHits = Balance(first.nHits, second.nHits);
333 balance.energy = Balance(first.energy, second.energy);
334 balance.sigma = balance.firstAxis < 0
335 ? kNaN
336 : Balance(first.sigma[balance.firstAxis], second.sigma[balance.secondAxis]);
337 }
338
339 RESTDebug << "TRestDetectorHitsAnalysisProcess: event " << event->GetID() << ", " << fNHits
340 << " hits, energy " << fEnergy << RESTendl;
341 return true;
342}
