125 TrackLine line;
126 line.type = hits.GetType(0);

129 const auto energies = hits.GetEnergies();

135// Reverses the tracks going along the coordinates they share with the reference track (clearest Bragg
136// peak: largest asymmetry x energy) in the opposite way
137void OrientTracks(std::vector<TrackLine>& lines) {
138 size_t reference = 0;
139 for (size_t t = 1; t < lines.size(); t++) {
140 if (lines[t].asymmetry * lines[t].energy > lines[reference].asymmetry * lines[reference].energy) {
141 reference = t;
142 }
143 }
144 const XYZVector referenceDirection = lines[reference].end - lines[reference].origin;
145 for (size_t t = 0; t < lines.size(); t++) {
146 if (t == reference) continue;
147 const XYZVector direction = lines[t].end - lines[t].origin;
148 double dot = 0;
149 for (int a = 0; a < 3; a++) {
150 if (TRestHitsData::IsDefined(lines[t].type, a) &&
151 TRestHitsData::IsDefined(lines[reference].type, a)) {
152 dot += Get(direction, a) * Get(referenceDirection, a);
153 }
154 }
155 if (dot < 0) std::swap(lines[t].origin, lines[t].end);
156 }
157}

159// Each coordinate the mean over the tracks that define it (0 if none)
160void GetCombinedEnds(const std::vector<TrackLine>& lines, XYZVector& origin, XYZVector& end) {
161 double originSum[3] = {}, endSum[3] = {};
162 int count[3] = {};
163 for (const auto& line : lines) {
164 for (int a = 0; a < 3; a++) {
165 if (!TRestHitsData::IsDefined(line.type, a)) continue;
166 originSum[a] += Get(line.origin, a);
167 endSum[a] += Get(line.end, a);
168 count[a]++;
169 }
170 }
171 for (int a = 0; a < 3; a++) {
172 if (count[a] == 0) continue;
173 originSum[a] /= count[a];
174 endSum[a] /= count[a];
175 }
176 origin = Make(originSum);
177 end = Make(endSum);
178}

180// Coordinate `axis` of the hits at `value` of the coordinate `common`, linearly interpolated between the
181// hits sorted along `common` (the extreme hit beyond them)
182double Interpolate(const TRestHits& hits, int common, int axis, double value) {
183 std::vector<std::pair<double, double>> nodes;
184 for (size_t i = 0; i < hits.GetNumberOfHits(); i++) {
185 nodes.emplace_back(hits.GetCoordinate(i, common), hits.GetCoordinate(i, axis));
186 }
187 std::sort(nodes.begin(), nodes.end());
188 if (value <= nodes.front().first) return nodes.front().second;
189 if (value >= nodes.back().first) return nodes.back().second;
190 const auto upper = std::upper_bound(nodes.begin(), nodes.end(), std::make_pair(value, kNaN),
191 [](const auto& a, const auto& b) { return a.first < b.first; });
192 const auto lower = upper - 1;
193 const double width = upper->first - lower->first;
194 if (width <= 0) return lower->second;
195 return lower->second + (value - lower->first) / width * (upper->second - lower->second);
196}

198// The tracks paired along their common coordinate (the one with the largest extent of the oriented ends).
199// False if they share no coordinate
200bool GetPairedEnds(const std::vector<TRestHits>& hits, const std::vector<TrackLine>& lines, XYZVector& origin,
201 XYZVector& end) {
202 int common = -1;
203 double maxExtent = -1;
204 for (int a = 0; a < 3; a++) {
205 bool shared = true;
206 double extent = 0;
207 for (const auto& line : lines) {
208 shared &= TRestHitsData::IsDefined(line.type, a);
209 extent += std::abs(Get(line.end, a) - Get(line.origin, a));
210 }
211 if (shared && extent > maxExtent) {
212 maxExtent = extent;
213 common = a;
214 }
215 }
216 if (common < 0) return false;
217
218 double commonOrigin = 0, commonEnd = 0;
219 for (const auto& line : lines) {
220 commonOrigin += Get(line.origin, common);
221 commonEnd += Get(line.end, common);
222 }
223 commonOrigin /= lines.size();
224 commonEnd /= lines.size();
225
226 double o[3] = {}, e[3] = {};
227 for (int a = 0; a < 3; a++) {
228 if (a == common) {
229 o[a] = commonOrigin;
230 e[a] = commonEnd;
231 continue;
232 }
233 int count = 0;
234 for (size_t t = 0; t < lines.size(); t++) {
235 if (!TRestHitsData::IsDefined(lines[t].type, a)) continue;
236 o[a] += Interpolate(hits[t], common, a, commonOrigin);
237 e[a] += Interpolate(hits[t], common, a, commonEnd);
238 count++;
239 }
240 if (count > 0) {
241 o[a] /= count;
242 e[a] /= count;
243 }
244 }
245 origin = Make(o);
246 end = Make(e);
247 return true;
248}

256// Position of the half of the energy along Z, relative to the Z extent and from the minimum Z, as legacy
257// TRestTrackEvent::GetMaxTrackRelativeZ. Only the hits that define Z
258double GetRelativeZ(const std::vector<TRestHits>& hits) {
259 std::vector<std::pair<double, double>> zEnergy;
260 double totalEnergy = 0;
261 for (const auto& h : hits) {
262 for (size_t i = 0; i < h.GetNumberOfHits(); i++) {
263 if (!TRestHitsData::IsDefined(h.GetType(i), 2)) continue;
264 zEnergy.emplace_back(h.GetZ(i), h.GetEnergy(i));
265 totalEnergy += h.GetEnergy(i);
266 }
267 }
268 if (zEnergy.empty()) return kNaN;
269 std::sort(zEnergy.begin(), zEnergy.end());
270
271 double integral = 0;
272 size_t half = 0;
273 for (; half + 1 < zEnergy.size(); half++) {
274 integral += zEnergy[half].second;
275 if (integral >= totalEnergy / 2.) break;
276 }
277 const double length = zEnergy.front().first - zEnergy.back().first;
278 return length == 0 ? 0 : (zEnergy.front().first - zEnergy[half].first) / length;
279}

354bool TRestTrackLineAnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
355 const auto* inputEvent = dynamic_cast<const TRestTrackEvent*>(&input);
356 auto* outputEvent = dynamic_cast<TRestTrackEvent*>(&output);
357 if (!inputEvent || !outputEvent) {
358 throw std::runtime_error("TRestTrackLineAnalysisProcess: input/output is not a TRestTrackEvent");
359 }
360
361 outputEvent->Initialize();
362 outputEvent->SetEventInfo(inputEvent);
363
364 // Main tracks: the most energetic one or the most energetic of each projection (type of the first hit)
365 for (auto& projection : fProjectionTracks) {
366 projection.index = -1;
367 projection.maxEnergy = projection.energy = projection.balance = 0;
368 projection.length = projection.asymmetry = projection.residual = 0;
369 projection.angle = kNaN;
370 }
371 int maxIndex = -1;
372 double maxEnergy = 0, totalEnergy = 0;
373 for (int t = 0; t < inputEvent->GetNumberOfTracks(); t++) {
374 if (inputEvent->GetNumberOfHits(t) == 0) continue;
375 const double energy = inputEvent->GetTrackEnergy(t);
376 totalEnergy += energy;
377 if (energy > maxEnergy) {
378 maxEnergy = energy;
379 maxIndex = t;
380 }
381 const int type = inputEvent->GetTrack(t).GetHits().GetType(0);
382 for (auto& projection : fProjectionTracks) {
383 if (type != projection.type) continue;
384 projection.energy += energy;
385 if (energy > projection.maxEnergy) {
386 projection.maxEnergy = energy;
387 projection.index = t;
388 }
389 }
390 }
391
392 std::vector<int> mainTracks;
393 if (fProjectionTracks.empty()) {
394 if (maxIndex < 0) return false;
395 mainTracks.push_back(maxIndex);
396 fEnergy = maxEnergy;
397 fTrackBalance = totalEnergy > 0 ? maxEnergy / totalEnergy : 0;
398 } else {
399 double projectionsEnergy = 0;
400 fEnergy = 0;
401 for (auto& projection : fProjectionTracks) {
402 if (projection.index < 0) return false;
403 mainTracks.push_back(projection.index);
404 fEnergy += projection.maxEnergy;
405 projectionsEnergy += projection.energy;
406 projection.balance = projection.energy > 0 ? projection.maxEnergy / projection.energy : 0;
407 }
408 fTrackBalance = projectionsEnergy > 0 ? fEnergy / projectionsEnergy : 0;
409 }
410
411 std::vector<TRestHits> hits;
412 std::vector<TrackLine> lines;
413 int definedType = 1; // Coordinates any track of the main track defines
414 for (int t : mainTracks) {
415 hits.push_back(inputEvent->GetTrack(t).GetHits());
416 lines.push_back(AnalyseTrack(hits.back()));
417 for (int a = 0; a < 3; a++) {
418 if (TRestHitsData::IsDefined(lines.back().type, a) && !TRestHitsData::IsDefined(definedType, a)) {
419 definedType *= TRestHitsData::kAxisTypes[a];
420 }
421 }
422 }
423
424 XYZVector origin, end;
425 if (lines.size() == 1) {
426 origin = lines[0].origin;
427 end = lines[0].end;
428 } else {
429 OrientTracks(lines);
430 bool paired = false;
431 if (fLineAnaMethod == "3D") {
432 paired = GetPairedEnds(hits, lines, origin, end);
433 if (!paired && !fMethodWarned) {
434 RESTWarning
435 << "TRestTrackLineAnalysisProcess: the projections share no coordinate, the 3D method "
436 "cannot pair them, using default"
437 << RESTendl;
438 fMethodWarned = true;
439 }
440 }
441 if (!paired) GetCombinedEnds(lines, origin, end);
442 }
443
444 // Undefined coordinates: NaN in origin and end, 0 in the direction (not in the length)
445 const bool defined[3] = {TRestHitsData::IsDefined(definedType, 0),
446 TRestHitsData::IsDefined(definedType, 1),
447 TRestHitsData::IsDefined(definedType, 2)};
448 fOriginX = defined[0] ? origin.X() : kNaN;
449 fOriginY = defined[1] ? origin.Y() : kNaN;
450 fOriginZ = defined[2] ? origin.Z() : kNaN;
451 fEndX = defined[0] ? end.X() : kNaN;
452 fEndY = defined[1] ? end.Y() : kNaN;
453 fEndZ = defined[2] ? end.Z() : kNaN;
454 const XYZVector direction(defined[0] ? origin.X() - end.X() : 0, defined[1] ? origin.Y() - end.Y() : 0,
455 defined[2] ? origin.Z() - end.Z() : 0);
456 fLength = direction.R();
457 fAngle = GetPolarAngle(direction, defined[2]);
458 fPhi = defined[0] && defined[1] && direction.Rho() > 0 ? direction.Phi() : kNaN;
459 fDownwards = !defined[2] || direction.Z() == 0 ? 0 : direction.Z() > 0 ? 1 : -1;
460 fRelativeZ = GetRelativeZ(hits);
461
462 double energySum = 0, asymmetrySum = 0, residualSum = 0;
463 for (const auto& line : lines) {
464 energySum += line.energy;
465 asymmetrySum += line.energy * line.asymmetry;
466 residualSum += line.energy * line.residual;
467 }
468 fAsymmetry = energySum > 0 ? asymmetrySum / energySum : 0;
469 fResidual = energySum > 0 ? residualSum / energySum : 0;
470
471 // Main track of each projection, with its own (oriented) origin and end
472 for (size_t p = 0; p < fProjectionTracks.size(); p++) {
473 auto& projection = fProjectionTracks[p];
474 const auto& line = lines[p];
475 const XYZVector d = line.origin - line.end;
476 const XYZVector projected(TRestHitsData::IsDefined(line.type, 0) ? d.X() : 0,
477 TRestHitsData::IsDefined(line.type, 1) ? d.Y() : 0,
478 TRestHitsData::IsDefined(line.type, 2) ? d.Z() : 0);
479 projection.length = projected.R();
480 projection.angle = GetPolarAngle(projected, TRestHitsData::IsDefined(line.type, 2));
481 projection.asymmetry = line.asymmetry;
482 projection.residual = line.residual;
483 }
484
485 RESTDebug << "TRestTrackLineAnalysisProcess: origin (" << fOriginX << ", " << fOriginY << ", " << fOriginZ
486 << "), end (" << fEndX << ", " << fEndY << ", " << fEndZ << "), length " << fLength
487 << ", asymmetry " << fAsymmetry << RESTendl;
488
489 // The tracks of the main track, as they are
490 for (size_t m = 0; m < mainTracks.size(); m++) {
491 std::vector<XYZVector> sigmas;
492 for (int h = 0; h < (int)hits[m].GetNumberOfHits(); h++)
493 sigmas.push_back(inputEvent->GetSigma(mainTracks[m], h));
494 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, inputEvent->GetTrackID(mainTracks[m]),
495 hits[m], sigmas);
496 }
497 return true;
498}
