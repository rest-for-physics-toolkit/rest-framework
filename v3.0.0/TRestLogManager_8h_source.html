30 void Reset(int totalEvents, int barWidth = 30, const std::string& countUnit = "Events") {
31 total = totalEvents;
32 width = barWidth;
33 unit = countUnit;
34 current = 0;
35 started = true;
36 startTime = std::chrono::steady_clock::now();

42 void Update(int currentEvent) {
43 if (!started || total <= 0) return;
44 current = currentEvent;
45 if (current > total) current = total;
46
47 double percentage = (double)current / total;
48 int progress = width * percentage;
49
50 std::string etaStr = "inf";
51 if (current > 0) {
52 auto currentTime = std::chrono::steady_clock::now();
53 auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(currentTime - startTime).count();
54 double secondsPerEvent = (double)elapsed / current;
55 int remainingEvents = total - current;
56 int etaSeconds = remainingEvents * secondsPerEvent;
57
58 int etaMinutes = etaSeconds / 60;
59 etaStr = std::to_string(etaMinutes) + "." + std::to_string((etaSeconds % 60) * 10 / 60);
60 }
61
62 std::cout << "\r\033[K" << current << " " << unit << ", " << etaStr << " min ETA, "
63 << (int)(percentage * 100) << ".0%[";
64
65 for (int i = 0; i < width; ++i) {
66 if (i < progress) {
67 std::cout << "=";
68 } else if (i == progress && current < total) {
69 std::cout << ">";
70 } else {
71 std::cout << " ";
72 }
73 }
74 std::cout << "]";
75 std::cout.flush();
76 }
77
79 void Finish() {
80 if (!started) return;
81 Update(total);
82 std::cout << std::endl;
83 started = false;
84 }
85
86 private:
87 int total;
88 int current;
89 int width;
90 bool started;
91 std::string unit = "Events";
92 std::chrono::steady_clock::time_point startTime;
93};
