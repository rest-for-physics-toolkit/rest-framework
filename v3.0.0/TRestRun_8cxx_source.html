92 const YAML::Node& inputRun = fInputFileNode;
93 if (!inputRun || !inputRun.IsMap()) return;
94 // The run of the file: its parameters, the times of the data taking and its runInfo
95 YAML::Node parameters(YAML::NodeType::Map);
96 for (const auto& key : kRunParameters)

99 for (const std::string key :
100 {"startTime", "endTime", "mergedSubRuns", "entriesSaved", "codeVersion", "codeCommit", "codeClean"})
101 if (inputRun[key]) parameters[key] = inputRun[key];
102 UpdateParamsFromYAML<TRestRun>(parameters);
103 if (inputRun["runInfo"] && inputRun["runInfo"].IsMap())
104 fRunInformation = inputRun["runInfo"].as<std::map<std::string, std::string>>();

107void TRestRun::LoadConfig() {
108 if (!fNode || fNode.IsNull()) {
109 RESTError << "TRestRun::LoadConfig - node is missing or null" << RESTendl;
110 return;
111 }
112
113 // What is written in the configuration always wins
114 const YAML::Node config = YAML::Clone(fNode);
115 for (const auto& key : kRunParameters) {
116 if (config[key] && config[key].IsScalar() && config[key].Scalar() == "preserve") {
117 throw std::runtime_error(
118 "TRestRun::LoadConfig - " + key +
119 ": \"preserve\" is not used, the parameters not given are taken from the name of the input "
120 "file (inputFormat) or its run (inheritInputRun: true)");
121 }
122 }
123 const bool autoRunNumber =
124 config["runNumber"] && config["runNumber"].IsScalar() && config["runNumber"].Scalar() == "auto";
125 YAML::Node fields = YAML::Clone(config);
126 if (autoRunNumber) fields.remove("runNumber");
127 UpdateParamsFromYAML<TRestRun>(fields);
128 // A new run number ($REST_HOME/.rest/runNumber)
129 if (autoRunNumber) fRunNumber = GetRunNumberAuto();
130
131 // Empty: the output file relative to the launch directory (restG4 reads the configuration from its own
132 // directory, which GetFullPath("") would give)
133 if (!fMainDataPath.empty()) fMainDataPath = GetFullPath(fMainDataPath);
134
135 if ((fInputFileName.empty() || fInputFileName == "Null") && !fInputFormat.empty()) {
136 fInputFileName = ResolveFilePattern(fInputFormat);
137 }
138
139 // A pattern of several files (e.g. R*_ANA.root) is not opened: it is only for the tasks
140 // (TRestAnalysisPlot), the pipeline reads one file
141 const bool isPattern = fInputFileName.find_first_of("*?") != std::string::npos;
142 if (!fInputFileName.empty() && fInputFileName != "Null" && !isPattern) {
143 // Other files (e.g. a DAQ file) are not opened: they are the input of a process reading it
144 if (isValidTRestRun(GetFullPath(fInputFileName))) {
145 OpenInputFile(fInputFileName);
146 } else {
147 fInputFileName = GetFullPath(fInputFileName);
148 if (!fileExists(fInputFileName)) {
149 RESTWarning << "TRestRun: input file " << fInputFileName << " not found" << RESTendl;
150 } else {
151 RESTInfo << "TRestRun: " << fInputFileName << " is not a REST run file, not opened (input "
152 << "of a process)" << RESTendl;
153 }
154 }
155 }
156
157 // The parameters not given: from the name of the input file (inputFormat), otherwise from its run
158 // (inheritInputRun). The times of the data taking and the runInfo always from its run
159 const YAML::Node& inputRun = fInputFileNode;
160 const bool hasInputRun = inputRun && inputRun.IsMap();
161 const auto tags = ParseInputFormat();
162 YAML::Node inherited(YAML::NodeType::Map);
163 for (const auto& key : kRunParameters) {
164 if (config[key]) continue;
165 const auto tag = tags.find(key);
166 if (tag != tags.end()) {
167 if (key != "runNumber" && key != "subRunNumber") {
168 inherited[key] = tag->second;
169 continue;
170 }
171 // Decimal (as a YAML integer 00012 would be octal)
172 try {
173 inherited[key] = std::stoi(tag->second);
174 } catch (...) {
175 RESTWarning << "TRestRun: " << key << " " << tag->second << " of the input file name is not "
176 << "a number" << RESTendl;
177 }
178 } else if (fInheritInputRun && hasInputRun && inputRun[key]) {
179 inherited[key] = inputRun[key];
180 }
181 }
182 // A process reading raw data sets them again; the merged subruns of the data
183 for (const std::string key : {"startTime", "endTime", "mergedSubRuns"}) {
184 if (!config[key] && hasInputRun && inputRun[key]) inherited[key] = inputRun[key];
185 }
186 UpdateParamsFromYAML<TRestRun>(inherited);
187
188 // runInfo: the keys of the configuration, of the input file name ([runInfo.key]) and of its run
189 std::map<std::string, std::string> runInformation;
190 if (hasInputRun && inputRun["runInfo"] && inputRun["runInfo"].IsMap())
191 runInformation = inputRun["runInfo"].as<std::map<std::string, std::string>>();
192 for (const auto& [name, value] : tags) {
193 if (name.rfind("runInfo.", 0) == 0) runInformation[name.substr(8)] = value;
194 }
195 for (const auto& [key, value] : fRunInformation) runInformation[key] = value;
196 fRunInformation = runInformation;
197
198 ResolveOutputFileName();
199
200 // Sync resolved parameters and file names to the node
201 UpdateYAMLFromParams<TRestRun>(fNode);
202}

219void TRestRun::OpenInputFile(const std::string& filename) {
220 if (fInputFile || filename.empty()) return;
221
222 fInputFileName = GetFullPath(filename);
223
224 if (!isValidTRestRun(fInputFileName)) {
225 throw std::runtime_error("TRestRun: " + filename +
226 " is not a valid REST run file (not found, not a ROOT file, or without "
227 "AnalysisTree or TRestRun metadata)");
228 }
229
230 fInputFile = std::make_unique<TFile>(fInputFileName.c_str(), "READ");
231 if (!fInputFile || fInputFile->IsZombie()) {
232 throw std::runtime_error("TRestRun: Cannot open file " + filename);
233 }
234
235 fInputFile->GetObject("AnalysisTree", fAnalysisTree);
236
237 fInputEntries = fAnalysisTree->GetEntries();
238
239 YAML::Node selfConfig;
240
241 // Its TRestRun is this run; the other metadata are created, with the objects stored in the file
242 const auto [runInstance, runConfig] =
243 TRestYAMLUtils::GetMetadataClass(TRestYAMLUtils::ReadMetadataStore(fInputFile.get()), "TRestRun");
244 if (!runInstance.empty()) {
245 SetName(runInstance);
246 selfConfig = runConfig;
247 }
248 fMetadataStore.Load(fInputFile.get(), this, {"TRestRun"});
249
250 if (selfConfig && !selfConfig.IsNull()) {
251 fInputFileNode = selfConfig;
252 // Preserved unless the processes of this run produce another (TRestManager::SaveMetadata)
253 fOutputEvent =
254 TRestYAMLUtils::ReadParamOrDefault<std::string>(fInputFileNode, "outputEvent", fOutputEvent);
255 } else {
256 RESTWarning << "Not valid TRestRun Metadata in " << filename << RESTendl;
257 }
258
259 TList* keys = fInputFile->GetListOfKeys();
260 if (keys) {
261 for (int i = 0; i < keys->GetEntries(); ++i) {
262 TKey* key = dynamic_cast<TKey*>(keys->At(i));
263 if (!key) continue;
264 std::string className = key->GetName();
265 if (className == "AnalysisTree" || className == "RESTMetadataStore") continue;
266
267 if (EventRegistry::Instance().Contains(className)) {
268 TTree* tree = nullptr;
269 fInputFile->GetObject(className.c_str(), tree);
270 if (!tree) {
271 throw std::runtime_error("TRestRun: Tree not found: " + className);
272 }
273 fInputEventTrees[className] = tree;
274
275 auto eventObj = EventRegistry::Instance().Create(className, className);
276 eventObj->Initialize();
277 // The branch layout of the file must be one this version of the event class reads
278 const int version = TRestEvent::ReadFormatVersion(tree);
279 if (version < eventObj->GetMinimumFormatVersion() || version > eventObj->GetFormatVersion()) {
280 throw std::runtime_error("TRestRun: the " + className + " of " + fInputFile->GetName() +
281 " has format version " + std::to_string(version) +
282 (version == 0 ? " (no version, older format)" : "") +
283 ", this REST reads versions " +
284 std::to_string(eventObj->GetMinimumFormatVersion()) + " to " +
285 std::to_string(eventObj->GetFormatVersion()));
286 }
287 eventObj->SetInputFormatVersion(version);
288 eventObj->SetBranchAddresses(tree);
289 eventObj->SetRestRun(this);
290
291 if (fAnalysisTree && fInputEvents.empty()) {
292 eventObj->TRestEvent::SetBranchAddresses(fAnalysisTree);
293 }
294
295 fInputEvents[className] = eventObj.release();
296 fEvent = fInputEvents[className];
297 }
298 }
299 }
300
301 // The last saved event type by default, otherwise the last event tree
302 const auto outputIt = fInputEvents.find(fOutputEvent);
303 if (outputIt != fInputEvents.end()) fEvent = outputIt->second;
304}

374long long TRestRun::GetEntryWithID(int eventID, int subEventID, const std::string& tag) {
375 if (!fAnalysisTree) return -1;
376
377 TTreeReader reader(fAnalysisTree);
378
379 if (!fAnalysisTree->GetBranch("eventID") || !fAnalysisTree->GetBranch("subEventID")) {
380 RESTError << "Branches 'eventID' or 'subEventID' not found in AnalysisTree." << RESTendl;
381 return -1;
382 }
383
384 if (!tag.empty() && !fAnalysisTree->GetBranch("subEventTag")) {
385 RESTError << "Branch 'subEventTag' not found in AnalysisTree" << RESTendl;
386 return -1;
387 }
388
389 TTreeReaderValue<int> rvEventID(reader, "eventID");
390 TTreeReaderValue<int> rvSubEventID(reader, "subEventID");
391
392 std::unique_ptr<TTreeReaderValue<std::string>> rvSubEventTag = nullptr;
393 if (fAnalysisTree->GetBranch("subEventTag")) {
394 rvSubEventTag = std::make_unique<TTreeReaderValue<std::string>>(reader, "subEventTag");
395 }
396
397 while (reader.Next()) {
398 if (*rvEventID == eventID) {
399 if (subEventID != -1 && *rvSubEventID != subEventID) {
400 continue;
401 }
402 if (!tag.empty() && rvSubEventTag) {
403 if (**rvSubEventTag != tag) {
404 continue;
405 }
406 }
407
408 long long currentEntry = reader.GetCurrentEntry();
409 this->GetEntry(currentEntry);
410 return currentEntry;
411 }
412 }
413
414 return -1;
415}

488const void* TRestRun::GetObservableAddress(const std::string& name, const std::type_info& type) const {
489 const std::string prefix = "TRestRun: observable '" + name + "' ";
490 // While processing the output tree (input and previous processes), reading a file the input tree
491 TTree* tree = fOutputAnalysisTree ? fOutputAnalysisTree : fAnalysisTree;
492 if (!tree) throw std::runtime_error(prefix + "requested without analysis tree");
493 TBranch* branch = tree->GetBranch(name.c_str());
494 if (!branch) {
495 throw std::runtime_error(
496 prefix +
497 "not in the analysis tree (of the input file or of a previous process: is it "
498 "in the observables of that process?)");
499 }
500
501 TClass* branchClass = nullptr;
502 EDataType branchType = kNoType_t;
503 if (branch->GetExpectedType(branchClass, branchType) != 0) {
504 throw std::runtime_error(prefix + "has an unknown type");
505 }
506 TClass* requestedClass = TClass::GetClass(type);
507 const std::string requestedName =
508 requestedClass ? requestedClass->GetName() : TDataType::GetTypeName(TDataType::GetType(type));
509
510 const void* address = nullptr;
511 if (branchClass) {
512 // Object (std::string, std::vector...): the object bound to the branch
513 if (branchClass != requestedClass) {
514 throw std::runtime_error(prefix + "is " + branchClass->GetName() + ", not " + requestedName);
515 }
516 auto* element = dynamic_cast<TBranchElement*>(branch);
517 address = element ? element->GetObject() : nullptr;
518 } else {
519 // Number or boolean: the buffer of its single leaf (the variable of the process or the input buffer)
520 if (requestedClass || branchType != TDataType::GetType(type)) {
521 throw std::runtime_error(prefix + "is " + TDataType::GetTypeName(branchType) + ", not " +
522 requestedName);
523 }
524 auto* leaf = static_cast<TLeaf*>(branch->GetListOfLeaves()->At(0));
525 if (branch->GetListOfLeaves()->GetEntries() != 1 || !leaf || leaf->GetLenStatic() != 1) {
526 throw std::runtime_error(prefix + "is not a single value");
527 }
528 address = leaf->GetValuePointer();
529 }
530 if (!address) throw std::runtime_error(prefix + "has no data bound to its branch");
531 return address;
532}

534void TRestRun::SyncAnalysisTreeBranches() {
535 if (!fAnalysisTree || !fOutputAnalysisTree) return;
536
537 TObjArray* inputBranches = fAnalysisTree->GetListOfBranches();
538
539 for (int i = 0; i < inputBranches->GetEntries(); ++i) {
540 auto* inputBranch = dynamic_cast<TBranch*>(inputBranches->At(i));
541
542 if (!inputBranch) continue;
543
544 const char* branchName = inputBranch->GetName();
545
546 TBranch* outputBranch = fOutputAnalysisTree->GetBranch(branchName);
547
548 if (!outputBranch) {
549 continue;
550 }
551
552 if (outputBranch->GetAddress() != nullptr) {
553 continue;
554 }
555
556 // A branch of numbers not read by any process has no address: CopyAddresses has bound the leaves of
557 // the output branch to those of the input one, and SetAddress(nullptr) would give them a new buffer
558 // (the observable copied as 0)
559 if (inputBranch->GetAddress() == nullptr) {
560 continue;
561 }
562
563 outputBranch->SetAddress(inputBranch->GetAddress());
564 }

628std::map<std::string, std::string> TRestRun::ParseInputFormat() const {
629 std::map<std::string, std::string> parameters;
630 if (fInputFormat.empty() || fInputFileName.empty() || fInputFileName == "Null") return parameters;
631
632 // The pattern of the file name (without directories nor the extension of the file) as a regex: each
633 // [tag] a part without "_", the rest literal
634 std::string pattern = std::filesystem::path(fInputFormat).filename().string();
635 const std::string extension = std::filesystem::path(fInputFileName).extension().string();
636 if (!extension.empty() && pattern.size() > extension.size() &&
637 pattern.compare(pattern.size() - extension.size(), extension.size(), extension) == 0)
638 pattern.resize(pattern.size() - extension.size());
639 const std::regex bracketRe(R"(\[([\w.]+)\])");
640 const std::regex special(R"([.^$|()\[\]{}*+?\\])");
641 std::vector<std::string> tags;
642 std::string regexStr;
643 std::string rest = pattern;
644 std::smatch m;
645 while (std::regex_search(rest, m, bracketRe)) {
646 const std::string tag = m[1].str();
647 if (std::find(kRunParameters.begin(), kRunParameters.end(), tag) == kRunParameters.end() &&
648 tag.rfind("runInfo.", 0) != 0) {
649 throw std::runtime_error("TRestRun: [" + tag + "] in inputFormat \"" + fInputFormat +
