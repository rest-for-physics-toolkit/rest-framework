36bool AreSimilar(const std::string& a, const std::string& b) {
37 constexpr size_t k = kMaxDifferences, kOut = k + 1;
38 const size_t n = a.size(), m = b.size();
39 if ((n > m ? n - m : m - n) > k) return false;
40 // Levenshtein distance by rows, only in the band |i - j| <= k (the cells out of it are > k), so the cost
41 // is n (2k + 1) whatever the length; stopped when no path can stay within the limit
42 std::vector<size_t> previous(m + 1, kOut), current(m + 1, kOut);
43 for (size_t j = 0; j <= std::min(m, k); j++) previous[j] = j;
44 for (size_t i = 1; i <= n; i++) {
45 const size_t from = i > k ? i - k : 1, to = std::min(m, i + k);
46 current[from - 1] = from == 1 ? i : kOut;
47 size_t rowMin = current[from - 1];
48 for (size_t j = from; j <= to; j++) {
49 const size_t substitution = previous[j - 1] + (a[i - 1] == b[j - 1] ? 0 : 1);
50 current[j] = std::min({previous[j] + 1, current[j - 1] + 1, substitution, kOut});
51 rowMin = std::min(rowMin, current[j]);
52 }
53 // Out of the band for the next row
54 if (to < m) current[to + 1] = kOut;
55 if (rowMin > k) return false;
56 std::swap(previous, current);
57 }
58 return previous[m] <= k;
59}

80void TRestLogManager::Write(const std::string_view& color, const std::string& prefix, const std::string& name,
81 const std::string& msg, const char* file, int line) {
82 std::lock_guard<std::mutex> lock(fMutex);
83 const int limit = maxRepeated;
84 const double interval = repeatedInterval;
85 const auto now = std::chrono::steady_clock::now();
86
87 if (!file || limit <= 0) {
88 WriteText(color, prefix, name, msg);
89 return;
90 }
91
92 auto& groups = fRepeated->sites[{file, line}];
93 auto group = std::find_if(groups.begin(), groups.end(), [&msg](const RepeatedMessages::Repeated& g) {
94 return AreSimilar(g.first, msg);
95 });
96 if (group == groups.end() && groups.size() >= kMaxGroups) group = std::prev(groups.end());
97 if (group == groups.end()) {
98 groups.emplace_back();
99 group = std::prev(groups.end());
100 group->first = msg;
101 group->windowStart = now;
102 }
103 auto& repeated = *group;
104 if (repeated.suppressed > 0) {
105 repeated.suppressed++;
106 repeated.color = color;
107 repeated.prefix = prefix;
108 repeated.name = name;
109 repeated.lastMessage = msg;
110 return;
111 }
112 // A new time window (interval > 0): the messages of the previous one were fewer than the limit
113 if (interval > 0 && std::chrono::duration<double>(now - repeated.windowStart).count() >= interval) {
114 repeated.windowStart = now;
115 repeated.count = 0;
116 }
117 repeated.count++;
118 if (repeated.count < limit) {
119 WriteText(color, prefix, name, msg);
120 return;
121 }
122 // The limit reached in a window: this is the last one written, marked before its final line breaks, and
123 // the next ones are suppressed for ever (counted from 1 here, the summary subtracts it)
124 repeated.suppressed = 1;
125 std::string text = msg;
126 text.insert(text.find_last_not_of('\n') + 1, " (further similar messages suppressed)");
127 WriteText(color, prefix, name, text);
128}
