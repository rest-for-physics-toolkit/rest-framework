186std::pair<std::string, YAML::Node> TRestDAQViewer::FindSection(const YAML::Node& node,
187 const std::string& className) {
188 if (!node.IsMap()) return {};
189 for (const auto& element : node) {
190 const YAML::Node value = element.second;
191 if (!value.IsMap()) continue;
192 if (value["class"] && value["class"].IsScalar() && value["class"].Scalar() == className)
193 return {element.first.as<std::string>(), value};
194 auto found = FindSection(value, className);
195 if (found.second) return found;
196 }
197 return {};

200void TRestDAQViewer::LoadConfigFile(const char* path) {
201 if (gSystem->AccessPathName(path)) {
202 std::cout << "TRestDAQViewer: configuration file " << path << " not found" << std::endl;
203 return;
204 }
205
206 try {
207 const YAML::Node config = TRestYAMLUtils::OpenConfigFile(path);
208
209 // Binning of the spectra: parameters spectrumBins/Min/Max, and with _<readout> for a readout
210 const auto parameters = TRestYAMLUtils::GetParameters();
211 auto readBinning = [&](const std::string& suffix, const SpectrumBinning& defaults) {
212 SpectrumBinning binning = defaults;
213 auto read = [&](const std::string& name, double& value) {
214 const auto it = parameters.find(name + suffix);
215 if (it == parameters.end()) return;
216 TRestYAMLUtils::MarkParameterUsed(it->first);
217 value = std::stod(it->second);
218 };
219 double bins = binning.bins;
220 read("spectrumBins", bins);
221 read("spectrumMin", binning.min);
222 read("spectrumMax", binning.max);
223 binning.bins = static_cast<int>(bins);
224 if (binning.bins <= 0 || binning.max <= binning.min)
225 throw std::runtime_error("spectrum" + suffix + ": bins > 0 and max > min needed");
226 return binning;
227 };
228 const SpectrumBinning defaultSpectrum = readBinning("", SpectrumBinning());
229 std::map<std::string, SpectrumBinning> spectrumOfReadout;
230
231 // Pulse parameters, the defaults of the process if not given
232 const auto analysis = FindSection(config, "TRestRawSignalAnalysisProcess");
233 if (analysis.second)
234 fPulseParameters =
235 std::make_unique<TRestRawSignalAnalysisProcess>(analysis.first, analysis.second);
236 else
237 fPulseParameters = std::make_unique<TRestRawSignalAnalysisProcess>();
238
239 fReadoutManager.reset();
240 const auto readouts = FindSection(config, "TRestDetectorReadoutManager");
241 if (readouts.second) {
242 fReadoutManager = std::make_unique<TRestDetectorReadoutManager>(readouts.first, readouts.second);
243 fReadoutManager->LoadReadout();
244 for (const auto& info : fReadoutManager->fReadoutInfo)
245 spectrumOfReadout[info.fInstanceName] =
246 readBinning("_" + info.fInstanceName, defaultSpectrum);
247 }
248 fDefaultSpectrum = defaultSpectrum;
249 fSpectrumOfReadout = spectrumOfReadout;
250
251 // A parameter not used, e.g. the name of a readout misspelt: warned, the configuration is used
252 try {
253 TRestYAMLUtils::CheckParametersUsed();
254 } catch (const std::exception& e) {
255 std::cout << "TRestDAQViewer: WARNING " << e.what() << std::endl;
256 }
257 } catch (const std::exception& e) {
258 std::cout << "TRestDAQViewer: error reading " << path << ": " << e.what() << std::endl;
259 return;
260 }
261
262 fConfigFile = path;
263 fConfigPathEntry->SetText(gSystem->BaseName(path));
264 fConfigPathEntry->SetToolTipText(path);
265 std::cout << "TRestDAQViewer: pulse parameters: baselineRange [" << fPulseParameters->fBaselineRange.first
266 << ", " << fPulseParameters->fBaselineRange.second << "], integralRange ["
267 << fPulseParameters->fIntegralRange.first << ", " << fPulseParameters->fIntegralRange.second
268 << "], pointsThreshold " << fPulseParameters->fPointsThreshold << ", signalThreshold "
269 << fPulseParameters->fSignalThreshold << ", pointsOverThreshold "
270 << fPulseParameters->fPointsOverThreshold << std::endl;
271
272 BuildViews();
273}

288void TRestDAQViewer::BuildViews() {
289 DeleteViews();
290
291 auto addView = [&](const std::string& name, TRestDetectorReadout* readout) {
292 ReadoutView view;
293 view.name = name;
294 view.readout = readout;
295 const int n = fViews.size();
296 const auto binning = fSpectrumOfReadout.find(name);
297 const SpectrumBinning& spectrum =
298 binning == fSpectrumOfReadout.end() ? fDefaultSpectrum : binning->second;
299 view.spectrum = new TH1F(Form("hSpec%d", n), Form("%s;Amplitude (sum);Counts", name.c_str()),
300 spectrum.bins, spectrum.min, spectrum.max);
301 view.spectrum->SetDirectory(nullptr);
302 fViews.push_back(view);
303 };
304
305 if (fReadoutManager) {
306 for (const auto& info : fReadoutManager->fReadoutInfo) {
307 auto readout = fReadoutManager->GetReadout(info.fInstanceName);
308 if (readout) addView(info.fInstanceName, readout);
309 }
310 }
311 // No readout: all the signals together
312 if (fViews.empty()) addView("All signals", nullptr);
313
314 // Hit map of each readout, from the channels of its decoding: strips on two axes (a single coordinate
315 // of GetPositionFromChannel), otherwise the projections of the volumes of the channels
316 for (size_t n = 0; n < fViews.size(); n++) {
317 auto& view = fViews[n];
318 if (!view.readout) continue;
319 std::map<int, int> physicalToDAQ;
320 std::map<int, std::pair<int, double>> strips;
321 std::set<double> stripPositions[3];
322 std::istringstream decoding(view.readout->GetDecodingAsString());
323 std::string line;
324 while (std::getline(decoding, line)) {
325 if (line.empty() || line[0] == '#') continue;
326 std::istringstream fields(line);
327 int daqID, physicalID;
328 if (!(fields >> daqID >> physicalID)) continue;
329 fViewOfSignal[daqID] = n;
330 physicalToDAQ[physicalID] = daqID;
331 const auto position = view.readout->GetPositionFromChannel(daqID);
332 const double coordinates[3] = {position.X(), position.Y(), position.Z()};
333 int axis = -1, nFinite = 0;
334 for (int a = 0; a < 3; a++)
335 if (std::isfinite(coordinates[a])) {
336 axis = a;
337 nFinite++;
338 }
339 if (nFinite != 1) continue; // Not a strip
340 strips[daqID] = {axis, coordinates[axis]};
341 stripPositions[axis].insert(coordinates[axis]);
342 }
343
344 // The (first) two axes with strips
345 int nAxes = 0;
346 for (int a = 0; a < 3 && nAxes < 2; a++)
347 if (!stripPositions[a].empty()) view.stripAxes[nAxes++] = a;
348 if (nAxes < 2) {
349 view.stripAxes[0] = view.stripAxes[1] = -1;
350 BuildProjections(view, n, physicalToDAQ);
351 continue;
352 }
353
354 // Bins centred on the strips
355 auto range = [](const std::set<double>& values) {
356 const double pitch =
357 values.size() > 1 ? (*values.rbegin() - *values.begin()) / (values.size() - 1) : 1;
358 return std::make_pair(*values.begin() - pitch / 2, *values.rbegin() + pitch / 2);
359 };
360 const auto& positionsA = stripPositions[view.stripAxes[0]];
361 const auto& positionsB = stripPositions[view.stripAxes[1]];
362 const auto [minA, maxA] = range(positionsA);
363 const auto [minB, maxB] = range(positionsB);
364 view.hitmap = new TH2F(Form("hHit%zu", n),
365 Form("%s hit map;%s [mm];%s [mm]", view.name.c_str(),
366 kAxisNames[view.stripAxes[0]], kAxisNames[view.stripAxes[1]]),
367 positionsA.size(), minA, maxA, positionsB.size(), minB, maxB);
368 view.hitmap->SetDirectory(nullptr);
369 for (const auto& [daqID, strip] : strips)
370 if (strip.first == view.stripAxes[0] || strip.first == view.stripAxes[1])
371 fStripPosition[daqID] = strip;
372 }
373
374 UpdateCanvasLayout();
375}

377void TRestDAQViewer::BuildProjections(ReadoutView& view, int index, const std::map<int, int>& physicalToDAQ) {
378 std::vector<TRestDetectorReadout::ViewItem> items;
379 view.readout->GetViewItems(items);
380
381 // Bounding box (global frame) of each volume of a channel of the decoding
382 struct Box {
383 int daqID;
384 double minimum[3], maximum[3];
385 };
386 std::vector<Box> boxes;
387 double minimum[3] = {INFINITY, INFINITY, INFINITY}, maximum[3] = {-INFINITY, -INFINITY, -INFINITY};
388 for (const auto& item : items) {
389 const auto channel = physicalToDAQ.find(item.physicalID);
390 if (channel == physicalToDAQ.end()) continue;
391 auto* bbox = dynamic_cast<TGeoBBox*>(item.shape);
392 if (!bbox || !item.matrix) continue;
393 Box box{channel->second, {INFINITY, INFINITY, INFINITY}, {-INFINITY, -INFINITY, -INFINITY}};
394 const double* origin = bbox->GetOrigin();
395 for (int sx = -1; sx <= 1; sx += 2)
396 for (int sy = -1; sy <= 1; sy += 2)
397 for (int sz = -1; sz <= 1; sz += 2) {
398 const double local[3] = {origin[0] + sx * bbox->GetDX(), origin[1] + sy * bbox->GetDY(),
399 origin[2] + sz * bbox->GetDZ()};
400 double global[3];
401 item.matrix->LocalToMaster(local, global);
402 for (int a = 0; a < 3; a++) {
403 box.minimum[a] = std::min(box.minimum[a], global[a]);
404 box.maximum[a] = std::max(box.maximum[a], global[a]);
405 }
406 }
407 for (int a = 0; a < 3; a++) {
408 minimum[a] = std::min(minimum[a], box.minimum[a]);
409 maximum[a] = std::max(maximum[a], box.maximum[a]);
410 }
411 boxes.push_back(box);
412 }
413 if (boxes.empty()) return;
414
415 // The same bins (one per volume) in the three projections
416 for (int p = 0; p < 3; p++) {
417 const int a = kProjectionAxes[p][0], b = kProjectionAxes[p][1];
418 const double marginA = 0.05 * (maximum[a] - minimum[a]), marginB = 0.05 * (maximum[b] - minimum[b]);
419 auto projection = new TH2Poly(Form("hProj%d_%d", index, p),
420 Form("%s %s%s;%s [mm];%s [mm]", view.name.c_str(), kAxisNames[a],
421 kAxisNames[b], kAxisNames[a], kAxisNames[b]),
422 minimum[a] - marginA, maximum[a] + marginA, minimum[b] - marginB,
423 maximum[b] + marginB);
424 projection->SetDirectory(nullptr);
425 projection->SetStats(false);
426 for (const auto& box : boxes) {
427 const int bin =
428 projection->AddBin(box.minimum[a], box.minimum[b], box.maximum[a], box.maximum[b]);
429 if (p == 0) fChannelBins[box.daqID].push_back(bin);
430 }
431 view.projections[p] = projection;
432 }
433}

442double TRestDAQViewer::GetAmplitude(const std::vector<short>& pulse) const {
443 const auto& p = *fPulseParameters;
444 double baseline = 0, baselineSigma = 0;
445 const auto data = CalculateBaselineAndSigma(pulse, p.fBaselineRange.first, p.fBaselineRange.second,
446 baseline, baselineSigma, p.fBaselineOption);
447 auto range = p.fIntegralRange;
448 const auto points = GetPointsOverThreshold(data, range, {p.fPointsThreshold, p.fSignalThreshold},
449 p.fPointsOverThreshold, p.fNPointsFlat, baselineSigma);
450 // A signal: at least 2 points over threshold (NGoodSignals of TRestRawSignalAnalysisProcess)
451 if (points.size() < 2) return 0;
452 auto graph = GetGraphPair(points);
453 const auto maxPeak = GetMax(graph, p.fMaxOption);
454 if (maxPeak.first < 0 || maxPeak.first >= pulse.size()) return 0;
455 return maxPeak.second;
456}

504void TRestDAQViewer::UpdateCanvasLayout() {
505 TCanvas* c = fMainCanvas->GetCanvas();
506 c->Clear();
507 // A column per readout, plus the rate
508 c->Divide(fViews.size() + 1, kRows);
509 // The hit map of the readouts with projections: a pad for each one
510 for (size_t n = 0; n < fViews.size(); n++) {
511 if (!fViews[n].projections[0]) continue;
512 c->cd((fViews.size() + 1) + n + 1);
513 gPad->Divide(3, 1);
514 }
515 DrawCanvas();
516}

574void TRestDAQViewer::NextEvent() {
575 if (!fReader || !fTree) return;
576
577 const Long64_t totalEntries = RefreshEntries();
578
579 if (fEntry >= totalEntries) {
580 if (fIsRunning) {
581 fStatusLabel->SetText(Form("Status: Waiting DAQ... (Processed: %lld/%lld)",
582 fCumulativeEntries + fEntry, fCumulativeEntries + totalEntries));
583 } else {
584 fStatusLabel->SetText(Form("Status: End of File (%lld/%lld)", fCumulativeEntries + fEntry,
585 fCumulativeEntries + totalEntries));
586 }
587 return;
588 }
589
590 const int stepSize = fStepSize->GetNumber();
591
592 int eventsToProcess = (fIsRunning) ? stepSize : 1;
593 if (fEntry + eventsToProcess >= totalEntries) eventsToProcess = totalEntries - fEntry;
594
595 // Per view in an event: total amplitude, and the amplitude and weighted position of the strips of
596 // each axis (0 X, 1 Y, 2 Z)
597 struct ViewSums {
598 double total = 0, amplitude[3] = {0, 0, 0}, position[3] = {0, 0, 0};
599 };
600 std::vector<ViewSums> sums(fViews.size());
601 std::vector<short> pulse;
602
603 for (int i = 0; i < eventsToProcess; ++i) {
604 // Check if we reached the end of the available data
605 if (fEntry >= totalEntries) break;
606
607 if (fReader->SetEntry(fEntry) != TTreeReader::kEntryValid) break;
608
609 const int eventID = **fReaderEventID;
610 const double timestamp = **fReaderSeconds + **fReaderNanoSeconds * 1e-9;
611 const std::vector<int>& signalIDs = **fReaderSignalsID;
612 const std::vector<short>& samples = **fReaderPulses;
613 const std::vector<int>& offsets = **fReaderOffsets;
614
615 // Rate logic
616 if (fEntry == 0) {
617 fTimeRateStart = timestamp;
618 fRateGraph->SetPoint(fRatePoints, timestamp, 0.);
619 } else if (fEntry % stepSize == 0) {
620 double dt = timestamp - fTimeRateStart;
621 if (dt > 0) fRateGraph->SetPoint(fRatePoints++, (timestamp + fTimeRateStart) / 2., stepSize / dt);
622 fTimeRateStart = timestamp;
623 }
624
625 const bool shouldDraw = (i == eventsToProcess - 1);
626 if (shouldDraw) {
627 ClearEvent();
628 fStatusLabel->SetText(Form("Status: Running (Ev: %lld / %lld)", fCumulativeEntries + fEntry,
629 fCumulativeEntries + totalEntries));
630 }
631
632 std::fill(sums.begin(), sums.end(), ViewSums());
633 for (size_t s = 0; s < signalIDs.size() && s < offsets.size(); s++) {
634 const int sID = signalIDs[s];
635 const int view = GetViewOfSignal(sID);
636 if (view < 0) continue;
637
638 // Samples of the signal s: from its offset to the next one
639 const int first = offsets[s];
640 const int last = s + 1 < offsets.size() ? offsets[s + 1] : (int)samples.size();
641 if (first < 0 || last > (int)samples.size() || first >= last) continue;
642 pulse.assign(samples.begin() + first, samples.begin() + last);
643
644 if (shouldDraw) {
645 auto& stack = fViews[view].pulses;
646 if (!stack)
647 stack = new THStack(Form("hs%d", view),
648 Form("%s, event %d", fViews[view].name.c_str(), eventID));
649 const int nPoints = pulse.size();
650 TH1S* h = new TH1S(Form("h_s%d", sID), "", nPoints, 0, nPoints);
651 h->SetDirectory(nullptr);
652 for (int p = 0; p < nPoints; p++) h->SetBinContent(p + 1, pulse[p]);
653 h->SetLineColor((sID % 72) + 1);
654 fHistos.push_back(h);
655 stack->Add(h);
656 }
657
658 const double amplitude = GetAmplitude(pulse);
659 if (amplitude <= 0) continue;
660 auto& sum = sums[view];
661 sum.total += amplitude;
662 const auto strip = fStripPosition.find(sID);
663 if (strip != fStripPosition.end()) {
664 sum.position[strip->second.first] += amplitude * strip->second.second;
665 sum.amplitude[strip->second.first] += amplitude;
666 }
667 // Projections: one more event with a signal in the volumes of the channel
668 const auto bins = fChannelBins.find(sID);
669 if (bins != fChannelBins.end()) {
670 for (auto projection : fViews[view].projections)
671 for (int bin : bins->second)
672 projection->SetBinContent(bin, projection->GetBinContent(bin) + 1);
673 }
674 }
675
676 for (size_t n = 0; n < fViews.size(); n++) {
677 const auto& sum = sums[n];
678 const auto& view = fViews[n];
679 if (sum.total > 0) view.spectrum->Fill(sum.total);
680 if (!view.hitmap) continue;
681 const int a = view.stripAxes[0], b = view.stripAxes[1];
682 if (sum.amplitude[a] > 0 && sum.amplitude[b] > 0)
683 view.hitmap->Fill(sum.position[a] / sum.amplitude[a], sum.position[b] / sum.amplitude[b]);
684 }
685
686 fEntry++;
687 }
688
689 // Update the canvas to show the latest processed data
690 DrawCanvas();
691}

766void TRestDAQViewer::CheckNewFile(int offset) {
767 // Extract directory and build search patterns
768 TString dirName = gSystem->DirName(fBaseFileName);
769 void* dir = gSystem->OpenDirectory(dirName);
770 if (!dir) return;
771
772 TString nextSubrunPattern;
773 nextSubrunPattern.Form("Run%05d_.*_%03d.root", fCurrentRun, fCurrentSubrun + 1);
774
775 TString nextRunPattern;
776 nextRunPattern.Form("Run%05d_.*_000.root", fCurrentRun + offset);
777
778 const char* entry;
779 TString foundNextSubrun = "";
780 TString foundNextRun = "";
781
782 // Scan directory for matches
783 TPRegexp reSub(nextSubrunPattern);
784 TPRegexp reRun(nextRunPattern);
785
786 while ((entry = gSystem->GetDirEntry(dir))) {
787 TString fileName = entry;
788
789 // TPRegexp::Match returns the number of matches found. With offset 0 (ReloadRun) the current run is
790 // reloaded from its first subrun, the next subrun is not looked for
791 if (offset != 0 && reSub.Match(fileName) > 0) {
792 foundNextSubrun = dirName + "/" + fileName;
793 break; // Subrun takes priority
794 }
795 if (reRun.Match(fileName) > 0) {
796 foundNextRun = dirName + "/" + fileName;
797 }
798 }
799 gSystem->FreeDirectory(dir);
800
801 // Load found files
802 if (foundNextSubrun != "") {
803 std::cout << ">>> New Subrun found: " << std::endl;
804 fCurrentSubrun++;
805 fBaseFileName = foundNextSubrun;
806 fCumulativeEntries += fTree ? fTree->GetEntries() : 0;
807 LoadDataFile(foundNextSubrun, false);
808 fDataPathEntry->SetText(gSystem->BaseName(fBaseFileName));
809 fDataPathEntry->SetToolTipText(fBaseFileName);
810
811 } else if (foundNextRun != "") {
812 std::cout << ">>> New Run detected: " << std::endl;
813 ManualReset(); // Reset histograms for new main Run
814 fCurrentRun += offset;
815 fCumulativeEntries = 0;
816 fCurrentSubrun = 0;
817 fBaseFileName = foundNextRun;
818 LoadDataFile(foundNextRun);
819 }
820}
