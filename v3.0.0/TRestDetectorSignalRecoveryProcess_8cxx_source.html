67 fManager->GetMetadataClass("TRestDetectorReadoutManager"))
68 : nullptr;

71 }

73 // The stored metadata keeps the readout used

76 // Groups of consecutive dead channels, from the first one (its previous channel is alive) to the last
77 const std::set<int> dead(fChannelIds.begin(), fChannelIds.end());
78 std::set<int> grouped;
79 fGroups.clear();
80 for (const int id : fChannelIds) {
81 if (grouped.count(id)) continue;
82 if (fReadout->GetPhysicalID(id) < 0) {
83 RESTWarning << "TRestDetectorSignalRecoveryProcess: channel " << id
84 << " is not in the readout, not recovered" << RESTendl;
85 grouped.insert(id);
86 continue;
87 }
88
89 int first = id;
90 for (size_t n = 0; n < dead.size(); n++) {
91 const int previous = fReadout->GetAdjacentChannels(first).first;
92 if (previous < 0 || !dead.count(previous) || previous == id) break;
93 first = previous;
94 }
95
96 DeadGroup group;
97 group.left = fReadout->GetAdjacentChannels(first).first;
98 int channel = first;
99 while (true) {
100 group.dead.push_back(channel);
101 grouped.insert(channel);
102 const int next = fReadout->GetAdjacentChannels(channel).second;
103 if (next >= 0 && dead.count(next) && !grouped.count(next)) {
104 channel = next;
105 } else {
106 group.right = next;
107 break;
108 }
109 }
110
111 if (group.left < 0 || group.right < 0) {
112 RESTWarning << "TRestDetectorSignalRecoveryProcess: dead channels from " << group.dead.front()
113 << " to " << group.dead.back()
114 << " at the edge of the readout (no adjacent channel), not recovered" << RESTendl;
115 continue;
116 }
117 RESTDebug << "Dead channels from " << group.dead.front() << " to " << group.dead.back()
118 << ", adjacent channels " << group.left << " and " << group.right << RESTendl;
119 fGroups.push_back(group);
120 }

123bool TRestDetectorSignalRecoveryProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
124 const auto* inEvent = dynamic_cast<const TRestDetectorSignalEvent*>(&input);
125 auto* outEvent = dynamic_cast<TRestDetectorSignalEvent*>(&output);
126 if (!inEvent || !outEvent) {
127 throw std::runtime_error(
128 "TRestDetectorSignalRecoveryProcess: input/output is not a TRestDetectorSignalEvent");
129 }
130 outEvent->CopyFrom(inEvent);
131
132 auto points = [inEvent](int id) {
133 return inEvent->HasSignalID(id) ? inEvent->GetSignalByID(id).GetPointsVector()
134 : std::vector<std::pair<double, double>>{};
135 };
136
137 // Number of groups each neighbour gives half of its charge to
138 std::map<int, int> uses;
139 for (const auto& group : fGroups) {
140 const auto left = points(group.left);
141 const auto right = points(group.right);
142 if (left.empty() && right.empty()) continue;
143
144 const int k = static_cast<int>(group.dead.size());
145 const double sum = k * (k + 1) / 2.;
146 for (int j = 1; j <= k; j++) {
147 const double leftWeight = 0.5 * (k + 1 - j) / sum;
148 const double rightWeight = 0.5 * j / sum;
149 std::map<double, double> recovered;
150 for (const auto& [time, charge] : left) recovered[time] += leftWeight * charge;
151 for (const auto& [time, charge] : right) recovered[time] += rightWeight * charge;
152
153 const int id = group.dead[j - 1];
154 outEvent->RemoveSignalByID(id);
155 outEvent->AddSignal(id,
156 std::vector<std::pair<double, double>>(recovered.begin(), recovered.end()));
157 }
158 if (!left.empty()) uses[group.left]++;
159 if (!right.empty()) uses[group.right]++;
160
161 RESTDebug << "Event " << inEvent->GetID() << ": channels " << group.dead.front() << " to "
162 << group.dead.back() << " recovered" << RESTendl;
163 }
164
165 // The neighbours keep the charge not given to the dead channels
166 for (const auto& [id, n] : uses) outEvent->GetSignalByID(id).MultiplySignalBy(1 - 0.5 * n);
167
168 return true;
169}
