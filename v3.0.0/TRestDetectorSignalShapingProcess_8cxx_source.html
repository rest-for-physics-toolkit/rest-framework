95void TRestDetectorSignalShapingProcess::InitProcess() {
96 if (fShapingTime <= 0) {
97 throw std::runtime_error("TRestDetectorSignalShapingProcess: shapingTime must be > 0");
98 }
99 // Throws for an unknown type
100 TRestPulseShapeAnalysis::ShapingFunction(fShapingType, 0);
101 fSupport = TRestPulseShapeAnalysis::ShapingSupport(fShapingType);
102
103 // Tabulated response and its maximum
104 const double du = (fSupport.second - fSupport.first) / kSteps;
105 fResponse.assign(kSteps + 1, 0);
106 double maximum = 0;
107 for (int i = 0; i <= kSteps; i++) {
108 fResponse[i] = TRestPulseShapeAnalysis::ShapingFunction(fShapingType, fSupport.first + i * du);
109 maximum = std::max(maximum, fResponse[i]);
110 }
111 if (!(maximum > 0)) {
112 throw std::runtime_error("TRestDetectorSignalShapingProcess: the response '" + fShapingType +
113 "' has no positive maximum");
114 }
115 fNormalization = fGain / maximum;
116}

118double TRestDetectorSignalShapingProcess::Response(double u) const {
119 const double x = (u - fSupport.first) / (fSupport.second - fSupport.first) * kSteps;
120 if (x < 0 || x > kSteps) return 0;
121 const int i = std::min((int)x, kSteps - 1);
122 return fResponse[i] + (x - i) * (fResponse[i + 1] - fResponse[i]);
123}

125bool TRestDetectorSignalShapingProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
126 const auto* inputEvent = dynamic_cast<const TRestDetectorSignalEvent*>(&input);
127 auto* outputEvent = dynamic_cast<TRestDetectorSignalEvent*>(&output);
128 if (!inputEvent || !outputEvent) {
129 throw std::runtime_error(
130 "TRestDetectorSignalShapingProcess: input/output is not a TRestDetectorSignalEvent");
131 }
132
133 outputEvent->Initialize();
134 outputEvent->SetEventInfo(inputEvent);
135
136 for (int s = 0; s < inputEvent->GetNumberOfSignals(); s++) {
137 const auto& signal = inputEvent->GetSignal(s);
138 auto points = signal.GetPointsVector();
139 if (points.empty()) continue;
140 std::sort(points.begin(), points.end());
141
142 // Grid of multiples of step covering the response of each point, only on the union of their
143 // supports: points far apart in time (e.g. delayed decays) do not fill the time between them
144 const double step = fShapingTime / kPointsPerShapingTime;
145 auto firstIndex = [&](double time) {
146 return std::lround(std::floor((time + fSupport.first * fShapingTime) / step));
147 };
148 auto lastIndex = [&](double time) {
149 return std::lround(std::ceil((time + fSupport.second * fShapingTime) / step));
150 };
151
152 std::vector<double> times, values;
153 size_t begin = 0;
154 while (begin < points.size()) {
155 // Group of consecutive points whose supports overlap (or touch)
156 const long first = firstIndex(points[begin].first);
157 long last = lastIndex(points[begin].first);
158 size_t end = begin + 1;
159 while (end < points.size() && firstIndex(points[end].first) <= last + 1) {
160 last = std::max(last, lastIndex(points[end].first));
161 end++;
162 }
163 for (long k = first; k <= last; k++) {
164 const double t = k * step;
165 double value = 0;
166 for (size_t i = begin; i < end; i++) {
167 const double u = (t - points[i].first) / fShapingTime;
168 if (u < fSupport.first || u > fSupport.second) continue;
169 value += points[i].second * Response(u);
170 }
171 times.push_back(t);
172 values.push_back(fNormalization * value);
173 }
174 begin = end;
175 }
176 outputEvent->AddSignal(signal.GetSignalID(), times, values);
177 }
178
179 return outputEvent->GetNumberOfSignals() > 0;
180}
