30 ROOT::Math::XYZVector SampleGdmlPosition() override {
31 const auto* detector = static_cast<const DetectorConstruction*>(
32 G4RunManager::GetRunManager()->GetUserDetectorConstruction());
33 G4ThreeVector position;
34 if (GetInfo().IsGdmlSurface()) {
35 // TODO: G4VSolid::GetPointOnSurface is not uniform for every solid
36 position = detector->GetGeneratorSolid()->GetPointOnSurface();
37 } else {
38 auto& random = GetRandom();
39 do {
40 position = {random.Uniform(detector->GetBoundBoxXMin(), detector->GetBoundBoxXMax()),
41 random.Uniform(detector->GetBoundBoxYMin(), detector->GetBoundBoxYMax()),
42 random.Uniform(detector->GetBoundBoxZMin(), detector->GetBoundBoxZMax())};
43 } while (
44 detector->GetGeneratorSolid()->Inside(position) != kInside ||
45 detector->IsPointInsideAnyDaughterVolume(detector->GetGeneratorLogicalVolume(), position));
46 }
47 // The solid has no placement, rotate and translate it as its physical volume
48 position = detector->GetGeneratorRotation() * position + detector->GetGeneratorTranslation();
49 return {position.x(), position.y(), position.z()};
50 }

60G4ParticleDefinition* PrimaryGeneratorAction::FindGeant4Particle(const string& name, double excitation) {
61 G4ParticleDefinition* definition = nullptr;
62 if (excitation == 0) definition = G4ParticleTable::GetParticleTable()->FindParticle(name);
63 if (!definition) {
64 // An ion by its name, e.g. Pb210
65 auto* ionTable = G4IonTable::GetIonTable();
66 for (int Z = 1; Z <= 110 && !definition; Z++) {
67 for (int A = 2 * Z - 1; A <= 3 * Z && !definition; A++) {
68 if (name == ionTable->GetIonName(Z, A)) definition = ionTable->GetIon(Z, A, excitation * keV);
69 }
70 }
71 }
72 return definition;
73}

75G4ParticleDefinition* PrimaryGeneratorAction::FindParticle(const TRestGeant4ParticleState& particle) {
76 const string name = particle.GetParticleName();
77 const double excitation = particle.GetExcitationLevel();
78 const auto key = make_pair(name, excitation);
79 const auto cached = fParticles.find(key);
80 if (cached != fParticles.end()) return cached->second;
81
82 G4ParticleDefinition* definition = FindGeant4Particle(name, excitation);
83 if (!definition) {
84 ostringstream message;
85 message << "Primary particle '" << name << "' not found";
86 G4Exception("PrimaryGeneratorAction::FindParticle", "REST_G4_PARTICLE_SOURCE", FatalException,
87 message.str().c_str());
88 }
89 fParticles[key] = definition;
90 return definition;
91}

94 const auto start = chrono::high_resolution_clock::now();
95 auto outputManager = SimulationManager::GetOutputManager();
96 outputManager->SetEventTimeStart(start);
97
98 const auto* metadata = fSimulationManager->GetRestMetadata();
99
100 // Volume biasing, after the first stage: the event replays a particle stored by the previous stage
101 if (fSimulationManager->GetBiasingStage() > 0) {
102 const auto& particle = fSimulationManager->GetBiasingReplayParticle(event->GetEventID());
103 fParticleGun.SetParticleDefinition(const_cast<G4ParticleDefinition*>(particle.definition));
104 fParticleGun.SetParticleCharge(particle.charge); // after the definition, which sets its charge
105 fParticleGun.SetParticleEnergy(particle.energy);
106 fParticleGun.SetParticleMomentumDirection(particle.direction);
107 fParticleGun.SetParticlePolarization(particle.polarization);
108 fParticleGun.SetParticlePosition(particle.position);
109 fParticleGun.SetParticleTime(particle.time);
110 fParticleGun.GeneratePrimaryVertex(event);
111
112 const chrono::duration<double> elapsed = chrono::high_resolution_clock::now() - start;
113 outputManager->SetEventTimeWallPrimaryGeneration(elapsed.count());
114 outputManager->UpdatePrimaryData(event);
115 fSimulationManager->SetLastEventGenerated();
116 return;
117 }
118
119 if (!fGenerator) {
120 const auto seed = static_cast<unsigned int>(metadata->GetSeed() + G4Threading::G4GetThreadId() + 1);
121 fGenerator = make_unique<Geant4PrimaryGenerator>(metadata->GetGeant4PrimaryGeneratorInfo(), seed);
122 }
123
124 vector<TRestGeant4ParticleState> particles;
125 try {
126 particles = fGenerator->Generate();
127 } catch (const exception& error) {
128 G4Exception("PrimaryGeneratorAction::GeneratePrimaries", "REST_G4_GENERATOR", FatalException,
129 error.what());
130 }
131
132 // cry generator: its maxExposureTime was reached
133 if (particles.empty() && metadata->GetGeant4PrimaryGeneratorInfo().IsCRY()) {
134 if (!fSimulationManager->GetAbortFlag()) {
135 G4cout << "Stopping Run! The maxExposureTime of CRY was reached" << G4endl;
136 fSimulationManager->SetGeneratorExhausted();
137 fSimulationManager->StopSimulation();
138 }
139 event->SetEventAborted();
140 return;
141 }
142
143 // Volume biasing: a primary inside the outermost volume is never stopped by it (a warning at the end)
144 const auto* outermostVolume = fSimulationManager->GetStoppingBiasingVolume();
145
146 for (const auto& particle : particles) {
147 if (outermostVolume != nullptr) {
148 const auto& origin = particle.GetOrigin();
149 if (outermostVolume->isInside(origin.X(), origin.Y(), origin.Z())) {
150 fSimulationManager->CountBiasingPrimaryInside();
151 }
152 }
153 auto* definition = FindParticle(particle);
154 fParticleGun.SetParticleDefinition(definition);
155 if (definition->GetParticleType() == "nucleus") {
156 fParticleGun.SetParticleCharge(particle.GetParticleCharge() * eplus);
157 }
158 fParticleGun.SetParticleEnergy(particle.GetEnergy() * keV);
159 const auto& direction = particle.GetDirection();
160 fParticleGun.SetParticleMomentumDirection({direction.X(), direction.Y(), direction.Z()});
161 const auto& origin = particle.GetOrigin();
162 fParticleGun.SetParticlePosition(G4ThreeVector(origin.X(), origin.Y(), origin.Z()) * mm);
163 fParticleGun.SetParticleTime(particle.GetTime() * microsecond);
164 fParticleGun.GeneratePrimaryVertex(event);
165 }
166
167 const chrono::duration<double> elapsed = chrono::high_resolution_clock::now() - start;
168 outputManager->SetEventTimeWallPrimaryGeneration(elapsed.count());
169 outputManager->UpdatePrimaryData(event);
170 fSimulationManager->SetLastEventGenerated();
171}
