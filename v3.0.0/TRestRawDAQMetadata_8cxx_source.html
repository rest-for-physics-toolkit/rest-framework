64uint64_t TRestRawDAQMetadata::ParseSizeToBytes(const std::string& size) {
65 std::string text;
66 for (const char c : size) {
67 if (!std::isspace(static_cast<unsigned char>(c))) text += std::tolower(static_cast<unsigned char>(c));
68 }
69 // Number: digits with an optional decimal part (e.g. "1.5GB"), parsed here (independent of the locale)
70 size_t pos = 0;
71 size_t nDigits = 0;
72 bool point = false;
73 long double value = 0;
74 long double scale = 1;
75 while (pos < text.size()) {
76 const unsigned char c = text[pos];
77 if (std::isdigit(c)) {
78 nDigits++;
79 if (point) {
80 scale /= 10;
81 value += (c - '0') * scale;
82 } else {
83 value = value * 10 + (c - '0');
84 }
85 } else if (c == '.' && !point) {
86 point = true;
87 } else {
88 break;
89 }
90 pos++;
91 }
92 if (nDigits == 0) throw std::runtime_error("TRestRawDAQMetadata: no number in the size \"" + size + "\"");
93
94 const std::string unit = text.substr(pos);
95 long double factor = 0;
96 if (unit.empty() || unit == "b") {
97 factor = 1;
98 } else if (unit == "kb" || unit == "k") {
99 factor = 1024.;
100 } else if (unit == "mb" || unit == "m") {
101 factor = 1024. * 1024.;
102 } else if (unit == "gb" || unit == "g") {
103 factor = 1024. * 1024. * 1024.;
104 }
105 // Rounded to bytes
106 if (factor > 0) return static_cast<uint64_t>(std::llround(value * factor));
107 throw std::runtime_error("TRestRawDAQMetadata: unknown unit \"" + unit + "\" in the size \"" + size +
108 "\" (B, kB, MB or GB)");
109}
