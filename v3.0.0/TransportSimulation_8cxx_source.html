40std::unique_ptr<Garfield::Medium> CreateMedium(const TRestDetectorMedium& definition,
41 const TRestDetectorTransportTable& simulation,
42 const std::vector<double>& fields, double pressure,
43 double temperature, bool verbose) {
44 const std::string error = "restGarfield: medium '" + definition.GetName() + "': ";
45 if (definition.IsGas()) {
46 auto gas = CreateGas(definition, pressure, temperature);
47 // The nodes of the table (or a contiguous part of them, with the same spacing)
48 gas->SetFieldGrid(fields.front() * kVcmPerVmm, fields.back() * kVcmPerVmm, fields.size(),
49 simulation.fLogField && fields.size() > 1);
50 gas->SetMaxElectronEnergy(simulation.fMaxElectronEnergy / kEV);
51 gas->GenerateGasTable(simulation.fCollisions, verbose);
52 // Adjusts the Townsend coefficient of the table with its excitation and ionisation rates (lambda
53 // only matters in microscopic tracking)
54 if (simulation.fUsePenning && !gas->EnablePenningTransfer(simulation.fPenningRate, 0.)) {
55 throw std::runtime_error(error + "cannot enable the Penning transfer");
56 }
57 return gas;
58 }
59 const auto& components = definition.fComponents;
60 const std::string material = components.size() == 1 ? components.begin()->first : "";
61 std::unique_ptr<Garfield::Medium> medium;
62 if (definition.fState == "solid") {
63 if (material == "Si") medium = std::make_unique<Garfield::MediumSilicon>();
64 if (material == "GaAs") medium = std::make_unique<Garfield::MediumGaAs>();
65 if (material == "GaN") medium = std::make_unique<Garfield::MediumGaN>();
66 if (material == "CdTe") medium = std::make_unique<Garfield::MediumCdTe>();
67 if (material == "diamond") medium = std::make_unique<Garfield::MediumDiamond>();
68 }
69 if (!medium) {
70 throw std::runtime_error(error +
71 "not supported by Garfield (gases, or solids Si, GaAs, GaN, CdTe or "
72 "diamond with a single component)");
73 }
74 medium->SetTemperature(temperature);
75 return medium;
76}

102Carriers ComputeCarriers(const TRestDetectorMedium& definition, const TRestDetectorTransportTable& simulation,
103 const std::vector<double>& fields, double pressure, double temperature,
104 bool verbose) {
105 auto garfieldMedium = CreateMedium(definition, simulation, fields, pressure, temperature, verbose);
106 Carriers carriers;
107 for (const std::string carrier : {"electron", "hole", "ion"}) {
108 TRestDetectorTransportTable::Carrier values;
109 bool available = true;
110 for (const double field : fields) {
111 const double e = field * kVcmPerVmm;
112 double vx = 0, vy = 0, vz = 0, dl = 0, dt = 0, alpha = 0, eta = 0;
113 if (carrier == "electron") {
114 available &= garfieldMedium->ElectronVelocity(0, 0, e, 0, 0, 0, vx, vy, vz);
115 garfieldMedium->ElectronDiffusion(0, 0, e, 0, 0, 0, dl, dt);
116 garfieldMedium->ElectronTownsend(0, 0, e, 0, 0, 0, alpha);
117 garfieldMedium->ElectronAttachment(0, 0, e, 0, 0, 0, eta);
118 } else if (carrier == "hole") {
119 available &= garfieldMedium->HoleVelocity(0, 0, e, 0, 0, 0, vx, vy, vz);
120 garfieldMedium->HoleDiffusion(0, 0, e, 0, 0, 0, dl, dt);
121 garfieldMedium->HoleTownsend(0, 0, e, 0, 0, 0, alpha);
122 garfieldMedium->HoleAttachment(0, 0, e, 0, 0, 0, eta);
123 } else {
124 available &= garfieldMedium->IonVelocity(0, 0, e, 0, 0, 0, vx, vy, vz);
125 garfieldMedium->IonDiffusion(0, 0, e, 0, 0, 0, dl, dt);
126 }
127 if (!available) break;
128 // cm/ns -> mm/us
129 values.fDriftVelocity.push_back(std::sqrt(vx * vx + vy * vy + vz * vz) * 1e4);
130 values.fLongitudinalDiffusion.push_back(dl);
131 values.fTransversalDiffusion.push_back(dt);
132 values.fTownsend.push_back(alpha);
133 values.fAttachment.push_back(eta);
134 }
135 if (available) carriers[carrier] = values;
136 }
137 return carriers;
138}

147bool WriteCarriers(FILE* pipe, Carriers& carriers, size_t nFields) {
148 for (const std::string name : {"electron", "hole", "ion"}) {
149 const char available = carriers.count(name);
150 if (fwrite(&available, 1, 1, pipe) != 1) return false;
151 if (!available) continue;
152 for (auto* column : Columns(carriers[name])) {
153 if (column->size() != nFields) return false;
154 if (fwrite(column->data(), sizeof(double), nFields, pipe) != nFields) return false;
155 }
156 }
157 return true;
158}

161bool ReadCarriers(FILE* pipe, Carriers& carriers, size_t nFields, bool first) {
162 for (const std::string name : {"electron", "hole", "ion"}) {
163 char available = 0;
164 if (fread(&available, 1, 1, pipe) != 1) return false;
165 // A carrier is kept if it is available at all the fields
166 if (!available) {
167 carriers.erase(name);
168 continue;
169 }
170 if (!first && !carriers.count(name)) {
171 TRestDetectorTransportTable::Carrier ignored;
172 for (auto* column : Columns(ignored)) {
173 column->resize(nFields);
174 if (fread(column->data(), sizeof(double), nFields, pipe) != nFields) return false;
175 }
176 continue;
177 }
178 for (auto* column : Columns(carriers[name])) {
179 const size_t offset = column->size();
180 column->resize(offset + nFields);
181 if (fread(column->data() + offset, sizeof(double), nFields, pipe) != nFields) return false;
182 }
183 }
184 return true;
185}

190Carriers ComputeInParallel(const TRestDetectorMedium& definition,
191 const TRestDetectorTransportTable& simulation, const std::vector<double>& fields,
192 double pressure, double temperature, bool verbose, int jobs) {
193 const size_t nodes = fields.size();
194 const size_t parts = std::max<size_t>(1, std::min<size_t>(jobs, nodes / 2));
195 struct Child {
196 pid_t pid;
197 FILE* pipe;
198 size_t nFields;
199 };
200 std::vector<Child> children;
201 size_t begin = 0;
202 for (size_t part = 0; part < parts; part++) {
203 // Parts of nodes / parts nodes (the first ones one more): at least 2
204 const size_t size = nodes / parts + (part < nodes % parts ? 1 : 0);
205 const std::vector<double> partFields(fields.begin() + begin, fields.begin() + begin + size);
206 begin += size;
207 int fd[2];
208 if (pipe(fd) != 0) throw std::runtime_error("restGarfield: cannot create a pipe");
209 std::cout.flush();
210 std::fflush(nullptr);
211 const pid_t pid = fork();
212 if (pid < 0) throw std::runtime_error("restGarfield: cannot create a process");
213 if (pid == 0) {
214 close(fd[0]);
215 int status = 1;
216 try {
217 Carriers carriers =
218 ComputeCarriers(definition, simulation, partFields, pressure, temperature, verbose);
219 FILE* output = fdopen(fd[1], "wb");
220 if (output && WriteCarriers(output, carriers, partFields.size()) && fclose(output) == 0) {
221 status = 0;
222 }
223 } catch (const std::exception& error) {
224 std::cerr << "restGarfield: " << error.what() << std::endl;
225 }
226 std::cout.flush();
227 // Without the exit handlers of the parent (ROOT)
228 _exit(status);
229 }
230 close(fd[1]);
231 children.push_back({pid, fdopen(fd[0], "rb"), partFields.size()});
232 }
233 RESTInfo << "restGarfield: " << nodes << " field nodes in " << parts << " processes" << RESTendl;
234
235 Carriers carriers;
236 bool ok = true;
237 for (size_t i = 0; i < children.size(); i++) {
238 auto& child = children[i];
239 ok &= child.pipe && ReadCarriers(child.pipe, carriers, child.nFields, i == 0);
240 if (child.pipe) fclose(child.pipe);
241 int status = 0;
242 waitpid(child.pid, &status, 0);
243 ok &= WIFEXITED(status) && WEXITSTATUS(status) == 0;
244 }
245 if (!ok) throw std::runtime_error("restGarfield: a process computing the field nodes failed");
246 return carriers;
247}

274std::unique_ptr<Garfield::MediumGas> CreateGasFromTable(const TRestDetectorMedium& medium,
275 const TRestDetectorTransportTable& table,
276 double pressure) {
277 const std::string error = "restGarfield: medium '" + medium.GetName() + "': ";
278 if (!table.HasCarrier("electron")) {
279 throw std::runtime_error(error + "the transport table " + table.GetKey() + " has no electrons");
280 }
281 auto gas = std::make_unique<Garfield::MediumGas>();
282 SetComposition(*gas, medium);
283
284 // The field nodes of the table at the pressure (E/p scaling), V/cm
285 const double scale = table.fPressureScaling ? pressure / table.fPressure : 1;
286 std::vector<double> fields;
287 for (const double field : table.fFields) fields.push_back(field * scale * kVcmPerVmm);
288 gas->SetFieldGrid(fields, {0.}, {Garfield::HalfPi});
289 // Garfield keeps the logarithm of the Townsend and attachment coefficients (-30: 0)
290 auto logarithm = [](double value) { return value > 0 ? std::log(value) : -30.; };
291 for (size_t i = 0; i < fields.size(); i++) {
292 const double field = table.fFields[i] * scale; // V/mm
293 // mm/us -> cm/ns
294 gas->SetElectronVelocityE(i, 0, 0, table.GetDriftVelocity(field, pressure) * 1e-4);
295 gas->SetElectronLongitudinalDiffusion(i, 0, 0, table.GetLongitudinalDiffusion(field, pressure));
296 gas->SetElectronTransverseDiffusion(i, 0, 0, table.GetTransversalDiffusion(field, pressure));
297 gas->SetElectronTownsend(i, 0, 0, logarithm(table.GetTownsend(field, pressure)));
298 gas->SetElectronAttachment(i, 0, 0, logarithm(table.GetAttachment(field, pressure)));
299 if (table.HasCarrier("ion")) {
300 gas->SetIonMobility(i, 0, 0, table.GetDriftVelocity(field, pressure, "ion") * 1e-4 / fields[i]);
301 gas->SetIonLongitudinalDiffusion(i, 0, 0, table.GetLongitudinalDiffusion(field, pressure, "ion"));
302 gas->SetIonTransverseDiffusion(i, 0, 0, table.GetTransversalDiffusion(field, pressure, "ion"));
303 }
304 }
305 return gas;
306}

308void RunTransportSimulation(const TRestDetectorMedium& medium, TRestDetectorTransportTable& simulation,
309 bool verbose, int jobs) {
310 const auto mediumFile = medium.GetMediumFile();
311 if (mediumFile.has_parent_path()) std::filesystem::create_directories(mediumFile.parent_path());
312 const std::string fileName = mediumFile.string();
313 const double pressure = simulation.fPressure;
314 const auto fields = simulation.GetFieldGrid();
315 if (simulation.fUsePenning && !medium.IsGas()) {
316 throw std::runtime_error("restGarfield: usePenning is only for gases (" + medium.GetName() + ")");
317 }
318 const double penning = simulation.fUsePenning ? simulation.fPenningRate : 0;
319
320 for (const double temperature : simulation.GetTemperatures(medium)) {
321 const std::string key = TRestDetectorTransportTable::GetKey(pressure, temperature, fields.front(),
322 fields.back(), penning);
323 // The file is created by the first table: TFile::Open would print an error if it does not exist
324 bool exists = false;
325 if (std::filesystem::exists(fileName)) {
326 const MediumFileLock lock(mediumFile);
327 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "READ"));
328 exists = file && !file->IsZombie() &&
329 TRestDetectorTransportTable::HasTable(file.get(), medium.GetName(), key);
330 }
331 if (!simulation.fOverwrite && exists) {
332 RESTInfo << "restGarfield: table " << key << " already in " << fileName
333 << " (set overwrite: true to compute it again)" << RESTendl;
334 continue;
335 }
336
337 RESTInfo << "restGarfield: computing table " << key << " of " << medium.GetName() << RESTendl;
338
339 TRestDetectorTransportTable table;
340 table.fMedium = medium.GetName();
341 table.fPressure = pressure;
342 table.fTemperature = temperature;
343 table.fPressureScaling = medium.IsGas();
344 table.fPenning = penning;
345 for (const double field : fields) table.fFields.push_back(field);
346 // Magboltz (gases) in parallel processes; the solids are fast
347 if (jobs > 1 && medium.IsGas()) {
348 table.fCarriers =
349 ComputeInParallel(medium, simulation, fields, pressure, temperature, verbose, jobs);
350 } else {
351 table.fCarriers = ComputeCarriers(medium, simulation, fields, pressure, temperature, verbose);
352 }
353 if (!table.HasCarrier("electron")) {
354 throw std::runtime_error("restGarfield: no electron transport for " + medium.GetName());
355 }
356
357 // The configuration used: the simulation section (all its conditions) and the program
358 YAML::Node generator = YAML::Clone(simulation.GetYAMLNode());
359 generator["program"] = medium.IsGas() ? "Magboltz (Garfield++)" : "Garfield++";
360 const std::time_t now = std::time(nullptr);
361 char date[32];
362 std::strftime(date, sizeof(date), "%Y-%m-%d %H:%M:%S", std::localtime(&now));
363 generator["date"] = date;
364 generator["code"] = REST_Version::GetSummary();
365
366 // Locked (other processes may write other media or tables) until the file is closed
367 const MediumFileLock lock(mediumFile);
368 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "UPDATE"));
369 if (!file || file->IsZombie()) throw std::runtime_error("restGarfield: cannot open " + fileName);
370 table.Write(file.get(), medium.GetDefinition(), generator);
371 RESTInfo << "restGarfield: table " << key << " written to " << fileName << RESTendl;
372 }
