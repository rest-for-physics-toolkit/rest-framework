182YAML::Node WithoutVerbose(const YAML::Node& node) {
183 YAML::Node copy = TRestYAMLUtils::Detach(node);
184 std::function<void(YAML::Node)> strip = [&strip](YAML::Node n) {
185 if (n.IsMap()) {
186 n.remove("verbose");
187 for (auto element : n) strip(element.second);
188 } else if (n.IsSequence()) {
189 for (auto element : n) strip(element);
190 }
191 };
192 strip(copy);
193 return copy;
194}

198YAML::Node Processing(const YAML::Node& store) {
199 YAML::Node processing(YAML::NodeType::Map);
200 for (const auto& entry : store) {
201 const YAML::Node& config = entry.second;
202 if (!config.IsMap() || !config["class"] ||
203 config["class"].as<std::string>() != "TRestEventProcessManager")
204 continue;
205 YAML::Node manager = WithoutVerbose(config);
206 manager.remove("eventsToProcess");
207 manager.remove("nThreads");
208 processing[entry.first.as<std::string>()] = manager;
209 }
210 return processing;
211}

217std::map<std::string, std::string> DescribeSetup(TFile* file, const YAML::Node& store) {
218 std::map<std::string, std::string> setup;
219 for (const auto& manager : Processing(store)) {
220 std::string processes;
221 for (const auto& element : manager.second) {
222 if (element.second.IsMap() && element.second["class"]) {
223 processes +=
224 element.first.as<std::string>() + ": " + element.second["class"].as<std::string>() + "\n";
225 }
226 }
227 setup["processes of '" + manager.first.as<std::string>() + "'"] = processes;
228 }
229 if (auto* tree = file->Get<TTree>("AnalysisTree")) {
230 std::string branches;
231 for (const auto* object : *tree->GetListOfLeaves()) {
232 const auto* leaf = static_cast<const TLeaf*>(object);
233 branches += std::string(leaf->GetName()) + "/" + leaf->GetTypeName() + "\n";
234 }
235 setup["branches of the analysis tree"] = branches;
236 }
237 if (TDirectory* readouts = file->GetDirectory(kReadoutsDirectory.c_str())) {
238 for (auto* object : *readouts->GetListOfKeys()) {
239 const std::string name = object->GetName();
240 TDirectory* folder = readouts->GetDirectory(name.c_str());
241 if (!folder) continue;
242 std::string text = store[name] ? YAML::Dump(WithoutVerbose(store[name])) : "";
243 if (TDirectory* decodings = folder->GetDirectory("Decodings")) {
244 for (auto* key : *decodings->GetListOfKeys()) {
245 std::unique_ptr<TObjString> decoding(decodings->Get<TObjString>(key->GetName()));
246 if (decoding) text += std::string(key->GetName()) + ": " + decoding->GetString().Data();
247 }
248 }
249 setup["readout '" + name + "'"] = text + TRestTools::ReadGeometryHash(folder, kGeometryKey);
250 }
251 }
252 const std::string geometry = TRestTools::ReadGeometryHash(file, kGeometryKey);
253 if (!geometry.empty()) setup["geometry"] = geometry;
254 return setup;
255}

257std::vector<std::string> RunKeys(const TRestDataSet::Run& run) {
258 const int number = run.run->GetRunNumber();
259 if (number < 0) return {"file " + run.fileName};
260 auto subRuns = run.run->fMergedSubRuns;
261 if (subRuns.empty()) subRuns.push_back(run.run->GetSubRunNumber());
262 std::vector<std::string> keys;
263 for (const int subRun : subRuns)
264 keys.push_back("run " + std::to_string(number) + "." + std::to_string(subRun));
265 return keys;

317std::string TRestDataSet::DefineRunParameters(ROOT::RDF::RNode& frame, const std::string& expression) const {
318 std::set<std::string> columns;
319 for (const auto& column : frame.GetColumnNames()) columns.insert(column);
320 std::vector<YAML::Node> parameters;
321 for (const auto& run : fRuns) parameters.push_back(run.run->GetParameters());
322
323 // Dates become numbers, text literals are kept, a number with units (300 V) becomes the number in REST
324 // units (as the run parameters); a path followed by a member (runTag.size()) is the longest run
325 // parameter
326 const std::string identifier =
327 "date\\(\\s*\"[^\"]*\"\\s*\\)|\"(?:[^\"\\\\]|\\\\.)*\"|"
328 "(?:[0-9]+\\.?[0-9]*|\\.[0-9]+)(?:[eE][+-]?[0-9]+)?\\s*[A-Za-z][A-Za-z0-9/^]*(?![+-]?[0-9])|"
329 "[A-Za-z_][A-Za-z0-9_]*(?:\\.[A-Za-z_][A-Za-z0-9_]*)*";
330 return TRestTools::ReplaceIdentifiers(
331 expression, identifier, [&](const std::string& match) -> std::optional<std::string> {
332 if (match.front() == '"') return std::nullopt;
333 // A number followed by units (not an exponent: 2.5e-3 is not 2.5 e)
334 if (std::isdigit(static_cast<unsigned char>(match.front())) || match.front() == '.') {
335 // Not units (e.g. the operator "and" after a number): kept as written
336 const auto value = REST_Units::ParseValueWithUnits(match);
337 if (!value) return std::nullopt;
338 char number[32];
339 snprintf(number, sizeof(number), "%.17g", *value);
340 return std::string(number);
341 }
342 if (match.rfind("date", 0) == 0 && match.find('(') != std::string::npos) {
343 const size_t open = match.find('"');
344 const std::string date = match.substr(open + 1, match.rfind('"') - open - 1);
345 char number[32];
346 snprintf(number, sizeof(number), "%.0f", TRestTools::StringToTimeStamp(date));
347 return std::string(number);
348 }
349 const auto parts = TRestTools::Split(match, '.');
350 if (columns.count(parts.front())) return std::nullopt;
351 for (size_t length = parts.size(); length > 0; length--) {
352 std::string path = parts[0];
353 for (size_t i = 1; i < length; i++) path += "." + parts[i];
354 std::string remainder;
355 for (size_t i = length; i < parts.size(); i++) remainder += "." + parts[i];
356
357 // A run parameter if some run has it; then all the runs must have it
358 std::vector<std::string> texts;
359 std::optional<size_t> missing;
360 for (size_t r = 0; r < parameters.size(); r++) {
361 const YAML::Node node = GetNodeAtPath(parameters[r], path);
362 if (!node || !node.IsScalar()) {
363 if (!missing) missing = r;
364 continue;
365 }
366 texts.push_back(node.Scalar());
367 }
368 if (texts.empty()) continue;
369 if (missing) {
370 throw std::runtime_error("TRestDataSet '" + fName + "': run parameter " + path +
371 " is not in the run " + RunKeys(fRuns[*missing]).front());
372 }
373
374 std::string column = "runParameter_" + path;
375 std::replace(column.begin(), column.end(), '.', '_');
376 // The set of columns, not frame.HasColumn: it crashes (ROOT 6.36) on a frame without data
377 // source, as the frame of the runs (RDataFrame with a number of entries)
378 if (columns.insert(column).second) {
379 std::vector<double> values;
380 for (const auto& text : texts) {
381 const auto value = REST_Units::ParseValueWithUnits(text);
382 if (!value) break;
383 values.push_back(*value);
384 }
385 if (values.size() == texts.size()) {
386 frame =
387 frame.Define(column, [values](int index) { return values[index]; }, {"runIndex"});
388 } else {
389 frame =
390 frame.Define(column, [texts](int index) { return texts[index]; }, {"runIndex"});
391 }
392 }
393 return column + remainder;
394 }
395 return std::nullopt;
396 });
397}

399void TRestDataSet::Generate(const std::vector<TRestCut*>& cuts) {
400 if (fFilePattern.empty()) throw std::runtime_error("TRestDataSet '" + fName + "': no filePattern");
401 fOpened = false;
402 fInputFiles.clear();
403 for (const auto& file : TRestTools::GetFilesMatchingPattern(fFilePattern)) {
404 fInputFiles.push_back(TRestTools::GetFullPath(file));
405 }
406 std::sort(fInputFiles.begin(), fInputFiles.end());
407 if (fInputFiles.empty()) {
408 throw std::runtime_error("TRestDataSet '" + fName + "': no files match " + fFilePattern);
409 }
410
411 // Runs of each input file: its TRestRun, or the runs of a dataset
412 const bool showProgress = fVerboseLevel >= TRestLogManager::REST_Verbose_Level::REST_Info;
413 RESTInfo << "TRestDataSet '" << fName << "': reading the metadata of " << fInputFiles.size() << " files"
414 << RESTendl;
415 TRestProgressBar progress;
416 if (showProgress) progress.Reset(static_cast<int>(fInputFiles.size()), 30, "Files");
417 std::vector<std::vector<Run>> inputRuns;
418 std::vector<YAML::Node> stores;
419 std::vector<std::map<std::string, std::string>> setups;
420 // The processings of each input file: its own, or those of a dataset (Run::processing)
421 std::vector<std::vector<YAML::Node>> inputProcessings;
422 std::optional<bool> dataSets;
423 fInputEntries = 0;
424 fInputFileNode.reset();
425 fInputMetadata.reset();
426 fMetadataFile.clear();
427 fMetadataStore.Clear();
428 fMetadataStoreLoaded = false;
429 for (const auto& fileName : fInputFiles) {
430 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "READ"));
431 if (!file || file->IsZombie()) {
432 throw std::runtime_error("TRestDataSet '" + fName + "': cannot open " + fileName);
433 }
434 auto* tree = file->Get<TTree>("AnalysisTree");
435 if (tree) fInputEntries += tree->GetEntries();
436 auto runs = ReadRuns(file.get());
437 const bool isDataSet = runs.has_value();
438 // The metadata of the first input file with runs selected are kept and written, the runs must have
439 // the same setup (readouts, geometry, processes and observables; the parameters can differ)
440 const YAML::Node store = TRestYAMLUtils::ReadMetadataStore(file.get());
441 stores.push_back(store);
442 setups.push_back(DescribeSetup(file.get(), store));
443 auto processings = isDataSet ? ReadProcessings(file.get()) : std::vector<YAML::Node>{};
444 if (processings.empty()) processings.push_back(Processing(store));
445 inputProcessings.push_back(std::move(processings));
446 if (!isDataSet) {
447 const auto [instance, runNode] = TRestYAMLUtils::GetMetadataClass(store, "TRestRun");
448 if (!runNode || !tree) {
449 throw std::runtime_error("TRestDataSet '" + fName + "': " + fileName +
450 " is not a run file (AnalysisTree and TRestRun) nor a dataset");
451 }
452 runs = std::vector<Run>{
453 {TRestRunInfo::FromConfig(instance, runNode), fileName, tree->GetEntries(), 0}};
454 }
455 if (dataSets && *dataSets != isDataSet) {
456 throw std::runtime_error("TRestDataSet '" + fName +
457 "': the inputs must be all run files or all datasets, " + fileName);
458 }
459 dataSets = isDataSet;
460 inputRuns.push_back(std::move(*runs));
461 if (showProgress) progress.Update(static_cast<int>(inputRuns.size()));
462 }
463 if (showProgress) progress.Finish();
464
465 // Selection of the runs (runConditions), on a frame with an entry per run
466 fRuns.clear();
467 for (const auto& runs : inputRuns) fRuns.insert(fRuns.end(), runs.begin(), runs.end());
468 std::set<int> selected;
469 for (size_t i = 0; i < fRuns.size(); i++) selected.insert(i);
470 if (!fRunConditions.empty()) {
471 ROOT::RDF::RNode runFrame =
472 ROOT::RDataFrame(fRuns.size())
473 .Define("runIndex", [](unsigned long long entry) { return static_cast<int>(entry); },
474 {"rdfentry_"});
475 for (const auto& [name, condition] : fRunConditions) {
476 runFrame = runFrame.Filter(DefineRunParameters(runFrame, condition), name);
477 }
478 auto passed = runFrame.Take<int>("runIndex");
479 selected = std::set<int>(passed->begin(), passed->end());
480 }
481
482 // New index of the runs of each input file, -1 if not selected
483 std::vector<Run> allRuns = std::move(fRuns);
484 fRuns.clear();
485 std::vector<std::vector<int>> newIndex;
486 std::map<std::string, std::string> keys;
487 fProcessings.clear();
488 std::map<std::string, int> processingIndex;
489 size_t position = 0;
490 for (size_t f = 0; f < inputRuns.size(); f++) {
491 newIndex.emplace_back(inputRuns[f].size(), -1);
492 for (size_t r = 0; r < inputRuns[f].size(); r++, position++) {
493 if (selected.count(position) == 0) continue;
494 const Run& run = allRuns[position];
495 for (const auto& key : RunKeys(run)) {
496 const auto [it, inserted] = keys.emplace(key, run.fileName);
497 if (!inserted) {
498 throw std::runtime_error("TRestDataSet '" + fName + "': " + it->first + " is twice, in " +
499 it->second + " and " + run.fileName);
500 }
501 }
502 newIndex[f][r] = fRuns.size();
503 fRuns.push_back(run);
504 // Its processing, once in fProcessings
505 const auto& processings = inputProcessings[f];
506 if (run.processing < 0 || run.processing >= static_cast<int>(processings.size())) {
507 throw std::runtime_error("TRestDataSet '" + fName + "': processing " +
508 std::to_string(run.processing) + " of a run of " + fInputFiles[f] +
509 " not found");
510 }
511 const std::string text = YAML::Dump(processings[run.processing]);
512 const auto it = processingIndex.find(text);
513 if (it != processingIndex.end()) {
514 fRuns.back().processing = it->second;
515 } else {
516 fRuns.back().processing = fProcessings.size();
517 processingIndex[text] = fProcessings.size();
518 fProcessings.push_back(processings[run.processing]);
519 }
520 }
521 }
522 if (fRuns.empty()) throw std::runtime_error("TRestDataSet '" + fName + "': no run selected");
523
524 // The setup of the files with runs selected, the same as the first one
525 std::optional<size_t> first;
526 for (size_t f = 0; f < inputRuns.size(); f++) {
527 if (std::all_of(newIndex[f].begin(), newIndex[f].end(), [](int index) { return index < 0; }))
528 continue;
529 if (!first) {
530 first = f;
531 continue;
532 }
533 const auto& reference = setups[*first];
534 const auto& current = setups[f];
535 for (const auto* names : {&reference, &current}) {
536 for (const auto& [name, description] : *names) {
537 const auto it = setups[f].find(name);
538 const auto referenceIt = reference.find(name);
539 if (it != setups[f].end() && referenceIt != reference.end() &&
540 it->second == referenceIt->second)
541 continue;
542 throw std::runtime_error("TRestDataSet '" + fName + "': the " + name + " of " +
543 fInputFiles[f] + " is not the same as in " + fInputFiles[*first] +
544 " (the runs of a dataset must have the same readouts, geometry, "
545 "processes and observables)");
546 }
547 }
548 }
549 WarnProcessingDifferences();
550 fInputMetadata = stores[*first];
551 fMetadataFile = fInputFiles[*first];
552 RESTInfo << "TRestDataSet '" << fName << "': " << fRuns.size() << " runs selected of " << allRuns.size()
553 << RESTendl;
554
555 fFrame = std::make_unique<ROOT::RDataFrame>("AnalysisTree", fInputFiles);
556 std::vector<std::string> treeColumns = fFrame->GetColumnNames();
557 ROOT::RDF::RNode frame(*fFrame);
558
559 // Input file of each sample (file), from its id "fileName/AnalysisTree"
560 std::map<std::string, int> sampleIds;
561 for (size_t i = 0; i < fInputFiles.size(); i++) sampleIds[fInputFiles[i] + "/AnalysisTree"] = i;
562 frame = frame.DefinePerSample(
563 kInputFileColumn, [sampleIds](unsigned int, const ROOT::RDF::RSampleInfo& info) {
564 const auto it = sampleIds.find(info.AsString());
565 if (it == sampleIds.end()) {
566 throw std::runtime_error("TRestDataSet: unknown input file " + info.AsString());
567 }
568 return it->second;
569 });
570 if (*dataSets) {
571 frame = frame.Redefine("runIndex", [newIndex](int file, int run) { return newIndex[file][run]; },
572 {kInputFileColumn, "runIndex"});
573 } else {
574 frame =
575 frame.Define("runIndex", [newIndex](int file) { return newIndex[file][0]; }, {kInputFileColumn});
576 }
577 if (fRuns.size() < allRuns.size())
578 frame = frame.Filter([](int index) { return index >= 0; }, {"runIndex"});
579
580 // Cuts on the entries
581 fCutConfigs.clear();
582 for (const auto& cutName : fCutNames) {
583 TRestCut* cut = nullptr;
584 for (auto* c : cuts) {
585 if (c && c->GetName() == cutName) cut = c;
586 }
587 if (!cut)
588 throw std::runtime_error("TRestDataSet '" + fName + "': TRestCut '" + cutName + "' not found");
589 fCutConfigs.emplace_back(cut->GetName(), cut->GetYAMLNode());
590 const std::string expression = cut->GetExpression();
591 if (!expression.empty()) frame = frame.Filter(DefineRunParameters(frame, expression), cutName);
592 }
593 fDataFrame = frame;
594
595 // Columns written: those of every event, runIndex, the observables and the new columns
596 fColumns = kEventColumns;
597 if (std::find(treeColumns.begin(), treeColumns.end(), "timeNanoSecs") != treeColumns.end())
598 fColumns.push_back("timeNanoSecs");
599 fColumns.push_back("runIndex");
600 for (const auto& column : treeColumns) {
601 if (std::find(fColumns.begin(), fColumns.end(), column) != fColumns.end()) continue;
602 bool match = fObservables.empty();
603 for (const auto& pattern : fObservables) {
604 if (std::regex_match(column, std::regex(TRestTools::PatternToRegex(pattern)))) match = true;
605 }
606 if (match) fColumns.push_back(column);
607 }
608 for (const auto& [name, expression] : fColumnExpressions) Define(name, expression);
609}

611void TRestDataSet::Open(const std::string& fileName) {
612 const std::string path = TRestTools::GetFullPath(fileName);
613 std::unique_ptr<TFile> file(TFile::Open(path.c_str(), "READ"));
614 if (!file || file->IsZombie()) throw std::runtime_error("TRestDataSet: cannot open " + fileName);
615 auto runs = ReadRuns(file.get());
616 if (!runs || !file->Get<TTree>("AnalysisTree")) {
617 throw std::runtime_error("TRestDataSet: " + fileName + " is not a dataset");
618 }
619
620 // Its configuration and the TRestCut it applied, kept apart as TRestRun::fInputFileNode
621 fInputMetadata = TRestYAMLUtils::ReadMetadataStore(file.get());
622 const auto [instance, node] = TRestYAMLUtils::GetMetadataClass(fInputMetadata, "TRestDataSet");
623 fInputFileNode.reset(node);
624 if ((!fNode || !fNode.IsMap()) && !instance.empty()) SetName(instance);
625
626 fRuns = std::move(*runs);
627 fProcessings = ReadProcessings(file.get());
628 if (fProcessings.empty()) fProcessings.push_back(Processing(fInputMetadata));
629 fOpened = true;
630 fMetadataFile = path;
631 fMetadataStore.Clear();
632 fMetadataStoreLoaded = false;
633 fInputFiles = {path};
634 fInputEntries = file->Get<TTree>("AnalysisTree")->GetEntries();
635 fFrame = std::make_unique<ROOT::RDataFrame>("AnalysisTree", path);
636 fDataFrame = *fFrame;
637 fColumns = fFrame->GetColumnNames();
638}

656std::string TRestDataSet::ResolveFilePattern(const std::string& pattern) const {
657 return TRestTools::ReplaceDelimited(
658 pattern, '[', ']', [&](const std::string& content) -> std::optional<std::string> {
659 // [key] or [key|default]: the default when the runs do not share the value
660 const size_t bar = content.find('|');
661 const std::string key = content.substr(0, bar);
662 const std::optional<std::string> fallback =
663 bar == std::string::npos ? std::nullopt : std::optional<std::string>(content.substr(bar + 1));
664 if (fRuns.empty()) {
665 throw std::runtime_error("TRestDataSet '" + fName + "': [" + content + "] in \"" + pattern +
666 "\" but the dataset has no runs");
667 }
668 if (key == "runStart" || key == "runEnd" || key == "runRange") {
669 int start = fRuns.front().run->GetRunNumber(), end = start;
670 for (const auto& run : fRuns) {
671 start = std::min(start, run.run->GetRunNumber());
672 end = std::max(end, run.run->GetRunNumber());
673 }
674 const std::string first = TRestTools::Format("%05d", start);
675 const std::string last = TRestTools::Format("%05d", end);
676 if (key == "runStart") return first;
677 if (key == "runEnd") return last;
678 return start == end ? first : first + "_" + last;
679 }
680 // A parameter of the runs: the same in all of them, otherwise the default or an error
681 std::optional<std::string> value;
682 for (const auto& run : fRuns) {
683 std::string runValue;
684 try {
685 runValue = run.run->ResolveFilePattern("[" + key + "]");
686 } catch (const std::exception&) {
687 if (fallback) return fallback;
688 throw;
689 }
690 if (!value) value = runValue;
691 if (runValue == *value) continue;
692 if (fallback) return fallback;
693 throw std::runtime_error("TRestDataSet '" + fName + "': [" + key + "] in \"" + pattern +
694 "\" is not the same in all the runs: give a default, [" + key +
695 "|Mixed], or e.g. [runRange] for the run numbers");
696 }
697 return value;
698 });
699}

701void TRestDataSet::Write(const std::string& fileNamePattern) {
702 if (!fFrame) throw std::runtime_error("TRestDataSet '" + fName + "': Generate before Write");
703 if (fileNamePattern.empty())
704 throw std::runtime_error("TRestDataSet '" + fName + "': no output file name");
705 const std::string fileName = ResolveFilePattern(fileNamePattern);
706 const std::string path = TRestTools::GetFullPath(fileName);
707 if (std::find(fInputFiles.begin(), fInputFiles.end(), path) != fInputFiles.end()) {
708 throw std::runtime_error("TRestDataSet '" + fName + "': the output file " + fileName +
709 " is an input file");
710 }
711
712 // Booked before the Snapshot: filled in its event loop
713 const int nRuns = fRuns.size();
714 auto runEntries = fDataFrame.Histo1D<int>({"runEntries", "", nRuns, -0.5, nRuns - 0.5}, "runIndex");
715 // Progress: entries read (before the cuts), from a count of the input in the same loop
716 const bool showProgress = fVerboseLevel >= TRestLogManager::REST_Verbose_Level::REST_Info;
717 TRestProgressBar progress;
718 auto readEntries = ROOT::RDF::RNode(*fFrame).Count();
719 if (showProgress && fInputEntries > 0) {
720 const unsigned long long step = std::max<unsigned long long>(fInputEntries / 200, 1);
721 readEntries.OnPartialResult(
722 step, [&progress](unsigned long long& entries) { progress.Update(static_cast<int>(entries)); });
723 RESTInfo << "TRestDataSet '" << fName << "': writing " << fileName << RESTendl;
724 progress.Reset(static_cast<int>(fInputEntries), 30, "Entries");
725 }
726 fDataFrame.Snapshot("AnalysisTree", fileName, fColumns);
727 if (showProgress && fInputEntries > 0) progress.Finish();
728 for (int i = 0; i < nRuns; i++) fRuns[i].selectedEntries = std::llround(runEntries->GetBinContent(i + 1));
729
730 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "UPDATE"));
731 if (!file || file->IsZombie())
732 throw std::runtime_error("TRestDataSet '" + fName + "': cannot open " + fileName);
733 fOutputFileName = path;
734 // The code writing the dataset
735 fCodeVersion = REST_Version::GetVersion();
736 fCodeCommit = REST_Version::GetCommit();
737 fCodeClean = REST_Version::IsClean();
738 if (fOpened) {
739 // A dataset file opened: its metadata (its TRestCut, the dataset processes applied...) and its
740 // configuration, with this output and this code
741 for (const auto& entry : fInputMetadata) {
742 const std::string instance = entry.first.as<std::string>();
743 YAML::Node config = TRestYAMLUtils::Detach(entry.second);
744 if (config.IsMap() && config["class"] && config["class"].as<std::string>() == GetClassName()) {
745 config["outputFileName"] = path;
746 config["codeVersion"] = fCodeVersion;
747 config["codeCommit"] = fCodeCommit;
748 config["codeClean"] = fCodeClean;
749 }
750 TRestMetadata::WriteMetadata(file.get(), instance, config);
751 }
752 } else {
753 // The metadata of the input file (processing, readouts...) but its run (in RESTDataSetInfo) and the
754 // dataset it comes from (replaced by this one)
755 for (const auto& entry : fInputMetadata) {
756 const YAML::Node& config = entry.second;
757 const std::string className =
758 config.IsMap() && config["class"] ? config["class"].as<std::string>() : "";
759 if (className == "TRestRun" || className == GetClassName()) continue;
760 TRestMetadata::WriteMetadata(file.get(), entry.first.as<std::string>(), config);
761 }
762 // Its configuration, with its input (the files or dataset it comes from) and this output
763 if (!fNode || !fNode.IsMap()) {
764 fNode = YAML::Node(YAML::NodeType::Map);
765 fNode["class"] = GetClassName();
766 }
767 WriteMetadata(file.get());
768 for (const auto& [name, config] : fCutConfigs) TRestMetadata::WriteMetadata(file.get(), name, config);
769 }
770
771 // The readouts and the geometry of the input file
772 if (!fMetadataFile.empty()) {
773 std::unique_ptr<TFile> input(TFile::Open(fMetadataFile.c_str(), "READ"));
774 if (input && !input->IsZombie()) {
775 if (TDirectory* readouts = input->GetDirectory(kReadoutsDirectory.c_str())) {
776 TDirectory* target = file->mkdir(kReadoutsDirectory.c_str(), "", true);
777 TRestTools::CopyDirectory(readouts, target);
778 }
779 if (input->GetKey(kGeometryKey.c_str())) {
780 // A copy read to write it, with its hash (also for files without it); gGeoManager kept
781 std::unique_ptr<TGeoManager, void (*)(TGeoManager*)> geometry(
782 TRestTools::ReadGeometryCopy(input.get(), kGeometryKey), &TRestTools::DeleteGeometry);
783 TRestTools::WriteGeometry(file.get(), geometry.get(), kGeometryKey);
784 }
785 }
786 file->cd();
787 }
788
789 // The runs, in the order of runIndex: a YAML sequence in a TObjString (RESTDataSetInfo)
790 YAML::Node runs(YAML::NodeType::Sequence);
791 for (const auto& run : fRuns) {
792 YAML::Node node;
793 node["fileName"] = run.fileName;
794 node["entries"] = run.entries;
795 node["selectedEntries"] = run.selectedEntries;
796 node["processing"] = run.processing;
797 node["run"] = TRestYAMLUtils::Detach(run.run->GetYAMLNode());
798 runs.push_back(node);
799 }
800 file->cd();
801 TObjString text(YAML::Dump(runs).c_str());
802 text.Write(kRunsKey.c_str(), TObject::kOverwrite);
803 // The different processings, in the order of Run::processing (RESTDataSetProcessing)
804 YAML::Node processings(YAML::NodeType::Sequence);
805 for (const auto& processing : fProcessings) processings.push_back(TRestYAMLUtils::Detach(processing));
806 TObjString processingsText(YAML::Dump(processings).c_str());
807 processingsText.Write(kProcessingsKey.c_str(), TObject::kOverwrite);
808 file->Close();
809
810 RESTInfo << "TRestDataSet '" << fName << "': " << fRuns.size() << " runs written in " << fileName
811 << RESTendl;
812}

863void TRestDataSet::WarnProcessingDifferences() const {
864 if (fProcessings.size() < 2) return;
865 // Each parameter (path of keys) of each processing, the sequences in a line
866 std::function<void(const YAML::Node&, const std::string&, std::map<std::string, std::string>&)> flatten =
867 [&flatten](const YAML::Node& node, const std::string& path, std::map<std::string, std::string>& out) {
868 if (node.IsMap()) {
869 for (const auto& element : node) {
870 const std::string key = element.first.as<std::string>();
871 flatten(element.second, path.empty() ? key : path + "." + key, out);
872 }
873 } else if (node.IsScalar()) {
874 out[path] = node.Scalar();
875 } else {
876 YAML::Emitter emitter;
877 emitter << YAML::Flow << node;
878 out[path] = emitter.c_str();
879 }
880 };
881 std::vector<std::map<std::string, std::string>> parameters(fProcessings.size());
882 std::set<std::string> paths;
883 for (size_t i = 0; i < fProcessings.size(); i++) {
884 flatten(fProcessings[i], "", parameters[i]);
885 for (const auto& [path, value] : parameters[i]) paths.insert(path);
886 }
887 // The runs of each processing
888 std::vector<std::vector<std::string>> runs(fProcessings.size());
889 for (const auto& run : fRuns) runs[run.processing].push_back(RunKeys(run).front());
890
891 RESTWarning
892 << "TRestDataSet '" << fName << "': the runs have " << fProcessings.size()
893 << " different processings (the same processes, different parameters); all of them are written "
894 "(GetProcessings, RESTDataSetProcessing). Parameters that differ:"
895 << RESTendl;
896 for (const auto& path : paths) {
897 // Value -> runs with it
898 std::map<std::string, std::vector<std::string>> values;
899 for (size_t i = 0; i < fProcessings.size(); i++) {
900 const auto it = parameters[i].find(path);
901 auto& list = values[it == parameters[i].end() ? "(not given)" : it->second];
902 list.insert(list.end(), runs[i].begin(), runs[i].end());
903 }
904 if (values.size() < 2) continue;
905 std::string text = " - " + path + ":";
906 for (const auto& [value, runList] : values) {
907 text += " " + value + " (" + std::to_string(runList.size()) + " runs: ";
908 for (size_t r = 0; r < runList.size() && r < 3; r++) text += (r ? ", " : "") + runList[r];
909 text += runList.size() > 3 ? ", ...)" : ")";
910 }
911 RESTWarning << text << RESTendl;
912 }
913}
