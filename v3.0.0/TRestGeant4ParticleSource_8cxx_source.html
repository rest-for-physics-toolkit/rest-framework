58vector<vector<TRestGeant4ParticleState>> ReadDecay0(const string& fileName) {
59 const string error = "TRestGeant4ParticleSource: Decay0 file '" + fileName + "': ";
60 ifstream input(TRestTools::GetFullPath(fileName));
61 if (!input.is_open()) throw runtime_error(error + "cannot be opened");
62
63 // Geant3 particle IDs: mass (MeV), name and charge
64 struct Particle {
65 double mass;
66 string name;
67 int charge;
68 };
69 const map<int, Particle> particles = {{1, {0, "gamma", 0}}, {2, {0.511, "e+", 1}},
70 {3, {0.511, "e-", -1}}, {5, {105.658, "mu+", 1}},
71 {6, {105.658, "mu-", -1}}, {14, {938.272, "proton", 1}},
72 {47, {3727.379, "alpha", 2}}};
73 auto readParticle = [&](int id, double px, double py, double pz) {
74 const auto particle = particles.find(id);
75 if (particle == particles.end()) {
76 throw runtime_error(error + "particle ID " + to_string(id) + " not supported");
77 }
78 const double mass = particle->second.mass;
79 TRestGeant4ParticleState state;
80 state.SetParticleName(particle->second.name);
81 state.SetParticleCharge(particle->second.charge);
82 state.SetEnergy(1000.0 * (sqrt(px * px + py * py + pz * pz + mass * mass) - mass));
83 state.SetDirection(XYZVector(px, py, pz));
84 return state;
85 };
86
87 // The header: the number of events of either format
88 int nEvents = 0;
89 bool oldFormat = false;
90 string line;
91 for (int i = 0; i < 30 && getline(input, line); ++i) {
92 const auto position = line.find("@nevents=");
93 if (position != string::npos) {
94 nEvents = stoi(line.substr(position + 9));
95 break;
96 }
97 if (line.find("First event and full number of events:") != string::npos) {
98 int firstEvent = 0;
99 input >> firstEvent >> nEvents;
100 oldFormat = true;
101 break;
102 }
103 }
104 if (nEvents <= 0) {
105 throw runtime_error(error +
106 "no '@nevents=' (bxdecay0) nor 'First event and full number of events:' "
107 "(old Decay0) in its header");
108 }
109
110 vector<vector<TRestGeant4ParticleState>> events;
111 for (int event = 0; event < nEvents; ++event) {
112 int nParticles = 0;
113 if (oldFormat) {
114 int eventNumber = 0;
115 double time = 0;
116 input >> eventNumber >> time >> nParticles;
117 } else {
118 while (getline(input, line) && line.find("@event_start") == string::npos) {
119 }
120 getline(input, line);
121 input >> nParticles;
122 }
123 if (!input) throw runtime_error(error + "less events than given in its header");
124 vector<TRestGeant4ParticleState> eventParticles;
125 for (int i = 0; i < nParticles; ++i) {
126 int id = 0;
127 double time = 0, px = 0, py = 0, pz = 0;
128 if (oldFormat) {
129 input >> id >> px >> py >> pz >> time;
130 } else {
131 string name;
132 input >> id >> time >> px >> py >> pz >> name;
133 }
134 eventParticles.push_back(readParticle(id, px, py, pz));
135 }
136 events.push_back(std::move(eventParticles));
137 }
138 return events;
139}

171void TRestGeant4ParticleSource::CheckJoint() {
172 const auto* energy = dynamic_cast<const TRestGeant4EnergyDistributionJoint*>(fEnergy.get());
173 const auto* angular = dynamic_cast<const TRestGeant4AngularDistributionJoint*>(fAngular.get());
174 if (!energy || !angular || energy->fType != angular->fType) {
175 throw runtime_error(GetClassName() +
176 ": formula2, TH2D and cosmic must be the type of both energy and angular");
177 }
178 fJointType = energy->fType;
179
180 if (fJointType == "formula2") {
181 fJointFormula = Pick(energy->fFormula, angular->fFormula, "formula");
182 // An unknown formula name throws
183 StringToEnergyAndAngularDistributionFormulas(fJointFormula);
184 return;
185 }
186
187 fJointFile = Pick(energy->fFilename, angular->fFilename, "filename");
188 fJointName = Pick(energy->fNameInFile, angular->fNameInFile, "nameInFile");
189 if (fJointFile.empty() || fJointName.empty()) {
190 throw runtime_error(GetClassName() + ": 'filename' and 'nameInFile' are needed for " + fJointType);
191 }
192 if (fJointType == "cosmic") {
193 // As legacy, nameInFile is the particle key, the histogram <key>_energy_zenith
194 const auto particle = kCosmicParticles.find(fJointName);
195 if (particle == kCosmicParticles.end()) {
196 throw runtime_error(GetClassName() + ": unknown cosmic particle key '" + fJointName + "'");
197 }
198 if (fParticleName.empty()) fParticleName = particle->second;
199 if (fParticleName != particle->second) {
200 throw runtime_error(GetClassName() + ": the particle of the cosmic key '" + fJointName + "' is " +
201 particle->second + ", not " + fParticleName);
202 }
203 fJointName += "_energy_zenith";
204 }

229void TRestGeant4ParticleSource::BuildJointCumulative() {
230 const int nx = fJointFunction->GetNpx(), ny = fJointFunction->GetNpy();
231 const double xMin = fJointFunction->GetXmin(), yMin = fJointFunction->GetYmin();
232 const double dx = (fJointFunction->GetXmax() - xMin) / nx, dy = (fJointFunction->GetYmax() - yMin) / ny;
233 // Integral of each cell with 2 x 2 Gauss-Legendre points (exact for cubic polynomials)
234 const double node = 0.5 / std::sqrt(3.0);
235 auto cumulative = make_shared<vector<double>>(static_cast<size_t>(nx) * ny + 1, 0.0);
236 size_t cell = 0;
237 for (int j = 0; j < ny; j++) {
238 const double yCenter = yMin + (j + 0.5) * dy;
239 for (int i = 0; i < nx; i++) {
240 const double xCenter = xMin + (i + 0.5) * dx;
241 double integral = 0;
242 for (const double u : {-node, node})
243 for (const double v : {-node, node})
244 integral += fJointFunction->Eval(xCenter + u * dx, yCenter + v * dy);
245 // Negative values taken as positive, as TF2::GetRandom2
246 (*cumulative)[cell + 1] = (*cumulative)[cell] + std::abs(integral) * 0.25 * dx * dy;
247 cell++;
248 }
249 }
250 const double total = cumulative->back();
251 if (!(total > 0)) throw runtime_error(GetClassName() + ": the integral of the formula is zero");
252 for (auto& value : *cumulative) value /= total;
253 fJointCumulative = cumulative;
254}

256void TRestGeant4ParticleSource::SampleJointFunction(TRandom& random, double& x, double& y) const {
257 const auto& cumulative = *fJointCumulative;
258 const int nx = fJointFunction->GetNpx();
259 const double xMin = fJointFunction->GetXmin(), yMin = fJointFunction->GetYmin();
260 const double dx = (fJointFunction->GetXmax() - xMin) / nx;
261 const double dy = (fJointFunction->GetYmax() - yMin) / fJointFunction->GetNpy();
262 const double r = random.Rndm();
263 // The cell of r, as TF2::GetRandom2: x linear in the cell with r, y uniform
264 size_t cell = std::upper_bound(cumulative.begin(), cumulative.end(), r) - cumulative.begin();
265 cell = std::min(std::max<size_t>(cell, 1), cumulative.size() - 1) - 1;
266 const double width = cumulative[cell + 1] - cumulative[cell];
267 const double xInCell = width > 0 ? (r - cumulative[cell]) / width : 0.5;
268 x = xMin + dx * (cell % nx + xInCell);
269 y = yMin + dy * (cell / nx + random.Rndm());
270}

272void TRestGeant4ParticleSource::LoadJointDistribution() {
273 const auto* energy = static_cast<const TRestGeant4EnergyDistributionJoint*>(fEnergy.get());
274 const auto& energyRange = energy->fRange;
275 const auto* angular = static_cast<const TRestGeant4AngularDistributionJoint*>(fAngular.get());
276 const auto& angleRange = angular->fRange;
277
278 if (fJointType == "formula2") {
279 // Energy in keV and angle in rad
280 fJointFunction = make_unique<TF2>(EnergyAndAngularDistributionFormulasToRootFormula(
281 StringToEnergyAndAngularDistributionFormulas(fJointFormula)));
282 const double energyMin = std::max<double>(energyRange.first, fJointFunction->GetXmin());
283 const double energyMax = std::min<double>(energyRange.second, fJointFunction->GetXmax());
284 const double angleMin = std::max<double>(angleRange.first, fJointFunction->GetYmin());
285 const double angleMax = std::min<double>(angleRange.second, fJointFunction->GetYmax());
286 if (energyMin >= energyMax || angleMin >= angleMax) {
287 throw runtime_error(GetClassName() + ": 'range' is outside the formula one");
288 }
289 fJointFunction->SetRange(energyMin, angleMin, energyMax, angleMax);
290 fJointFunction->SetNpx(energy->fFormulaNPoints);
291 fJointFunction->SetNpy(angular->fFormulaNPoints);
292 if (!fJointCumulative) BuildJointCumulative();
293 fJointEnergyUnit = 1;
294 fJointAngleUnit = 1;
295 return;
296 }
297
298 // As legacy, the histograms have the energy in MeV and the angle in deg
299 fJointHistogram = ReadHistogram2D(fJointFile, fJointName); // Any 2D histogram, as a TH2D
300 fJointEnergyUnit = 1000;
301 fJointAngleUnit = TMath::DegToRad();
302 for (int i = 1; i <= fJointHistogram->GetNbinsX(); i++) {
303 const double energy = fJointHistogram->GetXaxis()->GetBinCenter(i) * fJointEnergyUnit;
304 for (int j = 1; j <= fJointHistogram->GetNbinsY(); j++) {
305 const double angle = fJointHistogram->GetYaxis()->GetBinCenter(j) * fJointAngleUnit;
306 double content = fJointHistogram->GetBinContent(i, j);
307 if (energy < energyRange.first || energy > energyRange.second || angle < angleRange.first ||
308 angle > angleRange.second) {
309 content = 0;
310 } else if (fJointType == "cosmic" && std::abs(std::cos(angle)) > 1e-6) {
311 // As legacy, the flux through a horizontal surface to the one through the disk normal to the
312 // direction of the cosmic generator
313 content /= std::abs(std::cos(angle));
314 }
315 fJointHistogram->SetBinContent(i, j, content);
316 }
317 }
318 if (fJointHistogram->Integral() <= 0) {
319 throw runtime_error(GetClassName() + ": the histogram is empty in 'range'");
320 }
321 fJointHistogram->ComputeIntegral();
322}

324vector<TRestGeant4ParticleState> TRestGeant4ParticleSource::Generate(TRandom& random, const XYZVector& origin,
325 const XYZVector& center, bool inward) {
326 if (IsDecay0()) {
327 if (!fDecay0Events) throw runtime_error(GetClassName() + ": the Decay0 file is not loaded");
328 auto particles = (*fDecay0Events)[random.Integer(static_cast<unsigned int>(fDecay0Events->size()))];
329 for (auto& particle : particles) particle.SetOrigin(origin);
330 return particles;
331 }
332
333 XYZVector axis = fAngular->GetDirection();
334 if (inward && (center - origin).R() > 0) axis = (center - origin).Unit();
335
336 double energy = 0;
337 XYZVector direction;
338 if (!fJointType.empty()) {
339 double x = 0, angle = 0;
340 if (fJointFunction) {
341 SampleJointFunction(random, x, angle);
342 } else if (fJointHistogram) {
343 fJointHistogram->GetRandom2(x, angle, &random);
344 } else {
345 throw runtime_error(GetClassName() + ": the distribution is not loaded");
346 }
347 energy = x * fJointEnergyUnit;
348 direction = DirectionFromAxis(random, axis, angle * fJointAngleUnit);
349 } else {
350 energy = fEnergy->Sample(random);
351 direction = fAngular->Sample(random, axis);
352 }
353
354 TRestGeant4ParticleState particle;
355 particle.SetParticleName(fParticleName);
356 particle.SetExcitationLevel(fExcitedLevel);
357 particle.SetParticleCharge(fCharge);
358 particle.SetEnergy(energy);
359 particle.SetDirection(direction);
360 particle.SetOrigin(origin);
361 return {particle};
362}
