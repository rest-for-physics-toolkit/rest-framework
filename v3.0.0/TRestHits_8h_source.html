26template <typename... Arrays>
27void ApplyPermutation(const std::vector<int>& order, int start, Arrays&... arrays) {
28 const int n = (int)order.size();
29 auto permuteOne = [&order, n, start](auto& values) {
30 using T = typename std::decay_t<decltype(values)>::value_type;
31 std::vector<T> sorted(n);
32 for (int i = 0; i < n; ++i) sorted[i] = values[start + order[i]];
33 for (int i = 0; i < n; ++i) values[start + i] = sorted[i];
34 };
35 (permuteOne(arrays), ...);
36}

41inline std::vector<int> BuildSortOrder(int n, std::function<bool(int, int)> compareCondition,
42 std::function<bool(int, int)> defaultCompare) {
43 std::vector<int> order(n);
44 std::iota(order.begin(), order.end(), 0);
45 std::stable_sort(order.begin(), order.end(), compareCondition ? compareCondition : defaultCompare);

51inline std::vector<int> BuildShuffleOrder(int n, int NLoop) {
52 std::vector<int> order(n);
53 std::iota(order.begin(), order.end(), 0);
54 for (int k = 0; k < NLoop; ++k) {
55 // Integer(n) gives [0, n): (int)(n * Uniform(0, 1)) could give n if the generator returns 1
56 int i = (int)gRandom->Integer(n);
57 int j = (int)gRandom->Integer(n);
58 std::swap(order[i], order[j]);
59 }
60 return order;
61}
