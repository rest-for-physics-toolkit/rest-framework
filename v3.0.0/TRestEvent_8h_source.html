101 static DrawOption ParseDrawOption(const std::string& option) {
102 const size_t open = option.find('('), close = option.find(')');
103 const size_t openValue = option.find('['), closeValue = option.find(']');
104 DrawOption parsed;
105 parsed.name = option.substr(0, std::min(open, openValue));
106 if (open != std::string::npos && close != std::string::npos)
107 parsed.rootOption = option.substr(open + 1, close - open - 1);
108 if (openValue != std::string::npos && closeValue != std::string::npos)
109 parsed.value = std::stod(option.substr(openValue + 1, closeValue - openValue - 1));

216 static unsigned long EventSeed(unsigned long seed, const TRestEvent& event) {
217 auto mix = [](uint64_t x) {
218 x += 0x9e3779b97f4a7c15ULL;
219 x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
220 x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
221 return x ^ (x >> 31);
222 };
223 uint64_t h = mix(seed);
224 h = mix(h ^ static_cast<uint32_t>(event.GetID()));
225 h = mix(h ^ static_cast<uint32_t>(event.GetSubID()));
226 const unsigned long folded = (h ^ (h >> 32)) & 0xffffffffULL;
227 return folded != 0 ? folded : 1;
228 }
