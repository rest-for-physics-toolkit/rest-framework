92int Bin(double value, double min, double max, int n) {
93 if (!(max > min)) return 0;
94 const int bin = static_cast<int>(std::floor((value - min) / (max - min) * n));
95 return std::clamp(bin, 0, n - 1);

99double Sample(const std::vector<int>& counts, double min, double max, TRandom& random) {
100 long total = 0;
101 for (const int count : counts) total += count;
102 if (total <= 0 || !(max > min)) return min;
103 long target = static_cast<long>(random.Uniform() * total);
104 size_t bin = 0;
105 for (; bin + 1 < counts.size(); bin++) {
106 if (target < counts[bin]) break;
107 target -= counts[bin];
108 }
109 const double width = (max - min) / counts.size();
110 return min + (bin + random.Uniform()) * width;
111}

179void TRestDetectorDriftResponse::SetNode(size_t index, int primaries,
180 const std::vector<std::array<double, 3>>& arrivals) {
181 if (fNodes.size() != GetNumberOfNodes()) fNodes.assign(GetNumberOfNodes(), Node());
182 Node& node = fNodes.at(index);
183 node.fPrimaries = primaries;
184 node.fArrived = arrivals.size();
185 node.fPositionCounts.assign(static_cast<size_t>(fPositionBins) * fPositionBins, 0);
186 node.fTimeCounts.assign(fTimeBins, 0);
187 if (arrivals.empty()) return;
188 node.fUMin = node.fUMax = arrivals[0][0];
189 node.fVMin = node.fVMax = arrivals[0][1];
190 node.fTMin = node.fTMax = arrivals[0][2];
191 for (const auto& [u, v, t] : arrivals) {
192 node.fUMin = std::min(node.fUMin, u);
193 node.fUMax = std::max(node.fUMax, u);
194 node.fVMin = std::min(node.fVMin, v);
195 node.fVMax = std::max(node.fVMax, v);
196 node.fTMin = std::min(node.fTMin, t);
197 node.fTMax = std::max(node.fTMax, t);
198 }
199 for (const auto& [u, v, t] : arrivals) {
200 const int uBin = Bin(u, node.fUMin, node.fUMax, fPositionBins);
201 const int vBin = Bin(v, node.fVMin, node.fVMax, fPositionBins);
202 node.fPositionCounts[uBin + static_cast<size_t>(fPositionBins) * vBin]++;
203 node.fTimeCounts[Bin(t, node.fTMin, node.fTMax, fTimeBins)]++;
204 }
205}

207bool TRestDetectorDriftResponse::Neighbours(const XYZVector& position, std::array<size_t, 8>& nodes,
208 std::array<double, 8>& weights) const {
209 if (fNodes.size() != GetNumberOfNodes()) return false;
210 const double coordinates[3] = {position.X(), position.Y(), position.Z()};
211 const std::pair<TRestWithUnits, TRestWithUnits>* ranges[3] = {&fXRange, &fYRange, &fZRange};
212 int first[3];
213 double fraction[3];
214 for (int axis = 0; axis < 3; axis++) {
215 const int n = fGridNodes[axis];
216 const double min = ranges[axis]->first, max = ranges[axis]->second;
217 const double f = (coordinates[axis] - min) / (max - min) * (n - 1);
218 constexpr double tolerance = 1e-9;
219 if (!(f >= -tolerance && f <= n - 1 + tolerance)) return false;
220 first[axis] = std::clamp(static_cast<int>(std::floor(f)), 0, n - 2);
221 fraction[axis] = std::clamp(f - first[axis], 0.0, 1.0);
222 }
223 for (int corner = 0; corner < 8; corner++) {
224 const int di = corner & 1, dj = (corner >> 1) & 1, dk = (corner >> 2) & 1;
225 nodes[corner] = NodeIndex(first[0] + di, first[1] + dj, first[2] + dk);
226 weights[corner] = (di ? fraction[0] : 1 - fraction[0]) * (dj ? fraction[1] : 1 - fraction[1]) *
227 (dk ? fraction[2] : 1 - fraction[2]);
228 }
229 return true;
230}

232double TRestDetectorDriftResponse::GetSurvival(const XYZVector& position) const {
233 std::array<size_t, 8> nodes;
234 std::array<double, 8> weights;
235 if (!Neighbours(position, nodes, weights)) return 0;
236 double survival = 0;
237 for (int corner = 0; corner < 8; corner++) {
238 const Node& node = fNodes[nodes[corner]];
239 if (node.fPrimaries > 0) survival += weights[corner] * node.fArrived / node.fPrimaries;
240 }
241 return survival;
242}

244bool TRestDetectorDriftResponse::SampleArrival(const XYZVector& position, TRandom& random, double& u,
245 double& v, double& time) const {
246 std::array<size_t, 8> nodes;
247 std::array<double, 8> weights;
248 if (!Neighbours(position, nodes, weights)) return false;
249 // Weight of each node: its interpolation weight times its survival
250 double total = 0;
251 for (int corner = 0; corner < 8; corner++) {
252 const Node& node = fNodes[nodes[corner]];
253 weights[corner] *= node.fPrimaries > 0 ? static_cast<double>(node.fArrived) / node.fPrimaries : 0;
254 total += weights[corner];
255 }
256 if (total <= 0) return false;
257 double target = random.Uniform() * total;
258 int corner = 0;
259 for (; corner < 7; corner++) {
260 if (target < weights[corner]) break;
261 target -= weights[corner];
262 }
263 // The last corners can have no weight (rounding): the last one with arrivals
264 while (corner > 0 && weights[corner] <= 0) corner--;
265 const Node& node = fNodes[nodes[corner]];
266
267 // Position bin from the 2D counts, then uniform within it
268 long count = 0;
269 for (const int c : node.fPositionCounts) count += c;
270 long bin = 0;
271 if (count > 0) {
272 long t = static_cast<long>(random.Uniform() * count);
273 for (; bin + 1 < static_cast<long>(node.fPositionCounts.size()); bin++) {
274 if (t < node.fPositionCounts[bin]) break;
275 t -= node.fPositionCounts[bin];
276 }
277 }
278 const int uBin = bin % fPositionBins, vBin = bin / fPositionBins;
279 const double uWidth = (node.fUMax - node.fUMin) / fPositionBins;
280 const double vWidth = (node.fVMax - node.fVMin) / fPositionBins;
281 u = node.fUMin + (uBin + random.Uniform()) * uWidth;
282 v = node.fVMin + (vBin + random.Uniform()) * vWidth;
283 time = Sample(node.fTimeCounts, node.fTMin, node.fTMax, random);
284 return true;
285}

317 std::copy(values, values + 6, range);
318 positionCounts = node.fPositionCounts;
319 timeCounts = node.fTimeCounts;

321 }

325TRestDetectorDriftResponse TRestDetectorDriftResponse::Read(TFile* file) {
326 const std::string error = "TRestDetectorDriftResponse: file " + std::string(file->GetName()) + ": ";
327 const YAML::Node metadata = TRestMetadata::ReadMetadata(file, kName);
328 if (!metadata || !metadata.IsMap() || !metadata["configuration"]) {
329 throw std::runtime_error(error + "no " + kName + " metadata");
330 }
331 TRestDetectorDriftResponse response("drift", YAML::Clone(metadata["configuration"]));
332 response.fMedium = metadata["medium"].as<std::string>("");
333 response.fPressure = metadata["pressure"].as<double>(0);
334 response.fTemperature = metadata["temperature"].as<double>(0);
335 response.fTransportTable = metadata["transportTable"].as<std::string>("");
336
337 auto* tree = file->Get<TTree>(kName.c_str());
338 if (!tree) throw std::runtime_error(error + "no " + kName + " tree");
339 if (static_cast<size_t>(tree->GetEntries()) != response.GetNumberOfNodes()) {
340 throw std::runtime_error(error + "the tree has not an entry per node of the grid");
341 }
342 int primaries = 0, arrived = 0;
343 double range[6];
344 std::vector<int>* positionCounts = nullptr;
345 std::vector<int>* timeCounts = nullptr;
346 tree->SetBranchAddress("primaries", &primaries);
347 tree->SetBranchAddress("arrived", &arrived);
348 tree->SetBranchAddress("range", range);
349 tree->SetBranchAddress("positionCounts", &positionCounts);
350 tree->SetBranchAddress("timeCounts", &timeCounts);
351 const size_t positionSize = static_cast<size_t>(response.fPositionBins) * response.fPositionBins;
352 response.fNodes.resize(response.GetNumberOfNodes());
353 for (Long64_t entry = 0; entry < tree->GetEntries(); entry++) {
354 tree->GetEntry(entry);
355 if (!positionCounts || !timeCounts || positionCounts->size() != positionSize ||
356 timeCounts->size() != static_cast<size_t>(response.fTimeBins)) {
357 throw std::runtime_error(error + "node " + std::to_string(entry) +
358 " has not the bins of the grid");
359 }
360 Node& node = response.fNodes[entry];
361 node.fPrimaries = primaries;
362 node.fArrived = arrived;
363 node.fUMin = range[0];
364 node.fUMax = range[1];
365 node.fVMin = range[2];
366 node.fVMax = range[3];
367 node.fTMin = range[4];
368 node.fTMax = range[5];
369 node.fPositionCounts = *positionCounts;
370 node.fTimeCounts = *timeCounts;
371 }
372 tree->ResetBranchAddresses();
373 delete positionCounts;
374 delete timeCounts;
375 return response;
376}
