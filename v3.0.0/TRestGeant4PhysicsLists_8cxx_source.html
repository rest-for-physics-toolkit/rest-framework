76void TRestGeant4PhysicsLists::LoadConfig() {
77 UpdateParamsFromYAML<TRestGeant4PhysicsLists>(fNode);
78 // A name outside fValidPhysicsLists would be ignored by restG4 (a typo, or a list not supported)
79 for (const auto& physicsList : fPhysicsLists) {
80 if (PhysicsListExists(physicsList.name)) continue;
81 std::string valid;
82 for (const auto& name : fValidPhysicsLists) valid += (valid.empty() ? "" : ", ") + name;
83 throw std::runtime_error("TRestGeant4PhysicsLists '" + GetName() + "': physics list '" +
84 physicsList.name + "' of 'physicsList' is not valid. Valid ones: " + valid);
85 }
86 UpdateYAMLFromParams<TRestGeant4PhysicsLists>(fNode);
87}
