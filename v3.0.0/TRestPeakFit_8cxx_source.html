53Peak FitPeak(TH1& histogram, double start, double end, const std::string& name, Model model,
54 Background background) {
55 Peak peak;
56 RemoveFunction(histogram, name);
57
58 // The bins with their center in [start, end]
59 const TAxis* axis = histogram.GetXaxis();
60 int first = std::max(axis->FindFixBin(start), 1);
61 int last = std::min(axis->FindFixBin(end), axis->GetNbins());
62 if (axis->GetBinCenter(first) < start) first++;
63 if (axis->GetBinCenter(last) > end) last--;
64 if (last - first < 2) return peak;
65 const double low = axis->GetBinLowEdge(first), high = axis->GetBinUpEdge(last);
66 const int nBins = last - first + 1;
67 TH1D window("window", "", nBins, low, high);
68 window.SetDirectory(nullptr);
69 int maximumBin = first;
70 for (int bin = first; bin <= last; bin++) {
71 window.SetBinContent(bin - first + 1, histogram.GetBinContent(bin));
72 window.SetBinError(bin - first + 1, histogram.GetBinError(bin));
73 if (histogram.GetBinContent(bin) > histogram.GetBinContent(maximumBin)) maximumBin = bin;
74 }
75 const double total = window.Integral();
76 if (total <= 0) return peak;
77 const double binWidth = (high - low) / nBins;
78
79 // The model: the peak (gaussian or Crystal Ball) and the background (Chebychev polynomial: flat or
80 // linear), extended
81 RooRealVar x("x", "x", low, high);
82 x.setBins(nBins);
83 RooDataHist data("data", "", RooArgList(x), RooFit::Import(window));
84 RooRealVar mean("mean", "", axis->GetBinCenter(maximumBin), low, high);
85 RooRealVar sigma("sigma", "", (high - low) / 6, binWidth / 10, high - low);
86 RooRealVar alpha("alpha", "", 1.5, 0.1, 10);
87 RooRealVar n("n", "", 3, 1.01, 100);
88 std::unique_ptr<RooAbsPdf> signalPdf;
89 if (model == Model::CrystalBall) {
90 signalPdf = std::make_unique<RooCBShape>("signal", "", x, mean, sigma, alpha, n);
91 } else {
92 signalPdf = std::make_unique<RooGaussian>("signal", "", x, mean, sigma);
93 }
94 RooRealVar slope("backgroundSlope", "", 0, -1, 1);
95 std::unique_ptr<RooAbsPdf> backgroundPdf;
96 if (background == Background::Constant) {
97 backgroundPdf = std::make_unique<RooChebychev>("background", "", x, RooArgList());
98 } else if (background == Background::Linear) {
99 backgroundPdf = std::make_unique<RooChebychev>("background", "", x, RooArgList(slope));
100 }
101 RooRealVar nSignal("signalCounts", "", backgroundPdf ? 0.8 * total : total, 0, 2 * total);
102 RooRealVar nBackground("backgroundCounts", "", 0.2 * total, 0, 2 * total);
103 std::unique_ptr<RooAbsPdf> fitPdf;
104 if (backgroundPdf) {
105 fitPdf = std::make_unique<RooAddPdf>("model", "", RooArgList(*signalPdf, *backgroundPdf),
106 RooArgList(nSignal, nBackground));
107 } else {
108 fitPdf = std::make_unique<RooExtendPdf>("model", "", *signalPdf, nSignal);
109 }
110
111 // Quiet fit
112 RooMsgService& messages = RooMsgService::instance();
113 const RooFit::MsgLevel level = messages.globalKillBelow();
114 messages.setGlobalKillBelow(RooFit::WARNING);
115 std::unique_ptr<RooFitResult> result(fitPdf->fitTo(data, RooFit::Save(), RooFit::PrintLevel(-1),
116 RooFit::Extended(true), RooFit::Warnings(false)));
117 messages.setGlobalKillBelow(level);
118
119 peak.valid = result && result->status() == 0;
120 peak.mean = mean.getVal();
121 peak.sigma = sigma.getVal();
122 peak.meanError = mean.getError();
123 peak.sigmaError = sigma.getError();
124 peak.signal = nSignal.getVal();
125 peak.background = backgroundPdf ? nBackground.getVal() : 0;
126 if (result) {
127 peak.status = result->status();
128 for (const auto* parameter : result->floatParsFinal()) {
129 if (const auto* variable = dynamic_cast<const RooRealVar*>(parameter))
130 peak.parameters[variable->GetName()] = {variable->getVal(), variable->getError()};
131 }
132 }
133
134 // The fit as a TF1 with the scale of the histogram (counts per bin), kept in it
135 std::string formula = "[Constant]*exp(-0.5*((x-[Mean])/[Sigma])^2)";
136 if (model == Model::CrystalBall) {
137 formula =
138 "[Constant]*(((x-[Mean])/[Sigma] > -[Alpha]) ? exp(-0.5*((x-[Mean])/[Sigma])^2) : "
139 "pow([N]/[Alpha],[N])*exp(-0.5*[Alpha]*[Alpha])*pow([N]/[Alpha]-[Alpha]-(x-[Mean])/"
140 "[Sigma],-[N]))";
141 }
142 if (backgroundPdf) formula += "+[Background]*(1+[Slope]*(2*x-[Low]-[High])/([High]-[Low]))";
143 auto* fit = new TF1(name.c_str(), formula.c_str(), low, high, TF1::EAddToList::kNo);
144 fit->SetParameter("Constant", 1);
145 fit->SetParameter("Mean", peak.mean);
146 fit->SetParameter("Sigma", peak.sigma);
147 fit->SetParError(fit->GetParNumber("Mean"), peak.meanError);
148 fit->SetParError(fit->GetParNumber("Sigma"), peak.sigmaError);
149 if (model == Model::CrystalBall) {
150 fit->SetParameter("Alpha", alpha.getVal());
151 fit->SetParameter("N", n.getVal());
152 }
153 if (backgroundPdf) {
154 fit->SetParameter("Low", low);
155 fit->SetParameter("High", high);
156 fit->SetParameter("Slope", background == Background::Linear ? slope.getVal() : 0);
157 fit->SetParameter("Background", 0);
158 }
159 // Normalized as RooFit, in the fit range
160 const double shapeIntegral = fit->Integral(low, high);
161 fit->SetParameter("Constant", shapeIntegral > 0 ? peak.signal * binWidth / shapeIntegral : 0);
162 if (backgroundPdf) fit->SetParameter("Background", peak.background * binWidth / (high - low));
163 histogram.GetListOfFunctions()->Add(fit);
164 return peak;
165}

173std::optional<double> FindPeak(const TH1& histogram, double start, double end, double smoothing,
174 double significance) {
175 // The bins with their center in [start, end]
176 const TAxis* axis = histogram.GetXaxis();
177 std::vector<double> counts, centers;
178 for (int bin = 1; bin <= axis->GetNbins(); bin++) {
179 const double center = axis->GetBinCenter(bin);
180 if (center < start || center > end) continue;
181 counts.push_back(histogram.GetBinContent(bin));
182 centers.push_back(center);
183 }
184 const int n = static_cast<int>(counts.size());
185 if (n < 3) return std::nullopt;
186
187 // Gaussian smoothing (kernel up to 3 sigma, normalized with the bins inside the range)
188 std::vector<double> smoothed = counts;
189 if (smoothing > 0) {
190 const int reach = static_cast<int>(std::ceil(3 * smoothing));
191 for (int i = 0; i < n; i++) {
192 double sum = 0, weights = 0;
193 for (int j = std::max(0, i - reach); j <= std::min(n - 1, i + reach); j++) {
194 const double weight = std::exp(-0.5 * (i - j) * (i - j) / (smoothing * smoothing));
195 sum += weight * counts[j];
196 weights += weight;
197 }
198 smoothed[i] = sum / weights;
199 }
200 }
201
202 // The most prominent significant local maximum
203 std::optional<double> position;
204 double bestProminence = 0;
205 for (int i = 1; i < n - 1; i++) {
206 if (!(smoothed[i] > smoothed[i - 1] && smoothed[i] >= smoothed[i + 1])) continue;
207 // Lowest value on each side before a higher maximum (or the edge)
208 double left = smoothed[i], right = smoothed[i];
209 for (int j = i - 1; j >= 0 && smoothed[j] <= smoothed[i]; j--) left = std::min(left, smoothed[j]);
210 for (int j = i + 1; j < n && smoothed[j] <= smoothed[i]; j++) right = std::min(right, smoothed[j]);
211 const double prominence = smoothed[i] - std::max(left, right);
212 if (prominence < significance * std::sqrt(std::max(smoothed[i], 1.))) continue;
213 if (prominence > bestProminence) {
214 bestProminence = prominence;
215 position = centers[i];
216 }
217 }
218 return position;
219}

222 const std::pair<double, double>& zone,
223 double fraction) {

225 if (!peak) return std::nullopt;
226 return std::make_pair(std::max(zone.first, *peak * (1 - fraction)),
227 std::min(zone.second, *peak * (1 + fraction)));

230std::pair<double, double> EstimateWindow(const TH1& histogram, double maximumEnergy, double energy,
231 double fraction) {
232 const double maximum = histogram.GetBinCenter(histogram.GetMaximumBin());
233 const double position = energy * maximum / maximumEnergy;
234 return {position * (1 - fraction), position * (1 + fraction)};
235}

237std::pair<double, double> FitLine(TGraph& graph) {
238 // The points not finite (e.g. the mean of a failed peak fit, NaN) are removed
239 for (int i = graph.GetN() - 1; i >= 0; i--) {
240 if (std::isfinite(graph.GetPointX(i)) && std::isfinite(graph.GetPointY(i))) continue;
241 RESTWarning << "TRestPeakFit::FitLine: point (" << graph.GetPointX(i) << ", " << graph.GetPointY(i)
242 << ") of " << graph.GetName() << " not finite, removed from the linear fit" << RESTendl;
243 graph.RemovePoint(i);
244 }
245 if (graph.GetN() == 0) {
246 throw std::runtime_error(std::string("TRestPeakFit::FitLine: no valid points in ") + graph.GetName() +
247 " for the linear fit");
248 }
249 while (graph.GetN() < 2) graph.SetPoint(graph.GetN(), 0, 0);
250 const double xMin = *std::min_element(graph.GetX(), graph.GetX() + graph.GetN());
251 const double xMax = *std::max_element(graph.GetX(), graph.GetX() + graph.GetN());
252 if (xMin == xMax) {
253 throw std::runtime_error(std::string("TRestPeakFit::FitLine: all the points of ") + graph.GetName() +
254 " have the same x (" + std::to_string(xMin) + "): no line can be fitted");
255 }
256 TF1 line("linearFit", "pol1", 0, 1, TF1::EAddToList::kNo);
257 const TFitResultPtr result = graph.Fit(&line, "SQ");
258 if (static_cast<int>(result) != 0 || !result.Get() || !result->IsValid()) {
259 throw std::runtime_error(std::string("TRestPeakFit::FitLine: the linear fit of ") + graph.GetName() +
260 " failed (status " + std::to_string(static_cast<int>(result)) + ")");
261 }
262 return {line.GetParameter(0), line.GetParameter(1)};
263}
