171void TRestMetadataFieldRegistry::CheckFields(const TRestMetadata* instance, const YAML::Node& node) const {
172 if (!instance || !node || !node.IsMap()) return;
173 std::set<std::string> keys = GetFieldKeys(instance);
174 for (const auto& key : instance->GetSharedNodeKeys()) keys.insert(key);
175 bool hasChildren = false;
176 for (const auto& isContainer : fChildContainers) hasChildren = hasChildren || isContainer(instance);
177
178 std::string unknown;
179 for (const auto& element : node) {
180 const auto key = element.first.as<std::string>();
181 if (keys.count(key) || key == "class" || key == "name" || key == "verbose") continue;
182 if (hasChildren && element.second.IsMap() && element.second["class"]) continue;
183 unknown += " '" + key + "'";
184 }
185 if (unknown.empty()) return;
186
187 std::string valid;
188 for (const auto& key : keys) valid += " " + key;
189 throw std::runtime_error(instance->GetClassName() + " '" + instance->GetName() +
190 "': unknown parameter(s)" + unknown + ". Valid parameters: class name verbose" +
191 valid + (hasChildren ? " and metadata (maps with class)" : ""));
192}
