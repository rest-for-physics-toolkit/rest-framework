303 template <typename T>
304 void RegisterEvent(const std::string& className, T& eventObject) {
305 static_assert(std::is_base_of_v<TRestEvent, T>, "T must inherit from TRestEvent");
306
307 if (!fOutputFile) {
308 throw std::runtime_error("TRestRun: No output file added");
309 }
310
311 if (fOutputEventTrees.find(className) == fOutputEventTrees.end()) {
312 fOutputFile->cd();
313 auto* tree = new TTree(className.c_str(), (className + " AOD Event Tree").c_str());
314 tree->SetAutoSave(0);
315 fOutputEventTrees[className] = tree;
316 }
317
318 fOutputEvents[className] = &eventObject;
319
320 eventObject.CreateBranches(fOutputEventTrees[className]);
321 eventObject.WriteFormatVersion(fOutputEventTrees[className]);
322 // Larger baskets mean fewer, bigger writes, which helps a lot on network filesystems
323 if (fOutputBasketSize > 0) fOutputEventTrees[className]->SetBasketSize("*", fOutputBasketSize);
324
325 // Event info branches of the analysis tree only once, bound to the first registered event (the
326 // following ones copy its info); with an input analysis tree copied they are cloned from it
327 if (fOutputAnalysisTree && !CopiesInputAnalysisTree() && !fOutputAnalysisTree->GetBranch("eventID")) {
328 eventObject.TRestEvent::CreateBranches(fOutputAnalysisTree);
329 }
330 }

336 template <typename T>
337 void SetObservable(const std::string& name, T& variable) {
338 if (!fOutputAnalysisTree) throw std::runtime_error("TRestRun: Output file not open");
339 TBranch* branch = fOutputAnalysisTree->GetBranch(name.c_str());
340
341 if constexpr (std::is_fundamental_v<T>) {
342 if (!branch)
343 fOutputAnalysisTree->Branch(name.c_str(), &variable);
344 else
345 fOutputAnalysisTree->SetBranchAddress(name.c_str(), &variable);
346 } else {
347 auto holder = std::make_shared<T*>(&variable);
348 fObservablePointers.push_back(holder);
349 T** ptr = holder.get();
350 if (!branch) {
351 fOutputAnalysisTree->Branch(name.c_str(), ptr);
352 } else {
353 fOutputAnalysisTree->SetBranchAddress(name.c_str(), ptr);
354 }
355 }
