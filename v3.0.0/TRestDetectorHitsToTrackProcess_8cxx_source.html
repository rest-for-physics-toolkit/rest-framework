127bool TRestDetectorHitsToTrackProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
128 const auto* hitsEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
129 auto* trackEvent = dynamic_cast<TRestTrackEvent*>(&output);
130 if (!hitsEvent || !trackEvent) {
131 throw std::runtime_error(
132 "TRestDetectorHitsToTrackProcess: input/output is not a TRestDetectorHitsEvent/TRestTrackEvent");
133 }
134
135 trackEvent->Initialize();
136 trackEvent->SetEventInfo(&input);
137
138 const TRestHits& hits = *hitsEvent->GetHits();
139 std::map<int, std::vector<int>> hitsByType;
140 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) hitsByType[hits.GetType(n)].push_back(n);
141
142 for (const auto& [type, indices] : hitsByType) {
143 std::array<bool, 3> coordinates;
144 bool definesCoordinate = false;
145 for (int a = 0; a < 3; a++) {
146 coordinates[a] = TRestHitsData::IsDefined(type, a);
147 definesCoordinate |= coordinates[a];
148 }
149 if (!definesCoordinate) continue;
150
151 // Groups of hits (indices in the event) closer than clusterDistance, as without attachEnergy
152 std::vector<ROOT::Math::XYZVector> points;
153 points.reserve(indices.size());
154 for (int n : indices) points.push_back(hits.GetPosition(n));
155 const auto clusters = TRestHitsAnalysis::FindClusters(points, fClusterDistance, coordinates);
156
157 // Group of the track of each group: itself, or the group an isolated hit is attached to
158 std::vector<size_t> root(clusters.size());
159 std::iota(root.begin(), root.end(), 0);
160
161 // With attachEnergy, each isolated low energy hit (a group of one hit) joins the group of the
162 // closest hit, relative to attachDistance, closer than attachDistance in every coordinate of the
163 // type among the hits of the other groups (any energy): isolated hits are not attached to each other
164 size_t nAttached = 0;
165 if (fAttachEnergy > 0) {
166 auto isolated = [&](size_t c) {
167 return clusters[c].size() == 1 && hits.GetEnergy(indices[clusters[c][0]]) < fAttachEnergy;
168 };
169 std::vector<std::pair<int, size_t>> targets; // hit and its group
170 for (size_t c = 0; c < clusters.size(); c++) {
171 if (isolated(c)) continue;
172 for (size_t k : clusters[c]) targets.emplace_back(indices[k], c);
173 }
174 // (d / attachDistance)^2 summed over the coordinates, -1 if outside attachDistance in one
175 auto relative2 = [&](int a, int b) {
176 double r2 = 0;
177 for (int axis = 0; axis < 3; axis++) {
178 if (!coordinates[axis]) continue;
179 const double d = std::abs(hits.GetCoordinate(a, axis) - hits.GetCoordinate(b, axis));
180 if (d >= fAttachDistance[axis]) return -1.;
181 r2 += d * d / (fAttachDistance[axis] * fAttachDistance[axis]);
182 }
183 return r2;
184 };
185 for (size_t c = 0; c < clusters.size(); c++) {
186 if (!isolated(c)) continue;
187 const int n = indices[clusters[c][0]];
188 double closestR2 = std::numeric_limits<double>::max();
189 for (const auto& [m, group] : targets) {
190 const double r2 = relative2(n, m);
191 if (r2 >= 0 && r2 < closestR2) {
192 closestR2 = r2;
193 root[c] = group;
194 }
195 }
196 if (root[c] == c) {
197 fNNotAttached++;
198 } else {
199 nAttached++;
200 }
201 }
202 fNAttached += nAttached;
203 }
204
205 // Tracks in order of their first group, the hits keep the input order
206 std::vector<std::vector<int>> tracks;
207 std::map<size_t, size_t> trackOfRoot;
208 for (size_t c = 0; c < clusters.size(); c++) {
209 const size_t r = root[c];
210 if (!trackOfRoot.count(r)) {
211 trackOfRoot[r] = tracks.size();
212 tracks.emplace_back();
213 }
214 for (size_t k : clusters[c]) tracks[trackOfRoot[r]].push_back(indices[k]);
215 }
216 for (auto& track : tracks) {
217 std::sort(track.begin(), track.end());
218 trackEvent->AddTrack(trackEvent->GetNumberOfTracks() + 1, 0);
219 for (int n : track) {
220 trackEvent->AddHitToTrack(hits.GetPosition(n), hits.GetEnergy(n), hits.GetTime(n),
221 hits.GetType(n));
222 }
223 }
224 RESTDebug << "TRestDetectorHitsToTrackProcess: " << indices.size() << " hits of type " << type
225 << " give " << tracks.size() << " tracks (" << nAttached << " isolated hits attached)"
226 << RESTendl;
227 }
228
229 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) trackEvent->PrintOnlyTracks();
230
231 return trackEvent->GetNumberOfTracks() > 0;
232}
