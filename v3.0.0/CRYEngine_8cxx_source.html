33std::string ParticleName(CRYParticle& particle) {
34 const int charge = particle.charge();
35 switch (particle.id()) {
36 case CRYParticle::Neutron:
37 return "neutron";
38 case CRYParticle::Proton:
39 return "proton";
40 case CRYParticle::Pion:
41 return charge > 0 ? "pi+" : charge < 0 ? "pi-" : "pi0";
42 case CRYParticle::Kaon:
43 return charge > 0 ? "kaon+" : charge < 0 ? "kaon-" : "kaon0";
44 case CRYParticle::Muon:
45 return charge > 0 ? "mu+" : "mu-";
46 case CRYParticle::Electron:
47 return charge > 0 ? "e+" : "e-";
48 case CRYParticle::Gamma:
49 return "gamma";
50 }
51 throw std::runtime_error("CRY: unknown particle " + std::to_string(particle.id()));
52}

56 CRYEngine(const std::string& setupString, long seed) {
57 gCRYRandom.SetSeed(seed);
58 const char* dataPath = std::getenv("CRYDATAPATH");
59 fSetup = std::make_unique<CRYSetup>(setupString, dataPath && *dataPath ? dataPath : CRY_DATA_PATH);
60 fSetup->setRandomFunction(CRYRandom);
61 fGenerator = std::make_unique<CRYGenerator>(fSetup.get());
62 }

64 std::vector<TRestGeant4ParticleState> Generate() override {
65 constexpr double kUsPerS = 1e6;
66 std::vector<CRYParticle*> cryParticles;
67 fGenerator->genEvent(&cryParticles);
68
69 double firstTime = 0;
70 for (size_t i = 0; i < cryParticles.size(); i++) {
71 firstTime = i == 0 ? cryParticles[i]->t() : std::min(firstTime, cryParticles[i]->t());
72 }
73 std::vector<TRestGeant4ParticleState> particles;
74 for (auto* cryParticle : cryParticles) {
75 TRestGeant4ParticleState particle;
76 particle.SetParticleName(ParticleName(*cryParticle));
77 particle.SetParticleCharge(cryParticle->charge());
78 particle.SetEnergy(cryParticle->ke() * 1000); // MeV -> keV
79 particle.SetTime((cryParticle->t() - firstTime) * kUsPerS);
80 particle.SetOrigin({cryParticle->x() * 1000, cryParticle->y() * 1000, cryParticle->z() * 1000});
81 particle.SetDirection({cryParticle->u(), cryParticle->v(), cryParticle->w()});
82 particles.push_back(particle);
83 delete cryParticle;
84 }
85 return particles;
86 }
87
88 double GetTimeSimulated() const override { return fGenerator->timeSimulated() * 1e6; }
89
90 private:
91 std::unique_ptr<CRYSetup> fSetup;
92 std::unique_ptr<CRYGenerator> fGenerator;
93};
