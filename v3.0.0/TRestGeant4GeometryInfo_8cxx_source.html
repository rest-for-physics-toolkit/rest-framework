26void VisitPhysicalNodes(TGeoManager* geoManager, const string& separator, const NodeVisitor& visit) {
27 TGeoNode* top = geoManager ? geoManager->GetTopNode() : nullptr;
28 if (!top) return;
29 NodeVisitor walk = [&](TGeoNode* node, const string& path, const TGeoHMatrix& placement) {
30 if (visit(node, path, placement)) return true;
31 for (int i = 0; i < node->GetNdaughters(); i++) {
32 TGeoNode* daughter = node->GetDaughter(i);
33 TGeoHMatrix daughterPlacement = placement;
34 daughterPlacement.Multiply(daughter->GetMatrix());
35 const string name = daughter->GetName();
36 if (walk(daughter, node == top ? name : path + separator + name, daughterPlacement)) return true;
37 }
38 return false;
39 };
40 walk(top, top->GetName(), TGeoHMatrix());
41}

44TGeoNode* FindPhysicalNode(TGeoManager* geoManager, const string& separator, const string& name,
45 TGeoHMatrix& placement) {
46 TGeoNode* found = nullptr;
47 const auto visit = [&](TGeoNode* node, const string& path, const TGeoHMatrix& matrix) {
48 if (name != node->GetName() && name != path) return false;
49 found = node;
50 placement = matrix;
51 return true;
52 };
53 VisitPhysicalNodes(geoManager, separator, visit);
54 return found;
55}

140std::vector<std::string> TRestGeant4GeometryInfo::GetAllPhysicalVolumesMatchingExpression(
141 const std::string& expression) const {
142 std::vector<std::string> matched;
143 TPRegexp regex(expression.c_str());
144
145 // The Geant4 physical names (with their logical names), the ones of the hits, when the Geant4
146 // geometry is known (restG4): inside an assembly they differ from the GDML/TGeo ones (e.g.
147 // av_1_impr_1_<logical>_pv_0 instead of the physvol name)
148 for (const auto& [physName, alternativeNames] : fG4ToAltNames) {
149 const std::string cleanName = alternativeNames.empty() ? physName : *alternativeNames.begin();
150 if (regex.MatchB(cleanName.c_str()) || regex.MatchB(physName.c_str())) matched.push_back(physName);
151 }
152 if (!matched.empty() || !fG4ToAltNames.empty()) return matched;
153
154 // Otherwise the TGeo node names (the GDML physvol names)
155 for (const auto& nodeName : GetAllPhysicalVolumes()) {
156 if (regex.MatchB(nodeName.c_str())) matched.push_back(nodeName);
157 }
158 return matched;
159}

213std::string TRestGeant4GeometryInfo::GetGeant4NameFiltered(const std::string& g4Name) {
214 if (g4Name.find("av_") != std::string::npos && g4Name.find("_impr_") != std::string::npos) {
215 size_t imprPos = g4Name.find("_impr_");
216 size_t startPos = g4Name.find("_", imprPos + 6);
217 size_t endPos = g4Name.rfind("_pv_");
218
219 if (startPos != std::string::npos && endPos != std::string::npos && endPos > startPos) {
220 return g4Name.substr(startPos + 1, endPos - startPos - 1);

313std::vector<std::string> TRestGeant4GeometryInfo::GetAllPhysicalVolumesFromLogical(
314 const std::string& logicalVolumeName) const {
315 std::vector<std::string> result;
316 // The Geant4 physical names when the Geant4 geometry is known (restG4), as in
317 // GetAllPhysicalVolumesMatchingExpression: inside an assembly they differ from the TGeo ones
318 for (const auto& [physName, logicalNames] : fG4ToAltNames) {
319 if (logicalNames.count(logicalVolumeName)) result.push_back(physName);
320 }
321 if (!fG4ToAltNames.empty()) return result;
322
323 // Otherwise the TGeo node names (the GDML physvol names)
324 const auto visit = [&](TGeoNode* node, const string&, const TGeoHMatrix&) {
325 if (node->GetVolume() && logicalVolumeName == node->GetVolume()->GetName()) {
326 result.emplace_back(node->GetName());
327 }
328 return false;
329 };
330 VisitPhysicalNodes(fGeoManager, fPathSeparator, visit);
331 return result;
