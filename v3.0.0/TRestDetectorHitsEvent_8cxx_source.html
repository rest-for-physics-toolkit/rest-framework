27void TRestDetectorHitsEvent::Sort(std::function<bool(int, int)> compareCondition) {
28 const int n = static_cast<int>(GetNumberOfHits());
29 if (n < 2) return;

32 TRestHitsUtils::ApplyPermutation(order, 0, fHitsData.x, fHitsData.y, fHitsData.z, fHitsData.time,
33 fHitsData.energy, fHitsData.type, fParticleIDs);

128 fPad = new TPad(Form("pad_hits_event_%d", GetID()), "REST Detector Hits Viewer", 0, 0, 1, 1);
129 fPad->Divide(1, options.size());
130 fPad->Draw();
131
132 for (size_t i = 0; i < options.size(); i++) {
133 // name(rootOption)[pitch]
134 const auto [name, histOption, pitch] = ParseDrawOption(options[i]);
135
136 TVirtualPad* optionPad = fPad->cd(i + 1);
137 if (name == "graph") {
138 fHits.DrawGraphs(optionPad, Form("Event ID %d", GetID()));
139 } else if (name == "hist" || name == "markers") {
140 fHits.DrawHistograms(optionPad, histOption, pitch, name == "markers");
141 } else {
142 std::cout << "TRestDetectorHitsEvent::DrawEvent: unknown option '" << options[i] << "'"
143 << std::endl;
144 }
145 }
146
147 fPad->Modified();
148 if (safePad) safePad->cd();
