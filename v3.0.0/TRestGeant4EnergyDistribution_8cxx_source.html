55unique_ptr<TRestGeant4EnergyDistribution> TRestGeant4EnergyDistribution::Create(const string& name,
56 const YAML::Node& node) {
57 if (!node || !node["type"]) {
58 throw runtime_error("TRestGeant4EnergyDistribution: the energy 'type' is missing");
59 }
60 const auto type = ToLower(node["type"].as<string>());
61 if (type == "mono") return make_unique<TRestGeant4EnergyDistributionMono>(name, node);
62 if (type == "flat") return make_unique<TRestGeant4EnergyDistributionFlat>(name, node);
63 if (type == "log") return make_unique<TRestGeant4EnergyDistributionLog>(name, node);
64 if (type == "formula") return make_unique<TRestGeant4EnergyDistributionFormula>(name, node);
65 if (type == "th1d") return make_unique<TRestGeant4EnergyDistributionTH1D>(name, node);
66 if (type == "formula2" || type == "th2d" || type == "cosmic") {
67 return make_unique<TRestGeant4EnergyDistributionJoint>(name, node);
68 }
69 throw runtime_error("TRestGeant4EnergyDistribution: unknown energy type '" + type +
70 "', it must be mono, flat, log, formula, TH1D, formula2, TH2D or cosmic");
71}

150void TRestGeant4EnergyDistributionFormula::LoadDistribution() {
151 fFunction = make_shared<TF1>(
152 EnergyDistributionFormulasToRootFormula(StringToEnergyDistributionFormulas(fFormula)));
153 // The range cannot be larger than the formula one
154 const double min = std::max<double>(fRange.first, fFunction->GetXmin());
155 const double max = std::min<double>(fRange.second, fFunction->GetXmax());
156 if (min >= max) throw runtime_error(GetClassName() + ": 'range' is outside the formula one");
157 fFunction->SetRange(min, max);

189void TRestGeant4EnergyDistributionTH1D::LoadDistribution() {
190 fHistogram = ReadHistogram1D(fFilename, fNameInFile); // Any 1D histogram, as a TH1D
191
192 // As legacy, the unit from the axis title
193 const std::string title = fHistogram->GetXaxis()->GetTitle();
194 const auto contains = [&title](const char* unit) { return title.find(unit) != std::string::npos; };
195 fUnit = contains("GeV") ? 1e6 : contains("MeV") ? 1e3 : 1;
196
197 for (int i = 1; i <= fHistogram->GetNbinsX(); i++) {
198 const double energy = fHistogram->GetBinCenter(i) * fUnit;
199 if (energy < fRange.first || energy > fRange.second) fHistogram->SetBinContent(i, 0);
200 }
201 if (fHistogram->Integral() <= 0) {
202 throw runtime_error(GetClassName() + ": the histogram is empty in 'range'");
203 }
204 fHistogram->ComputeIntegral();
205}

207double TRestGeant4EnergyDistributionTH1D::Sample(TRandom& random) {
208 if (!fHistogram) throw runtime_error(GetClassName() + ": the distribution is not loaded");
209 return fHistogram->GetRandom(&random) * fUnit;
210}
