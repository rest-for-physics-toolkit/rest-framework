258int DCCPacket::Arg12ToFecAsicChannel(unsigned short arg1, unsigned short arg2, unsigned short& fec,
259 unsigned short& asic, unsigned short& channel) {
260 fec = (10 * (arg1 % 6) / 2 + arg2) / 4;
261 asic = (10 * (arg1 % 6) / 2 + arg2) % 4;
262 channel = arg1 / 6;
263 if ((fec > 5) || (asic > 3)) {
264 fec = arg2 - 4;
265 asic = 4;
266 channel = (arg1 - 4) / 6;
267 }
268
269 int physChannel = -10;
270 if (channel > 2 && channel < 15) {
271 physChannel = channel - 3;
272 } else if (channel > 15 && channel < 28) {
273 physChannel = channel - 4;
274 } else if (channel > 28 && channel < 53) {
275 physChannel = channel - 5;
276 } else if (channel > 53 && channel < 66) {
277 physChannel = channel - 6;
278 } else if (channel > 66) {
279 physChannel = channel - 7;
280 }
281
282 if (physChannel < 0) return physChannel;
283
284 physChannel = fec * 72 * 4 + asic * 72 + physChannel;
285
286 return physChannel;
287}

322bool DCCPacket::TryExtractNextEvent(std::deque<uint16_t>& buffer, size_t& idx, std::vector<uint16_t>& out) {
323 if (buffer.empty()) return false;
324
325 bool endOfEvent = false;
326 const size_t buffSize = buffer.size();
327
328 while (idx < buffSize) {
329 if (buffer[idx] == 0xFFFF) {
330 idx += 4;
331 endOfEvent = true;
332 break;
333 }
334
335 DataPacket* pck = (DCCPacket::DataPacket*)&buffer[idx];
336 size_t pckSize = (ntohs(pck->size)) / sizeof(uint16_t);
337 if (pckSize == 0) {
338 std::cout << "Warning packet size is zero, data may be not aligned due "
339 "to missing packets?"
340 << std::endl;
341 idx++;
342 } else
343 idx += pckSize;
344 // std::cout<<"Data packet pos "<<idx<<"/"<<buffSize<<"
345 // "<<pckSize<<std::endl;
346 }
347
348 if (!endOfEvent) {
349 return false; // incomplete event
350 }
351
352 out.assign(buffer.begin(), buffer.begin() + idx);
353 buffer.erase(buffer.begin(), buffer.begin() + idx);
354
355 idx = 0;
356
357 return true;
358}

360void DCCPacket::ParseEventFromWords(std::vector<uint16_t>& event, TRestRawSignalEvent& sEvent, uint64_t& tS,
361 uint32_t& ev_count) {
362 if (event.empty()) return;
363
364 size_t idx = 0;
365 const size_t buffSize = event.size();
366
367 while (idx < buffSize) {
368 if (event[idx] == 0xFFFF) {
369 if (idx + 3 >= buffSize) break; // Incomplete timestamp
370 // Decode 48 bit timestamp
371 uint16_t t1 = event[idx + 1]; // Bits 47-32 (MSB)
372 uint16_t t2 = event[idx + 2]; // Bits 31-16
373 uint16_t t3 = event[idx + 3]; // Bits 15-0 (LSB)
374
375 tS = (static_cast<uint64_t>(t1) << 32) | (static_cast<uint64_t>(t2) << 16) |
376 static_cast<uint64_t>(t3);
377 idx += 4;
378 break;
379 }
380
381 // Words of the header of a data packet, before the samples
382 constexpr size_t kHeaderWords = 8;
383 if (idx + kHeaderWords > buffSize) break; // Incomplete packet
384 const size_t pckStart = idx;
385 DataPacket* pck = (DCCPacket::DataPacket*)&event[idx];
386 const size_t pckSize = ntohs(pck->size) / sizeof(uint16_t);
387 // A packet of size 0 would never advance
388 idx += std::max<size_t>(pckSize, 1);
389
390 // The samples of the packet that are in the event
391 const unsigned int scnt = std::min<size_t>(ntohs(pck->scnt), buffSize - pckStart - kHeaderWords);
392 if (scnt < 2) continue;
393 if ((scnt <= 8) && (ntohs(pck->samp[0]) == 0) && (ntohs(pck->samp[1]) == 0)) continue; // empty data
394 if ((scnt <= 12) && ((ntohs(pck->samp[0]) == 0x11ff) || (ntohs(pck->samp[1]) == 0x11ff)))
395 continue; // Data starting at 511 bin
396
397 ev_count = GET_EVENT_COUNT(ntohs(pck->ecnt));
398
399 unsigned short fec, asic, channel;
400 const unsigned short arg1 = GET_RB_ARG1(ntohs(pck->args));
401 const unsigned short arg2 = GET_RB_ARG2(ntohs(pck->args));
402
403 int physChannel = Arg12ToFecAsicChannel(arg1, arg2, fec, asic, channel);
404
405 if (physChannel < 0) continue;
406
407 int timeBin = 0;
408 std::vector<short> sData(512, 0);
409
410 for (unsigned int i = 0; i < scnt && timeBin <= 511; i++) {
411 uint16_t val = ntohs(pck->samp[i]);
412 if ((val & 0xFE00) == 0x1000) {
413 timeBin = GET_CELL_INDEX(val);
414 } else if ((val & 0xF000) == 0) {
415 if (timeBin < 512) sData[timeBin] = val;
416 timeBin++;
417 }
418 }
419
420 sEvent.AddSignal(physChannel, sData);
421 }
