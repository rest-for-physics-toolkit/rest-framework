90void TRestDetectorReadoutManager::LoadReadout() {
91 for (auto& [name, ptr] : fReadoutMap) {
92 delete ptr;
93 }
94
95 fReadoutMap.clear();
96
97 for (const auto& readoutInfo : fReadoutInfo) {
98 // kSelfFile: the readout is stored in the file the manager is read from, or the input file of the run
99 std::unique_ptr<TFile> ownedFile;
100 TFile* fIn = nullptr;
101 if (readoutInfo.fInputFileName == kSelfFile) {
102 fIn = fSelfFile ? fSelfFile : fRunInfo ? fRunInfo->GetInputFile() : nullptr;
103 if (!fIn) {
104 RESTError << "Readout '" << readoutInfo.fInstanceName << "' has inputFileName '" << kSelfFile
105 << "' but the manager is not read from a file nor has a run with an input file"
106 << RESTendl;
107 continue;
108 }
109 } else {
110 // A URL (e.g. a versioned readout of rest-common) is downloaded and cached
111 const std::string inputFile = TRestTools::GetLocalFile(readoutInfo.fInputFileName);
112 if (!TRestTools::isRootFile(inputFile)) {
113 RESTError << "Input file " << readoutInfo.fInputFileName
114 << " not found, readout will not be loaded" << RESTendl;
115 continue;
116 }
117
118 ownedFile.reset(TFile::Open(inputFile.c_str(), "READ"));
119 if (!ownedFile || ownedFile->IsZombie()) {
120 RESTError << "Could not open ROOT file: " << readoutInfo.fInputFileName << RESTendl;
121 continue;
122 }
123 fIn = ownedFile.get();
124 }
125
126 auto yamlConfig = TRestMetadata::ReadMetadata(fIn, readoutInfo.fInstanceName);
127 if (!yamlConfig) {
128 RESTError << "Metadata instance " << readoutInfo.fInstanceName << " not found in file "
129 << readoutInfo.fInputFileName << RESTendl;
130 continue;
131 }
132
133 std::string className = yamlConfig["class"].as<std::string>();
134 // Written by REST (Export): not checked
135 std::unique_ptr<TRestMetadata> metadata =
136 MetadataClassRegistry::Instance().Create(className, readoutInfo.fInstanceName, yamlConfig, false);
137
138 if (!metadata) {
139 RESTError << className << " not found in MetadataRegistry." << RESTendl;
140 continue;
141 }
142
143 std::unique_ptr<TRestDetectorReadout> readout(dynamic_cast<TRestDetectorReadout*>(metadata.get()));
144 if (!readout) {
145 RESTError << "Cannot cast metadata to TRestDetectorReadout." << RESTendl;
146 continue;
147 }
148
149 metadata.release();
150
151 // The decoding name is the one given to Export when the readout file was written
152 const std::string& decodingName = readoutInfo.fDecodingName;
153 if (!readout->Import(fIn, readoutInfo.fInstanceName, decodingName)) {
154 // Not kept: the readout would silently use the decoding file read from disk at construction
155 std::string available;
156 for (const auto& name : TRestDetectorReadout::GetDecodingNames(fIn, readoutInfo.fInstanceName)) {
157 available += " '" + name + "'";
158 }
159 RESTError << "Decoding '" << decodingName << "' of readout '" << readoutInfo.fInstanceName
160 << "' (or its geometry) not found in " << readoutInfo.fInputFileName
161 << ". Available:" << available << RESTendl;
162 continue;
163 }
164
165 fReadoutMap[readoutInfo.fInstanceName] = readout.release();
166 RESTInfo << "Successfully imported readout instance '" << readoutInfo.fInstanceName
167 << "' with decoding '" << decodingName << "'" << RESTendl;
168
169 fReadoutMap[readoutInfo.fInstanceName]->PrintMetadata();
170 }
171}

185void TRestDetectorReadoutManager::WriteMetadata(TFile* file) {
186 UpdateYAMLFromParams<TRestDetectorReadoutManager>(fNode);
187 YAML::Node node = YAML::Clone(fNode);
188
189 // With a data readout (`readout`), only that one is stored: the data of the file are those of a single
190 // readout, the others were loaded for this processing (e.g. to view the whole detector)
191 YAML::Node readoutsNode = node["readouts"];
192 YAML::Node stored(YAML::NodeType::Sequence);
193 for (size_t i = 0; i < fReadoutInfo.size() && i < readoutsNode.size(); ++i) {
194 const auto& readoutInfo = fReadoutInfo[i];
195 if (!fDataReadoutName.empty() && readoutInfo.fInstanceName != fDataReadoutName) continue;
196 stored.push_back(readoutsNode[i]);
197 auto* readout = GetReadout(readoutInfo.fInstanceName);
198 // Readouts not imported keep pointing to their original file
199 if (!readout) continue;
200
201 if (!readout->Export(file, readoutInfo.fInstanceName, readoutInfo.fDecodingName)) {
202 RESTError << "Readout '" << readoutInfo.fInstanceName << "' could not be exported to "
203 << file->GetName() << RESTendl;
204 continue;
205 }
206 stored[stored.size() - 1]["inputFileName"] = kSelfFile;
207 }
208 node["readouts"] = stored;
209
210 TRestMetadata::WriteMetadata(file, fName, node);
211}

229TRestDetectorReadout* TRestDetectorReadoutManager::SelectReadout(std::string& name,
230 const std::string& user) const {
231 const std::string data = GetDataReadoutName();
232 std::string available;
233 for (const auto& [readoutName, readout] : fReadoutMap) available += " '" + readoutName + "'";
234 if (available.empty()) available = " none";
235 if (name.empty()) {
236 if (data.empty()) {
237 throw std::runtime_error(
238 user + ": several readouts loaded (" + available.substr(1) +
239 ") and no data readout: set 'readout' of the TRestDetectorReadoutManager");
240 }
241 name = data;
242 } else if (!data.empty() && name != data) {
243 RESTWarning << user << ": readout '" << name << "' instead of the data readout '" << data
244 << "' of the TRestDetectorReadoutManager" << RESTendl;
245 }
246 TRestDetectorReadout* readout = GetReadout(name);
247 if (!readout) {
248 throw std::runtime_error(user + ": readout '" + name + "' not loaded (readouts:" + available + ")");
249 }
250 return readout;
251}
