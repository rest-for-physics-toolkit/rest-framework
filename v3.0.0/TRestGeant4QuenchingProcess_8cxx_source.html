78double LindhardQuenchingFactor(double recoilEnergy, int A, int Z) {
79 if (recoilEnergy <= 0 || A <= 0 || Z <= 0) {
80 return 1.0;
81 }
82
83 const double gamma = 11.5 * recoilEnergy * TMath::Power(Z, -7.0 / 3.0);
84 const double g = 3 * TMath::Power(gamma, 0.15) + 0.7 * TMath::Power(gamma, 0.6) + gamma;
85 const double k = 0.133 * TMath::Power(Z, 2.0 / 3.0) * TMath::Power(A, -1.0 / 2.0);
86 return Clamp01(k * g / (1 + k * g));
87}

98double EstimateStepLength(const TRestGeant4Track& track, int hitIndex, double fallbackStepLength) {
99 if (hitIndex <= 0 || hitIndex >= static_cast<int>(track.GetNumberOfHits())) {
100 return fallbackStepLength;
101 }
102
103 if (track.GetHitVolumeID(hitIndex - 1) != track.GetHitVolumeID(hitIndex)) {
104 return fallbackStepLength;
105 }
106
107 const double stepLength = (track.GetHitPosition(hitIndex) - track.GetHitPosition(hitIndex - 1)).R();
108
109 if (!std::isfinite(stepLength) || stepLength <= 0) {
110 return fallbackStepLength;
111 }
112
113 return stepLength;
114}

116double BirksQuenchingFactor(double energy, double stepLength, double birksConstant) {
117 if (energy <= 0 || stepLength <= 0 || birksConstant <= 0) {
118 return 1.0;
119 }
120
121 const double dEdx = energy / stepLength;
122 return Clamp01(1.0 / (1.0 + birksConstant * dEdx));
123}

170void TRestGeant4QuenchingProcess::InitProcess() {
171 if (fRunInfo) fGeant4Metadata = (TRestGeant4Metadata*)fRunInfo->GetMetadataClass("TRestGeant4Metadata");
172 if (!fGeant4Metadata) {
173 throw std::runtime_error("TRestGeant4QuenchingProcess: no TRestGeant4Metadata found");
174 }
175 const auto& geometryInfo = fGeant4Metadata->GetGeant4GeometryInfo();
176
177 fVolumes.clear();
178 fVolumeModels.clear();
179 fVolumeBirksConstants.clear();
180 fVolumeBirksFallbackStepLengths.clear();
181
182 RegisterObservable("sensitiveQuenched", fObs.sensitiveQuenched);
183 RegisterObservable("sensitiveVolumeEnergyBefore", fObs.sensitiveVolumeEnergyBefore);
184 RegisterObservable("sensitiveVolumeEnergyAfter", fObs.sensitiveVolumeEnergyAfter);
185
186 // check all the user volume expressions are valid and correspond to at least a volume
187 for (auto& userVolume : fUserVolumes) {
188 // physical volumes matching the expression or, if none, those of the matching logical volumes
189 const auto volumeNames = geometryInfo.GetVolumesMatchingExpression(userVolume.fVolume);
190
191 if (volumeNames.empty()) {
192 RESTError << "TRestGeant4QuenchingProcess: No volume found matching expression: "
193 << userVolume.fVolume << RESTendl;
194 continue;
195 }
196
197 for (const auto& volumeNameString : volumeNames) {
198 RESTDebug << "Volume: " << volumeNameString << RESTendl;
199 fVolumes.insert(volumeNameString);
200 fVolumeModels[volumeNameString] = userVolume.fModel;
201 fVolumeBirksConstants[volumeNameString] = userVolume.fBirksConstant;
202 fVolumeBirksFallbackStepLengths[volumeNameString] = userVolume.fBirksFallbackStepLength;
203 }
204 }
205
206 // `model`: all the other volumes of the geometry, with the Birks parameters of the process
207 if (!fModel.empty()) {
208 for (const auto& volumeName : geometryInfo.GetAllPhysicalVolumes()) {
209 if (fVolumes.count(volumeName)) continue;
210 fVolumes.insert(volumeName);
211 fVolumeModels[volumeName] = fModel;
212 fVolumeBirksConstants[volumeName] = fBirksConstant;
213 fVolumeBirksFallbackStepLengths[volumeName] = fBirksFallbackStepLength;
214 }
215 }
216
217 // The Lindhard factor needs the target nucleus of the hits: without it nothing would be quenched
218 if (!fGeant4Metadata->GetStoreHadronicTargetInfo()) {
219 for (const auto& [volume, model] : fVolumeModels) {
220 if (model != "lindhard") continue;
221 throw std::runtime_error(
222 "TRestGeant4QuenchingProcess: the volume '" + volume +
223 "' uses the lindhard model, which needs the hadronic target "
224 "nucleus of the hits, but the simulation was made without storeHadronicTargetInfo: true in "
225 "the TRestGeant4Metadata. Simulate with it, or use the birks model for this volume");
226 }
227 }
228
229 RESTDebug << "TRestGeant4QuenchingProcess initialized with volumes" << RESTendl;
230 for (const auto& volume : fVolumes) {
231 RESTDebug << " " << volume << " (" << fVolumeModels[volume] << ")" << RESTendl;
232 }
233}

236bool TRestGeant4QuenchingProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
237 output.CopyFrom(&input);
238
239 auto* g4Event = dynamic_cast<TRestGeant4Event*>(&output);
240 if (!g4Event) {
241 throw std::runtime_error("TRestGeant4QuenchingProcess: output is not a TRestGeant4Event");
242 }
243 g4Event->SetGeant4Metadata(fGeant4Metadata);
244
245 fObs.clear();
246 fObs.sensitiveVolumeEnergyBefore = 0.0;
247 fObs.sensitiveVolumeEnergyAfter = 0.0;
248
249 const auto sensitiveVolumes = fGeant4Metadata->GetSensitiveVolumes();
250 for (const auto& sensitiveVolumeName : sensitiveVolumes) {
251 fObs.sensitiveVolumeEnergyBefore += g4Event->GetEnergyInVolume(sensitiveVolumeName);
252 }
253
254 // The deposited energies (total, per volume and per volume, particle and process) are recomputed
255 // from the visible energy of the hits, as legacy
256 auto& eventData = g4Event->fEventData;
257 eventData.totalDepositedEnergy = 0.0;
258 std::fill(eventData.volumeDepositedEnergy.begin(), eventData.volumeDepositedEnergy.end(), 0.0);
259 eventData.crossVolumeIDs.clear();
260 eventData.crossParticleIDs.clear();
261 eventData.crossProcessIDs.clear();
262 eventData.crossDepositedEnergies.clear();
263 g4Event->RebuildCrossIndex();
264
265 g4Event->ClearCache();
266
267 bool missingHadronicInfoWarningPrinted = false;
268
269 for (int trackIndex = 0; trackIndex < static_cast<int>(g4Event->GetNumberOfTracks()); ++trackIndex) {
270 auto track = g4Event->GetTrack(trackIndex);
271
272 const std::string particleName = track.GetParticleName();
273
274 const size_t nHits = track.GetNumberOfHits();
275
276 double trackNewDepositedEnergy = 0.0;
277
278 for (size_t hitIndex = 0; hitIndex < nHits; ++hitIndex) {
279 const std::string physicalVolumeName = track.GetHitVolumeName(hitIndex);
280
281 std::string cleanVolumeName =
282 fGeant4Metadata->GetGeant4GeometryInfo().GetGeant4NameFiltered(physicalVolumeName);
283
284 const double depositedEnergy = track.GetHitEnergy(hitIndex);
285
286 double quenchingFactor = 1.0;
287
288 if ((fVolumes.count(cleanVolumeName) || fVolumes.count(physicalVolumeName)) &&
289 depositedEnergy > 0) {
290 const std::string targetVol =
291 fVolumes.count(cleanVolumeName) ? cleanVolumeName : physicalVolumeName;
292
293 const auto model =
294 fVolumeModels.count(targetVol) ? fVolumeModels[targetVol] : std::string("lindhard");
295
296 if (model == "birks") {
297 if (!IsNeutralParticleWithoutBirksQuenching(particleName)) {
298 const double stepLength = EstimateStepLength(
299 track, static_cast<int>(hitIndex), fVolumeBirksFallbackStepLengths[targetVol]);
300
301 quenchingFactor = BirksQuenchingFactor(depositedEnergy, stepLength,
302 fVolumeBirksConstants[targetVol]);
303 }
304
305 } else if (model == "lindhard") {
306 // storeHadronicTargetInfo is checked in InitProcess, the event may still lack the
307 // information (e.g. written before it was stored)
308 if (track.GetHadronicOk()) {
309 const int isotopeA = track.GetHitHadronicTargetIsotopeA(hitIndex);
310
311 const int isotopeZ = track.GetHitHadronicTargetIsotopeZ(hitIndex);
312
313 // A = 0: no hadronic target nucleus in this hit
314 if (isotopeA > 0) {
315 quenchingFactor = LindhardQuenchingFactor(depositedEnergy, isotopeA, isotopeZ);
316 }
317
318 } else if (!missingHadronicInfoWarningPrinted) {
319 RESTWarning << "TRestGeant4QuenchingProcess: "
320 "Lindhard quenching requested but "
321 "hadronic target info not available."
322 << RESTendl;
323
324 missingHadronicInfoWarningPrinted = true;
325 }
326 }
327 }
328
329 const auto visibleEnergy = depositedEnergy * quenchingFactor;
330
331 if (fApplyToHitEnergies) {
332 track.SetHitEnergy(hitIndex, visibleEnergy);
333 }
334
335 if (visibleEnergy > 0) {
336 trackNewDepositedEnergy += visibleEnergy;
337
338 g4Event->AddEnergyInVolumeForParticleForProcess(visibleEnergy, track.GetHitVolumeID(hitIndex),
339 track.GetParticleID(),
340 track.GetHitProcess(hitIndex));
341 }
342 }
343
344 g4Event->fEventData.trackDepositedEnergy[trackIndex] = trackNewDepositedEnergy;
345 }
346
347 g4Event->ClearCache();
348 g4Event->RefreshViews();
349
350 for (const auto& sensitiveVolumeName : sensitiveVolumes) {
351 fObs.sensitiveVolumeEnergyAfter += g4Event->GetEnergyInVolume(sensitiveVolumeName);
352 }
353 g4Event->fEventData.sensitiveVolumeEnergy = fObs.sensitiveVolumeEnergyAfter;
354
355 fObs.sensitiveQuenched =
356 TMath::Abs(fObs.sensitiveVolumeEnergyAfter - fObs.sensitiveVolumeEnergyBefore) > 1e-2;
357 RESTDebug << fObs.sensitiveVolumeEnergyBefore << " " << fObs.sensitiveVolumeEnergyAfter << " "
358 << fObs.sensitiveQuenched << RESTendl;
359
360 return true;
361}
