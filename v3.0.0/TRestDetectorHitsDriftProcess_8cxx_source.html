111void TRestDetectorHitsDriftProcess::InitProcess() {
112 if (!fManager) throw std::runtime_error("TRestDetectorHitsDriftProcess: no TRestManager");
113
114 auto* readouts =
115 dynamic_cast<TRestDetectorReadoutManager*>(fManager->GetMetadataClass("TRestDetectorReadoutManager"));
116 if (!readouts) {
117 throw std::runtime_error("TRestDetectorHitsDriftProcess: no TRestDetectorReadoutManager found");
118 }
119 fReadout = readouts->SelectReadout(fReadoutName, "TRestDetectorHitsDriftProcess");
120
121 const std::string user = "TRestDetectorHitsDriftProcess";
122 const auto* medium = TRestDetectorMedium::Find(*fManager, fMediumName, "charge", user);
123 if (!medium) throw std::runtime_error(user + ": needs a TRestDetectorMedium with charge");
124 fCharge = &medium->GetChargeResponse(user);
125 if (fCharge->GetWValue() <= 0) {
126 throw std::runtime_error(user + ": the charge of the medium '" + medium->GetName() +
127 "' needs wValue > 0");
128 }
129 if (fFluctuations == "fano" && fCharge->GetFanoFactor() <= 0) {
130 throw std::runtime_error(user + ": fluctuations 'fano' need fanoFactor > 0 in the charge of '" +
131 medium->GetName() + "'");
132 }
133 fMediumName = medium->GetName();
134 fLongitudinalDiffusion = fCharge->GetLongitudinalDiffusion();
135 fTransversalDiffusion = fCharge->GetTransversalDiffusion();
136 fAttachment = fCharge->GetAttachment();
137 fDriftVelocity = fCharge->GetDriftVelocity();
138 if (fDriftVelocity <= 0) {
139 throw std::runtime_error(user + ": the charge of the medium '" + medium->GetName() +
140 "' gives no drift velocity");
141 }
142 // The stored metadata keeps the medium and the transport used
143 UpdateYAMLFromParams<TRestDetectorHitsDriftProcess>(fNode);
144
145 // Seed 0: random seed. Otherwise re-seeded at each event in ProcessEvent
146 fRandom.SetSeed(fSeed);
147}

149bool TRestDetectorHitsDriftProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
150 const auto* inputEvent = dynamic_cast<const TRestDetectorHitsEvent*>(&input);
151 auto* outputEvent = dynamic_cast<TRestDetectorHitsEvent*>(&output);
152 if (!inputEvent || !outputEvent) {
153 throw std::runtime_error(
154 "TRestDetectorHitsDriftProcess: input/output is not a TRestDetectorHitsEvent");
155 }
156
157 outputEvent->Initialize();
158 outputEvent->SetEventInfo(inputEvent);
159 if (fSeed != 0) fRandom.SetSeed(TRestEvent::EventSeed(fSeed, *inputEvent));
160
161 double totalEnergy = 0;
162 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
163 if (inputEvent->GetEnergy(n) > 0) totalEnergy += inputEvent->GetEnergy(n);
164 }
165 const double wValue = fCharge->GetWValue();
166 // Effective W-value: each electron carries the energy of several if there are more than fMaxHits
167 const double effectiveW =
168 fMaxHits > 0 && totalEnergy / wValue > fMaxHits ? totalEnergy / fMaxHits : wValue;
169
170 for (size_t n = 0; n < inputEvent->GetNumberOfHits(); n++) {
171 const double energy = inputEvent->GetEnergy(n);
172 if (energy <= 0) continue;
173
174 const XYZVector position = inputEvent->GetPosition(n);
175 const double distance = fReadout->GetDistanceToReadout(position);
176 // NaN maximum distance: no cathode, no upper cut
177 const double maxDistance = fReadout->GetMaxDistanceToReadout(position);
178 const bool hasCathode = !std::isnan(maxDistance);
179 if (std::isnan(distance) || distance < 0 || (hasCathode && distance > maxDistance)) continue;
180 const XYZVector normal = fReadout->GetReadoutNormal(position);
181
182 double electrons = energy / wValue;
183 if (fFluctuations == "fano") {
184 electrons = fRandom.Gaus(electrons, std::sqrt(fCharge->GetFanoFactor() * electrons));
185 } else if (fFluctuations == "poisson") {
186 electrons = fRandom.Poisson(electrons);
187 }
188 if (!(electrons > 0)) continue;
189 if (!fDiffusion) {
190 // The whole hit on the readout, its charge attenuated by the mean attachment (1/cm, d in mm)
191 const double charge = electrons * wValue * std::exp(-fAttachment * distance / 10.);
192 outputEvent->AddHit(fReadout->ProjectToReadout(position), charge,
193 inputEvent->GetTime(n) + distance / fDriftVelocity, inputEvent->GetType(n),
194 inputEvent->GetParticleID(n));
195 continue;
196 }
197 // The electrons of the hit carry the fluctuated charge (electrons * W)
198 const long nElectrons = std::max(1L, std::lround(electrons * wValue / effectiveW));
199 const double electronEnergy = electrons * wValue / nElectrons;
200
201 for (long e = 0; e < nElectrons; e++) {
202 XYZVector electron = position;
203 if (!TRestDetectorChargeResponse::DiffuseElectron(electron, normal, distance,
204 fLongitudinalDiffusion, fTransversalDiffusion,
205 fAttachment, fRandom)) {
206 continue;
207 }
208 // Kept between the readout plane and the cathode, then collected on the readout plane
209 double diffusedDistance = std::max(0.0, fReadout->GetDistanceToReadout(electron));
210 if (hasCathode) diffusedDistance = std::min(diffusedDistance, maxDistance);
211 const double arrival = inputEvent->GetTime(n) + diffusedDistance / fDriftVelocity;
212 outputEvent->AddHit(fReadout->ProjectToReadout(electron), electronEnergy, arrival,
213 inputEvent->GetType(n), inputEvent->GetParticleID(n));
214 }
215 }
216
217 return outputEvent->GetNumberOfHits() > 0;
218}
