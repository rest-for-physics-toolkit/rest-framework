125bool TRestRawSignalAnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
126 output.CopyFrom(&input);
127
128 auto* sigEvent = dynamic_cast<TRestRawSignalEvent*>(&output);
129 if (!sigEvent) {
130 throw std::runtime_error("TRestRawSignalAnalysisProcess: output is not a TRestRawSignalEvent");
131 }
132
133 // Reset observable values
134 fObs.clear();
135 const auto thrParams = std::make_pair(fPointsThreshold, fSignalThreshold);
136
137 fObs.MaxPeakAmplitude = 0.0;
138 fObs.MinPeakAmplitude = std::numeric_limits<double>::max();
139 fObs.MinEventValue = std::numeric_limits<double>::max();
140
141 fObs.MinPeakTime = std::numeric_limits<double>::max();
142 fObs.MaxPeakTime = 0.0;
143 fObs.AveragePeakTime = 0.0;
144
145 // A signal is considered "good" (legacy criterion) when it has at least 2 points over threshold
146 constexpr size_t kMinGoodPoints = 2;
147
148 for (int s = 0; s < sigEvent->GetNumberOfSignals(); s++) {
149 const auto& signal = sigEvent->GetSignal(s);
150 const int sID = signal.GetSignalID();
151
152 double baseline, baselineSigma;
153 auto data = CalculateBaselineAndSigma(signal.GetData(), fBaselineRange.first, fBaselineRange.second,
154 baseline, baselineSigma, fBaselineOption);
155
156 // GetPointsOverThreshold clamps the range to the signal length: use a per-signal copy
157 auto range = fIntegralRange;
158 auto pointsOverThr =
159 GetPointsOverThreshold(data, range, thrParams, fPointsOverThreshold, fNPointsFlat, baselineSigma);
160 const bool good = pointsOverThr.size() >= kMinGoodPoints;
161
162 fObs.NSignals++;
163 if (good) fObs.NGoodSignals++;
164
165 double thrIntegral = 0;
166 for (const auto& [bin, value] : pointsOverThr) {
167 thrIntegral += value;
168 }
169
170 // Neutral values when there are no points over threshold
171 std::pair<double, double> maxPeak{0., 0.};
172 std::pair<double, double> minVal{0., 0.};
173 if (!pointsOverThr.empty()) {
174 const std::string title = "Signal ID: " + std::to_string(sID);
175 auto sGraph = GetGraphPair(pointsOverThr, title);
176 maxPeak = GetMax(sGraph, fMaxOption);
177 minVal = GetMin(sGraph);
178 // A fit-based maximum can fall outside the signal: treat it as invalid
179 if (maxPeak.first < 0 || maxPeak.first >= signal.GetNPoints()) maxPeak = {0., 0.};
180 }
181 const auto riseTime = GetRiseTime(pointsOverThr);
182 const auto riseSlope = GetRiseSlope(pointsOverThr);
183 const bool saturated = signal.IsADCSaturation();
184 // Full width at half maximum of the baseline-subtracted pulse, in bins
185 const double width = pointsOverThr.empty() ? 0. : GetMaxPeakWidth(data);
186
187 fObs.SignalsID.push_back(sID);
188 fObs.Baseline.push_back(baseline);
189 fObs.BaselineSigma.push_back(baselineSigma);
190 fObs.AmpSgnMaxMethod.push_back(maxPeak.second);
191 fObs.PeakTime.push_back(maxPeak.first);
192 fObs.Width.push_back(width);
193 fObs.AmpSgnIntMethod.push_back(thrIntegral);
194 fObs.RiseTime.push_back(riseTime);
195 fObs.RiseSlope.push_back(riseSlope);
196 fObs.PointsOverThr.push_back(pointsOverThr.size());
197 fObs.Saturated.push_back(saturated);
198
199 if (good) {
200 fObs.FullIntegralSum += GetIntegral(data, range.first, range.second);
201 fObs.ThresholdIntegralSum += thrIntegral;
202 fObs.MaxIntegral += maxPeak.second;
203 fObs.SlopeIntegral += GetSlopeIntegral(pointsOverThr);
204 if (maxPeak.second > fObs.MaxPeakAmplitude) fObs.MaxPeakAmplitude = maxPeak.second;
205 if (maxPeak.second < fObs.MinPeakAmplitude) fObs.MinPeakAmplitude = maxPeak.second;
206
207 fObs.AveragePeakTime += maxPeak.first;
208 if (maxPeak.first < fObs.MinPeakTime) fObs.MinPeakTime = maxPeak.first;
209 if (maxPeak.first > fObs.MaxPeakTime) fObs.MaxPeakTime = maxPeak.first;
210 if (minVal.second < fObs.MinEventValue) {
211 fObs.MinEventValue = minVal.second;
212 }
213 }
214 }
215
216 if (fObs.NSignals <= 0) {
217 // Empty event: do not leave the accumulation sentinels (DBL_MAX) in the observables
218 fObs.MinPeakAmplitude = 0.0;
219 fObs.MinEventValue = 0.0;
220 fObs.MinPeakTime = 0.0;
221 return true;
222 }
223 fObs.BaselineAvg = std::accumulate(fObs.Baseline.begin(), fObs.Baseline.end(), 0.0) / fObs.NSignals;
224 fObs.BaselineSigmaAvg =
225 std::accumulate(fObs.BaselineSigma.begin(), fObs.BaselineSigma.end(), 0.0) / fObs.NSignals;
226
227 auto calcAvgOverThr = [](const auto& data, const std::vector<int>& pointsThr,
228 size_t minPoints) -> double {
229 if (data.size() != pointsThr.size()) return 0.0;
230 double sum = 0.0;
231 int count = 0;
232 for (size_t i = 0; i < data.size(); ++i) {
233 if ((size_t)pointsThr[i] >= minPoints) {
234 sum += data[i];
235 count++;
236 }
237 }
238 return (count > 0) ? (sum / count) : 0.0;
239 };
240
241 fObs.RiseSlopeAvg = calcAvgOverThr(fObs.RiseSlope, fObs.PointsOverThr, kMinGoodPoints);
242 fObs.RiseTimeAvg = calcAvgOverThr(fObs.RiseTime, fObs.PointsOverThr, kMinGoodPoints);
243 fObs.AmplitudeAvg = calcAvgOverThr(fObs.AmpSgnMaxMethod, fObs.PointsOverThr, kMinGoodPoints);
244 fObs.IntegralAvg = calcAvgOverThr(fObs.AmpSgnIntMethod, fObs.PointsOverThr, kMinGoodPoints);
245 fObs.WidthAvg = calcAvgOverThr(fObs.Width, fObs.PointsOverThr, kMinGoodPoints);
246 fObs.PointsOverThrAvg = calcAvgOverThr(fObs.PointsOverThr, fObs.PointsOverThr, kMinGoodPoints);
247 fObs.TimeBinLength = sigEvent->GetSignal(0).GetNPoints();
248 fObs.RateOfChangeAvg = (fObs.SlopeIntegral != 0) ? fObs.RiseSlopeAvg / fObs.SlopeIntegral : 0.0;
249 const double integralTotal = fObs.FullIntegralSum + fObs.ThresholdIntegralSum;
250 fObs.IntegralBalance =
251 (integralTotal != 0) ? (fObs.FullIntegralSum - fObs.ThresholdIntegralSum) / integralTotal : 0.0;
252
253 if (fObs.MaxIntegral > 0) fObs.AmplitudeIntegralRatio = fObs.ThresholdIntegralSum / fObs.MaxIntegral;
254 if (fObs.MaxPeakAmplitude > 0) fObs.AmplitudeRatio = fObs.MaxIntegral / fObs.MaxPeakAmplitude;
255 fObs.PeakAmplitudeIntegral = fObs.MaxIntegral;
256
257 if (fObs.NGoodSignals > 0) {
258 fObs.AveragePeakTime /= fObs.NGoodSignals;
259 fObs.MaxPeakTimeDelay = fObs.MaxPeakTime - fObs.MinPeakTime;
260 } else {
261 fObs.MinPeakAmplitude = 0.0;
262 fObs.MinEventValue = 0.0;
263 fObs.MinPeakTime = 0.0;
264 }
265
266 return true;
267}
