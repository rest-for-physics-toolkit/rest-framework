42std::string TRestDataSetProcess::ResolveReference(const TRestDataSet& dataSet) const {
43 const std::string context = GetClassName() + " '" + fName + "'";
44 if (fReferenceFileName.empty()) return dataSet.fInputFileName;
45 if (!fReferenceSelection.empty() && fReferenceSelection != "closest") {
46 throw std::runtime_error(context + ": referenceSelection '" + fReferenceSelection +
47 "' is not valid (\"closest\" or empty)");
48 }
49 std::vector<std::string> files;
50 for (const auto& file : TRestTools::GetFilesMatchingPattern(fReferenceFileName))
51 files.push_back(TRestTools::GetFullPath(file));
52 if (files.empty()) throw std::runtime_error(context + ": no file matches " + fReferenceFileName);
53 if (files.size() == 1) return files.front();
54 if (fReferenceSelection.empty()) {
55 throw std::runtime_error(context + ": " + std::to_string(files.size()) + " files match " +
56 fReferenceFileName + " (give one, or referenceSelection: closest)");
57 }
58
59 // The closest in time: the smallest gap between the times of the datasets, then the closest middles
60 const double start = dataSet.GetStartTime(), end = dataSet.GetEndTime();
61 std::string closest;
62 std::pair<double, double> best = {std::numeric_limits<double>::infinity(), 0};
63 for (const auto& fileName : files) {
64 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "READ"));
65 const auto runs = file && !file->IsZombie() ? TRestDataSet::ReadRuns(file.get()) : std::nullopt;
66 if (!runs || runs->empty()) throw std::runtime_error(context + ": " + fileName + " is not a dataset");
67 double referenceStart = runs->front().run->GetStartTime(),
68 referenceEnd = runs->front().run->GetEndTime();
69 for (const auto& run : *runs) {
70 referenceStart = std::min(referenceStart, run.run->GetStartTime());
71 referenceEnd = std::max(referenceEnd, run.run->GetEndTime());
72 }
73 const double gap = std::max({0., referenceStart - end, start - referenceEnd});
74 const double middles = std::abs((referenceStart + referenceEnd) / 2 - (start + end) / 2);
75 if (std::make_pair(gap, middles) < best) {
76 best = {gap, middles};
77 closest = fileName;
78 }
79 }
80 RESTInfo << context << ": reference " << closest << ", the closest in time of " << files.size()
81 << " (gap " << best.first << " s)" << RESTendl;
82 return closest;
83}

85std::string TRestDataSetProcess::GetCutExpression(const std::vector<std::string>& cutNames) const {
86 std::string expression;
87 for (const auto& name : cutNames) {
88 const TRestCut* cut = nullptr;
89 if (fManager) {
90 for (const auto* c : fManager->GetMetadataOfType<TRestCut>()) {
91 if (c->GetName() == name) cut = c;
92 }
93 }
94 if (!cut) {
95 throw std::runtime_error(GetClassName() + " '" + fName + "': TRestCut '" + name +
96 "' not found in the manager");
97 }
98 const std::string condition = cut->GetExpression();
99 if (condition.empty()) continue;
100 if (!expression.empty()) expression += " && ";
101 expression += "(" + condition + ")";
102 }
103 return expression;
104}

106void TRestDataSetProcess::ObtainParameters(const TRestDataSet& dataSet, const TRestManager* manager) {
107 fManager = manager;
108 // The cuts applied to the reference
109 const std::string cutExpression = GetCutExpression(fCutNames);
110
111 // A dataset generated in memory (not in a file yet) is its own reference: computed, it has no processes;
112 // the reference is the file it will be written to
113 const std::string referenceFile = ResolveReference(dataSet);
114 const bool inMemory = referenceFile.empty();
115 fReferenceUsed = inMemory ? dataSet.fOutputFileName : referenceFile;
116
117 // The reference processed by this process: its parameters, not computed
118 std::unique_ptr<TFile> file;
119 YAML::Node stored;
120 if (!inMemory) {
121 file.reset(TFile::Open(referenceFile.c_str(), "READ"));
122 if (!file || file->IsZombie()) {
123 throw std::runtime_error(GetClassName() + " '" + fName + "': cannot open " + referenceFile);
124 }
125 stored = TRestYAMLUtils::ReadMetadataStore(file.get())[fName];
126 }
127 if (stored && stored.IsMap() && stored["class"] && stored["class"].as<std::string>() == GetClassName()) {
128 RESTWarning << GetClassName() << " '" << fName << "': parameters read from " << referenceFile
129 << " (processed by it), not computed" << RESTendl;
130 const std::string reference = fReferenceFileName;
131 const std::string selection = fReferenceSelection;
132 const std::vector<std::string> cuts = fCutNames;
133 const std::string verbose = TRestLogManager::GetStringFromVerbose(fVerboseLevel);
134 YAML::Node config = TRestYAMLUtils::Detach(stored);
135 config["verbose"] = verbose;
136 LoadStoredConfig(config);
137 ReadObjects(GetStoreDirectory(file.get(), GetClassName(), fName));
138 // This configuration, with the stored results
139 fReferenceFileName = reference;
140 fReferenceSelection = selection;
141 fReferenceUsed = referenceFile;
142 fCutNames = cuts;
143 UpdateYAMLFromParams<TRestDataSetProcess>(fNode);
144 return;
145 }
146 file.reset();
147
148 RESTInfo << GetClassName() << " '" << fName << "': computing the parameters on " << fReferenceUsed
149 << RESTendl;
150 if (fReferenceFileName.empty()) {
151 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
152 if (!cutExpression.empty()) frame = frame.Filter(cutExpression);
153 ComputeParameters(dataSet, frame);
154 } else {
155 const TRestDataSet reference(referenceFile);
156 ROOT::RDF::RNode frame = reference.GetDataFrame();
157 if (!cutExpression.empty()) frame = frame.Filter(cutExpression);
158 ComputeParameters(reference, frame);
159 }
160 fParametersFileName = fReferenceUsed;
161 UpdateYAMLFromParams<TRestDataSetProcess>(fNode);
162}
