159int SegmentIndex(const std::vector<double>& split, double value) {
160 const int last = static_cast<int>(split.size()) - 2;
161 if (last < 0) return 0;
162 const auto upper = std::upper_bound(split.begin(), split.end(), value);
163 if (upper == split.end()) return last;
164 return std::max(static_cast<int>(std::distance(split.begin(), upper)) - 1, 0);
165}

168std::vector<double> EqualSplit(int segments, double first, double last) {
169 std::vector<double> split;
170 for (int i = 0; i <= segments; i++) split.push_back(first + (last - first) / segments * i);
171 return split;
172}

302std::pair<double, double> TRestDataSetGainMap::FitPeaks(TH1D& spectrum, TGraph& graph) const {
303 graph.Set(0);
304 if (spectrum.Integral() <= 0) {
305 RESTError << "TRestDataSetGainMap '" << fName << "': empty spectrum " << spectrum.GetName()
306 << RESTendl;
307 return {0, 0};
308 }
309
310 // The position of the peaks from the maximum of the spectrum, the first peak
311 const double maximumEnergy = fPeaks.front().fEnergy;
312 for (size_t c = 0; c < fPeaks.size(); c++) {
313 const double energy = fPeaks[c].fEnergy;
314 const auto zone = fPeaks[c].HasRange()
315 ? fPeaks[c].fRange
316 : TRestPeakFit::EstimateWindow(spectrum, maximumEnergy, energy, fWindow);
317 auto window = zone;
318 if (fPeakSearch) {
319 if (const auto found = TRestPeakFit::FindPeakWindow(spectrum, zone, fWindow)) {
320 window = *found;
321 } else {
322 RESTWarning << "TRestDataSetGainMap '" << fName << "': no peak found for " << energy << " keV"
323 << " of " << spectrum.GetName() << " in (" << zone.first << ", " << zone.second
324 << "), fitted in it" << RESTendl;
325 }
326 }
327 const auto [start, end] = window;
328 const auto fit =
329 TRestPeakFit::FitPeak(spectrum, start, end, FitName(c), TRestPeakFit::ModelFromString(fPeakModel),
330 TRestPeakFit::BackgroundFromString(fBackground));
331 if (!fit.valid) {
332 RESTWarning << "TRestDataSetGainMap '" << fName << "': the fit of the peak " << energy
333 << " keV of " << spectrum.GetName() << " in (" << start << ", " << end
334 << ") did not converge" << RESTendl;
335 }
336 RESTDebug << " " << spectrum.GetName() << ": peak " << energy << " keV fitted in (" << start << ", "
337 << end << "): " << fit.mean << RESTendl;
338 graph.SetPoint(graph.GetN(), fit.mean, energy);
339 }
340 if (fZeroPoint) graph.SetPoint(graph.GetN(), 0, 0);
341 return FitLineOrNaN(graph, spectrum.GetName());
342}

344std::pair<double, double> TRestDataSetGainMap::UpdateLinearFit(TH1D& spectrum, TGraph& graph) const {
345 graph.Set(0);
346 for (size_t i = 0; i < fPeaks.size(); i++) {
347 const double mean = TRestPeakFit::GetPeakMean(spectrum, FitName(i));
348 if (std::isnan(mean)) {
349 RESTWarning << "TRestDataSetGainMap '" << fName << "': no fit " << FitName(i) << " of the peak "
350 << static_cast<double>(fPeaks[i].fEnergy) << " keV in " << spectrum.GetName()
351 << RESTendl;
352 continue;
353 }
354 graph.SetPoint(graph.GetN(), mean, static_cast<double>(fPeaks[i].fEnergy));
355 }
356 if (fZeroPoint) graph.SetPoint(graph.GetN(), 0, 0);
357 return FitLineOrNaN(graph, spectrum.GetName());
358}

360void TRestDataSetGainMap::ComputeParameters(const TRestDataSet& reference, ROOT::RDF::RNode frame) {
361 // The extent of the readout, divided in segments
362 if (fSplitX.empty() || fSplitY.empty()) {
363 const auto [minimum, maximum] = FindReadout(reference).GetChannelsBoundingBox();
364 const double low[3] = {minimum.X(), minimum.Y(), minimum.Z()};
365 const double high[3] = {maximum.X(), maximum.Y(), maximum.Z()};
366 // The plane of the readout: its normal is the axis of smallest extent (Z for a plane XY), X and Y
367 // of the map the two other axes in cyclic order (those of the strips of a Micromegas)
368 int normal = 2;
369 for (int a = 0; a < 3; a++)
370 if (high[a] - low[a] < high[normal] - low[normal]) normal = a;
371 const int axisX = (normal + 1) % 3, axisY = (normal + 2) % 3;
372 if (!std::isfinite(low[axisX]) || !std::isfinite(high[axisX])) {
373 throw std::runtime_error("TRestDataSetGainMap '" + fName +
374 "': the readout has no channels (give splitX and splitY)");
375 }
376 if (fSplitX.empty()) fSplitX = EqualSplit(fSegments.first, low[axisX], high[axisX]);
377 if (fSplitY.empty()) fSplitY = EqualSplit(fSegments.second, low[axisY], high[axisY]);
378 const char* names = "xyz";
379 RESTInfo << "TRestDataSetGainMap '" << fName << "': readout extent " << names[axisX] << " ["
380 << low[axisX] << ", " << high[axisX] << "], " << names[axisY] << " [" << low[axisY] << ", "
381 << high[axisY] << "]" << RESTendl;
382 }
383
384 // Automatic range: up to where the last 1% of the spectrum starts (as legacy)
385 if (!(fRange.first < fRange.second)) {
386 auto automatic = frame.Histo1D({"automatic", "", fNBins, 0, 0}, fObservable);
387 const TH1D& histogram = *automatic;
388 const double integral = histogram.Integral();
389 const double xMax = histogram.GetXaxis()->GetXmax();
390 double fraction = 1, atEnd = 0;
391 const double step = 0.66;
392 while (integral > 0 && atEnd / integral < 0.01 && fraction > 0.001) {
393 fraction *= step;
394 atEnd = histogram.Integral(histogram.FindFixBin(xMax * fraction), histogram.FindFixBin(xMax));
395 }
396 fRange.second = xMax * fraction / step;
397 RESTDebug << "TRestDataSetGainMap '" << fName << "': range (" << fRange.first << ", " << fRange.second
398 << ")" << RESTendl;
399 }
400
401 // All the spectra booked before the event loop
402 auto full = frame.Histo1D({"spectrum", "", fNBins, fRange.first, fRange.second}, fObservable);
403 const int nX = GetNumberOfSegmentsX(), nY = GetNumberOfSegmentsY();
404 std::vector<std::vector<ROOT::RDF::RResultPtr<TH1D>>> segments(nX);
405 auto inside = [](const std::string& column, double lower, double upper) {
406 return TRestTools::Format("%s >= %.17g && %s < %.17g", column.c_str(), lower, column.c_str(), upper);
407 };
408 for (int i = 0; i < nX; i++) {
409 for (int j = 0; j < nY; j++) {
410 std::string cut = inside(fSpatialObservableX, fSplitX[i], fSplitX[i + 1]) + " && " +
411 inside(fSpatialObservableY, fSplitY[j], fSplitY[j + 1]);
412 if (!fSpatialObservableXSecondary.empty())
413 cut += " && " + inside(fSpatialObservableXSecondary, fSplitX[i], fSplitX[i + 1]);
414 if (!fSpatialObservableYSecondary.empty())
415 cut += " && " + inside(fSpatialObservableYSecondary, fSplitY[j], fSplitY[j + 1]);
416 const std::string name = SegmentName("spectrum", i, j);
417 segments[i].push_back(frame.Filter(cut).Histo1D(
418 {name.c_str(), "", fNBins, fRange.first, fRange.second}, fObservable));
419 }
420 }
421
422 auto keep = [this](const TH1D& histogram) {
423 std::unique_ptr<TH1D> copy(static_cast<TH1D*>(histogram.Clone()));
424 copy->SetDirectory(nullptr);
425 copy->SetTitle((";" + fObservable + ";Counts").c_str());
426 return copy;
427 };
428 auto newGraph = [this](const std::string& name) {
429 auto graph = std::make_unique<TGraph>();
430 graph->SetName(name.c_str());
431 graph->SetTitle((";" + fObservable + ";Energy (keV)").c_str());
432 return graph;
433 };
434
435 fFullSpectrum = keep(*full);
436 fFullLinearFit = newGraph("linearFit");
437 std::tie(fFullIntercept, fFullSlope) = FitPeaks(*fFullSpectrum, *fFullLinearFit);
438
439 fSegmentSpectra.clear();
440 fSegmentSpectra.resize(nX);
441 fSegmentLinearFits.clear();
442 fSegmentLinearFits.resize(nX);
443 fSlope.assign(nX, std::vector<double>(nY, 0));
444 fIntercept.assign(nX, std::vector<double>(nY, 0));
445 for (int i = 0; i < nX; i++) {
446 for (int j = 0; j < nY; j++) {
447 fSegmentSpectra[i].push_back(keep(*segments[i][j]));
448 fSegmentLinearFits[i].push_back(newGraph(SegmentName("linearFit", i, j)));
449 std::tie(fIntercept[i][j], fSlope[i][j]) =
450 FitPeaks(*fSegmentSpectra[i][j], *fSegmentLinearFits[i][j]);
451 }
452 }
453 UpdateYAMLFromParams<TRestDataSetGainMap>(fNode);
454}

456void TRestDataSetGainMap::Apply(TRestDataSet& dataSet) const {
457 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
458
459 // The calibration copied: the data frame can be used after this process
460 const std::vector<double> splitX = fSplitX, splitY = fSplitY;
461 const std::vector<std::vector<double>> slope = fSlope, intercept = fIntercept;
462 const std::string observable = AsDouble(frame, fObservable);
463 const std::string x = AsDouble(frame, fSpatialObservableX);
464 const std::string y = AsDouble(frame, fSpatialObservableY);
465 auto calibrate = [splitX, splitY, slope, intercept](double value, double posX, double posY) {
466 // Without position (NaN) the event cannot be calibrated with a segment
467 if (!std::isfinite(posX) || !std::isfinite(posY)) return std::numeric_limits<double>::quiet_NaN();
468 const size_t i = SegmentIndex(splitX, posX), j = SegmentIndex(splitY, posY);
469 if (i >= slope.size() || j >= slope[i].size()) return std::numeric_limits<double>::quiet_NaN();
470 return slope[i][j] * value + intercept[i][j];
471 };
472 frame = DefineColumn(frame, GetEnergyColumn(), calibrate, {observable, x, y});
473 frame = DefineColumn(
474 frame, GetEnergyNoSegmentationColumn(),
475 TRestTools::Format("%.17g * (%s) + %.17g", fFullSlope, fObservable.c_str(), fFullIntercept));
476
477 dataSet.SetDataFrame(frame, {GetEnergyColumn(), GetEnergyNoSegmentationColumn()});
478}

613void TRestDataSetGainMap::DrawGainMap(size_t peak, bool fullReadoutAsReference, bool showText) const {
614 if (peak >= fPeaks.size()) {
615 RESTError << "TRestDataSetGainMap '" << fName << "': no peak " << peak << RESTendl;
616 return;
617 }
618 const int nX = static_cast<int>(fSegmentSpectra.size());
619 if (nX == 0 || !fFullSpectrum) {
620 RESTError << "TRestDataSetGainMap '" << fName << "': no spectra (fitted or read?)" << RESTendl;
621 return;
622 }
623 const int nY = static_cast<int>(fSegmentSpectra[0].size());
624 // Position of the peak: mean of its fit
625 auto position = [&](const TH1D* spectrum) {
626 return spectrum ? TRestPeakFit::GetPeakMean(*spectrum, FitName(peak))
627 : std::numeric_limits<double>::quiet_NaN();
628 };
629 const double reference = fullReadoutAsReference
630 ? position(fFullSpectrum.get())
631 : position(GetSegmentSpectrum((nX - 1) / 2, (nY - 1) / 2));
632
633 const std::string name = fName + "_gainMap" + std::to_string(peak);
634 auto* canvas = new TCanvas(name.c_str(), name.c_str());
635 canvas->cd();
636 const std::string title =
637 TRestTools::Format("Gain map for energy %g keV;%s;%s", static_cast<double>(fPeaks[peak].fEnergy),
638 fSpatialObservableX.c_str(), fSpatialObservableY.c_str());
639 auto* map = new TH2D(("h" + name).c_str(), title.c_str(), nX, fSplitX.data(), nY, fSplitY.data());
640 for (int i = 0; i < nX; i++) {
641 for (int j = 0; j < nY; j++) {
642 map->SetBinContent(i + 1, j + 1, position(GetSegmentSpectrum(i, j)) / reference);
643 }
644 }
645 map->SetStats(false);
646 map->Draw("colz");
647 map->SetBarOffset(0.2);
648 if (showText) map->Draw("TEXT SAME");
649}

662void TRestDataSetGainMap::ReadObjects(TDirectory* directory) {
663 if (!directory) return;
664 auto readHistogram = [directory](const std::string& key) -> std::unique_ptr<TH1D> {
665 auto* object = directory->Get<TH1D>(key.c_str());
666 if (!object) return nullptr;
667 std::unique_ptr<TH1D> copy(static_cast<TH1D*>(object->Clone()));
668 copy->SetDirectory(nullptr);
669 return copy;
670 };
671 auto readGraph = [directory](const std::string& key) -> std::unique_ptr<TGraph> {
672 auto* object = directory->Get<TGraph>(key.c_str());
673 return object ? std::unique_ptr<TGraph>(static_cast<TGraph*>(object->Clone())) : nullptr;
674 };
675 fFullSpectrum = readHistogram("spectrum");
676 fFullLinearFit = readGraph("linearFit");
677 const int nX = GetNumberOfSegmentsX(), nY = GetNumberOfSegmentsY();
678 fSegmentSpectra.clear();
679 fSegmentSpectra.resize(std::max(nX, 0));
680 fSegmentLinearFits.clear();
681 fSegmentLinearFits.resize(std::max(nX, 0));
682 for (int i = 0; i < nX; i++) {
683 for (int j = 0; j < nY; j++) {
684 fSegmentSpectra[i].push_back(readHistogram(SegmentName("spectrum", i, j)));
685 fSegmentLinearFits[i].push_back(readGraph(SegmentName("linearFit", i, j)));
686 }
687 }
