68void TRestEventViewer::Plot(const char* option) {
69 ClearCanvasSelection();
70 // The pad is the one of the event (DrawEvent), owned by it: the viewer does not create nor delete it
71 if (fEvent != nullptr) {
72 fPad = fEvent->DrawEvent(option);
73 fCanvas->cd();
74 if (fPad != nullptr) {
75 fPad->Draw();
76 fPad->Update();
77 }
78 fCanvas->Update();
79 } else {
80 // Without event the pad of the previous one may be deleted: only the canvas is cleared
81 fPad = nullptr;
82 fCanvas->cd();
83 fCanvas->Clear();
84 fCanvas->Update();
85 }
86}
