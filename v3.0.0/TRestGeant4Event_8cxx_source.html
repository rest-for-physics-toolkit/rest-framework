385void TRestGeant4Event::RebuildSecondaries() {
386 auto& d = fEventData;
387 const std::size_t n = d.trackIDs.size();
388
389 std::map<int, std::vector<int>> children;
390 for (std::size_t i = 0; i < n && i < d.parentIDs.size(); ++i) {
391 children[d.parentIDs[i]].push_back(d.trackIDs[i]);
392 }
393
394 d.trackSecondariesIDs.clear();
395 d.trackSecondariesIndices.assign(n, 0);
396 d.trackSecondariesOffsets.assign(n, 0);
397 for (std::size_t i = 0; i < n; ++i) {
398 d.trackSecondariesIndices[i] = static_cast<int>(d.trackSecondariesIDs.size());
399 const auto it = children.find(d.trackIDs[i]);
400 if (it == children.end()) continue;
401 d.trackSecondariesIDs.insert(d.trackSecondariesIDs.end(), it->second.begin(), it->second.end());
402 d.trackSecondariesOffsets[i] = static_cast<int>(it->second.size());
403 }
404}

406void TRestGeant4Event::RemoveTrackHits(std::size_t trackIndex) {
407 if (trackIndex >= fEventData.trackIDs.size()) return;
408 if (trackIndex >= fEventData.trackStartIndices.size() || trackIndex >= fEventData.trackNHits.size())
409 return;
410
411 const std::size_t start = static_cast<std::size_t>(fEventData.trackStartIndices[trackIndex]);
412 const std::size_t count = static_cast<std::size_t>(fEventData.trackNHits[trackIndex]);
413 if (count == 0) return;
414
415 auto eraseRange = [start, count](auto& values) {
416 if (start >= values.size()) return;
417 const std::size_t end = std::min(start + count, values.size());
418 values.erase(values.begin() + static_cast<std::ptrdiff_t>(start),
419 values.begin() + static_cast<std::ptrdiff_t>(end));
420 };
421
422 eraseRange(fEventData.hitsStorage.x);
423 eraseRange(fEventData.hitsStorage.y);
424 eraseRange(fEventData.hitsStorage.z);
425 eraseRange(fEventData.hitsStorage.time);
426 eraseRange(fEventData.hitsStorage.energy);
427 eraseRange(fEventData.hitsStorage.type);
428 eraseRange(fEventData.hitProcessID);
429 eraseRange(fEventData.hitVolumeID);
430 eraseRange(fEventData.hitKineticEnergy);
431 eraseRange(fEventData.hitMomentumDirectionX);
432 eraseRange(fEventData.hitMomentumDirectionY);
433 eraseRange(fEventData.hitMomentumDirectionZ);
434 eraseRange(fEventData.hitHadronicTargetIsotopeA);
435 eraseRange(fEventData.hitHadronicTargetIsotopeZ);
436
437 for (std::size_t i = trackIndex + 1; i < fEventData.trackStartIndices.size(); ++i) {
438 fEventData.trackStartIndices[i] -= static_cast<int>(count);
439 }
440
441 fEventData.trackNHits[trackIndex] = 0;
442}

444void TRestGeant4Event::AddEnergyInVolumeForParticleForProcess(double energy, int volumeID, int particleID,
445 int processID) {
446 if (energy <= 0) return;
447
448 fEventData.totalDepositedEnergy += energy;
449
450 const auto key = std::make_tuple(volumeID, particleID, processID);
451 const auto itCross = fCrossIndexMap.find(key);
452 if (itCross != fCrossIndexMap.end()) {
453 fEventData.crossDepositedEnergies[itCross->second] += energy;
454 } else {
455 fCrossIndexMap[key] = fEventData.crossVolumeIDs.size();
456 fEventData.crossVolumeIDs.push_back(volumeID);
457 fEventData.crossParticleIDs.push_back(particleID);
458 fEventData.crossProcessIDs.push_back(processID);
459 fEventData.crossDepositedEnergies.push_back(energy);
460 }
461
462 const auto itVolume = fVolumeIndexMap.find(volumeID);
463 if (itVolume != fVolumeIndexMap.end()) {
464 fEventData.volumeDepositedEnergy[itVolume->second] += energy;
465 } else {
466 fVolumeIndexMap[volumeID] = fEventData.volumeIDs.size();
467 fEventData.volumeIDs.push_back(volumeID);
468 fEventData.volumeDepositedEnergy.push_back(energy);
469 }
470}

551TPad* TRestGeant4Event::DrawEvent(const std::string& option) const {
552 std::vector<std::string> options;
553 for (const auto& opt : TRestTools::Split(option, ':')) {
554 if (opt == "print") {
555 PrintEvent();
556 } else if (!opt.empty()) {
557 options.push_back(opt);
558 }
559 }
560 if (options.empty()) options = {"graph", "hist(Cont0,colz)"};
561
562 if (fEventData.hitsStorage.x.empty()) {
563 std::cout << "Empty event " << std::endl;
564 return nullptr;
565 }
566
567 TVirtualPad* safePad = gPad;
568
569 delete fPad;
570 fPad = new TPad(Form("pad_geant4_event_%d", GetID()), "REST Geant4 Event Viewer", 0, 0, 1, 1);
571 fPad->Divide(1, options.size());
572 fPad->Draw();
573
574 for (size_t i = 0; i < options.size(); i++) {
575 // name(rootOption)[list]: the brackets are a list of processes (graph) or the bin size (hist), so
576 // they are not parsed as the DrawOption value
577 const auto open = options[i].find('['), close = options[i].find(']');
578 std::string brackets;
579 if (open != std::string::npos && close != std::string::npos && close > open) {
580 brackets = options[i].substr(open + 1, close - open - 1);
581 }
582 const auto drawOption = ParseDrawOption(options[i].substr(0, open));
583
584 TVirtualPad* optionPad = fPad->cd(i + 1);
585 if (drawOption.name == "graph") {
586 std::vector<std::string> processes;
587 for (const auto& process : TRestTools::Split(brackets, ',')) {
588 if (!process.empty()) processes.push_back(process);
589 }
590 DrawGraphs(optionPad, processes);
591 } else if (drawOption.name == "hist") {
592 // As legacy, also written binSize=<value>
593 const auto equal = brackets.find('=');
594 const auto value = brackets.substr(equal == std::string::npos ? 0 : equal + 1);
595 const double binSize = value.empty() ? kDefaultBinSize : std::stod(value);
596
597 // The hits with energy, the others (transportation) only change the ranges
598 TRestHitsData hits;
599 const auto& storage = fEventData.hitsStorage;
600 for (size_t n = 0; n < storage.energy.size(); n++) {
601 if (storage.energy[n] <= 0) continue;
602 hits.x.push_back(storage.x[n]);
603 hits.y.push_back(storage.y[n]);
604 hits.z.push_back(storage.z[n]);
605 hits.time.push_back(storage.time[n]);
606 hits.energy.push_back(storage.energy[n]);
607 hits.type.push_back(storage.type[n]);
608 }
609 if (hits.x.empty()) {
610 std::cout << "TRestGeant4Event::DrawEvent: no hit with energy" << std::endl;
611 continue;
612 }
613 TRestHits(&hits, 0, static_cast<int>(hits.x.size()))
614 .DrawHistograms(optionPad, drawOption.rootOption, binSize);
615 } else {
616 std::cout << "TRestGeant4Event::DrawEvent: unknown option '" << options[i] << "'" << std::endl;
617 }
618 }
619
620 fPad->Modified();
621 if (safePad) safePad->cd();
622 return fPad;
623}

625void TRestGeant4Event::DrawGraphs(TVirtualPad* pad, const std::vector<std::string>& processes) const {
626 const auto& data = fEventData;
627 const auto& hits = data.hitsStorage;
628
629 // Process IDs of the marked processes, from the physics information of the simulation
630 std::vector<int> processIDs;
631 if (!processes.empty()) {
632 const auto* metadata = GetGeant4Metadata();
633 for (const auto& process : processes) {
634 const int id = metadata ? metadata->GetGeant4PhysicsInfo().GetProcessID(process) : -1;
635 if (id < 0) {
636 std::cout << "TRestGeant4Event::DrawEvent: process '" << process
637 << "' not marked, unknown or no TRestGeant4Metadata" << std::endl;
638 }
639 processIDs.push_back(id);
640 }
641 }
642
643 // A colour per track, by particle (once per track, GetParticleColor warns for unknown particles)
644 std::vector<Color_t> colors;
645 std::vector<std::pair<std::string, Color_t>> particles;
646 for (size_t t = 0; t < GetNumberOfTracks(); t++) {
647 const auto track = GetTrack(t);
648 colors.push_back(track.GetParticleColor());
649 const auto particle = track.GetParticleName();
650 const auto isParticle = [&particle](const auto& p) { return p.first == particle; };
651 if (std::none_of(particles.begin(), particles.end(), isParticle)) {
652 particles.emplace_back(particle, colors.back());
653 }
654 }
655
656 double min[3], max[3];
657 double maxEnergy = 0;
658 for (int a = 0; a < 3; a++) {
659 min[a] = std::numeric_limits<double>::max();
660 max[a] = std::numeric_limits<double>::lowest();
661 }
662 for (size_t n = 0; n < hits.x.size(); n++) {
663 const double position[3] = {hits.x[n], hits.y[n], hits.z[n]};
664 for (int a = 0; a < 3; a++) {
665 min[a] = std::min(min[a], position[a]);
666 max[a] = std::max(max[a], position[a]);
667 }
668 maxEnergy = std::max(maxEnergy, static_cast<double>(hits.energy[n]));
669 }
670 // A margin around the hits, at least 1 mm (a single hit)
671 for (int a = 0; a < 3; a++) {
672 const double margin = std::max(0.05 * (max[a] - min[a]), 1.);
673 min[a] -= margin;
674 max[a] += margin;
675 }
676 auto markerSize = [maxEnergy](double energy) {
677 if (maxEnergy <= 0) return kMinMarkerSize;
678 return kMinMarkerSize + (kMaxMarkerSize - kMinMarkerSize) * std::max(energy, 0.) / maxEnergy;
679 };
680
681 pad->Divide(3, 1);
682 const std::string title = Form("Event ID %d", GetID());
683 for (int p = 0; p < 3; p++) {
684 const int u = TRestHitsData::kPlanes[p][0], v = TRestHitsData::kPlanes[p][1];
685 pad->cd(p + 1);
686 SetPadMargins(gPad);
687
688 auto* frame = new TH1F(Form("frame_%d", p), title.c_str(), 1, min[u], max[u]);
689 frame->SetMinimum(min[v]);
690 frame->SetMaximum(max[v]);
691 DrawOwned(frame, "AXIS", TRestHitsData::kAxisTitles[u], TRestHitsData::kAxisTitles[v]);
692
693 auto* legend = new TLegend(0.75, 0.7, 0.95, 0.9);
694 legend->SetBit(TObject::kCanDelete);
695 for (const auto& [particle, color] : particles) {
696 auto* entry = legend->AddEntry(static_cast<TObject*>(nullptr), particle.c_str(), "p");
697 entry->SetMarkerColor(color);
698 entry->SetMarkerStyle(20);
699 entry->SetMarkerSize(1.5);
700 }
701
702 std::vector<bool> processInLegend(processIDs.size(), false);
703 for (size_t t = 0; t < GetNumberOfTracks(); t++) {
704 const size_t start = data.trackStartIndices[t];
705 for (size_t n = start; n < start + data.trackNHits[t]; n++) {
706 const double position[3] = {hits.x[n], hits.y[n], hits.z[n]};
707 auto* marker = new TMarker(position[u], position[v], 20);
708 marker->SetMarkerColor(colors[t]);
709 marker->SetMarkerSize(markerSize(hits.energy[n]));
710 marker->SetBit(TObject::kCanDelete);
711 marker->Draw();
712
713 for (size_t k = 0; k < processIDs.size(); k++) {
714 if (processIDs[k] < 0 || data.hitProcessID[n] != processIDs[k]) continue;
715 auto* processMarker = new TMarker(position[u], position[v], 24 + k % 9);
716 processMarker->SetMarkerColor(kBlack);
717 processMarker->SetMarkerSize(2);
718 processMarker->SetBit(TObject::kCanDelete);
719 processMarker->Draw();
720 if (!processInLegend[k]) {
721 auto* entry =
722 legend->AddEntry(static_cast<TObject*>(nullptr), processes[k].c_str(), "p");
723 entry->SetMarkerColor(kBlack);
724 entry->SetMarkerStyle(processMarker->GetMarkerStyle());
725 entry->SetMarkerSize(1.5);
726 processInLegend[k] = true;
727 }
728 }
729 }
730 }
731 legend->Draw();
732 }
