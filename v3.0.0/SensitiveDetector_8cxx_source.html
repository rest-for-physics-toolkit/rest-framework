18G4bool SensitiveDetector::ProcessHits(G4Step* step, G4TouchableHistory*) {
19 // return value will always be ignored, its present for backwards compatibility (I guess)
20 TRestGeant4Metadata* restG4Metadata = fSimulationManager->GetRestMetadata();
21
22 /* Sensitive detector are assigned to logical volumes, but several physical volumes
23 can share the same logical volume. Now we get the physical volume where the pre-step
24 is located and filter if it is one of the declared sensitive volumes by the user*/
25
26 // The Geant4 physical volume name identifies it (the names are unique, DetectorConstruction checks it),
27 // as the sensitive volumes (the physical volumes matching them): also for nested volumes and assemblies
28 const std::string physicalName = step->GetPreStepPoint()->GetPhysicalVolume()->GetName();
29
30 // search and filter if this physical volume is sensitive
31 const auto& sensitiveVolumes = restG4Metadata->GetSensitiveVolumes();
32 if (std::find(sensitiveVolumes.begin(), sensitiveVolumes.end(), physicalName) == sensitiveVolumes.end()) {
33 return false;
34 }
35
36 const bool isGeantino = step->GetTrack()->GetParticleDefinition() == G4Geantino::Definition();
37
38 if (isGeantino) {
39 // Since geantinos don't deposit energy, the length traveled inside the volumes is stored as energy
40 // (mm as keV)
41 const auto length = step->GetStepLength() / CLHEP::mm;
42 fSimulationManager->GetOutputManager()->AddSensitiveEnergy(length);
43 return true;
44 } else {
45 auto energy = step->GetTotalEnergyDeposit() / keV;
46
47 if (energy <= 0) {
48 return true;
49 }
50 fSimulationManager->GetOutputManager()->AddSensitiveEnergy(energy);
51 return true;
52 }
53}
