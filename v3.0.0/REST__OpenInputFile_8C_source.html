62int REST_OpenRun(const std::string& filename) {
63 _ev = nullptr;
64 delete _run;
65 _run = nullptr;
66
67 TRestRun* run = nullptr;
68 try {
69 run = new TRestRun(filename);
70
71 if (!run->fInputFile || !run->fInputFile->IsOpen()) {
72 RESTError << "REST_OpenRun: could not open file " << filename << RESTendl;
73 delete run;
74 return -1;
75 }
76
77 run->GetEntry(0);
78 } catch (const std::exception& error) {
79 RESTError << "REST_OpenRun: " << error.what() << RESTendl;
80 delete run;
81 return -1;
82 }
83
84 TRestEvent* event = run->GetEvent();
85 if (!event) {
86 RESTError << "REST_OpenRun: failed to retrieve a valid TRestEvent from the file." << RESTendl;
87 delete run;
88 return -1;
89 }
90
91 _run = run;
92 _ev = event;
93
94 std::cout << "Successfully loaded TRestRun. Event type detected: " << _ev->GetClassName() << std::endl;
95 std::cout << "--> Global pointers '_run' and '_ev' are now fully accessible in your prompt!\n"
96 << std::endl;
97
98 return 0;
99}
