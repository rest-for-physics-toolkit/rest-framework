52unique_ptr<TRestGeant4SpatialGenerator> TRestGeant4SpatialGenerator::Create(const string& name,
53 const YAML::Node& node) {
54 if (!node["type"]) throw runtime_error("TRestGeant4SpatialGenerator: the generator 'type' is missing");
55 const auto type = ToLower(node["type"].as<string>());
56 if (type == "point") return make_unique<TRestGeant4SpatialGeneratorPoint>(name, node);
57 if (type == "volume") return make_unique<TRestGeant4SpatialGeneratorVolume>(name, node);
58 if (type == "surface") return make_unique<TRestGeant4SpatialGeneratorSurface>(name, node);
59 if (type == "cosmic") return make_unique<TRestGeant4SpatialGeneratorCosmic>(name, node);
60 if (type == "cry") return make_unique<TRestGeant4SpatialGeneratorCRY>(name, node);
61 throw runtime_error("TRestGeant4SpatialGenerator: unknown generator type '" + type +
62 "', it must be point, volume, surface, cosmic or cry");
63}

92void TRestGeant4SpatialGeneratorShape::CheckShape(const map<string, size_t>& shapes) {
93 // As legacy, a generator from a geometry volume has no shape
94 if (!fFrom.empty()) {
95 fShape = "";
96 return;
97 }
98 fShape = ToLower(fShape);
99 const auto shape = shapes.find(fShape);
100 if (shape == shapes.end()) {
101 string names;
102 for (const auto& [name, nSizes] : shapes) names += (names.empty() ? "" : ", ") + name;
103 throw runtime_error(GetClassName() + ": 'from' (a geometry volume) or 'shape' (" + names +
104 ") is needed, '" + fShape + "' is not valid");
105 }
106 for (size_t i = 0; i < shape->second; i++) {
107 if (fSize[i] <= 0) {
108 throw runtime_error(GetClassName() + ": the " + fShape + " needs " + to_string(shape->second) +
109 " positive 'size' values");
110 }
111 }
112 if (fRotationValue != 0 && ToVector(fRotationAxis).R() == 0) {
113 throw runtime_error(GetClassName() + ": 'rotationAxis' must not be zero");
114 }

137XYZVector TRestGeant4SpatialGeneratorVolume::Sample(TRandom& random) {
138 if (IsGdml()) throw runtime_error(GetClassName() + ": a GDML volume needs the Geant4 geometry");
139 const double phi = random.Uniform(0, 2 * M_PI);
140 XYZVector local;
141 if (fShape == "box") {
142 local = {fSize[0] * (random.Uniform() - 0.5), fSize[1] * (random.Uniform() - 0.5),
143 fSize[2] * (random.Uniform() - 0.5)};
144 } else if (fShape == "sphere") {
145 local = fSize[0] * cbrt(random.Uniform()) * IsotropicDirection(random);
146 } else {
147 const double r = fSize[0] * sqrt(random.Uniform());
148 local = {r * cos(phi), r * sin(phi), fSize[1] * (random.Uniform() - 0.5)};
149 }
150 return ToWorld(local);
151}

167XYZVector TRestGeant4SpatialGeneratorSurface::Sample(TRandom& random) {
168 if (IsGdml()) throw runtime_error(GetClassName() + ": a GDML surface needs the Geant4 geometry");
169 const double phi = random.Uniform(0, 2 * M_PI);
170 XYZVector local;
171 if (fShape == "sphere") {
172 local = fSize[0] * IsotropicDirection(random);
173 } else if (fShape == "cylinder") {
174 local = {fSize[0] * cos(phi), fSize[0] * sin(phi), fSize[1] * (random.Uniform() - 0.5)};
175 } else if (fShape == "circle") {
176 const double r = fSize[0] * sqrt(random.Uniform());
177 local = {r * cos(phi), r * sin(phi), 0};
178 } else {
179 local = {fSize[0] * (random.Uniform() - 0.5), fSize[1] * (random.Uniform() - 0.5), 0};
180 }
181 return ToWorld(local);

202XYZVector TRestGeant4SpatialGeneratorCosmic::SampleFromDirection(TRandom& random,
203 const XYZVector& direction) const {
204 if (fRadius <= 0) throw runtime_error(GetClassName() + ": the radius is not set");
205 // Two unit vectors normal to the direction
206 const XYZVector reference = abs(direction.Z()) < 0.9 ? XYZVector(0, 0, 1) : XYZVector(1, 0, 0);
207 const XYZVector v = direction.Cross(reference).Unit();
208 const XYZVector w = direction.Cross(v);
209
210 const double r = fRadius * sqrt(random.Uniform());
211 const double phi = random.Uniform(0, 2 * M_PI);
212 const XYZVector onDisk = r * (cos(phi) * v + sin(phi) * w);
213 return onDisk - sqrt(fRadius * fRadius - r * r) * direction;
214}

279bool TRestGeant4SpatialGeneratorCRY::NextShower(vector<TRestGeant4ParticleState>& particles) {
280 if (!fState) throw runtime_error(GetClassName() + ": CRY is not started");
281 particles.clear();
282 {
283 lock_guard<mutex> lock(fState->mutex);
284 if (fMaxExposureTime > 0 && fState->engine->GetTimeSimulated() >= fMaxExposureTime) return false;
285 particles = fState->engine->Generate();
286 fState->showers++;
287 }
288 const XYZVector center = GetCenter();
289 for (auto& particle : particles) {
290 particle.SetOrigin(RotateToWorld(particle.GetOrigin()) + center);
291 particle.SetDirection(RotateToWorld(particle.GetDirection()));
292 }
293 return true;
294}

296XYZVector TRestGeant4SpatialGeneratorCRY::RotateToWorld(const XYZVector& v) const {
297 const XYZVector up = ToVector(fVerticalAxis).Unit();
298 const XYZVector z(0, 0, 1);
299 const XYZVector axis = z.Cross(up);
300 const double angle = acos(clamp(z.Dot(up), -1.0, 1.0));
301 const ROOT::Math::AxisAngle rotation(axis.R() > 0 ? axis.Unit() : XYZVector(1, 0, 0), angle);
302 return rotation * v;
303}

305vector<XYZVector> TRestGeant4SpatialGeneratorCRY::GetPlaneCorners() const {
306 const double half = fSubBoxLength / 2;
307 vector<XYZVector> corners;
308 for (const double x : {-half, half}) {
309 for (const double y : {-half, half}) corners.push_back(RotateToWorld({x, y, 0}) + GetCenter());
310 }
