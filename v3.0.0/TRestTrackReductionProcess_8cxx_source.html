150std::vector<Node> TRestTrackReductionProcess::Reduce(const TRestHits& hits,
151 const std::vector<XYZVector>& sigmas) const {
152 if (hits.GetNumberOfHits() == 0) return {};
153 for (size_t h = 1; h < hits.GetNumberOfHits(); h++) {
154 if (hits.GetType(h) != hits.GetType(0)) return {};
155 }
156
157 const auto points = hits.GetPoints();
158 const auto energies = hits.GetEnergies();
159 auto merge = [this](const std::vector<XYZVector>& p, const std::vector<double>& e,
160 const std::vector<XYZVector>& s) {
161 return TRestHitsAnalysis::MergePoints(p, e, fStartingDistance, fMinimumDistance, fDistanceStepFactor,
162 fMaxNodes, s);
163 };
164
165 std::vector<Node> nodes = merge(points, energies, sigmas);
166 if (!fKMeans) return nodes;
167
168 // As legacy: k-means of the hits around the nodes, merging the nodes again until their number is stable
169 size_t nodesBefore = 0;
170 do {
171 nodesBefore = nodes.size();
172 auto moved =
173 TRestHitsAnalysis::KMeansClustering(points, energies, NodePositions(nodes), fMaxIterations);
174 if (moved.empty()) break; // less than 2 nodes
175 nodes = merge(NodePositions(moved), NodeEnergies(moved), NodeSigmas(moved));
176 } while (nodes.size() != nodesBefore);
177 return nodes;
178}

180bool TRestTrackReductionProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
181 const auto* inputEvent = dynamic_cast<const TRestTrackEvent*>(&input);
182 auto* outputEvent = dynamic_cast<TRestTrackEvent*>(&output);
183 if (!inputEvent || !outputEvent) {
184 throw std::runtime_error("TRestTrackReductionProcess: input/output is not a TRestTrackEvent");
185 }
186
187 outputEvent->Initialize();
188 outputEvent->SetEventInfo(inputEvent);
189
190 const double nan = std::numeric_limits<double>::quiet_NaN();
191 for (int t = 0; t < inputEvent->GetNumberOfTracks(); t++) {
192 const TRestHits hits = inputEvent->GetTrack(t).GetHits();
193 const int trackID = inputEvent->GetTrackID(t);
194 std::vector<XYZVector> sigmas;
195 for (size_t h = 0; h < hits.GetNumberOfHits(); h++) sigmas.push_back(inputEvent->GetSigma(t, (int)h));
196 const auto nodes = Reduce(hits, sigmas);
197 fNTracks++;
198
199 if (nodes.empty()) {
200 fNotReduced++;
201 RESTDebug << "TRestTrackReductionProcess: event " << inputEvent->GetID() << ", track " << trackID
202 << " not reduced (hits of different types), kept as it is" << RESTendl;
203 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, trackID, hits, sigmas);
204 continue;
205 }
206
207 const auto type = hits.GetType(0);
208 outputEvent->AddTrack(outputEvent->GetNumberOfTracks() + 1, trackID);
209 for (const auto& node : nodes) {
210 // Undefined coordinate back to NaN, as in the hits
211 const XYZVector position(TRestHitsData::IsDefined(type, 0) ? node.position.X() : nan,
212 TRestHitsData::IsDefined(type, 1) ? node.position.Y() : nan,
213 TRestHitsData::IsDefined(type, 2) ? node.position.Z() : nan);
214 outputEvent->AddHitToTrack(position, node.energy, 0, type, node.sigma);
215 }
216 RESTDebug << "TRestTrackReductionProcess: track " << trackID << " with " << hits.GetNumberOfHits()
217 << " hits reduced to " << nodes.size() << " nodes" << RESTendl;
218 }
219
220 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) outputEvent->PrintOnlyTracks();
221 return true;
222}
