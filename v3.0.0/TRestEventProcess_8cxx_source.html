39bool TRestEventProcess::ShouldRegisterObservable(const std::string& name) const {
40 // Two variables with the same name would share the branch, the last one replacing the other
41 if (std::find(fOfferedObservables.begin(), fOfferedObservables.end(), name) !=
42 fOfferedObservables.end()) {
43 throw std::runtime_error(GetName() + ": the observable " + name +
44 " is given twice (e.g. a volume and a particle with the same name)");
45 }
46 fOfferedObservables.push_back(name);
47 if (!fRunInfo) return false;
48 for (const auto& obs : fObservables) {
49 if (obs == "none") return false;
50 if (obs == "all" || obs == name) return true;
51 }
52 return false;
53}

55void TRestEventProcess::CheckObservables() const {
56 std::string unknown;
57 for (const auto& obs : fObservables) {
58 if (obs == "all" || obs == "none") continue;
59 if (std::find(fOfferedObservables.begin(), fOfferedObservables.end(), obs) !=
60 fOfferedObservables.end()) {
61 continue;
62 }
63 unknown += (unknown.empty() ? "" : ", ") + obs;
64 }
65 if (unknown.empty()) return;
66
67 std::string offered;
68 for (const auto& obs : fOfferedObservables) offered += (offered.empty() ? "" : ", ") + obs;
69 throw std::runtime_error(GetName() + ": unknown observables " + unknown +
70 " (not given by the process with this configuration; available: " +
71 (offered.empty() ? "none" : offered) + ")");
72}

74void TRestEventProcess::InitCuts() {
75 fCuts.clear();
76 if (fCutNames.empty()) return;
77 if (!fManager) throw std::runtime_error(GetName() + ": cuts need a TRestManager with the TRestCut");
78 if (!fRunInfo || !fRunInfo->fOutputAnalysisTree) {
79 throw std::runtime_error(GetName() + ": cuts need the output analysis tree");
80 }
81 const auto cuts = fManager->GetMetadataOfType<TRestCut>();
82 for (const auto& name : fCutNames) {
83 TRestCut* cut = nullptr;
84 for (auto* c : cuts) {
85 if (c->GetName() == name) cut = c;
86 }
87 if (!cut) throw std::runtime_error(GetName() + ": TRestCut '" + name + "' not found in the manager");
88 cut->Compile(fRunInfo->fOutputAnalysisTree);
89 fCuts.push_back(cut);
90 }
91}
