225int ParseColor(const std::string& value, int defaultValue) {
226 if (value.empty()) return defaultValue;
227 if (value[0] == '#') return TColor::GetColor(value.c_str());
228 if (TRestTools::isANumber(value)) return std::stoi(value);
229 const size_t sign = value.find_first_of("+-");
230 std::string name = TRestTools::ToLower(Trim(value.substr(0, sign)));
231 if (name.size() > 1 && name[0] == 'k') name = name.substr(1);
232 const auto it = kColors.find(name);
233 if (it == kColors.end()) {
234 RESTWarning << "TRestAnalysisPlot: unknown color " << value << RESTendl;
235 return defaultValue;
236 }
237 if (sign == std::string::npos) return it->second;
238 return it->second + std::stoi(value.substr(sign));

242class Evaluator {
243 public:
245 const std::map<std::string, YAML::Node>* metadata = nullptr;
246 YAML::Node run;
247 std::map<std::string, double> numbers;
248 std::map<std::string, std::string> texts;
250 std::map<std::string, double> statistics;
251 int precision = 3;
252
254 std::string Replace(const std::string& text) const {
255 return TRestTools::ReplaceDelimited(text, '{', '}', [this](const std::string& content) {
256 std::string expression = content;
257 std::string format;
258 // The last '|' that is not part of ||
259 const size_t bar = expression.rfind('|');
260 if (bar != std::string::npos && (bar == 0 || expression[bar - 1] != '|')) {
261 format = Trim(expression.substr(bar + 1));
262 expression = expression.substr(0, bar);
263 }
264 return std::optional<std::string>(Value(Trim(expression), format));
265 });
266 }
267
269 std::optional<double> Evaluate(const std::string& expression) const {
270 // Statistics first: their observable is not a reference
271 std::string withStatistics;
272 size_t last = 0;
273 for (auto it = std::sregex_iterator(expression.begin(), expression.end(), kStatistic);
274 it != std::sregex_iterator(); ++it) {
275 const auto value = statistics.find((*it)[1].str() + "(" + (*it)[2].str() + ")");
276 if (value == statistics.end()) return std::nullopt;
277 withStatistics += expression.substr(last, it->position() - last) + Number(value->second);
278 last = it->position() + it->length();
279 }
280 withStatistics += expression.substr(last);
281
282 // References with a value; the rest (functions, e.g. sqrt or TMath::Pi) are left to TFormula
283 bool valid = true;
284 const std::string formula =
285 TRestTools::ReplaceIdentifiers(withStatistics, kReferencePattern,
286 [&](const std::string& reference) -> std::optional<std::string> {
287 const auto number = numbers.find(reference);
288 if (number != numbers.end()) return Number(number->second);
289 const auto text = Lookup(reference);
290 // A metadata reference (instance.key) not found: not
291 // evaluated (TFormula would fail with errors of cling)
292 if (!text && reference.find('.') != std::string::npos)
293 valid = false;
294 if (!text) return std::nullopt;
295 const auto value = REST_Units::ParseValueWithUnits(*text);
296 if (!value) valid = false;
297 return value ? Number(*value) : *text;
298 });
299 if (!valid) return std::nullopt;
300
301 TFormula function("TRestAnalysisPlot", formula.c_str(), false);
302 if (!function.IsValid()) return std::nullopt;
303 return function.Eval(0);
304 }
305
306 private:
307 static std::string Number(double value) {
308 char buffer[64];
309 std::snprintf(buffer, sizeof(buffer), "(%.17g)", value);
310 return buffer;
311 }
312
314 std::optional<std::string> Lookup(const std::string& reference) const {
315 const auto text = texts.find(reference);
316 if (text != texts.end()) return text->second;
317 const size_t dot = reference.find('.');
318 if (dot == std::string::npos) return std::nullopt;
319
320 // The run of the file whatever its instance name (run.runInfo.key: its runInfo)
321 const std::string instance = reference.substr(0, dot);
322 YAML::Node node;
323 if (instance == "run") {
324 node.reset(run);
325 } else if (metadata && metadata->count(instance)) {
326 node.reset(metadata->at(instance));
327 }
328 if (!node) return std::nullopt;
329 node.reset(TRestYAMLUtils::GetNodeAtPath(node, reference.substr(dot + 1)));
330 if (!node || !node.IsScalar()) return std::nullopt;
331 return node.as<std::string>();
332 }

334 std::string Format(double value, const std::string& format) const {
335 char buffer[128];
336 if (format.find('%') != std::string::npos) {
337 std::snprintf(buffer, sizeof(buffer), format.c_str(), value);
338 } else if (format.empty() && value == std::round(value) && std::abs(value) < 1e15) {
339 // Integers (e.g. entries) without exponent
340 std::snprintf(buffer, sizeof(buffer), "%.0f", value);
341 } else {
342 int digits = precision;
343 if (!format.empty()) {
344 if (TRestTools::isANumber(format)) {
345 digits = std::stoi(format);
346 } else {
347 RESTWarning << "TRestAnalysisPlot: unknown format " << format << RESTendl;
348 }
349 }
350 std::snprintf(buffer, sizeof(buffer), "%.*g", digits, value);
351 }
352 return buffer;
353 }
354
355 std::string Value(const std::string& expression, const std::string& format) const {
356 // A single reference to a text is written as it is (formatted if it is a number and has format)
357 if (std::regex_match(expression, kReference) && !numbers.count(expression)) {
358 if (const auto text = Lookup(expression)) {
359 const auto number = format.empty() ? std::nullopt : REST_Units::ParseValueWithUnits(*text);
360 return number ? Format(*number, format) : *text;
361 }
362 }
363 if (const auto value = Evaluate(expression)) return Format(*value, format);
364 RESTWarning << "TRestAnalysisPlot: cannot evaluate {" << expression << "}" << RESTendl;
365 return "?";
366 }

609void TRestAnalysisPlot::DrawFrame(ROOT::RDF::RNode frame, const std::map<std::string, YAML::Node>& metadata,
610 const YAML::Node& run, double startTime, double endTime, double duration,
611 const TRestManager* manager) {
612 auto cutExpression = [&](const std::vector<std::string>& names) {
613 std::string expression;
614 for (const auto& name : names) {
615 const TRestCut* cut = nullptr;
616 if (manager) {
617 for (const auto* c : manager->GetMetadataOfType<TRestCut>()) {
618 if (c->GetName() == name) cut = c;
619 }
620 }
621 if (!cut) {
622 throw std::runtime_error("TRestAnalysisPlot '" + fName + "': TRestCut '" + name +
623 "' not found in the manager");
624 }
625 const std::string condition = cut->GetExpression();
626 if (condition.empty()) continue;
627 if (!expression.empty()) expression += " && ";
628 expression += "(" + condition + ")";
629 }
630 return expression;
631 };
632
633 // All the results are booked before the loop, run once when the first one is read
634 const std::string globalCuts = cutExpression(fCutNames);
635 ROOT::RDF::RNode all = frame;
636 if (!globalCuts.empty()) all = all.Filter(globalCuts);
637 auto withCuts = [&](const std::vector<std::string>& names) -> ROOT::RDF::RNode {
638 const std::string expression = cutExpression(names);
639 if (expression.empty()) return all;
640 return all.Filter(expression);
641 };
642 auto totalEntries = all.Count();
643 // Files without start and end times (e.g. not from raw data): time of the first and last event
644 const bool eventTimes = duration <= 0 && frame.HasColumn("timeSeconds");
645 ROOT::RDF::RResultPtr<double> firstEventTime, lastEventTime;
646 if (eventTimes) {
647 firstEventTime = frame.Min("timeSeconds");
648 lastEventTime = frame.Max("timeSeconds");
649 }
650
651 struct PanelResults {
652 ROOT::RDF::RResultPtr<unsigned long long> entries;
653 std::map<std::string, ROOT::RDF::RResultPtr<double>> statistics;
654 std::string cuts;
655 };
656 std::vector<PanelResults> panelResults;
657 for (const auto& panel : fPanels) {
658 auto node = withCuts(panel.cuts);
659 PanelResults results;
660 results.entries = node.Count();
661 std::vector<std::string> cuts = fCutNames;
662 cuts.insert(cuts.end(), panel.cuts.begin(), panel.cuts.end());
663 results.cuts = cutExpression(cuts);
664 for (const auto& label : panel.labels) {
665 for (auto it = std::sregex_iterator(label.text.begin(), label.text.end(), kStatistic);
666 it != std::sregex_iterator(); ++it) {
667 const std::string function = (*it)[1].str();
668 const std::string column = (*it)[2].str();
669 const std::string key = function + "(" + column + ")";
670 if (results.statistics.count(key)) continue;
671 try {
672 // Only the defined values (e.g. the line observables are NaN when not defined)
673 auto finite = node.Filter("std::isfinite(" + column + ")");
674 if (function == "mean") results.statistics[key] = finite.Mean(column);
675 if (function == "sigma") results.statistics[key] = finite.StdDev(column);

715 } else {
716 const auto& y = histo.variables.back();

718 x.max, y.nBins, y.min, y.max);
