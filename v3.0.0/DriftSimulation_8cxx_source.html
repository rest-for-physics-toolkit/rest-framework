42void ComputeDriftResponse(const TRestDetectorMedium& medium, const TRestDetectorTransportTable& table,
43 const TRestDetectorReadout& readout, TRestDetectorDriftResponse& response,
44 bool verbose) {
45 const std::string error = "restGarfield: drift response " + response.fResponseFile + ": ";
46 auto gas = CreateGasFromTable(medium, table, medium.GetPressure());
47 const std::array<double, 3> offset = {response.fFieldMapOffset[0], response.fFieldMapOffset[1],
48 response.fFieldMapOffset[2]};
49 auto map = LoadCSTFieldMap(response.fFieldMap, offset);
50 map->SetMedium(gas.get());
51 unsigned int nx = 0, ny = 0, nz = 0;
52 double low[3], high[3]; // cm
53 map->GetMesh(nx, ny, nz, low[0], high[0], low[1], high[1], low[2], high[2]);
54
55 // The readout normal at the centre of the grid, along an axis: the drift region is cut at the boundary
56 const XYZVector centre((response.fXRange.first + response.fXRange.second) / 2,
57 (response.fYRange.first + response.fYRange.second) / 2,
58 (response.fZRange.first + response.fZRange.second) / 2);
59 const XYZVector normal = readout.GetReadoutNormal(centre);
60 if (normal.R() == 0) throw std::runtime_error(error + "the readout has no distance to it");
61 int axis = -1;
62 for (int a = 0; a < 3; a++) {
63 if (std::abs(Component(normal, a)) > 1 - 1e-6) axis = a;
64 }
65 if (axis < 0) throw std::runtime_error(error + "the normal of the readout must be along x, y or z");
66 const double boundary = response.fBoundary;
67 const XYZVector onBoundary = centre - (readout.GetDistanceToReadout(centre) - boundary) * normal;
68 const double plane = Component(onBoundary, axis) * kCmPerMm;
69 if (Component(normal, axis) > 0) {
70 low[axis] = std::max(low[axis], plane);
71 } else {
72 high[axis] = std::min(high[axis], plane);
73 }
74 if (!(low[axis] < high[axis])) {
75 throw std::runtime_error(error + "the field map does not reach the drift region above the boundary");
76 }
77
78 Garfield::Sensor sensor;
79 sensor.AddComponent(map.get());
80 sensor.SetArea(low[0], low[1], low[2], high[0], high[1], high[2]);
81 Garfield::AvalancheMC drift;
82 drift.SetSensor(&sensor);
83 drift.SetDistanceSteps(response.fStepSize * kCmPerMm);
84
85 response.fMedium = medium.GetName();
86 response.fPressure = medium.GetPressure();
87 response.fTemperature = medium.GetTemperature();
88 response.fTransportTable = table.GetKey();
89 const auto [u, v] = TRestDetectorDriftResponse::PlaneAxes(normal);
90 const size_t nodes = response.GetNumberOfNodes();
91 response.fNodes.assign(nodes, TRestDetectorDriftResponse::Node());
92 size_t reported = 0;
93 for (size_t index = 0; index < nodes; index++) {
94 const XYZVector start = response.GetNodePosition(index);
95 const double distance = readout.GetDistanceToReadout(start);
96 const XYZVector startOnBoundary = start - (distance - boundary) * normal;
97 std::vector<std::array<double, 3>> arrivals;
98 const double x[3] = {start.X() * kCmPerMm, start.Y() * kCmPerMm, start.Z() * kCmPerMm};
99 const bool inArea = x[0] >= low[0] && x[0] <= high[0] && x[1] >= low[1] && x[1] <= high[1] &&
100 x[2] >= low[2] && x[2] <= high[2];
101 if (distance <= boundary + kArrivalTolerance) {
102 // On or below the boundary plane: the drift belongs to the amplification
103 arrivals.assign(response.fElectrons, std::array<double, 3>{0, 0, 0});
104 } else if (inArea) {
105 for (int e = 0; e < response.fElectrons; e++) {
106 if (!drift.DriftElectron(x[0], x[1], x[2], 0)) continue;
107 if (drift.GetNumberOfElectronEndpoints() == 0) continue;
108 double x0, y0, z0, t0, x1, y1, z1, t1;
109 int status = 0;
110 drift.GetElectronEndpoint(0, x0, y0, z0, t0, x1, y1, z1, t1, status);
111 const XYZVector end(x1 / kCmPerMm, y1 / kCmPerMm, z1 / kCmPerMm);
112 if (std::abs(readout.GetDistanceToReadout(end) - boundary) > kArrivalTolerance) continue;
113 const XYZVector displacement = end - startOnBoundary;
114 arrivals.push_back({displacement.Dot(u), displacement.Dot(v), t1 * kUsPerNs});
115 }
116 }
117 response.SetNode(index, response.fElectrons, arrivals);
118 if (verbose && (index + 1) * 10 / nodes > reported) {
119 reported = (index + 1) * 10 / nodes;
120 RESTInfo << "restGarfield: drift response, " << index + 1 << " of " << nodes << " nodes"
121 << RESTendl;
122 }
123 }
124}

126void RunDriftResponse(const TRestDetectorMedium& medium, TRestDetectorDriftResponse& response, bool verbose) {
127 const std::string error = "restGarfield: drift response " + response.fResponseFile + ": ";
128 const std::string fileName = response.fResponseFile;
129 if (!response.fOverwrite && std::filesystem::exists(fileName)) {
130 RESTInfo << "restGarfield: drift response " << fileName
131 << " already exists (set overwrite: true to compute it again)" << RESTendl;
132 return;
133 }
134 if (!medium.IsGas()) throw std::runtime_error(error + "the medium must be a gas");
135
136 // The readout of the readouts file
137 if (response.fReadoutsFile.empty()) throw std::runtime_error(error + "readouts (a YAML file) is needed");
138 const YAML::Node config = TRestYAMLUtils::OpenConfigFile(response.fReadoutsFile);
139 const auto section = TRestYAMLUtils::GetMetadataClass(config, "TRestDetectorReadoutManager");
140 if (!section.second) {
141 throw std::runtime_error(error + "no TRestDetectorReadoutManager in " + response.fReadoutsFile);
142 }
143 TRestDetectorReadoutManager readouts(section.first, section.second);
144 readouts.LoadReadout();
145 std::string readoutName = response.fReadoutName;
146 const TRestDetectorReadout* readout = readouts.SelectReadout(readoutName, "restGarfield drift response");
147
148 // The transport table of the medium file
149 const auto mediumFile = medium.GetMediumFile();
150 std::unique_ptr<TRestDetectorTransportTable> table;
151 {
152 const MediumFileLock lock(mediumFile);
153 std::unique_ptr<TFile> file(TFile::Open(mediumFile.string().c_str(), "READ"));
154 if (!file || file->IsZombie()) throw std::runtime_error(error + "cannot open " + mediumFile.string());
155 std::string key = response.fTransportTable;
156 if (key.empty()) {
157 const YAML::Node tables = TRestDetectorTransportTable::GetTables(file.get(), medium.GetName());
158 if (tables.size() != 1) {
159 throw std::runtime_error(error + "the medium file has " + std::to_string(tables.size()) +
160 " transport tables of " + medium.GetName() +
161 ": give transportTable (1 needed)");
162 }
163 key = tables[0]["key"].as<std::string>();
164 }
165 table = std::make_unique<TRestDetectorTransportTable>(
166 TRestDetectorTransportTable::Read(file.get(), medium.GetName(), key));
167 }
168
169 RESTInfo << "restGarfield: computing the drift response " << fileName << " ("
170 << response.GetNumberOfNodes() << " nodes, " << response.fElectrons << " electrons each)"
171 << RESTendl;
172 ComputeDriftResponse(medium, *table, *readout, response, verbose);
173
174 YAML::Node generator;
175 generator["program"] = "Garfield++ AvalancheMC (field map)";
176 const std::time_t now = std::time(nullptr);
177 char date[32];
178 std::strftime(date, sizeof(date), "%Y-%m-%d %H:%M:%S", std::localtime(&now));
179 generator["date"] = date;
180 generator["code"] = REST_Version::GetSummary();
181 const std::filesystem::path path(fileName);
182 if (path.has_parent_path()) std::filesystem::create_directories(path.parent_path());
183 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "RECREATE"));
184 if (!file || file->IsZombie()) throw std::runtime_error(error + "cannot create the file");
185 response.Write(file.get(), generator);
186 RESTInfo << "restGarfield: drift response written to " << fileName << RESTendl;
187}
