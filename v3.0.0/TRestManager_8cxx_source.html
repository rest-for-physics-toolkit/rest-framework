48void TRestManager::LoadConfig() {
49 fMetaObjects.clear();
50 fProcessManager = nullptr;
51 fDataSetProcessManager = nullptr;
52 fDataSet = nullptr;
53 fRunInfo = nullptr;
54
55 for (const auto& element : fNode) {
56 const auto key = element.first.as<std::string>();
57 auto value = element.second;
58
59 if (!value || value.IsScalar() || !value.IsMap()) continue;
60
61 auto meta = CreateChildMetadata(key, value);
62 if (!meta) continue;
63
64 if (auto* processManager = dynamic_cast<TRestEventProcessManager*>(meta.get())) {
65 fProcessManager = processManager;
66 }
67 if (auto* dataSetProcessManager = dynamic_cast<TRestDataSetProcessManager*>(meta.get())) {
68 fDataSetProcessManager = dataSetProcessManager;
69 }
70 if (auto* dataSet = dynamic_cast<TRestDataSet*>(meta.get())) {
71 fDataSet = dataSet;
72 }
73 if (auto* run = dynamic_cast<TRestRun*>(meta.get())) {
74 fRunInfo = run;
75 }
76
77 meta->PrintMetadata();
78 fMetaObjects.emplace_back(std::move(meta));
79 }
80
81 // After the loop: the run may be declared after the metadata that uses it
82 if (fRunInfo) {
83 for (const auto& meta : fMetaObjects) {
84 if (meta.get() != fRunInfo) meta->SetRunInfo(fRunInfo);
85 }
86 }

89void TRestManager::Run() {
90 const auto plotList = GetMetadataOfType<TRestAnalysisPlot>();
91 const bool plots = !plotList.empty();
92 // Datasets: generation (TRestDataSet) and/or pipeline (TRestDataSetProcessManager), no TRestRun nor
93 // event pipeline
94 if (fDataSet || fDataSetProcessManager) {
95 if (fProcessManager || fRunInfo) {
96 throw std::runtime_error(
97 "TRestManager::Run - a TRestDataSet or a TRestDataSetProcessManager works on datasets: the "
98 "manager cannot have a TRestRun nor a TRestEventProcessManager");
99 }
100 if (fDataSet) {
101 if (fDataSet->fFilePattern.empty()) {
102 throw std::runtime_error("TRestManager::Run - the TRestDataSet '" + fDataSet->GetName() +
103 "' has no filePattern (or --i)");
104 }
105 if (fDataSet->fOutputFileName.empty() && !fDataSetProcessManager) {
106 throw std::runtime_error("TRestManager::Run - the TRestDataSet '" + fDataSet->GetName() +
107 "' has no outputFileName (or --o)");
108 }
109 fDataSet->Generate(GetMetadataOfType<TRestCut>());
110 // Written with the name resolved ([runStart], [runTag|Mixed]...), kept in fOutputFileName; with a
111 // dataset pipeline only if it has an outputFileName (the dataset before the processes)
112 if (!fDataSet->fOutputFileName.empty()) fDataSet->Write(fDataSet->fOutputFileName);
113 }
114 if (fDataSetProcessManager) {
115 fDataSetProcessManager->SetManager(this);
116 // The dataset generated is processed in memory: only the processed one is written
117 if (fDataSet) {
118 fDataSetProcessManager->Run(*fDataSet);
119 } else {
120 fDataSetProcessManager->Run();
121 }
122 }
123 return;
124 }
125 // Only plots (RunPlots): each one with its own input, no TRestRun
126 if (fProcessManager == nullptr && fRunInfo == nullptr && plots) return;
127 if (fProcessManager == nullptr && plots) {
128 throw std::runtime_error(
129 "TRestManager::Run - a manager with only plots has no TRestRun: remove the TRestRun section '" +
130 fRunInfo->GetName() +
131 "', the input is the inputFileName of each TRestAnalysisPlot (--i sets it when there is no "
132 "TRestRun)");
133 }
134 if (fProcessManager == nullptr) {
135 throw std::runtime_error(
136 "TRestManager::Run - no TRestEventProcessManager configured under manager section (a TRestRun is "
137 "the "
138 "input of the pipeline; for only plots, the input is the inputFileName of each "
139 "TRestAnalysisPlot, "
140 "or --i)");
141 }
142 if (fRunInfo == nullptr) {
143 throw std::runtime_error("TRestManager::Run - no TRestRun configured under manager section.");
144 }
145
146 // The plots without inputFileName are made with the output of the pipeline
147 for (const auto* plot : plotList) {
148 if (plot->fInputFileName.empty() && fRunInfo->HasNullOutput()) {
149 throw std::runtime_error("TRestManager::Run - the TRestAnalysisPlot '" + plot->GetName() +
150 "' is made with the output of the pipeline: the TRestRun needs an "
151 "outputFileName");
152 }
153 }
154
155 if (!fileExists(fRunInfo->GetInputFileName())) {
156 throw std::runtime_error("TRestManager::Run - input file " + fRunInfo->GetInputFileName() +
157 " not found");
158 }
159
160 fProcessManager->SetManager(this);
161
162 const unsigned int nThreads = fProcessManager->fNThreads;
163 if (nThreads > 1) {
164 RunMultithreaded(nThreads);
165 } else {
166 fProcessManager->Run();
167 }
168
169 if (fRunInfo->HasNullOutput()) {
170 RESTError << "TRestManager: no outputFileName in the TRestRun, nothing has been saved" << RESTendl;
171 }

174void TRestManager::RunMultithreaded(unsigned int nThreads) {
175 if (nThreads <= 1) {
176 fProcessManager->Run();
177 return;
178 }
179
180 // A process creating the events (e.g. from a DAQ file) reads them sequentially from the start
181 if (fProcessManager->HasGenerator()) {
182 RESTWarning << "TRestManager::RunMultithreaded: a process creates the events from its input, running "
183 "single-thread."
184 << RESTendl;
185 fProcessManager->Run();
186 return;
187 }
188
189 long long totalEntries = fProcessManager->fEventsToProcess;
190
191 // Not beyond the entries of the input file
192 const long long inputEntries = fRunInfo->GetEntries();
193 if (totalEntries <= 0 || (inputEntries > 0 && totalEntries > inputEntries)) {
194 totalEntries = inputEntries;
195 }
196
197 if (totalEntries <= 0) {
198 RESTError << "TRestManager::RunMultithreaded: cannot determine the number of entries to run, falling "
199 "back to single-thread."
200 << RESTendl;
201 fProcessManager->Run();
202 return;
203 }
204
205 auto [runKeyOrig, runNodeOrig] = TRestYAMLUtils::GetMetadataClass(fNode, "TRestRun");
206 if (runNodeOrig && !runNodeOrig.IsNull()) {
207 TRestYAMLUtils::OverrideParam(runNodeOrig, "runNumber", std::to_string(fRunInfo->GetRunNumber()));
208 }
209
210 const std::string baseOutput = fRunInfo->GetOutputFileName();
211 // Without output (/dev/null) the threads do not write either, and there is nothing to merge
212 const bool nullOutput = fRunInfo->HasNullOutput();
213 std::filesystem::path outPath(baseOutput);
214 const std::string stem = outPath.stem().string();
215 const std::string ext = outPath.extension().string();
216 const std::string dir = outPath.parent_path().string();
217
218 const long long chunk = totalEntries / nThreads;
219 std::vector<std::pair<long long, long long>> ranges(nThreads);
220 for (unsigned int t = 0; t < nThreads; ++t) {
221 ranges[t] = {t * chunk, (t == nThreads - 1) ? totalEntries : (t + 1) * chunk};
222 }
223
224 std::vector<std::string> threadOutputs(nThreads);
225 // Entries saved by each thread (their runs fill, not this one), and the exception of each thread: an
226 // exception escaping a std::thread would terminate the program
227 std::vector<long long> threadEntriesSaved(nThreads, 0);
228 std::vector<std::exception_ptr> threadErrors(nThreads);
229 const bool stopRequestedBefore = StopRequested();
230 std::vector<std::thread> workers;
231
232 for (unsigned int t = 0; t < nThreads; ++t) {
233 threadOutputs[t] = (dir.empty() ? "" : dir + "/") + stem + ".thread" + std::to_string(t) + ext;
234 if (nullOutput) threadOutputs[t] = baseOutput;
235
236 workers.emplace_back([this, t, &ranges, &threadOutputs, &threadEntriesSaved, &threadErrors] {
237 try {
238 YAML::Node threadManagerNode = YAML::Clone(fNode);
239
240 auto [runKey, runNode] = TRestYAMLUtils::GetMetadataClass(threadManagerNode, "TRestRun");
241 auto [pipeKey, pipeNode] =
242 TRestYAMLUtils::GetMetadataClass(threadManagerNode, "TRestEventProcessManager");
243 if (!runNode || runNode.IsNull() || !pipeNode || pipeNode.IsNull()) {
244 throw std::runtime_error(
245 "TRestManager::RunMultithreaded - TRestRun or TRestEventProcessManager section not "
246 "found");
247 }
248
249 TRestYAMLUtils::OverrideParam(runNode, "outputFileName", threadOutputs[t]);
250 TRestYAMLUtils::OverrideParam(pipeNode, "nThreads", std::string("1"));
251
252 TRestManager threadManager(GetName(), threadManagerNode);
253
254 auto* threadRun = threadManager.GetRunInfo();
255 auto* threadPipeline = dynamic_cast<TRestEventProcessManager*>(
256 threadManager.GetMetadataClass("TRestEventProcessManager"));
257 if (!threadRun || !threadPipeline) {
258 throw std::runtime_error(
259 "TRestManager::RunMultithreaded - cannot retrieve the TRestRun or the "
260 "TRestEventProcessManager of the thread");
261 }
262
263 threadPipeline->SetEntryRange(ranges[t].first, ranges[t].second);
264 threadPipeline->SetManager(&threadManager);
265 threadPipeline->Run();
266
267 threadEntriesSaved[t] = threadRun->GetEntriesSaved();
268 threadRun->CloseFiles();
269 } catch (...) {
270 threadErrors[t] = std::current_exception();
271 // The other threads stop at their next entry
272 RequestStop();
273 }
274 });
275 }
276
277 for (auto& w : workers) w.join();
278
279 // The first error is thrown, before merging; the outputs of the threads are removed
280 std::exception_ptr firstError;
281 for (unsigned int t = 0; t < nThreads; ++t) {
282 if (!threadErrors[t]) continue;
283 if (!firstError) firstError = threadErrors[t];
284 try {
285 std::rethrow_exception(threadErrors[t]);
286 } catch (const std::exception& error) {
287 RESTError << "TRestManager::RunMultithreaded: thread " << t << " failed: " << error.what()
288 << RESTendl;
289 } catch (...) {
290 RESTError << "TRestManager::RunMultithreaded: thread " << t << " failed (unknown exception)"
291 << RESTendl;
292 }
293 }
294 if (firstError) {
295 // The stop was requested by the failing thread, not by the user
296 if (!stopRequestedBefore) ResetStopRequest();
297 if (!nullOutput) {
298 std::error_code error;
299 for (const auto& f : threadOutputs) std::filesystem::remove(f, error);
300 }
301 RESTError << "TRestManager::RunMultithreaded: a thread failed, the output of the threads is not "
302 "merged"
303 << RESTendl;
304 std::rethrow_exception(firstError);
305 }
306
307 // The entries saved by this run (written with its metadata) are the ones of the threads
308 long long entriesSaved = 0;
309 for (const auto entries : threadEntriesSaved) entriesSaved += entries;
310 fRunInfo->SetEntriesSaved(entriesSaved);
311
312 if (nullOutput) return;
313
314 TFileMerger merger(kFALSE);
315 merger.SetFastMethod(kFALSE);
316
317 merger.SetMergeOptions(std::string_view("no-fast unbosom"));
318 merger.OutputFile(baseOutput.c_str(), "RECREATE");
319 for (unsigned int t = 0; t < nThreads; ++t) merger.AddFile(threadOutputs[t].c_str());
320 if (!merger.Merge()) {
321 throw std::runtime_error("TRestManager::RunMultithreaded - error merging output files");
322 }
323 for (const auto& f : threadOutputs) std::filesystem::remove(f);
324
325 fRunInfo->OpenOutputFile(baseOutput, "UPDATE");
326
327 if (StopRequested()) {
328 RESTError << "TRestManager::RunMultithreaded: Stop requested, output file may be cropped" << RESTendl;
329 }
330
331 RESTLog << "TRestManager: " << nThreads << " file merged in " << baseOutput << RESTendl;
332}

349void TRestManager::RunPlots() {
350 const auto plots = GetMetadataOfType<TRestAnalysisPlot>();
351 if (plots.empty()) return;
352 // The output of the pipeline, for the plots without inputFileName; closed, so the plots can update it
353 // (outputFileName of each plot)
354 std::string pipelineOutput;
355 if (fProcessManager && fRunInfo) {
356 pipelineOutput = fRunInfo->GetOutputFileName();
357 fRunInfo->CloseFiles();
358 }
359 if (fDataSet) pipelineOutput = fDataSet->fOutputFileName;
360 if (fDataSetProcessManager) pipelineOutput = fDataSetProcessManager->fOutputFileName;
361 for (auto* plot : plots) {
362 const std::string fileName = plot->fInputFileName.empty() ? pipelineOutput : plot->fInputFileName;
363 if (fileName.empty()) {
364 throw std::runtime_error("TRestManager::RunPlots - the TRestAnalysisPlot '" + plot->GetName() +
365 "' has no input: its inputFileName (or --i), or a pipeline");
366 }
367 plot->PlotCombinedCanvas(fileName, this);
368 }
