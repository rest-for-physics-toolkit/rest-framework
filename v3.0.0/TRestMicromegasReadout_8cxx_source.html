107void TRestMicromegasReadout::BuildGeometry() {
108 if (!fNode || fNode.IsNull()) {
109 throw std::runtime_error(
110 "TRestMicromegasReadout: 'readoutParameters' section is missing or fNode is not initialized!");
111 }
112
113 InitializeReadout();
114 if (!LoadDecoding(fDecodingFile)) {
115 throw std::runtime_error("TRestMicromegasReadout::BuildGeometry - cannot load the decoding file '" +
116 fDecodingFile + "' (not found or without channels)");
117 }
118
119 const auto pitch = fReadoutParams.fPitch;
120 const auto nChannels = fReadoutParams.fNChannels;
121 double visibleThickness = fReadoutParams.fThickness;
122 // The plane is built in its local frame (X and Y strips along local X and Y, normal along local Z) and
123 // placed with the rotation of its axes (globalMatrix): positionRelative in that frame
124 const int axisX = fReadoutParams.GetStripAxisX(), axisY = fReadoutParams.GetStripAxisY();
125 const int axisNormal = fReadoutParams.GetNormalAxis();
126 const double posRel[3] = {fReadoutParams.fPositionRelative[axisX],
127 fReadoutParams.fPositionRelative[axisY],
128 fReadoutParams.fPositionRelative[axisNormal]};
129
130 double pixelSize = pitch / std::sqrt(2.0);
131
132 TGeoRotation* rot45 = new TGeoRotation("rot45");
133 rot45->RotateZ(45.0);
134 TGeoRotation* rotMinus135 = new TGeoRotation("rotMinus135");
135 rotMinus135->RotateZ(-135.0);
136
137 TGeoMedium* copperMedium = fGeoManager->GetMedium("copper_medium");
138 if (!copperMedium) {
139 TGeoMaterial* matCopper = new TGeoMaterial("Copper", 63.546, 29.0, 8.96);
140 copperMedium = new TGeoMedium("copper_medium", 1, matCopper);
141 }
142
143 TGeoBBox* pixelShape =
144 new TGeoBBox("pixel_shape", pixelSize / 2.0, pixelSize / 2.0, visibleThickness / 2.0);
145
146 double moduleSizeX = (fReadoutParams.fNChannels + 1) * pitch - 0.5 * pitch;
147 double moduleSizeY = (fReadoutParams.fNChannels + 1) * pitch - 0.75 * pitch;
148 double offsetX = -moduleSizeX / 2.0;
149 double offsetY = -moduleSizeY / 2.0;
150
151 int nodeCounter = 0;
152
153 TGeoVolumeAssembly* readoutGeom = new TGeoVolumeAssembly("readout_geom_rect");
154
155 double zGlobal = 0.0;
156
157 // Pixels are grouped in one assembly per strip: a single volume holding all 2*N*N pixels overflows
158 // the int buffer size computed in TGeoVoxelFinder::SortAll (heap corruption for N >~ 215)
159 std::map<int, TGeoVolumeAssembly*> stripAssemblies;
160 std::map<int, int> stripPixelCounter;
161 const unsigned int kNoPhysicalID = static_cast<unsigned int>(-1);
162
163 // Pixel centred at (x, y) in the module frame (before the offsets and positionRelative)
164 auto addPixel = [&](double x, double y, TGeoRotation* localRot, int channelID, bool isChannelX) {
165 double posX = x + offsetX + posRel[0];
166 double posY = y + offsetY + posRel[1];
167 double posZ = zGlobal + posRel[2];
168
169 TGeoCombiTrans* finalMatrix = new TGeoCombiTrans(posX, posY, posZ, localRot);
170
171 std::string baseName = isChannelX ? "STRIP_X_" : "STRIP_Y_";
172 std::string volName = baseName + std::to_string(channelID);
173
174 TGeoVolume* vol = fGeoManager->GetVolume(volName.c_str());
175 if (!vol) {
176 vol = new TGeoVolume(volName.c_str(), pixelShape, copperMedium);
177
178 if (isChannelX) {
179 vol->SetLineColor(kBlack);
180 } else {
181 vol->SetLineColor(kGray);
182 }
183 }
184
185 auto& strip = stripAssemblies[channelID];
186 if (!strip) {
187 strip = new TGeoVolumeAssembly((volName + "_ASSEMBLY").c_str());
188 TGeoNode* stripNode = readoutGeom->AddNode(strip, channelID);
189 stripNode->SetUniqueID(kNoPhysicalID);
190 }
191
192 TGeoNode* node = strip->AddNode(vol, stripPixelCounter[channelID]++, finalMatrix);
193 node->SetUniqueID(channelID);
194 nodeCounter++;
195 };
196
197 // The positions are the nominal ones of GetPositionFromChannel (those of the DAQ).
198 // X channel nCh (physical ID nChannels + nCh): column of pixels at x = nCh * pitch - pitch / 4
199 for (int nCh = 0; nCh < nChannels; ++nCh) {
200 TGeoRotation* rot = nCh == 0 ? rotMinus135 : rot45;
201 for (int nPix = 0; nPix < nChannels; ++nPix) {
202 addPixel(nCh * pitch - pitch / 4.0, (0.5 + nPix) * pitch, rot, nChannels + nCh, true);
203 }
204 }
205
206 // Y channel nCh (physical ID nCh): row of pixels at y = (1 + nCh) * pitch, between the X pixels
207 for (int nCh = 0; nCh < nChannels; ++nCh) {
208 for (int nPix = 0; nPix < nChannels; ++nPix) {
209 addPixel(pitch / 4.0 + nPix * pitch, (1.0 + nCh) * pitch, rot45, nCh, false);
210 }
211 }
212
213 const int gasID = 2 * nChannels;
214 const double gasH = fReadoutParams.fGasThickness;
215
216 if (gasH > 0.0) {
217 TGeoMedium* gasMedium = fGeoManager->GetMedium("gas_medium");
218 if (!gasMedium) {
219 TGeoMaterial* matGas = new TGeoMaterial("Argon", 39.948, 18.0, 1.662e-3);
220 gasMedium = new TGeoMedium("gas_medium", 2, matGas);
221 }
222
223 const double gasRadius = 0.5 * std::sqrt(moduleSizeX * moduleSizeX + moduleSizeY * moduleSizeY);
224
225 TGeoVolume* gasVol = fGeoManager->MakeTube("GAS", gasMedium, 0.0, gasRadius, gasH / 2.0);
226 gasVol->SetLineColor(kAzure);
227 gasVol->SetTransparency(65);
228
229 // The gas (drift volume) is on the cathode side of the plane
230 const double gasZ =
231 posRel[2] + fReadoutParams.GetNormalSign() * (visibleThickness / 2.0 + gasH / 2.0);
232 auto* gasMatrix = new TGeoTranslation(posRel[0], posRel[1], gasZ);
233
234 TGeoNode* gasNode = readoutGeom->AddNode(gasVol, 0, gasMatrix);
235 gasNode->SetUniqueID(gasID);
236 }
237
238 TGeoHMatrix* globalMatrix = new TGeoHMatrix();
239 if (fReadoutParams.fGlobalRotation != 0.0) {
240 TGeoRotation* globalRot = new TGeoRotation("globalRot");
241 globalRot->RotateZ(fReadoutParams.fGlobalRotation * TMath::RadToDeg());
242 globalMatrix->MultiplyLeft(globalRot);
243 }
244 // Local axes X, Y, Z to the global axes of the X strips, Y strips and normal (a cyclic permutation:
245 // a proper rotation)
246 if (axisNormal != 2) {
247 double axes[9] = {0, 0, 0, 0, 0, 0, 0, 0, 0}; // Row major: global = axes * local
248 axes[axisX * 3 + 0] = 1;
249 axes[axisY * 3 + 1] = 1;
250 axes[axisNormal * 3 + 2] = 1;
251 TGeoRotation* axesRotation = new TGeoRotation("axesRotation");
252 axesRotation->SetMatrix(axes);
253 globalMatrix->MultiplyLeft(axesRotation);
254 }
255
256 TGeoVolume* top = fGeoManager->GetTopVolume();
257 TGeoNode* readoutNode = top->AddNode(readoutGeom, 0, globalMatrix);
258 readoutNode->SetUniqueID(kNoPhysicalID);
259
260 fGeoManager->CloseGeometry();
261 std::cout << "[+] Geometry built: " << nodeCounter << " pixels in a single plane." << std::endl;
262
263 if (fReadoutParams.fCheckOverlaps) {
264 const int overlaps = CheckOverlaps();
265 if (overlaps > 0) {
266 throw std::runtime_error("TRestMicromegasReadout::BuildGeometry - " + std::to_string(overlaps) +
267 " geometry overlaps in '" + GetName() + "'");
268 }
269 }
270}

290ROOT::Math::XYZVector TRestMicromegasReadout::GetPositionFromChannel(int daqID) const {
291 if (!fGeoManager) {
292 RESTError << "Geometry not initialized in TRestMicromegasReadout" << RESTendl;
293 return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
294 }
295
296 const int targetPhysicalID = GetPhysicalID(daqID);
297 if (targetPhysicalID < 0) {
298 RESTWarning << "DaqID " << daqID << " not found in Micromegas map" << RESTendl;
299 return ROOT::Math::XYZVector(REST_nan, REST_nan, REST_nan);
300 }
301
302 double moduleSizeX =
303 (fReadoutParams.fNChannels + 1) * fReadoutParams.fPitch - 0.5 * fReadoutParams.fPitch;
304 double moduleSizeY =
305 (fReadoutParams.fNChannels + 1) * fReadoutParams.fPitch - 0.75 * fReadoutParams.fPitch;
306 double offsetX = -moduleSizeX / 2.0;
307 double offsetY = -moduleSizeY / 2.0;
308
309 // Coordinates along the X strips axis and the Y strips axis (in the frame of the plane)
310 double posX = REST_nan;
311 double posY = REST_nan;
312 const int axisX = fReadoutParams.GetStripAxisX(), axisY = fReadoutParams.GetStripAxisY();
313
314 bool isChannelX = (targetPhysicalID >= fReadoutParams.fNChannels);
315
316 if (isChannelX) {
317 int nCh = targetPhysicalID - fReadoutParams.fNChannels;
318 double localY_fixed = nCh * fReadoutParams.fPitch - fReadoutParams.fPitch / 4.0;
319 posX = localY_fixed + offsetX + fReadoutParams.fPositionRelative[axisX];
320 } else {
321 int nCh = targetPhysicalID;
322 double localX_fixed = (1.0 + nCh) * fReadoutParams.fPitch;
323 posY = localX_fixed + offsetY + fReadoutParams.fPositionRelative[axisY];
324 }
325
326 // Global axes: the strip coordinate on its axis, the others NaN
327 double position[3] = {REST_nan, REST_nan, REST_nan};
328 position[axisX] = posX;
329 position[axisY] = posY;
330 return ROOT::Math::XYZVector(position[0], position[1], position[2]);
331}

333std::pair<ROOT::Math::XYZVector, ROOT::Math::XYZVector> TRestMicromegasReadout::GetChannelsBoundingBox()
334 const {
335 if (fReadoutParams.fGlobalRotation == 0.0) return TRestDetectorReadout::GetChannelsBoundingBox();
336
337 const double inf = std::numeric_limits<double>::infinity();
338 const int axisX = fReadoutParams.GetStripAxisX(), axisY = fReadoutParams.GetStripAxisY();
339 const int axisNormal = fReadoutParams.GetNormalAxis();
340 double minimum[2] = {inf, inf}, maximum[2] = {-inf, -inf};
341 for (const auto& [physicalID, daqID] : fPhysicalToDAQMap) {
342 const auto position = GetPositionFromChannel(daqID);
343 const double global[3] = {position.X(), position.Y(), position.Z()};
344 const double coordinates[2] = {global[axisX], global[axisY]};
345 for (int a = 0; a < 2; a++) {
346 if (!std::isfinite(coordinates[a])) continue;
347 minimum[a] = std::min(minimum[a], coordinates[a]);
348 maximum[a] = std::max(maximum[a], coordinates[a]);
349 }
350 }
351 const double nan = std::numeric_limits<double>::quiet_NaN();
352 if (minimum[0] > maximum[0] || minimum[1] > maximum[1]) {
353 return {ROOT::Math::XYZVector(nan, nan, nan), ROOT::Math::XYZVector(nan, nan, nan)};
354 }
355 const double halfPitch = fReadoutParams.fPitch / 2.0;
356 double low[3], high[3];
357 low[axisX] = minimum[0] - halfPitch;
358 low[axisY] = minimum[1] - halfPitch;
359 high[axisX] = maximum[0] + halfPitch;
360 high[axisY] = maximum[1] + halfPitch;
361 low[axisNormal] = high[axisNormal] = fReadoutParams.fPositionRelative[axisNormal];
362 return {ROOT::Math::XYZVector(low[0], low[1], low[2]), ROOT::Math::XYZVector(high[0], high[1], high[2])};
363}

365double TRestMicromegasReadout::GetDistanceToReadout(const ROOT::Math::XYZVector& position) const {
366 // Single readout plane at positionRelative along the normal axis; the global rotation is around it
367 const int axis = fReadoutParams.GetNormalAxis();
368 const double coordinate = axis == 0 ? position.X() : axis == 1 ? position.Y() : position.Z();
369 return fReadoutParams.GetNormalSign() * (coordinate - fReadoutParams.fPositionRelative[axis]);
370}
