34RunFile ReadRunFile(const std::string& fileName) {
35 std::unique_ptr<TFile> file(TFile::Open(fileName.c_str(), "READ"));
36 if (!file || file->IsZombie()) throw std::runtime_error("cannot open " + fileName);
37
38 RunFile runFile;
39 runFile.fileName = fileName;
40 for (const auto& entry : TRestYAMLUtils::ReadMetadataStore(file.get())) {
41 const std::string instance = entry.first.as<std::string>();
42 const YAML::Node node = entry.second;
43 if (!node.IsMap() || !node["class"]) continue;
44 if (node["class"].as<std::string>() == "TRestRun") {
45 runFile.runInstance = instance;
46 runFile.run = node;
47 } else {
48 runFile.metadata[instance] = node;
49 }
50 }
51 if (!runFile.run) throw std::runtime_error(fileName + " has no TRestRun");
52 runFile.subRuns = TRestYAMLUtils::GetParam<std::vector<int>>(runFile.run, "mergedSubRuns");
53 if (runFile.subRuns.empty())
54 runFile.subRuns.push_back(TRestYAMLUtils::GetParam<int>(runFile.run, "subRunNumber", 0));
55
56 // Event trees and analysis tree: name and type of each leaf
57 for (const auto* object : *file->GetListOfKeys()) {
58 const auto* key = static_cast<const TKey*>(object);
59 if (std::string(key->GetClassName()) != "TTree") continue;
60 auto* tree = file->Get<TTree>(key->GetName());
61 if (!tree) continue;
62 auto& branches = runFile.trees[key->GetName()];
63 for (const auto* leafObject : *tree->GetListOfLeaves()) {
64 const auto* leaf = static_cast<const TLeaf*>(leafObject);
65 branches.push_back(std::string(leaf->GetName()) + "/" + leaf->GetTypeName());
66 }
67 }
68 if (!runFile.trees.count("AnalysisTree")) throw std::runtime_error(fileName + " has no AnalysisTree");
69 return runFile;
70}

74void CopyObjects(TDirectory* source, TDirectory* target) {
75 for (const auto* object : *source->GetListOfKeys()) {
76 auto* key = static_cast<TKey*>(const_cast<TObject*>(object));
77 const std::string name = key->GetName();
78 const std::string className = key->GetClassName();
79 if (className == "TTree" || name == "RESTMetadataStore") continue;
80 if (className == "TDirectoryFile" || className == "TDirectory") {
81 auto* sourceDirectory = source->GetDirectory(name.c_str());
82 TDirectory* targetDirectory = target->mkdir(name.c_str(), "", true);
83 if (sourceDirectory && targetDirectory) CopyObjects(sourceDirectory, targetDirectory);
84 continue;
85 }
86 std::unique_ptr<TObject> copy(key->ReadObj());
87 if (!copy) continue;
88 target->cd();
89 copy->Write(name.c_str(), TObject::kOverwrite);
90 }
91}

125void MergeSimulationTrees(const std::vector<RunFile>& runFiles, const std::vector<long long>& eventOffsets,
126 TFile* output) {
127 for (const auto& [treeName, branches] : runFiles.front().trees) {
128 TTree* merged = nullptr;
129 // Open until the next file sets its addresses in the merged tree (CopyAddresses)
130 std::unique_ptr<TFile> previous;
131 for (size_t i = 0; i < runFiles.size(); i++) {
132 std::unique_ptr<TFile> file(TFile::Open(runFiles[i].fileName.c_str(), "READ"));
133 if (!file || file->IsZombie()) throw std::runtime_error("cannot open " + runFiles[i].fileName);
134 auto* tree = file->Get<TTree>(treeName.c_str());
135 if (!tree) throw std::runtime_error(runFiles[i].fileName + " has no " + treeName);
136 if (!merged) {
137 output->cd();
138 merged = tree->CloneTree(0);
139 } else {
140 tree->CopyAddresses(merged);
141 }
142 TLeaf* eventID = tree->GetLeaf("eventID");
143 for (Long64_t entry = 0; entry < tree->GetEntries(); entry++) {
144 tree->GetEntry(entry);
145 if (eventID) *static_cast<int*>(eventID->GetValuePointer()) += eventOffsets[i];
146 merged->Fill();
147 }
148 previous = std::move(file);
149 }
150 output->cd();
151 merged->Write("", TObject::kOverwrite);
152 delete merged;
153 }
154}

184int REST_MergeRuns(const std::string& filePattern, const std::string& outputFile) {
185 using namespace REST_MergeRuns_Detail;
186 try {
187 std::vector<RunFile> runFiles;
188 for (const auto& fileName : TRestTools::GetFilesMatchingPattern(filePattern)) {
189 runFiles.push_back(ReadRunFile(TRestTools::GetFullPath(fileName)));
190 }
191 if (runFiles.size() < 2) throw std::runtime_error("less than two files match " + filePattern);
192 const std::string outputPath = TRestTools::GetFullPath(outputFile);
193 for (const auto& runFile : runFiles) {
194 if (runFile.fileName == outputPath) throw std::runtime_error("the output file is an input file");
195 }
196 // Simulations: merged as a single one, in the order of the pattern; otherwise subruns in order
197 const std::string geant4Instance = Geant4Instance(runFiles.front());
198 const bool simulation = !geant4Instance.empty();
199 if (!simulation) {
200 std::sort(runFiles.begin(), runFiles.end(), [](const RunFile& a, const RunFile& b) {
201 return a.subRuns.front() < b.subRuns.front();
202 });
203 }
204
205 const RunFile& first = runFiles.front();
206 const int runNumber = TRestYAMLUtils::GetParam<int>(first.run, "runNumber", -1);
207 if (runNumber < 0 && !simulation) throw std::runtime_error(first.fileName + " has no run number");
208 std::vector<int> subRuns;
209 double startTime = 0, endTime = 0;
210 long long entries = 0;
211 // Simulations: the seeds, the events simulated before each file and the sums of the counts
212 std::set<long long> seeds;
213 std::vector<long long> eventOffsets;
214 long long events = 0, requestedEntries = 0, showersUsed = 0;
215 double exposureTime = 0;
216 for (size_t i = 0; i < runFiles.size(); i++) {
217 const RunFile& runFile = runFiles[i];
218 if (!simulation && TRestYAMLUtils::GetParam<int>(runFile.run, "runNumber", -1) != runNumber) {
219 throw std::runtime_error(runFile.fileName + " is not of the run " +
220 std::to_string(runNumber));
221 }
222 if (!simulation && i > 0 && runFile.subRuns.front() != subRuns.back() + 1) {
223 throw std::runtime_error("subruns not consecutive: " + std::to_string(subRuns.back()) +
224 " and " + std::to_string(runFile.subRuns.front()) + " (" +
225 runFile.fileName + ")");
226 }
227 if (runFile.trees != first.trees) {
228 throw std::runtime_error(runFile.fileName + " has not the same trees and branches as " +
229 first.fileName);
230 }
231 if (runFile.metadata.size() != first.metadata.size()) {
232 throw std::runtime_error(runFile.fileName + " has not the same metadata as " +
233 first.fileName);
234 }
235 for (const auto& [instance, node] : first.metadata) {
236 const auto it = runFile.metadata.find(instance);
237 bool equal = it != runFile.metadata.end();
238 if (equal && simulation && instance == geant4Instance) {
239 equal = YAML::Dump(SimulationConfiguration(it->second)) ==
240 YAML::Dump(SimulationConfiguration(node));
241 } else if (equal) {
242 equal = YAML::Dump(it->second) == YAML::Dump(node);
243 }
244 if (!equal) {
245 throw std::runtime_error(runFile.fileName + ": metadata '" + instance +
246 "' not equal to the one of " + first.fileName +
247 (simulation ? " (not the same simulation and processing)"
248 : " (not the same processing)"));
249 }
250 }
251 if (simulation) {
252 const YAML::Node& geant4 = runFile.metadata.at(geant4Instance);
253 const auto seed = TRestYAMLUtils::GetParam<long long>(geant4, "seed", 0);
254 if (!seeds.insert(seed).second) {
255 throw std::runtime_error(runFile.fileName + ": seed " + std::to_string(seed) +
256 " repeated (the same events)");
257 }
258 eventOffsets.push_back(events);
259 events += TRestYAMLUtils::GetParam<long long>(geant4, "nEvents", 0);
260 requestedEntries += TRestYAMLUtils::GetParam<long long>(geant4, "nRequestedEntries", 0);
261 if (geant4["generator"] && geant4["generator"].IsMap()) {
262 showersUsed += static_cast<long long>(NumberOf(geant4["generator"]["showersUsed"]));
263 exposureTime += NumberOf(geant4["generator"]["exposureTime"]);
264 }
265 }
266 subRuns.insert(subRuns.end(), runFile.subRuns.begin(), runFile.subRuns.end());
267 const double start = TRestYAMLUtils::GetParam<double>(runFile.run, "startTime", 0);
268 const double end = TRestYAMLUtils::GetParam<double>(runFile.run, "endTime", 0);
269 if (i == 0 || start < startTime) startTime = start;
270 if (i == 0 || end > endTime) endTime = end;
271 entries += TRestYAMLUtils::GetParam<long long>(runFile.run, "entriesSaved", 0);
272 }
273
274 if (simulation) {
275 std::unique_ptr<TFile> output(TFile::Open(outputPath.c_str(), "RECREATE"));
276 if (!output || output->IsZombie()) throw std::runtime_error("cannot create " + outputPath);
277 MergeSimulationTrees(runFiles, eventOffsets, output.get());
278 output->Close();
279 } else {
280 // Only the trees, as the threads of TRestManager::RunMultithreaded
281 TFileMerger merger(kFALSE);
282 merger.SetFastMethod(kFALSE);
283 merger.OutputFile(outputPath.c_str(), "RECREATE");
284 for (const auto& runFile : runFiles) merger.AddFile(runFile.fileName.c_str());
285 for (const auto& [treeName, branches] : first.trees) merger.AddObjectNames(treeName.c_str());
286 if (!merger.PartialMerge(TFileMerger::kAll | TFileMerger::kRegular | TFileMerger::kOnlyListed)) {
287 throw std::runtime_error("error merging the files");
288 }
289 }
290
291 // The rest from the first subrun (equal in all of them: same metadata), and the metadata: those of
292 // the first subrun and the merged run
293 YAML::Node run = YAML::Clone(first.run);
294 if (!simulation) {
295 run["subRunNumber"] = subRuns.front();
296 run["mergedSubRuns"] = subRuns;
297 }
298 run["startTime"] = startTime;
299 run["endTime"] = endTime;
300 run["entriesSaved"] = entries;
301 run["outputFileName"] = outputPath;
302 std::map<std::string, YAML::Node> metadata = first.metadata;
303 if (simulation) {
304 YAML::Node geant4 = YAML::Clone(metadata.at(geant4Instance));
305 geant4["nEvents"] = events;
306 if (geant4["nRequestedEntries"]) geant4["nRequestedEntries"] = requestedEntries;
307 if (geant4["generator"] && geant4["generator"]["showersUsed"]) {
308 geant4["generator"]["showersUsed"] = showersUsed;
309 geant4["generator"]["exposureTime"] = exposureTime;
310 }
311 metadata[geant4Instance] = geant4;
312 }
313
314 std::unique_ptr<TFile> file(TFile::Open(outputPath.c_str(), "UPDATE"));
315 if (!file || file->IsZombie()) throw std::runtime_error("cannot open " + outputPath);
316 {
317 std::unique_ptr<TFile> firstFile(TFile::Open(first.fileName.c_str(), "READ"));
318 if (!firstFile || firstFile->IsZombie())
319 throw std::runtime_error("cannot open " + first.fileName);
320 CopyObjects(firstFile.get(), file.get());
321 }
322 for (const auto& [instance, node] : metadata)
323 TRestMetadata::WriteMetadata(file.get(), instance, node);
324 TRestMetadata::WriteMetadata(file.get(), first.runInstance, run);
325 file->Close();
326
327 if (simulation) {
328 std::cout << "REST_MergeRuns: " << runFiles.size() << " simulations (" << events << " events, "
329 << entries << " entries) merged in " << outputFile << std::endl;
330 } else {
331 std::cout << "REST_MergeRuns: run " << runNumber << ", subruns " << subRuns.front() << " to "
332 << subRuns.back() << " (" << runFiles.size() << " files) merged in " << outputFile
333 << std::endl;
334 }
335 } catch (const std::exception& error) {
336 RESTError << "REST_MergeRuns: " << error.what() << RESTendl;
337 return 1;
338 }
339 return 0;
340}
