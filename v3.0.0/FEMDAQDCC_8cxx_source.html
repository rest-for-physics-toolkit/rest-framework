15void FEMDAQDCC::Pedestals(const std::vector<std::string>& flags) {

71void FEMDAQDCC::startDAQ(const std::vector<std::string>& flags) {
72 int mode = 0; // Mode
73 bool internal = false;
74
75 for (const auto& flag : flags) {
76 if (flag == "internal")
77 internal = true; // Internal (random) trigger, otherwise external
78 // (default)
79 else if (flag == "ZS")
80 mode = 1; // Zero suppression
81 else
82 std::cout << "Unknown flag " << flag
83 << " supported flags are internal (trigger) or ZS (Zero suppression)" << std::endl;
84 }
85
86 // A previous run not stopped (stopDAQ): stopped now, its threads cannot be replaced while running
87 stopDAQ();
88
89 auto& FEM = FEMArray.front();
90 FEM.mutex_mem.lock();
91 FEM.tmpBuffer.clear();
92 FEM.mutex_mem.unlock();
93
94 runStartTime = TRestTools::GetCurrentTime();
95 stopRun = false;
96 storedEvents = 0;
97
98 stopEventBuilder = false;
99 eventBuilderThread = std::thread([this]() { RunThread("Event builder", [this]() { EventBuilder(); }); });
100 UpdateRunThread = std::thread([this]() { RunThread("Update thread", [this]() { UpdateThread(); }); });
101
102 const auto& fecs = runConfig.daq->fFEMs.front().fFECs;
103
104 std::cout << TRestTools::GetTimeStampFromUnixTime(runStartTime) << " Starting data taking ";
105 if (HasRootFile()) std::cout << fileNameRoot;
106 std::cout << std::endl;
107
108 auto rS = std::chrono::high_resolution_clock::now();
109
110 try {
111 AcquisitionLoop(FEM, fecs, mode, internal, rS);
112 } catch (const std::exception& e) {
113 std::cout << "[CRITICAL] DAQ command failed: " << e.what() << std::endl;
114 stopDAQ();
115 throw;
116 }
117
118 // The run ended (nEvents, time or CTRL-C): the event builder writes the data left and stops, so the
119 // files can be closed after it
120 stopDAQ();
121}

123void FEMDAQDCC::AcquisitionLoop(FEMProxy& FEM, const std::vector<int>& fecs, int mode, bool internal,
124 std::chrono::high_resolution_clock::time_point rS) {
125 char cmd[200];
126 while (!stopRun) {
127 // SendCommand("fem 0");
128
129 SendCommand("isobus 0x64", FEM); // SCA start
130 if (internal) SendCommand("isobus 0x14", FEM);
131 waitForTrigger();
132 if (abrt) break;
133 // Perform data acquisition phase, compress, accept size
134 auto now = std::chrono::high_resolution_clock::now();
135 for (auto fecID : fecs) {
136 for (int a = 0; a < 4; a++) {
137 sprintf(cmd, "areq %d %d %d %d %d", mode, fecID, a, 3, 78);
138 SendCommand(cmd, FEM, 0);
139 }
140 }
141
142 auto elapsed = std::chrono::duration_cast<std::chrono::nanoseconds>(now - rS);
143
144 uint64_t ns10 = static_cast<uint64_t>(elapsed.count() / 10) & 0xFFFFFFFFFFFFULL; // 48 bits mask
145
146 uint16_t t1 = static_cast<uint16_t>((ns10 >> 32) & 0xFFFF); // Bits 47-32
147 uint16_t t2 = static_cast<uint16_t>((ns10 >> 16) & 0xFFFF); // Bits 31-16
148 uint16_t t3 = static_cast<uint16_t>(ns10 & 0xFFFF); // Bits 15-0
149
150 FEM.mutex_mem.lock();
151 FEM.tmpBuffer.emplace_back(0xFFFF); // Insert word to mark EOE
152 FEM.tmpBuffer.emplace_back(t1); // Insert timeStamp 48 bits
153 FEM.tmpBuffer.emplace_back(t2);
154 FEM.tmpBuffer.emplace_back(t3);
155 FEM.mutex_mem.unlock();
156 }

159void FEMDAQDCC::EventBuilder() {
160 auto& FEM = FEMArray.front();
161
162 double lastTimeSaved = runStartTime;
163 uint32_t ev_count = 0;
164 uint64_t ts = 0x0;
165 double eventTime = 0; // Time of the last event built
166
167 if (HasRootFile()) WriteRunStartTime(runStartTime);
168
169 bool newEvent = false;
170 bool emptyBuffer = true;
171 int tC = 0;
172
173 std::vector<uint16_t> eventBuffer;
174 eventBuffer.reserve(6 * 72 * 4 * 600);
175
176 FEM.bufferIndex = 0;
177 FEM.buffer.clear();
178 // Signals of an incomplete event of a previous run
179 sEvent.Initialize();
180
181 std::cout << "Start of event builder" << std::endl;
182
183 while (!(emptyBuffer && stopEventBuilder)) {
184 FEM.mutex_mem.lock();
185 if (!FEM.tmpBuffer.empty()) {
186 FEM.buffer.insert(FEM.buffer.end(), std::make_move_iterator(FEM.tmpBuffer.begin()),
187 std::make_move_iterator(FEM.tmpBuffer.end()));
188 FEM.tmpBuffer.clear();
189 }
190 FEM.mutex_mem.unlock();
191
192 emptyBuffer = FEM.buffer.empty();
193
194 if (!emptyBuffer) {
195 newEvent = DCCPacket::TryExtractNextEvent(FEM.buffer, FEM.bufferIndex, eventBuffer);
196 }
197
198 if (newEvent) {
199 DCCPacket::ParseEventFromWords(eventBuffer, sEvent, ts, ev_count);
200 eventBuffer.clear();
201 sEvent.AddFEM(FEM.femID, ev_count, ts, sEvent.GetNumberOfSignals());
202 sEvent.SetID(static_cast<int>(ev_count));
203 eventTime = (double)ts * 1.E-8 + runStartTime;
204 sEvent.SetTime(eventTime);
205
206 // Do not fill empty events
207 if (HasRootFile() && sEvent.GetNumberOfSignals() > 0) {
208 FillTree(eventTime, lastTimeSaved);
209
210 if (storedEvents % 100 == 0) {
211 CheckFileSize(eventTime);
212 }
213 }
214
215 newEvent = false;
216 eventBuffer.clear();
217
218 if (sEvent.GetNumberOfSignals() > 0) ++storedEvents;
219 sEvent.Initialize();
220 }
221
222 if (stopEventBuilder) {
223 if (tC % 100 == 0) {
224 if (runConfig.verboseLevel >= RunConfig::Verbosity::REST_Debug)
225 std::cout << "FEM " << FEM.femID << " Buffer size " << FEM.buffer.size() << std::endl;
226 }
227 tC++;
228 if (tC >= 500) break;
229 }
230
231 std::this_thread::sleep_for(std::chrono::milliseconds(2));
232 }
233
234 std::cout << TRestTools::GetTimeStampFromUnixTime(runStartTime)
235 << " End of event builder: " << storedEvents << " events stored and " << ev_count
236 << " events triggered" << std::endl;
237
238 if (!FEM.buffer.empty()) {
239 std::cout << "FEM " << FEM.femID << " Buffer size left " << FEM.buffer.size() << std::endl;
240 if (runConfig.verboseLevel >= RunConfig::Verbosity::REST_Debug) {
241 std::ostringstream words;
242 words << "------DEBUG BUFFER FRAMES LEFT-------";
243 char word[8];
244 for (const auto& w : FEM.buffer) {
245 std::snprintf(word, sizeof(word), "\n%04X", w);
246 words << word;
247 }
248 std::cout << words.str() << std::endl;
249 WriteLog(FEM, words.str());
250 }
251 }
252
253 if (HasRootFile()) {
254 // The run ends at the stop of the DAQ or at the last event, if later (as in FEMDAQARCFEM)
255 if (eventTime > runEndTime) runEndTime = eventTime;
256
257 WriteRunEndTime(runEndTime);
258 }
259}

261void FEMDAQDCC::SendCommand(const char* cmd) {
262 auto& FEM = FEMArray.front();
263
264 if (std::strstr(cmd, "asic") && (std::strstr(cmd, "gain") || std::strstr(cmd, "time"))) {
265 uint32_t asicID = 0, gain = 0, time = 0;
266 if (std::sscanf(cmd, "asic %u gain 0x%x time 0x%x", &asicID, &gain, &time) == 3) {
267 uint8_t regValue = ((time & 0x0F) << 3) | ((gain & 0x03) << 1);
268 char finalCmd[64];
269 std::snprintf(finalCmd, sizeof(finalCmd), "asic %u write 1 0x%X", asicID, regValue);
270 SendCommand(finalCmd, FEM);
271 } else {
272 std::string error =
273 "Invalid cmd sintax: " + std::string(cmd) + " proper sintax is asic 0 gain 0x1 time 0x3";
274 throw std::runtime_error(error);
275 }
276 } else {
277 SendCommand(cmd, FEM);
278 }

281bool FEMDAQDCC::SendCommand(const char* cmd, FEMProxy& FEM, int pckType) {
282 // pckType = -1 ASCII, >=0 Binary
283
284 FEM.Send(cmd);
285
286 const bool logCmd =
287 strncmp(cmd, "wait", 4) != 0 && strncmp(cmd, "areq", 4) != 0 && strncmp(cmd, "isobus", 6) != 0;
288
289 if (logCmd && runConfig.verboseLevel >= RunConfig::Verbosity::REST_Info)
290 WriteLog(FEM, ">> FEM " + std::to_string(FEM.femID) + " Cmd sent " + cmd);
291
292 // wait for incoming messages
293 bool done = false;
294 int length = 0;
295 size_t pckCnt = 1;
296 // 8192 bytes received; two more words for the '\0' written after a message starting at buf_rcv[1]
297 uint16_t buf_rcv[8192 / sizeof(uint16_t) + 2];
298 uint8_t* buf_ual;
299
300 while (!done) {
301 // 10 s timeout; -1: aborted (CTRL-C)
302 length = FEM.Receive(buf_rcv, 8192, 10, &abrt);
303 if (length < 0 || abrt) return false;
304
305 // if the first 2 bytes are null, UDP datagram is aligned on next 32-bit
306 // boundary, so skip these first two bytes
307 int index = 0;
308 if ((buf_rcv[0] == 0)) {
309 length -= 2;
310 index = 1;
311 }
312
313 buf_ual = reinterpret_cast<uint8_t*>(&buf_rcv[1]);
314
315 // ERROR ASCII packet
316 if (*buf_ual == '-') {
317 if (strncmp(cmd, "wait", 4) == 0) return true;
318 *(buf_ual + length) = '\0';
319 fprintf(stdout, "--------ERROR packet---------: %s\n", buf_ual);
320 WriteLog(FEM, "--------ERROR packet---------: " + std::string(reinterpret_cast<char*>(buf_ual)));
321 return false;
322 }
323
324 if (pckType >= 0) { // Data packet
325
326 DCCPacket::DataPacket* data_pkt = (DCCPacket::DataPacket*)buf_ual;
327
328 if (GET_TYPE(ntohs(data_pkt->hdr)) == RESP_TYPE_ADC_DATA) {
329 // The size of the packet, at most the words received
330 const size_t received = (length + 2 * index) / sizeof(uint16_t);
331 const size_t pckSize = std::min<size_t>(ntohs(data_pkt->size) / sizeof(uint16_t),
332 received > (size_t)index ? received - index : 0);
333 FEM.mutex_mem.lock();
334 FEM.tmpBuffer.insert(FEM.tmpBuffer.end(), &buf_rcv[index], &buf_rcv[index + pckSize]);
335 FEM.mutex_mem.unlock();
336
337 if (runConfig.verboseLevel > RunConfig::Verbosity::REST_Info) PrintMonitoring(data_pkt);
338 } else {
339 PrintMonitoring(data_pkt);
340 }
341
342 // Check End Of Event
343 if (GET_FRAME_TY_V2(ntohs(data_pkt->dcchdr)) & FRAME_FLAG_EOEV ||
344 GET_FRAME_TY_V2(ntohs(data_pkt->dcchdr)) & FRAME_FLAG_EORQ ||
345 (pckType > 0 && pckCnt >= static_cast<size_t>(pckType))) {
346 done = true;
347 }
348
349 } else { // ASCII Packet
350 if (logCmd) {
351 *(buf_ual + length) = '\0';
352 if (runConfig.verboseLevel > RunConfig::Verbosity::REST_Info)
353 fprintf(stdout, "dcc().rep(): %s", buf_ual);
354 WriteLog(FEM, ">>> dcc().rep(): " + std::string(reinterpret_cast<char*>(buf_ual)));
355 }
356 done = true;
357 }
358 pckCnt++;
359 }
360 return false;
361}
