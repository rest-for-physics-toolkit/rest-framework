42void TRestGeant4Manager::LoadConfig() {
43 fG4Metadata = nullptr;
44 fRunInfo = nullptr;
45 fG4PhysicsLists = nullptr;
46 fCRY = nullptr;
47
48 for (const auto& element : fNode) {
49 const auto key = element.first.as<std::string>();
50 auto value = element.second;
51
52 if (!value || value.IsScalar() || !value.IsMap()) continue;
53
54 auto meta = MetadataClassRegistry::Instance().Create(key, value);
55 if (!meta) continue;
56
57 if (auto* run = dynamic_cast<TRestRun*>(meta.get())) {
58 fRunInfo = run;
59 meta.release();
60 }
61 if (auto* pL = dynamic_cast<TRestGeant4PhysicsLists*>(meta.get())) {
62 fG4PhysicsLists = pL;
63 meta.release();
64 }
65
66 if (auto* G4Meta = dynamic_cast<TRestGeant4Metadata*>(meta.get())) {
67 fG4Metadata = G4Meta;
68 meta.release();
69 }
70 if (auto* cry = dynamic_cast<TRestGeant4CRY*>(meta.get())) {
71 fCRY = cry;
72 meta.release();
73 }
74 }
75
76 // The cry generator takes the options of CRY of their own section, as a process its readout. With
77 // another generator the section is not used (nor stored): the same configuration for several generators
78 const bool cryGenerator = fG4Metadata && fG4Metadata->GetGeant4PrimaryGeneratorInfo().IsCRY();
79 if (cryGenerator && !fCRY) {
80 throw std::runtime_error(
81 "TRestGeant4Manager: the generator type cry needs a section of class TRestGeant4CRY (the options "
82 "of CRY) in the TRestGeant4Manager");
83 }
84 if (cryGenerator) {
85 fG4Metadata->GetGeant4PrimaryGeneratorInfo().SetCRY(*fCRY);
86 } else if (fCRY) {
87 delete fCRY;
88 fCRY = nullptr;
89 }
90}
