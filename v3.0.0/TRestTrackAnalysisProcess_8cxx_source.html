118// Sum and maximum of the node sizes of track t, with the sigmas of the coordinates each node defines
119void GetNodeSizes(const TRestTrackEvent& event, int t, double& size, double& maxNodeSize) {
120 size = maxNodeSize = 0;
121 const TRestHits& hits = event.GetTrack(t).GetHits();
122 for (size_t h = 0; h < hits.GetNumberOfHits(); h++) {
123 const auto sigma = event.GetSigma(t, (int)h);
124 const double sigmas[3] = {sigma.X(), sigma.Y(), sigma.Z()};
125 double size2 = 0;
126 for (int a = 0; a < 3; a++) {
127 if (TRestHitsData::IsDefined(hits.GetType(h), a)) size2 += sigmas[a] * sigmas[a];
128 }
129 const double nodeSize = std::sqrt(size2);
130 size += nodeSize;
131 maxNodeSize = std::max(maxNodeSize, nodeSize);
132 }
133}

135// Type of a set of hits: product of the coordinates any hit defines (the hit type for a track)
136int GetHitsType(const TRestHits& hits) {
137 bool defined[3] = {false, false, false};
138 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
139 for (int a = 0; a < 3; a++) defined[a] |= TRestHitsData::IsDefined(hits.GetType(n), a);
140 }
141 int type = 1;
142 for (int a = 0; a < 3; a++) {
143 if (defined[a]) type *= TRestHitsData::kAxisTypes[a];
144 }
145 return type;
146}

237void TRestTrackAnalysisProcess::InitProcess() {
238 // Projections and their pairs; the vectors do not change after this, so the registered references
239 // stay valid
240 const std::map<std::string, int> hitTypes = {{"X", TRestHitsData::X}, {"Y", TRestHitsData::Y},
241 {"Z", TRestHitsData::Z}, {"XY", TRestHitsData::XY},
242 {"XZ", TRestHitsData::XZ}, {"YZ", TRestHitsData::YZ},
243 {"XYZ", TRestHitsData::XYZ}};
244 fProjectionObs.clear();
245 for (const auto& name : fProjections) {
246 auto it = hitTypes.find(name);
247 if (it == hitTypes.end()) {
248 RESTError << "TRestTrackAnalysisProcess: unknown projection '" << name
249 << "' (X, Y, Z, XY, XZ, YZ or XYZ)" << RESTendl;
250 continue;
251 }
252 ProjectionObservables projection;
253 projection.name = name;
254 projection.type = it->second;
255 fProjectionObs.push_back(projection);
256 }
257
258 fBalances.clear();
259 for (size_t i = 0; i < fProjectionObs.size(); i++) {
260 for (size_t j = i + 1; j < fProjectionObs.size(); j++) {
261 BalanceObservables balance;
262 balance.first = i;
263 balance.second = j;
264 // The coordinate only each projection defines, if there is exactly one
265 int nFirst = 0, nSecond = 0;
266 for (int a = 0; a < 3; a++) {
267 const bool inFirst = TRestHitsData::IsDefined(fProjectionObs[i].type, a);
268 const bool inSecond = TRestHitsData::IsDefined(fProjectionObs[j].type, a);
269 if (inFirst && !inSecond) {
270 balance.firstAxis = a;
271 nFirst++;
272 }
273 if (inSecond && !inFirst) {
274 balance.secondAxis = a;
275 nSecond++;
276 }
277 }
278 if (nFirst != 1 || nSecond != 1) balance.firstAxis = balance.secondAxis = -1;
279 fBalances.push_back(balance);
280 }
281 }
282
283 RegisterObservable("NTracks", fNTracks);
284 RegisterObservable("TotalEnergy", fTotalEnergy);
285 RegisterObservable("MaxTrackEnergyRatio", fMaxTrackEnergyRatio);
286 if (fProjectionObs.empty()) {
287 // Any type: all the coordinates (NaN for those the track does not define)
288 RegisterTrackObservables("MaxTrack", fMaxTrack, TRestHitsData::XYZ);
289 } else {
290 // Hits of different projections together: no length, twist or size, they are those of
291 // MaxTrack_P_<name>.
292 // The coordinates any projection defines
293 int type = 1;
294 for (int a = 0; a < 3; a++) {
295 for (const auto& projection : fProjectionObs) {
296 if (!TRestHitsData::IsDefined(projection.type, a)) continue;
297 type *= TRestHitsData::kAxisTypes[a];
298 break;
299 }
300 }
301 RegisterObservable("MaxTrackEnergy", fMaxTrack.energy);
302 RegisterObservable("MaxTrackNHits", fMaxTrack.nHits);
303 RegisterCoordinateObservables("MaxTrack", fMaxTrack, type);
304 }
305 RegisterObservable("MaxTrackSkewXY", fMaxTrackSkewXY);
306 RegisterObservable("MaxTrackSkewZ", fMaxTrackSkewZ);
307
308 RegisterObservable("TracksEnergy", fTracks.energy);
309 RegisterObservable("TracksType", fTracks.type);
310 RegisterObservable("TracksNHits", fTracks.nHits);
311 RegisterObservable("TracksLength", fTracks.length);
312 RegisterObservable("TracksVolume", fTracks.volume);
313 RegisterObservable("TracksMeanX", fTracks.meanX);
314 RegisterObservable("TracksMeanY", fTracks.meanY);
315 RegisterObservable("TracksMeanZ", fTracks.meanZ);
316 RegisterObservable("TracksSigmaX", fTracks.sigmaX);
317 RegisterObservable("TracksSigmaY", fTracks.sigmaY);
318 RegisterObservable("TracksSigmaZ", fTracks.sigmaZ);
319 RegisterObservable("TracksGaussSigmaX", fTracks.gaussSigmaX);
320 RegisterObservable("TracksGaussSigmaY", fTracks.gaussSigmaY);
321 RegisterObservable("TracksGaussSigmaZ", fTracks.gaussSigmaZ);
322 RegisterObservable("TracksTwist", fTracks.twist);
323 RegisterObservable("TracksTwistWeighted", fTracks.twistWeighted);
324 RegisterObservable("TracksTwistLow", fTracks.twistLow);
325 RegisterObservable("TracksTwistHigh", fTracks.twistHigh);
326 RegisterObservable("TracksSize", fTracks.size);
327 RegisterObservable("TracksMaxNodeSize", fTracks.maxNodeSize);
328
329 for (auto& projection : fProjectionObs) {
330 RegisterObservable("NTracks_" + projection.name, projection.nTracks);
331 RegisterObservable("MaxTrack_" + projection.name + "_EnergyRatio", projection.maxTrackEnergyRatio);
332 RegisterTrackObservables("MaxTrack_" + projection.name + "_", projection.maxTrack, projection.type);
333 }
334 for (auto& balance : fBalances) {
335 const std::string pair =
336 fProjectionObs[balance.first].name + "_" + fProjectionObs[balance.second].name;
337 RegisterObservable("EnergyBalance_" + pair, balance.energy);
338 RegisterObservable("SigmaBalance_" + pair, balance.sigma);
339 }

342TRestTrackAnalysisProcess::TrackObservables TRestTrackAnalysisProcess::GetHitsObservables(
343 const TRestHits& hits) const {
344 TrackObservables obs;
345 const size_t nHits = hits.GetNumberOfHits();
346 obs.energy = hits.GetTotalEnergy();
347 obs.nHits = nHits;
348 if (nHits == 0) return obs;
349
350 obs.type = GetHitsType(hits);
351
352 const auto mean = hits.GetMeanPosition();
353 const double means[3] = {mean.X(), mean.Y(), mean.Z()};
354 const double sigmas[3] = {hits.GetSigmaX(), hits.GetSigmaY(), hits.GetSigmaZ()};
355 double* meanObs[3] = {&obs.meanX, &obs.meanY, &obs.meanZ};
356 double* sigmaObs[3] = {&obs.sigmaX, &obs.sigmaY, &obs.sigmaZ};
357 double* gaussObs[3] = {&obs.gaussSigmaX, &obs.gaussSigmaY, &obs.gaussSigmaZ};
358
359 const double nan = std::numeric_limits<double>::quiet_NaN();
360 for (int a = 0; a < 3; a++) {
361 const bool defined = TRestHitsData::IsDefined(obs.type, a);
362 *meanObs[a] = defined ? means[a] : nan;
363 *sigmaObs[a] = defined ? sigmas[a] : nan;
364 if (!defined) {
365 *gaussObs[a] = nan;
366 } else if (fGaussFit) {
367 // Only the hits that define the coordinate
368 std::vector<double> values, energies;
369 for (size_t n = 0; n < nHits; n++) {
370 if (!TRestHitsData::IsDefined(hits.GetType(n), a)) continue;
371 values.push_back(hits.GetCoordinate(n, a));
372 energies.push_back(hits.GetEnergy(n));
373 }
374 *gaussObs[a] = TRestHitsAnalysis::GetGaussSigma(values, energies);
375 }
376 }
377
378 obs.length = hits.GetTotalDistance();
379 obs.volume = hits.GetMaximumHitDistance2();
380
381 const auto points = hits.GetPoints();
382 const auto energies = hits.GetEnergies();
383 obs.twist = TRestHitsAnalysis::GetTwist(points);
384 obs.twistWeighted = TRestHitsAnalysis::GetTwistWeighted(points, energies);
385 // -1 if the tails have no hits (GetTwist would take the whole track)
386 const size_t nTail = (size_t)(fTwistTailPercentage * nHits / 100.);
387 const double twistStart = nTail > 0 ? TRestHitsAnalysis::GetTwist(points, 0, nTail) : -1;
388 const double twistEnd = nTail > 0 ? TRestHitsAnalysis::GetTwist(points, nHits - nTail, nHits) : -1;
389 obs.twistLow = std::min(twistStart, twistEnd);
390 obs.twistHigh = std::max(twistStart, twistEnd);
391 return obs;
392}

394bool TRestTrackAnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
395 output.CopyFrom(&input);
396 auto* event = dynamic_cast<TRestTrackEvent*>(&output);
397 if (!event) throw std::runtime_error("TRestTrackAnalysisProcess: output is not a TRestTrackEvent");
398
399 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) event->PrintOnlyTracks();
400
401 // Assignments keep the addresses of the registered members
402 fNTracks = event->GetNumberOfTracks();
403 // Rejected events skip the analysis: the cut applies to the number of tracks of each projection,
404 // or to the total without projections
405 if (fCutsEnabled) {
406 auto outOfRange = [this](int n) { return n < fNTracksCut.first || n > fNTracksCut.second; };
407 if (fProjectionObs.empty() && outOfRange(fNTracks)) return false;
408 for (const auto& projection : fProjectionObs) {
409 int nTracks = 0;
410 for (int t = 0; t < fNTracks; t++)
411 nTracks += GetHitsType(event->GetTrack(t).GetHits()) == projection.type;
412 if (outOfRange(nTracks)) return false;
413 }
414 }
415 fTotalEnergy = event->GetEnergy();
416 fTracks.clear();
417 for (auto& projection : fProjectionObs) {
418 projection.nTracks = 0;
419 projection.energy = 0;
420 projection.maxTrackEnergyRatio = 0;
421 projection.maxTrack = TrackObservables();
422 projection.maxTrackIndex = -1;
423 }
424
425 std::vector<int> order(fNTracks);
426 std::iota(order.begin(), order.end(), 0);
427 std::stable_sort(order.begin(), order.end(),
428 [event](int a, int b) { return event->GetTrackEnergy(a) > event->GetTrackEnergy(b); });
429
430 // Main tracks: the most energetic one or, with projections, the most energetic of each projection
431 std::vector<TrackObservables> trackObs(order.size());
432 std::vector<bool> isMain(order.size(), false);
433 if (fProjectionObs.empty() && !order.empty()) isMain[0] = true;
434 for (size_t i = 0; i < order.size(); i++) {
435 trackObs[i] = GetHitsObservables(event->GetTrack(order[i]).GetHits());
436 GetNodeSizes(*event, order[i], trackObs[i].size, trackObs[i].maxNodeSize);
437
438 // In decreasing energy: the first track of each projection is its main track
439 for (auto& projection : fProjectionObs) {
440 if (trackObs[i].type != projection.type) continue;
441 projection.energy += trackObs[i].energy;
442 if (projection.nTracks++ == 0) {
443 projection.maxTrack = trackObs[i];
444 projection.maxTrackIndex = order[i];
445 isMain[i] = true;
446 }
447 }
448 }
449 for (size_t i = 0; i < order.size(); i++) {
450 if (!isMain[i]) fTracks.push_back(trackObs[i]);
451 }
452
453 TRestHitsData data;
454 TRestHits mainHits(&data);
455 for (size_t i = 0; i < order.size(); i++) {
456 if (isMain[i]) mainHits.AddHits(event->GetTrack(order[i]).GetHits());
457 }
458 fMaxTrack = GetHitsObservables(mainHits);
459 // Without projections the main track is a single track: its node sizes
460 if (fProjectionObs.empty() && !order.empty()) {
461 fMaxTrack.size = trackObs[0].size;
462 fMaxTrack.maxNodeSize = trackObs[0].maxNodeSize;
463 }
464 const bool hasHits = mainHits.GetNumberOfHits() > 0;
465 fMaxTrackSkewXY = hasHits ? mainHits.GetSkewXY() : 0;
466 fMaxTrackSkewZ = hasHits ? mainHits.GetSkewZ() : 0;
467
468 auto ratio = [](double energy, double mainEnergy) {
469 return energy != 0 ? (energy - mainEnergy) / energy : 0;
470 };
471 fMaxTrackEnergyRatio = ratio(fTotalEnergy, fMaxTrack.energy);
472 for (auto& projection : fProjectionObs) {
473 projection.maxTrackEnergyRatio = ratio(projection.energy, projection.maxTrack.energy);
474 }
475
476 auto sigmaOf = [](const TrackObservables& obs, int axis) {
477 return axis == 0 ? obs.sigmaX : axis == 1 ? obs.sigmaY : obs.sigmaZ;
478 };
479 for (auto& balance : fBalances) {
480 const auto& a = fProjectionObs[balance.first].maxTrack;
481 const auto& b = fProjectionObs[balance.second].maxTrack;
482 balance.energy = Balance(a.energy, b.energy);
483 balance.sigma = balance.firstAxis < 0
484 ? std::numeric_limits<double>::quiet_NaN()
485 : Balance(sigmaOf(a, balance.firstAxis), sigmaOf(b, balance.secondAxis));
486 }
487
488 return true;
489}
