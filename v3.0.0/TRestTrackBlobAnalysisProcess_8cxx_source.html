75// Type of a set of hits: product of the coordinates any hit defines (the hit type for a track)
76int GetHitsType(const TRestHits& hits) {
77 bool defined[3] = {false, false, false};
78 for (size_t n = 0; n < hits.GetNumberOfHits(); n++) {
79 for (int a = 0; a < 3; a++) defined[a] |= TRestHitsData::IsDefined(hits.GetType(n), a);
80 }
81 int type = 1;
82 for (int a = 0; a < 3; a++) {
83 if (defined[a]) type *= TRestHitsData::kAxisTypes[a];
84 }
85 return type;
86}

120void TRestTrackBlobAnalysisProcess::InitProcess() {
121 if (fHitsToCheckFraction <= 0 || fHitsToCheckFraction > 0.5) {
122 throw std::runtime_error("TRestTrackBlobAnalysisProcess: hitsToCheckFraction must be in (0, 0.5]");
123 }
124 for (const auto& radius : fRadii) {
125 if (radius <= 0) throw std::runtime_error("TRestTrackBlobAnalysisProcess: the radii must be > 0");
126 }
127
128 const std::map<std::string, int> hitTypes = {{"X", TRestHitsData::X}, {"Y", TRestHitsData::Y},
129 {"Z", TRestHitsData::Z}, {"XY", TRestHitsData::XY},
130 {"XZ", TRestHitsData::XZ}, {"YZ", TRestHitsData::YZ},
131 {"XYZ", TRestHitsData::XYZ}};
132 // The vectors do not change after this, so the registered references stay valid
133 fBlobObs.clear();
134 if (fProjections.empty()) fBlobObs.emplace_back();
135 for (const auto& name : fProjections) {
136 auto it = hitTypes.find(name);
137 if (it == hitTypes.end()) {
138 throw std::runtime_error("TRestTrackBlobAnalysisProcess: unknown projection '" + name +
139 "' (X, Y, Z, XY, XZ, YZ or XYZ)");
140 }
141 BlobObservables obs;
142 obs.name = name;
143 obs.type = it->second;
144 fBlobObs.push_back(obs);
145 }
146
147 for (auto& obs : fBlobObs) {
148 for (auto* q : {&obs.q1, &obs.q2, &obs.qHigh, &obs.qLow, &obs.qBalance, &obs.qRatio})
149 q->assign(fRadii.size(), 0);
150
151 const std::string suffix = obs.name.empty() ? "" : "_" + obs.name;
152 RegisterObservable("X1" + suffix, obs.x1);
153 RegisterObservable("Y1" + suffix, obs.y1);
154 RegisterObservable("Z1" + suffix, obs.z1);
155 RegisterObservable("X2" + suffix, obs.x2);
156 RegisterObservable("Y2" + suffix, obs.y2);
157 RegisterObservable("Z2" + suffix, obs.z2);
158 RegisterObservable("Distance" + suffix, obs.distance);
159 RegisterObservable("Q1" + suffix, obs.q1);
160 RegisterObservable("Q2" + suffix, obs.q2);
161 RegisterObservable("QHigh" + suffix, obs.qHigh);
162 RegisterObservable("QLow" + suffix, obs.qLow);
163 RegisterObservable("QBalance" + suffix, obs.qBalance);
164 RegisterObservable("QRatio" + suffix, obs.qRatio);
165 }

168void TRestTrackBlobAnalysisProcess::FillObservables(const TRestHits* hits, BlobObservables& obs) const {
169 const double nan = std::numeric_limits<double>::quiet_NaN();
170 // Assignments keep the addresses of the registered members
171 obs.x1 = obs.y1 = obs.z1 = obs.x2 = obs.y2 = obs.z2 = obs.distance = nan;
172 for (auto* q : {&obs.q1, &obs.q2, &obs.qHigh, &obs.qLow, &obs.qBalance, &obs.qRatio})
173 std::fill(q->begin(), q->end(), 0);
174
175 const int nHits = hits ? (int)hits->GetNumberOfHits() : 0;
176 if (nHits == 0) return;
177
178 // Most energetic node of each end (the first or last node if none has energy)
179 const int nCheck = std::max(1, (int)(nHits * fHitsToCheckFraction));
180 int hit1 = hits->GetMostEnergeticHitInRange(0, nCheck);
181 int hit2 = hits->GetMostEnergeticHitInRange(nHits - nCheck, nHits);
182 if (hit1 < 0) hit1 = 0;
183 if (hit2 < 0) hit2 = nHits - 1;
184
185 const int type = GetHitsType(*hits);
186 // Blob 1 closer to z = 0
187 if (TRestHitsData::IsDefined(type, 2) && std::abs(hits->GetZ(hit2)) < std::abs(hits->GetZ(hit1)))
188 std::swap(hit1, hit2);
189
190 // Undefined coordinates are 0 in the points, so the distances are those of the projection
191 const auto points = hits->GetPoints();
192 const auto blob1 = points[hit1], blob2 = points[hit2];
193 obs.distance = std::sqrt((blob1 - blob2).Mag2());
194
195 const double coords1[3] = {blob1.X(), blob1.Y(), blob1.Z()};
196 const double coords2[3] = {blob2.X(), blob2.Y(), blob2.Z()};
197 double* obs1[3] = {&obs.x1, &obs.y1, &obs.z1};
198 double* obs2[3] = {&obs.x2, &obs.y2, &obs.z2};
199 for (int a = 0; a < 3; a++) {
200 if (!TRestHitsData::IsDefined(type, a)) continue;
201 *obs1[a] = coords1[a];
202 *obs2[a] = coords2[a];
203 }
204
205 for (size_t r = 0; r < fRadii.size(); r++) {
206 const double radius2 = fRadii[r] * fRadii[r];
207 double q1 = 0, q2 = 0;
208 for (int n = 0; n < nHits; n++) {
209 if ((points[n] - blob1).Mag2() <= radius2) q1 += hits->GetEnergy(n);
210 if ((points[n] - blob2).Mag2() <= radius2) q2 += hits->GetEnergy(n);
211 }
212 const double high = std::max(q1, q2), low = std::min(q1, q2);
213 obs.q1[r] = q1;
214 obs.q2[r] = q2;
215 obs.qHigh[r] = high;
216 obs.qLow[r] = low;
217 obs.qBalance[r] = high + low != 0 ? (high - low) / (high + low) : 0;
218 obs.qRatio[r] = low != 0 ? high / low : 0;
219 }
220}

222bool TRestTrackBlobAnalysisProcess::ProcessEvent(const TRestEvent& input, TRestEvent& output) {
223 output.CopyFrom(&input);
224 auto* event = dynamic_cast<TRestTrackEvent*>(&output);
225 if (!event) throw std::runtime_error("TRestTrackBlobAnalysisProcess: output is not a TRestTrackEvent");
226
227 if (GetVerboseLevel() >= TRestLogManager::REST_Verbose_Level::REST_Debug) event->PrintOnlyTracks();
228
229 for (auto& obs : fBlobObs) {
230 // Most energetic track of the projection type (of any type without projections)
231 int mainTrack = -1;
232 for (int t = 0; t < event->GetNumberOfTracks(); t++) {
233 if (obs.type != 0 && GetHitsType(event->GetTrack(t).GetHits()) != obs.type) continue;
234 if (mainTrack < 0 || event->GetTrackEnergy(t) > event->GetTrackEnergy(mainTrack)) mainTrack = t;
235 }
236 if (mainTrack < 0) {
237 FillObservables(nullptr, obs);
238 continue;
239 }
240 const TRestHits hits = event->GetTrack(mainTrack).GetHits();
241 FillObservables(&hits, obs);
242 }
243
244 return true;
245}
