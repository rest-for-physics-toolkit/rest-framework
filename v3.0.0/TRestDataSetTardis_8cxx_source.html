137class ChannelMap {
138 public:
139 ChannelMap(const TRestDetectorReadout& readout, int tolerance)
140 : fReadout(readout), fTolerance(tolerance) {}
141
144 int Group(int physicalID) { return Get(physicalID).group; }
146 const std::vector<int>& Near(int physicalID) { return Get(physicalID).near; }
147
148 private:
149 struct Channel {
150 int group = -1;
151 std::vector<int> near;
152 };
153 const Channel& Get(int physicalID) {
154 const auto found = fChannels.find(physicalID);
155 if (found != fChannels.end()) return found->second;
156 Channel& channel = fChannels[physicalID];
157 const int daqID = physicalID < 0 ? -1 : fReadout.GetDAQID(physicalID);
158 if (daqID < 0) return channel;
159 const auto position = fReadout.GetPositionFromChannel(daqID);
160 channel.group = (std::isfinite(position.X()) ? 1 : 0) | (std::isfinite(position.Y()) ? 2 : 0) |
161 (std::isfinite(position.Z()) ? 4 : 0);
162 // Neighbours in steps of GetAdjacentChannels (DAQ IDs)
163 std::set<int> near = {daqID};
164 std::vector<int> front = {daqID};
165 for (int step = 0; step < fTolerance && !front.empty(); step++) {
166 std::vector<int> next;
167 for (const int daq : front) {
168 const auto [previous, following] = fReadout.GetAdjacentChannels(daq);
169 for (const int adjacent : {previous, following}) {
170 if (adjacent >= 0 && near.insert(adjacent).second) next.push_back(adjacent);
171 }
172 }
173 front = std::move(next);
174 }
175 for (const int daq : near) channel.near.push_back(fReadout.GetPhysicalID(daq));
176 return channel;
177 }
178
179 const TRestDetectorReadout& fReadout;
180 const int fTolerance;
181 std::unordered_map<int, Channel> fChannels;
182};

185std::vector<double> TRestDataSetTardis::TimeDelays(const std::vector<Entry>& entries,
186 const TRestDetectorReadout* readout,
187 int channelTolerance) {
188 std::vector<double> delays(entries.size(), kNaN);
189 if (!readout) {
190 int last = -1; // the last entry that can be the previous one
191 for (size_t i = 0; i < entries.size(); i++) {
192 if (last >= 0) delays[i] = Delay(entries[last], entries[i]);
193 if (entries[i].canBePrevious) last = static_cast<int>(i);
194 }
195 return delays;
196 }
197
198 ChannelMap map(*readout, channelTolerance);
199 // Entries (indices, increasing) with each physical channel, only those that can be the previous one
200 std::unordered_map<int, std::vector<int>> history;
201 // The physical channels of the readout of each entry (sorted) and its groups
202 std::vector<std::vector<int>> channels(entries.size());
203 std::vector<std::set<int>> groups(entries.size());
204
205 // The last entry before `before` with a channel of `near`, -1 if none
206 auto lastWith = [&history](const std::vector<int>& near, int before) {
207 int last = -1;
208 for (const int channel : near) {
209 const auto found = history.find(channel);
210 if (found == history.end()) continue;
211 const auto& entriesOf = found->second;
212 auto it = std::lower_bound(entriesOf.begin(), entriesOf.end(), before);
213 if (it != entriesOf.begin()) last = std::max(last, *std::prev(it));
214 }
215 return last;
216 };
217 auto hasAny = [](const std::vector<int>& sorted, const std::vector<int>& near) {
218 for (const int channel : near) {
219 if (std::binary_search(sorted.begin(), sorted.end(), channel)) return true;
220 }
221 return false;
222 };
223
224 for (size_t e = 0; e < entries.size(); e++) {
225 // The channels of the readout of the entry and the near channels of each of its groups
226 std::map<int, std::vector<int>> nearByGroup;
227 for (const int physicalID : entries[e].channels) {
228 const int group = map.Group(physicalID);
229 if (group < 0) continue;
230 channels[e].push_back(physicalID);
231 groups[e].insert(group);
232 auto& near = nearByGroup[group];
233 const auto& channelNear = map.Near(physicalID);
234 near.insert(near.end(), channelNear.begin(), channelNear.end());
235 }
236 std::sort(channels[e].begin(), channels[e].end());
237 channels[e].erase(std::unique(channels[e].begin(), channels[e].end()), channels[e].end());
238 for (auto& [group, near] : nearByGroup) {
239 std::sort(near.begin(), near.end());
240 near.erase(std::unique(near.begin(), near.end()), near.end());
241 }
242
243 // The candidates, from the last one: an entry near in a group; accepted if it is near in all the
244 // groups that both entries have
245 auto lastCandidate = [&](int before) {
246 int last = -1;
247 for (const auto& [group, near] : nearByGroup) last = std::max(last, lastWith(near, before));
248 return last;
249 };
250 for (int k = lastCandidate(static_cast<int>(e)); k >= 0; k = lastCandidate(k)) {
251 bool same = true;
252 for (const auto& [group, near] : nearByGroup) {
253 if (groups[k].count(group) && !hasAny(channels[k], near)) {
254 same = false;
255 break;
256 }
257 }
258 if (same) {
259 delays[e] = Delay(entries[k], entries[e]);
260 break;
261 }
262 }
263 if (entries[e].canBePrevious) {
264 for (const int channel : channels[e]) history[channel].push_back(static_cast<int>(e));
265 }
266 }
267 return delays;
268}

296void TRestDataSetTardis::Apply(TRestDataSet& dataSet) const {
297 const std::string context = GetClassName() + " '" + fName + "'";
298 ROOT::RDF::RNode frame = dataSet.GetDataFrame();
299 for (const std::string column : {"runIndex", "eventID", "subEventID", "timeSeconds", "timeNanoSecs"}) {
300 if (!frame.HasColumn(column)) {
301 throw std::runtime_error(context + ": the dataset has no column " + column +
302 " (generate it again with timeSeconds and timeNanoSecs)");
303 }
304 }
305 const std::string cuts = GetCutExpression(fCutNames);
306 if (!cuts.empty()) frame = frame.Filter(cuts, fName);
307
308 // The readout of the channels
309 const TRestDetectorReadout* readout = nullptr;
310 if (!fChannels.empty()) {
311 const auto managers = dataSet.GetMetadataOfType<TRestDetectorReadoutManager>();
312 if (managers.size() != 1) {
313 throw std::runtime_error(context + ": the dataset has " + std::to_string(managers.size()) +
314 " TRestDetectorReadoutManager (one expected for channels)");
315 }
316 std::string name = fReadout;
317 readout = managers.front()->SelectReadout(name, context);
318 }
319
320 // The timestamps and the channels of the entries (a single event loop)
321 auto runs = frame.Take<int>("runIndex");
322 auto events = frame.Take<int>("eventID");
323 auto subEvents = frame.Take<int>("subEventID");
324 auto seconds = frame.Take<long long>("timeSeconds");
325 auto nanoseconds = frame.Take<int>("timeNanoSecs");
326 ROOT::RDF::RResultPtr<std::vector<double>> previousValues;
327 if (!fPreviousObservable.empty()) {
328 if (!frame.HasColumn(fPreviousObservable)) {
329 throw std::runtime_error(context + ": the dataset has no column " + fPreviousObservable);
330 }
331 previousValues = frame.Take<double>(AsDouble(frame, fPreviousObservable));
332 }
333 ROOT::RDF::RResultPtr<std::vector<ROOT::RVec<int>>> channels;
334 if (readout) {
335 for (const auto& column : {fChannels, fChannelObservable}) {
336 if (!column.empty() && !frame.HasColumn(column)) {
337 throw std::runtime_error(context + ": the dataset has no column " + column);
338 }
339 }
340 // The channels, those over the threshold with channelObservable (a mask of the vector)
341 std::string expression = fChannels;
342 if (!fChannelObservable.empty()) {
343 char threshold[64];
344 std::snprintf(threshold, sizeof(threshold), "%.17g", fChannelThreshold);
345 expression += "[" + fChannelObservable + " > " + threshold + "]";
346 }
347 const std::string channelsColumn = "__tardisChannels_" + fName;
348 frame = frame.Define(channelsColumn, expression);
349 channels = frame.Take<ROOT::RVec<int>>(channelsColumn);
350 }
351 const size_t nEntries = runs->size();
352
353 std::vector<size_t> order(nEntries);
354 std::iota(order.begin(), order.end(), 0);
355 for (size_t n = 0; n < nEntries; n++) {
356 if ((*nanoseconds)[n] < 0 || (*nanoseconds)[n] >= 1000000000) {
357 throw std::runtime_error(context + ": timeNanoSecs " + std::to_string((*nanoseconds)[n]) +
358 " outside [0, 1e9) in the event " + std::to_string((*events)[n]));
359 }
360 }
361 std::sort(order.begin(), order.end(), [&](size_t a, size_t b) {
362 return std::tie((*seconds)[a], (*nanoseconds)[a], (*runs)[a], (*events)[a], (*subEvents)[a]) <
363 std::tie((*seconds)[b], (*nanoseconds)[b], (*runs)[b], (*events)[b], (*subEvents)[b]);
364 });
365
366 std::vector<Entry> entries(nEntries);
367 for (size_t i = 0; i < nEntries; i++) {
368 const size_t n = order[i];
369 entries[i].seconds = (*seconds)[n];
370 entries[i].nanoseconds = (*nanoseconds)[n];
371 if (readout) entries[i].channels.assign((*channels)[n].begin(), (*channels)[n].end());
372 if (!fPreviousObservable.empty()) {
373 entries[i].canBePrevious = (*previousValues)[n] > fPreviousThreshold;
374 }
375 }
376 const auto timeDelays = TimeDelays(entries, readout, fChannelTolerance);
377
378 auto delays = std::make_shared<std::unordered_map<EventKey, double, EventKeyHash>>();
379 delays->reserve(nEntries);
380 for (size_t i = 0; i < nEntries; i++) {
381 const size_t n = order[i];
382 const EventKey key{(*runs)[n], (*events)[n], (*subEvents)[n]};
383 if (!delays->emplace(key, timeDelays[i]).second) {
384 throw std::runtime_error(context + ": the event " + std::to_string(key.event) + " (subevent " +
385 std::to_string(key.subEvent) + ") of the run index " +
386 std::to_string(key.run) + " is twice in the dataset");
387 }
388 }
389 RESTInfo << context << ": time delays of " << nEntries << " entries"
390 << (readout ? " in the same channels of " + readout->GetName() : std::string()) << RESTendl;
391
392 const std::string delayColumn = fName + "_timeDelay";
393 frame = DefineColumn(frame, delayColumn,
394 [delays](int run, int event, int subEvent) {
395 const auto found = delays->find({run, event, subEvent});
396 return found == delays->end() ? kNaN : found->second;
397 },
398 {"runIndex", "eventID", "subEventID"});
399 dataSet.SetDataFrame(frame, {delayColumn});
